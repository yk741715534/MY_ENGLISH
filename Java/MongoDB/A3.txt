@Configuration
public class AwsKmsConfig {

    @Value("${mongodb.uri:mongodb://localhost:27017}")
    private String mongoUri;

    @Value("${aws.kms.key.arn:}")
    private String kmsKeyArn;

    @Value("${aws.access.key:}")
    private String awsAccessKey;

    @Value("${aws.secret.key:}")
    private String awsSecretKey;

    @Value("${aws.region:us-east-1}")
    private String awsRegion;

    @Bean
    public MongoClient mongoClient() {
        // 1. 配置 AWS KMS 提供者
        Map<String, Map<String, Object>> kmsProviders = new HashMap<>();
        Map<String, Object> awsKmsConfig = new HashMap<>();

        if (awsAccessKey != null && !awsAccessKey.isEmpty()) {
            // 使用显式凭证
            awsKmsConfig.put("accessKeyId", awsAccessKey);
            awsKmsConfig.put("secretAccessKey", awsSecretKey);
        } else {
            // 从环境变量或 IAM 角色自动获取
            awsKmsConfig.put("accessKeyId", System.getenv("AWS_ACCESS_KEY_ID"));
            awsKmsConfig.put("secretAccessKey", System.getenv("AWS_SECRET_ACCESS_KEY"));
        }

        kmsProviders.put("aws", awsKmsConfig);

        // 2. 构建加密模式（JSON Schema）
        Map<String, BsonDocument> schemaMap = new HashMap<>();

        // 示例：用户表的加密模式
        String userSchema = """
        {
          "bsonType": "object",
          "encryptMetadata": {
            "keyId": [
              {
                "$binary": {
                  "base64": "%s",
                  "subType": "04"
                }
              }
            ]
          },
          "properties": {
            "pii": {
              "bsonType": "object",
              "properties": {
                "ssn": {
                  "encrypt": {
                    "bsonType": "string",
                    "algorithm": "AEAD_AES_256_CBC_HMAC_SHA_512-Random"
                  }
                },
                "phone": {
                  "encrypt": {
                    "bsonType": "string",
                    "algorithm": "AEAD_AES_256_CBC_HMAC_SHA_512-Deterministic"
                  }
                },
                "email": {
                  "encrypt": {
                    "bsonType": "string",
                    "algorithm": "AEAD_AES_256_CBC_HMAC_SHA_512-Deterministic"
                  }
                }
              }
            },
            "creditCard": {
              "encrypt": {
                "bsonType": "string",
                "algorithm": "AEAD_AES_256_CBC_HMAC_SHA_512-Random"
              }
            },
            "medical": {
              "bsonType": "object",
              "properties": {
                "diagnosis": {
                  "encrypt": {
                    "bsonType": "string",
                    "algorithm": "AEAD_AES_256_CBC_HMAC_SHA_512-Random"
                  }
                },
                "prescription": {
                  "encrypt": {
                    "bsonType": "string",
                    "algorithm": "AEAD_AES_256_CBC_HMAC_SHA_512-Deterministic"
                  }
                }
              }
            }
          }
        }
        """;

        // 生成或获取主密钥ID
        String keyId = getOrCreateMasterKey();
        String formattedSchema = String.format(userSchema,
            Base64.getEncoder().encodeToString(keyId.getBytes()));

        schemaMap.put("clinic.patients", BsonDocument.parse(formattedSchema));

        // 3. 创建自动加密设置
        AutoEncryptionSettings autoEncryptionSettings = AutoEncryptionSettings.builder()
            .keyVaultNamespace("encryption.__keyVault")
            .kmsProviders(kmsProviders)
            .schemaMap(schemaMap)
            .extraOptions(Map.of(
                "cryptSharedLibPath", "/usr/lib/mongo_crypt_v1.so"  // 加密库路径
            ))
            .build();

        // 4. 创建 MongoClient
        MongoClientSettings settings = MongoClientSettings.builder()
            .applyConnectionString(new ConnectionString(mongoUri))
            .autoEncryptionSettings(autoEncryptionSettings)
            .uuidRepresentation(UuidRepresentation.STANDARD)
            .build();

        return MongoClients.create(settings);
    }

    /**
     * 获取或创建主数据加密密钥
     */
    private String getOrCreateMasterKey() {
        try (MongoClient client = MongoClients.create(mongoUri)) {
            var keyVault = client.getDatabase("encryption")
                .getCollection("__keyVault");

            // 查找现有密钥
            var existingKey = keyVault.find()
                .filter(new Document("keyAltNames", "aws_master_key"))
                .first();

            if (existingKey != null) {
                return existingKey.get("_id").toString();
            }

            // 创建新密钥
            UUID keyUUID = UUID.randomUUID();
            BsonDocument masterKey = new BsonDocument();
            masterKey.append("provider", new BsonString("aws"));
            masterKey.append("key", new BsonString(kmsKeyArn));
            masterKey.append("region", new BsonString(awsRegion));

            BsonDocument keyDocument = new BsonDocument();
            keyDocument.append("_id", new BsonBinary(BsonBinarySubType.UUID_STANDARD,
                uuidToBytes(keyUUID)));
            keyDocument.append("keyMaterial", new BsonBinary(new byte[96])); // 占位
            keyDocument.append("creationDate", new BsonInt32((int) (System.currentTimeMillis() / 1000)));
            keyDocument.append("updateDate", new BsonInt32((int) (System.currentTimeMillis() / 1000)));
            keyDocument.append("status", new BsonInt32(0)); // 0 = enabled
            keyDocument.append("masterKey", masterKey);
            keyDocument.append("keyAltNames", new BsonArray(
                Arrays.asList(new BsonString("aws_master_key"))
            ));

            keyVault.insertOne(new Document(keyDocument));

            return keyUUID.toString();

        } catch (Exception e) {
            throw new RuntimeException("Failed to create master key", e);
        }
    }

    private static byte[] uuidToBytes(UUID uuid) {
        ByteBuffer bb = ByteBuffer.wrap(new byte[16]);
        bb.putLong(uuid.getMostSignificantBits());
        bb.putLong(uuid.getLeastSignificantBits());
        return bb.array();
    }
}

public class GcpKmsConfig {

    @Value("${gcp.project.id:}")
    private String projectId;

    @Value("${gcp.location:global}")
    private String location;

    @Value("${gcp.keyring:mongodb-fle-keyring}")
    private String keyRing;

    @Value("${gcp.key.name:mongodb-fle-key}")
    private String keyName;

    @Bean
    public MongoClient mongoClient() throws IOException {
        // 从服务账号密钥文件加载凭证
        String credentialsJson = System.getenv("GOOGLE_APPLICATION_CREDENTIALS");

        Map<String, Map<String, Object>> kmsProviders = new HashMap<>();
        Map<String, Object> gcpConfig = new HashMap<>();

        if (credentialsJson != null) {
            // 从JSON文件加载
            gcpConfig.put("privateKey", credentialsJson);
        } else {
            // 从默认位置加载
            gcpConfig.put("privateKey", "");
        }

        kmsProviders.put("gcp", gcpConfig);

        // 构建主密钥文档
        String keyPath = String.format(
            "projects/%s/locations/%s/keyRings/%s/cryptoKeys/%s",
            projectId, location, keyRing, keyName
        );

        BsonDocument masterKey = new BsonDocument();
        masterKey.append("projectId", new BsonString(projectId));
        masterKey.append("location", new BsonString(location));
        masterKey.append("keyRing", new BsonString(keyRing));
        masterKey.append("keyName", new BsonString(keyName));
        masterKey.append("keyVersion", new BsonString("1"));

        AutoEncryptionSettings autoEncryptionSettings = AutoEncryptionSettings.builder()
            .keyVaultNamespace("encryption.__keyVault")
            .kmsProviders(kmsProviders)
            .build();

        MongoClientSettings settings = MongoClientSettings.builder()
            .applyConnectionString(new ConnectionString(mongoUri))
            .autoEncryptionSettings(autoEncryptionSettings)
            .build();

        return MongoClients.create(settings);
    }

@Component
public class KMSProviderFactory {

    public enum KMSProviderType {
        AWS, AZURE, GCP, VAULT, LOCAL
    }

    public Map<String, Map<String, Object>> createKMSProvider(
            KMSProviderType type,
            Map<String, String> config) {

        Map<String, Map<String, Object>> kmsProviders = new HashMap<>();

        switch (type) {
            case AWS:
                kmsProviders.put("aws", createAWSProvider(config));
                break;
            case AZURE:
                kmsProviders.put("azure", createAzureProvider(config));
                break;
            case GCP:
                kmsProviders.put("gcp", createGCPProvider(config));
                break;
            case VAULT:
                kmsProviders.put("local", createVaultProvider(config));
                break;
            case LOCAL:
                kmsProviders.put("local", createLocalProvider(config));
                break;
            default:
                throw new IllegalArgumentException("Unsupported KMS provider: " + type);
        }

        return kmsProviders;
    }

    private Map<String, Object> createAWSProvider(Map<String, String> config) {
        Map<String, Object> awsConfig = new HashMap<>();

        // 优先使用环境变量
        String accessKey = System.getenv("AWS_ACCESS_KEY_ID");
        String secretKey = System.getenv("AWS_SECRET_ACCESS_KEY");
        String sessionToken = System.getenv("AWS_SESSION_TOKEN");

        if (accessKey != null && secretKey != null) {
            awsConfig.put("accessKeyId", accessKey);
            awsConfig.put("secretAccessKey", secretKey);
            if (sessionToken != null) {
                awsConfig.put("sessionToken", sessionToken);
            }
        } else {
            // 回退到配置文件
            awsConfig.put("accessKeyId", config.get("accessKey"));
            awsConfig.put("secretAccessKey", config.get("secretKey"));
        }

        return awsConfig;
    }

    private Map<String, Object> createAzureProvider(Map<String, String> config) {
        Map<String, Object> azureConfig = new HashMap<>();

        azureConfig.put("tenantId", config.get("tenantId"));
        azureConfig.put("clientId", config.get("clientId"));
        azureConfig.put("clientSecret", config.get("clientSecret"));

        // 可选：使用托管身份
        if ("true".equals(config.get("useManagedIdentity"))) {
            azureConfig.put("useManagedIdentity", true);
            azureConfig.put("identityPlatformEndpoint",
                config.getOrDefault("identityEndpoint", ""));
        }

        return azureConfig;
    }

    private Map<String, Object> createLocalProvider(Map<String, String> config) {
        Map<String, Object> localConfig = new HashMap<>();

        // 从Base64解码本地密钥
        String base64Key = config.get("key");
        if (base64Key != null && !base64Key.isEmpty()) {
            byte[] keyBytes = Base64.getDecoder().decode(base64Key);
            localConfig.put("key", keyBytes);
        } else {
            // 生成新的本地密钥
            SecureRandom random = new SecureRandom();
            byte[] key = new byte[96];
            random.nextBytes(key);
            localConfig.put("key", key);
        }

        return localConfig;
    }
}
