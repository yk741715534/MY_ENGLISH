@Service
public class UserService {

    // 加密器组件
    @Autowired
    private FieldEncryptor encryptor;

    public User saveUser(User user) {
        // 手动加密敏感字段
        user.setEncryptedSsn(encryptor.encrypt(user.getSsn()));
        user.setEncryptedEmail(encryptor.encrypt(user.getEmail()));

        // 保存到数据库（已经是密文）
        return userRepository.save(user);
    }

    public User getUser(String id) {
        User user = userRepository.findById(id).orElse(null);
        if (user != null) {
            // 手动解密
            user.setSsn(encryptor.decrypt(user.getEncryptedSsn()));
            user.setEmail(encryptor.decrypt(user.getEncryptedEmail()));
        }
        return user;
    }
}

// 应用层加密：可能泄露信息
public class TimingAttackVulnerable {
    public boolean verifyPassword(String input, String stored) {
        // ❌ 时序攻击漏洞
        for (int i = 0; i < input.length(); i++) {
            if (input.charAt(i) != stored.charAt(i)) {
                return false; // 不同字符位置导致不同返回时间
            }
        }
        return input.length() == stored.length();
    }
}

@Service
@Slf4j
public class MigrationService {

    @Autowired
    private OldEncryptedRepository oldRepo;

    @Autowired
    private NewFleRepository newRepo;

    @Autowired
    private OldDecryptor oldDecryptor;

    @Transactional
    public void migrateData() {
        // 阶段1: 双写
        log.info("开始数据迁移...");

        List<OldEntity> oldData = oldRepo.findAll();
        for (OldEntity old : oldData) {
            // 解密旧数据
            String decryptedSsn = oldDecryptor.decrypt(old.getEncryptedSsn());
            String decryptedEmail = oldDecryptor.decrypt(old.getEncryptedEmail());

            // 创建新实体（FLE自动加密）
            NewEntity newEntity = new NewEntity();
            newEntity.setId(old.getId());
            newEntity.setSsn(decryptedSsn);  // 保存时FLE自动加密
            newEntity.setEmail(decryptedEmail);

            // 双写到新集合
            newRepo.save(newEntity);
        }

        // 阶段2: 验证
        verifyMigration();

        // 阶段3: 切换读取
        // 更新应用读取新集合

        // 阶段4: 清理
        // 可选：删除旧数据
    }
}

最终建议

对于大多数企业应用，我推荐：

新项目：直接使用 FLE 2.0

省时省力

安全合规

长期维护成本低

现有项目：评估后决定

如果已有稳定的应用层加密，且无查询需求，可保持

如果需要合规或查询功能，逐步迁移到FLE

混合方案：最佳实践

复制
最敏感字段 → FLE 2.0随机加密
需要查询的敏感字段 → FLE 2.0确定性加密
非敏感字段 → 不加密或应用层简单加密

核心原则：不要重复发明轮子，除非你有特殊需求且知道自己在做什么。大多数情况下，FLE 2.0是更优选择。
