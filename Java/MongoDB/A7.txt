@Configuration
@Slf4j
public class SmartMongoConfig {

    @Bean
    public MongoClient mongoClient() {
        Map<String, Object> extraOptions = new HashMap<>();

        // 1. 查找加密库
        String libPath = findCryptSharedLib();

        if (libPath != null) {
            extraOptions.put("cryptSharedLibPath", libPath);
            log.info("✅ 使用加密库: {}", libPath);
        } else {
            log.warn("⚠️  未找到加密库，将使用纯Java实现（性能较差）");
            // 可以不设置，让驱动回退
        }

        // 2. 根据环境决定是否必需
        boolean isProduction = "prod".equals(System.getenv("APP_ENV"));
        if (isProduction) {
            extraOptions.put("cryptSharedLibRequired", true); // 生产环境必须
        }

        AutoEncryptionSettings autoEncryptionSettings = AutoEncryptionSettings.builder()
            .keyVaultNamespace("encryption.__keyVault")
            .kmsProviders(kmsProviders)
            .schemaMap(schemaMap)
            .extraOptions(extraOptions.isEmpty() ? null : extraOptions)
            .build();

        return MongoClients.create(settings);
    }

    private String findCryptSharedLib() {
        // 检查顺序：
        // 1. 环境变量
        String envPath = System.getenv("MONGODB_CRYPT_SHARED_LIB_PATH");
        if (envPath != null && new File(envPath).exists()) {
            return envPath;
        }

        // 2. Java系统属性
        String propPath = System.getProperty("mongodb.crypt.shared.lib.path");
        if (propPath != null && new File(propPath).exists()) {
            return propPath;
        }

        // 3. 常见路径
        String[] commonPaths = {
            "/usr/lib/mongo_crypt_v1.so",
            "/usr/lib64/mongo_crypt_v1.so",
            "/usr/local/lib/mongo_crypt_v1.so",
            "/opt/homebrew/lib/mongo_crypt_v1.dylib", // macOS
            "./lib/mongo_crypt_v1.so"  // 项目目录
        };

        for (String path : commonPaths) {
            if (new File(path).exists()) {
                return path;
            }
        }

        return null;
    }
}

@Configuration
@ConditionalOnProperty(name = "mongodb.encryption.enabled", havingValue = "true")
public class ConditionalEncryptionConfig {

    @Value("${mongodb.encryption.crypt-lib-path:}")
    private String cryptLibPath;

    @Value("${mongodb.encryption.crypt-lib-required:false}")
    private boolean cryptLibRequired;

    @Bean
    public MongoClient mongoClient() {
        Map<String, Object> extraOptions = new HashMap<>();

        // 如果有配置路径，则使用
        if (StringUtils.hasText(cryptLibPath)) {
            extraOptions.put("cryptSharedLibPath", cryptLibPath);
        }

        // 如果要求必须
        if (cryptLibRequired) {
            extraOptions.put("cryptSharedLibRequired", true);
        }

        AutoEncryptionSettings autoEncryptionSettings = AutoEncryptionSettings.builder()
            .keyVaultNamespace("encryption.__keyVault")
            .kmsProviders(kmsProviders)
            .schemaMap(schemaMap)
            .extraOptions(extraOptions.isEmpty() ? null : extraOptions)
            .build();

        return MongoClients.create(settings);
    }
}

@Configuration
public class FinalRecommendationConfig {

    @Bean
    public MongoClient mongoClient(
            @Value("${spring.data.mongodb.uri}") String mongoUri,
            @Value("${crypt.lib.path:#{null}}") String cryptLibPath) {

        AutoEncryptionSettings.Builder encryptionBuilder =
            AutoEncryptionSettings.builder()
                .keyVaultNamespace("encryption.__keyVault")
                .kmsProviders(getKmsProviders());

        // 关键：只有明确提供了路径才配置
        if (StringUtils.hasText(cryptLibPath)) {
            Map<String, Object> extraOptions = new HashMap<>();
            extraOptions.put("cryptSharedLibPath", cryptLibPath);
            encryptionBuilder.extraOptions(extraOptions);
        }

        MongoClientSettings settings = MongoClientSettings.builder()
            .applyConnectionString(new ConnectionString(mongoUri))
            .autoEncryptionSettings(encryptionBuilder.build())
            .build();

        return MongoClients.create(settings);
    }
}

直接回答你的问题：

问：必须要配置 .extraOptions(Map.of("cryptSharedLibPath", "/usr/lib/mongo_crypt_v1.so")) 吗？

答：不，不是必须的！有三种情况：

1. 不配置 + 设置环境变量 ✅（推荐）

- 需要指定非标准路径
- 需要强制使用加密库（cryptSharedLibRequired: true）
- 特定性能调优
时才需要在代码中明确配置。
