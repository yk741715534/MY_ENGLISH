一、核心原则：客户端加密

Field-Level Encryption (FLE) 的核心设计原则：

复制
数据在离开你的应用程序之前就已经被加密了！
↓
MongoDB 服务器接收和存储的都是密文
↓
只有你的应用程序（有正确密钥）才能解密

应用服务器配置（重要！）

@Configuration
public class AppServerMongoConfig {

    // 应用服务器的配置
    @Bean
    public MongoClient mongoClient() {
        // 1. 指定加密库路径（应用服务器的路径！）
        String cryptLibPath = "/usr/lib/mongo_crypt_v1.so";

        // 2. 验证库文件存在
        File libFile = new File(cryptLibPath);
        if (!libFile.exists()) {
            throw new IllegalStateException(
                "加密库未找到！请在应用服务器安装: sudo apt-get install libmongocrypt-dev\n" +
                "当前路径: " + cryptLibPath
            );
        }

        // 3. 配置Vault连接（从应用服务器连接Vault）
        Map<String, Object> vaultConfig = new HashMap<>();
        vaultConfig.put("serverUrl", "http://vault-server:8200"); // Vault服务器地址
        vaultConfig.put("token", "s.your-vault-token");           // 应用服务器的Vault token

        Map<String, Map<String, Object>> kmsProviders = new HashMap<>();
        kmsProviders.put("local", vaultConfig);  // 注意：这里用"local"但连接远程Vault

        // 4. 配置自动加密
        AutoEncryptionSettings autoEncryptionSettings = AutoEncryptionSettings.builder()
            .keyVaultNamespace("encryption.__keyVault")
            .kmsProviders(kmsProviders)
            .extraOptions(Map.of(
                "cryptSharedLibPath", cryptLibPath  // 应用服务器的加密库
            ))
            .schemaMap(getEncryptionSchema())
            .build();

        // 5. 创建MongoClient（连接MongoDB服务器）
        MongoClientSettings settings = MongoClientSettings.builder()
            .applyConnectionString(new ConnectionString("mongodb://mongodb-server:27017/mydb"))
            .autoEncryptionSettings(autoEncryptionSettings)
            .build();

        return MongoClients.create(settings);
    }
}

验证脚本

@Component
@Slf4j
public class EncryptionSetupValidator {

    @PostConstruct
    public void validateSetup() {
        log.info("=== 验证加密配置 ===");

        // 1. 检查加密库
        String libPath = System.getenv("MONGODB_CRYPT_SHARED_LIB_PATH");
        if (libPath == null) libPath = "/usr/lib/mongo_crypt_v1.so";

        File libFile = new File(libPath);
        if (!libFile.exists()) {
            log.error("❌ 加密库未找到: {}", libPath);
            log.error("   请在应用服务器运行: sudo apt-get install libmongocrypt-dev");
            throw new RuntimeException("加密库缺失");
        }
        log.info("✅ 加密库找到: {}", libPath);

        // 2. 检查Vault连接
        try {
            String vaultAddr = System.getenv("VAULT_ADDR");
            if (vaultAddr == null) {
                log.warn("⚠️  VAULT_ADDR 未设置，使用默认: http://localhost:8200");
            } else {
                log.info("✅ Vault地址: {}", vaultAddr);
            }
        } catch (Exception e) {
            log.error("❌ Vault连接失败: {}", e.getMessage());
        }

        // 3. 检查MongoDB连接
        try {
            String mongoUri = System.getenv("MONGODB_URI");
            log.info("✅ MongoDB连接: {}", mongoUri != null ? mongoUri : "使用默认配置");
        } catch (Exception e) {
            log.error("❌ MongoDB配置检查失败");
        }

        log.info("=== 验证完成 ===");
        log.info("总结：加密发生在应用服务器，使用本地加密库，密钥来自Vault，数据存MongoDB");
    }
}

@RestController
@RequestMapping("/test/encryption")
public class EncryptionTestController {

    @Autowired
    private MongoTemplate mongoTemplate;

    @PostMapping("/test")
    public String testEncryption() {
        // 创建测试数据
        Map<String, Object> testDoc = new HashMap<>();
        testDoc.put("name", "测试用户");
        testDoc.put("ssn", "123-45-6789");  // 这个应该被加密
        testDoc.put("timestamp", new Date());

        // 插入数据库
        mongoTemplate.insert(testDoc, "encryption_test");

        // 从数据库直接读取（绕过驱动解密）
        Document rawDoc = mongoTemplate.getCollection("encryption_test")
            .find(Filters.eq("name", "测试用户")).first();

        Object rawSsn = rawDoc.get("ssn");

        if (rawSsn instanceof String) {
            String ssnStr = (String) rawSsn;
            if (ssnStr.equals("123-45-6789")) {
                return "❌ 加密失败：SSN在数据库中是明文";
            } else if (ssnStr.startsWith("AQ==") || ssnStr.length() > 50) {
                return "✅ 加密成功：SSN在数据库中是密文: " + ssnStr.substring(0, 50) + "...";
            } else {
                return "⚠️  不确定：SSN值是: " + ssnStr;
            }
        }

        return "测试完成";
    }
}
八、常见错误澄清
错误理解1：在MongoDB服务器安装加密库
复制
错误：在MongoDB服务器运行 apt-get install libmongocrypt-dev
原因：以为加密发生在MongoDB端
正确：加密发生在应用端，MongoDB服务器不需要加密库
错误理解2：在Vault服务器安装加密库
复制
错误：在Vault服务器安装MongoDB加密库
原因：以为Vault需要加密库来加解密
正确：Vault只管理密钥，不参与实际数据加解密
错误理解3：所有服务器都需要相同配置
复制
错误：在所有服务器上配置相同的 cryptSharedLibPath
原因：以为配置需要同步
正确：只有应用服务器需要这个配置
总结要点

加密库安装位置：应用服务器（运行Java应用的地方）

加密发生地点：客户端（你的应用程序内）

MongoDB的角色：只存储已经加密的数据

Vault的角色：只管理密钥，不接触实际数据

网络连接：

应用服务器 → MongoDB服务器：传输加密数据

应用服务器 → Vault服务器：获取加密密钥

MongoDB服务器和Vault服务器之间不需要直接通信

一句话总结：在你的Spring Boot应用运行的服务器上安装libmongocrypt，配置好Vault连接，加密就会在数据离开你的应用前自动完成。
