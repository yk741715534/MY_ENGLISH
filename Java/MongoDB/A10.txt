        );

@Service
public class UserService {

    @Autowired
    private MongoTemplate mongoTemplate;

    // 方法1：整个操作使用指定的 ReadConcern
    public User findUserWithReadConcern(String userId, ReadConcern readConcern) {
        MongoCollection<User> collection = mongoTemplate.getCollection("users", User.class)
            .withReadConcern(readConcern);

        return collection.find(eq("_id", userId)).first();
    }

    // 方法2：每次查询指定
    public User findUserMajority(String userId) {
        return mongoTemplate.getCollection("users")
            .withReadConcern(ReadConcern.MAJORITY)
            .find(new Document("_id", userId))
            .map(doc -> mongoTemplate.getConverter().read(User.class, doc))
            .first();
    }

    // 方法3：在 Repository 查询方法中使用
    public List<User> findCriticalUsers() {
        // 创建带有 ReadConcern 的 Query
        Query query = Query.query(Criteria.where("role").is("admin"))
            .withReadConcern(ReadConcern.MAJORITY);

        return mongoTemplate.find(query, User.class);
    }
}

4.1 LOCAL vs AVAILABLE

@Service
public class ReadConcernComparison {

    public void compareLocalVsAvailable() {
        /*
        LOCAL (默认):
        - 读取节点本地最新的数据
        - 可能读取到未复制到多数节点的数据
        - 不保证数据已持久化
        - 性能最好

        AVAILABLE:
        - 读取节点可用的最新数据
        - 对分片集群有特殊行为
        - 不保证数据已持久化
        - 可能返回孤立文档
        */

        // LOCAL 示例
        Document localDoc = mongoTemplate.getCollection("data")
            .withReadConcern(ReadConcern.LOCAL)
            .find(new Document("_id", "1"))
            .first();

        // AVAILABLE 示例（主要用于分片集群）
        Document availableDoc = mongoTemplate.getCollection("data")
            .withReadConcern(ReadConcern.AVAILABLE)
            .find(new Document("_id", "1"))
            .first();
    }
}

@Service
public class MajorityReadConcernService {

    public void demonstrateMajorityRead() {
        /*
        MAJORITY 特性:
        1. 只读取已复制到大多数节点的数据
        2. 保证读取的数据不会被回滚
        3. 可能读取到较旧的数据
        4. 需要多数派副本集成员在线
        */

        // 场景：确保读取到已持久化的数据
        User criticalUser = mongoTemplate.getCollection("users", User.class)
            .withReadConcern(ReadConcern.MAJORITY)
            .withReadPreference(ReadPreference.primary())
            .find(eq("_id", "critical_user_123"))
            .first();

        // 金融交易场景
        List<Transaction> transactions = mongoTemplate.find(
            Query.query(Criteria.where("accountId").is("acc123"))
                .withReadConcern(ReadConcern.MAJORITY)
                .withReadPreference(ReadPreference.primary()),
            Transaction.class
        );
    }

    public void majorityReadWithRetry() {
        // MAJORITY 读取可能失败，需要重试逻辑
        int maxRetries = 3;
        for (int i = 0; i < maxRetries; i++) {
            try {
                Document result = mongoTemplate.getCollection("important_data")
                    .withReadConcern(ReadConcern.MAJORITY)
                    .find(new Document("_id", "data123"))
                    .first();

                if (result != null) {
                    return; // 成功读取
                }

                // 如果找不到数据，可能是还未复制到多数节点
                Thread.sleep(100 * (i + 1));

            } catch (MongoQueryException e) {
                if (e.getErrorCode() == 246  // ReadConcernMajorityNotAvailable
                    || e.getErrorCode() == 133) {  // NotMasterOrSecondary
                    // 等待后重试
                    try {
                        Thread.sleep(200 * (i + 1));
                    } catch (InterruptedException ie) {
                        Thread.currentThread().interrupt();
                        break;
                    }
                } else {
                    throw e;
                }
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                break;
            }
        }
    }
}

@Service
public class LinearizableReadService {

    public void demonstrateLinearizableRead() {
        /*
        LINEARIZABLE 特性:
        1. 最强一致性保证
        2. 读取操作在读取开始前完成的所有写入
        3. 只能用于主节点读取
        4. 性能开销大
        5. 需要 writeConcern: majority
        */

        // 线性化读取示例
        Document linearizableData = mongoTemplate.getCollection("critical_data")
            .withReadConcern(ReadConcern.LINEARIZABLE)
            .withReadPreference(ReadPreference.primary())  // 必须主节点
            .find(new Document("_id", "lock1"))
            .maxTime(5, TimeUnit.SECONDS)  // 设置超时
            .first();

        // 使用场景：分布式锁
        boolean acquiredLock = tryAcquireLock("resource1", 5000);
    }

    private boolean tryAcquireLock(String resource, long timeoutMs) {
        long endTime = System.currentTimeMillis() + timeoutMs;

        while (System.currentTimeMillis() < endTime) {
            try {
                // 尝试获取锁 - 使用线性化读取确保一致性
                Document lock = mongoTemplate.getCollection("distributed_locks")
                    .withReadConcern(ReadConcern.LINEARIZABLE)
                    .withWriteConcern(WriteConcern.MAJORITY)
                    .findOneAndUpdate(
                        Filters.and(
                            eq("resource", resource),
                            lt("expiresAt", new Date())
                        ),
                        Updates.combine(
                            Updates.set("owner", "server1"),
                            Updates.set("expiresAt",
                                new Date(System.currentTimeMillis() + 30000))
                        ),
                        new FindOneAndUpdateOptions()
                            .upsert(true)
                            .returnDocument(ReturnDocument.AFTER)
                    );

                return lock != null;

            } catch (MongoCommandException e) {
                if (e.getErrorCode() == 246) {  // LinearizableNotAvailable
                    // 线性化读取不可用，重试
                    try {
                        Thread.sleep(100);
                    } catch (InterruptedException ie) {
                        Thread.currentThread().interrupt();
                        return false;
                    }
                } else {
                    throw e;
                }
            }
        }

        return false;
    }
}

@Service
@Transactional
public class SnapshotReadService {

    public void demonstrateSnapshotRead() {
        /*
        SNAPSHOT 特性:
        1. 只能在多文档事务中使用
        2. 提供事务级别的快照隔离
        3. 读取事务开始时的数据一致性视图
        4. 避免不可重复读和幻读
        */

        ClientSession session = mongoTemplate.getDb().getClient().startSession();

        try {
            // 开始事务
            session.startTransaction(TransactionOptions.builder()
                .readConcern(ReadConcern.SNAPSHOT)
                .writeConcern(WriteConcern.MAJORITY)
                .build());

            // 在事务中读取 - 看到一致性快照
            User user = mongoTemplate.findById("user123", User.class, session);

            // 相关操作也看到同样的快照
            List<Order> orders = mongoTemplate.find(
                Query.query(Criteria.where("userId").is("user123")),
                Order.class,
                session
            );

            // 执行更新
            mongoTemplate.updateFirst(
                Query.query(Criteria.where("_id").is("user123")),
                new Update().set("lastLogin", new Date()),
                User.class,
                session
            );

            session.commitTransaction();

        } catch (Exception e) {
            session.abortTransaction();
            throw e;
        } finally {
            session.close();
        }
    }

    public void snapshotTransactionExample() {
        // 转账事务 - 保证一致性
        ClientSession session = mongoTemplate.getDb().getClient().startSession();

        TransactionOptions txnOptions = TransactionOptions.builder()
            .readConcern(ReadConcern.SNAPSHOT)
            .writeConcern(WriteConcern.MAJORITY)
            .maxCommitTime(10, TimeUnit.SECONDS)
            .build();

        try {
            session.startTransaction(txnOptions);

            // 读取账户余额（快照）
            Account fromAccount = mongoTemplate.findOne(
                Query.query(Criteria.where("accountNo").is("ACC001")),
                Account.class,
                session
            );

            Account toAccount = mongoTemplate.findOne(
                Query.query(Criteria.where("accountNo").is("ACC002")),
                Account.class,
                session
            );

            // 检查余额
            if (fromAccount.getBalance().compareTo(BigDecimal.valueOf(1000)) >= 0) {
                // 执行转账
                mongoTemplate.updateFirst(
                    Query.query(Criteria.where("accountNo").is("ACC001")),
                    new Update().inc("balance", -1000),
                    Account.class,
                    session
                );

                mongoTemplate.updateFirst(
                    Query.query(Criteria.where("accountNo").is("ACC002")),
                    new Update().inc("balance", 1000),
                    Account.class,
                    session
                );

                // 记录交易
                TransactionRecord record = new TransactionRecord(
                    "ACC001", "ACC002", BigDecimal.valueOf(1000));
                mongoTemplate.save(record, session);
            }

            session.commitTransaction();

        } catch (Exception e) {
            session.abortTransaction();
            throw e;
        } finally {
            session.close();
        }
    }
}

@Service
public class ReadConcernWithPreference {

    public void combineReadConcernAndPreference() {
        // 场景1：强一致性读取
        Document strongConsistency = mongoTemplate.getCollection("data")
            .withReadConcern(ReadConcern.MAJORITY)
            .withReadPreference(ReadPreference.primary())
            .find(new Document("_id", "1"))
            .first();

        // 场景2：最终一致性读取（性能优先）
        Document eventualConsistency = mongoTemplate.getCollection("data")
            .withReadConcern(ReadConcern.LOCAL)
            .withReadPreference(ReadPreference.nearest())
            .find(new Document("_id", "1"))
            .first();

        // 场景3：跨区域读取
        Document crossRegion = mongoTemplate.getCollection("data")
            .withReadConcern(ReadConcern.AVAILABLE)
            .withReadPreference(ReadPreference.secondaryPreferred(
                new TagSet(Arrays.asList(new Tag("region", "east")))
            ))
            .find(new Document("_id", "1"))
            .first();
    }
}

@Service
public class ReadWriteConcernCombination {

    public void readYourWrites() {

        实现"读己之所写"语义
        场景：写入后立即读取，确保能读到刚写入的数据

        // 2. 尝试读取 - 可能因为复制延迟读不到
        User readUser1 = mongoTemplate.findById(userId, User.class);
        System.out.println("普通读取: " + (readUser1 != null ? "成功" : "失败"));

        // 3. 使用 MAJORITY 读取
        User readUser2 = mongoTemplate.getCollection("users", User.class)
            .withReadConcern(ReadConcern.MAJORITY)
            .find(eq("_id", userId))
            .first();
        System.out.println("MAJORITY读取: " + (readUser2 != null ? "成功" : "失败"));

        // 4. 从主节点读取
        User readUser3 = mongoTemplate.getCollection("users", User.class)
            .withReadPreference(ReadPreference.primary())
            .find(eq("_id", userId))
            .first();
        System.out.println("主节点读取: " + (readUser3 != null ? "成功" : "失败"));
    }

    public void strongConsistencyWriteAndRead() {
        // 强一致性读写组合

@Service
@Slf4j
public class ReadConcernMonitor {

    @Autowired
    private MongoTemplate mongoTemplate;

    public void monitorReadConcernPerformance() {
        Map<ReadConcern, PerformanceStats> stats = new HashMap<>();

        ReadConcern[] readConcerns = {
            ReadConcern.LOCAL,
            ReadConcern.AVAILABLE,
            ReadConcern.MAJORITY,
            ReadConcern.LINEARIZABLE
        };

        for (ReadConcern rc : readConcerns) {
            long totalTime = 0;
            int successes = 0;
            int failures = 0;

            for (int i = 0; i < 10; i++) {
                long startTime = System.nanoTime();

                try {
                    Document result = mongoTemplate.getCollection("test")
                        .withReadConcern(rc)
                        .withMaxTime(2, TimeUnit.SECONDS)
                        .find(new Document("_id", "test_" + i))
                        .first();

                    long duration = System.nanoTime() - startTime;
                    totalTime += duration;

                    if (result != null) {
                        successes++;
                    } else {
                        failures++;
                    }

                } catch (Exception e) {
                    failures++;
                    log.warn("ReadConcern {} failed: {}", rc.getValue(), e.getMessage());
                }
            }

            PerformanceStats stat = new PerformanceStats(
                rc.getValue(),
                totalTime / 10_000_000.0,  // 转换为毫秒
                successes,
                failures
            );

            stats.put(rc, stat);
            log.info("ReadConcern {} stats: {}", rc.getValue(), stat);
        }

        return stats;
    }

    public void handleReadConcernErrors() {
        try {
            // 尝试线性化读取
            Document result = mongoTemplate.getCollection("data")
                .withReadConcern(ReadConcern.LINEARIZABLE)
                .find(new Document("_id", "1"))
                .maxTime(3, TimeUnit.SECONDS)
                .first();

        } catch (MongoCommandException e) {
            switch (e.getErrorCode()) {
                case 246:  // ReadConcernMajorityNotAvailable
                    log.warn("Majority read concern not available, falling back to local");
                    // 降级到 LOCAL
                    Document fallback = mongoTemplate.getCollection("data")
                        .withReadConcern(ReadConcern.LOCAL)
                        .find(new Document("_id", "1"))
                        .first();
                    break;

                case 133:  // NotMasterOrSecondary
                    log.warn("Not connected to primary, retrying with secondary");
                    // 重试从从节点读取
                    Document retry = mongoTemplate.getCollection("data")
                        .withReadPreference(ReadPreference.secondaryPreferred())
                        .find(new Document("_id", "1"))
                        .first();
                    break;

                case 50:  // ExceededTimeLimit
                    log.warn("Read timeout, operation too slow");
                    // 处理超时
                    break;

                default:
                    throw e;
            }
        }
    }

    @Data
    @AllArgsConstructor
    private static class PerformanceStats {
        private String readConcern;
        private double avgLatencyMs;
        private int successes;
        private int failures;
    }
}

@Component
public class ReadConcernSelector {

    public ReadConcern selectForScenario(Scenario scenario) {
        switch (scenario) {
            case FINANCIAL_TRANSACTION:
                // 金融交易：最强一致性
                return ReadConcern.LINEARIZABLE;

            case ECOMMERCE_ORDER:
                // 电商订单：强一致性
                return ReadConcern.MAJORITY;

            case USER_PROFILE:
                // 用户资料：最终一致性即可
                return ReadConcern.LOCAL;

            case ANALYTICS_QUERY:
                // 分析查询：最高性能
                return ReadConcern.AVAILABLE;

            case MULTI_DOCUMENT_TRANSACTION:
                // 多文档事务：快照隔离
                return ReadConcern.SNAPSHOT;

            default:
                return ReadConcern.LOCAL;
        }
    }

    public enum Scenario {
        FINANCIAL_TRANSACTION,
        ECOMMERCE_ORDER,
        USER_PROFILE,
        ANALYTICS_QUERY,
        MULTI_DOCUMENT_TRANSACTION
    }
}

@Configuration
public class ProductionMongoConfig {

    @Value("${spring.profiles.active}")
    private String activeProfile;

    @Bean
    public MongoClient mongoClient() {
        MongoClientSettings.Builder builder = MongoClientSettings.builder();

        // 生产环境使用更严格的配置
        if ("prod".equals(activeProfile)) {
            builder.readConcern(ReadConcern.MAJORITY)
                   .readPreference(ReadPreference.primaryPreferred())
                   .writeConcern(WriteConcern.MAJORITY
                       .withWTimeout(10, TimeUnit.SECONDS))
                   .retryWrites(true)
                   .retryReads(true);
        } else {
            // 开发环境使用宽松配置
            builder.readConcern(ReadConcern.LOCAL)
                   .readPreference(ReadPreference.primaryPreferred())
                   .writeConcern(WriteConcern.ACKNOWLEDGED);
        }

        ConnectionString connString = new ConnectionString(
            "mongodb://localhost:27017/test"
        );
        builder.applyConnectionString(connString);

        return MongoClient.create(builder.build());
    }
}

关键总结：

根据业务需求选择适当的 ReadConcern

强一致性场景使用 MAJORITY或 LINEARIZABLE

性能敏感场景使用 LOCAL或 AVAILABLE

多文档事务使用 SNAPSHOT

总是设置合理的超时时间
