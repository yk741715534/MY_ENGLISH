MongoDB 支持对特定字段进行加密，主要通过 客户端字段级加密（Client-Side Field Level Encryption, FLE）​ 实现。以下是针对 Java 客户端（Spring Boot）​ 的详细方案：

一、核心方案：FLE 2.0（推荐）
特点

客户端加密：数据在客户端加密后存储，MongoDB 服务端看到的是密文

选择性加密：只加密敏感字段（如 ssn、creditCard），非敏感字段保持明文

查询支持：支持对加密字段进行等值查询（随机加密）和范围查询（确定性加密）

透明加解密：Spring Boot 应用中自动加解密，代码改动小

@Configuration
public class MongoEncryptionConfig {

    @Value("${mongodb.uri}")
    private String mongoUri;

    // 本地主密钥（生产环境用KMS）
    @Value("${encryption.local.key:}")
    private String localKeyBase64;

    @Bean
    public MongoClient mongoClient() {
        // 1. 准备密钥
        byte[] localKey = Base64.getDecoder().decode(localKeyBase64);
        if (localKey.length == 0) {
            localKey = generateLocalKey(); // 生成测试密钥
        }

        // 2. 密钥保管库集合
        Map<String, Object> keyMap = new HashMap<>();
        UUID keyId = UUID.fromString("12345678-1234-1234-1234-123456789012");
        keyMap.put(keyId.toString(), new SecretKeySpec(localKey, "AES"));

        // 3. 构建加密模式（JSON Schema）
        Map<String, BsonDocument> schemaMap = new HashMap<>();
        BsonDocument schema = BsonDocument.parse(
            "{" +
            "  \"properties\": {" +
            "    \"ssn\": {" +
            "      \"encrypt\": {" +
            "        \"keyId\": [{" +
            "          \"$binary\": {" +
            "            \"base64\": \"" + Base64.getEncoder().encodeToString(keyId.toString().getBytes()) + "\"," +
            "            \"subType\": \"04\"" +
            "          }" +
            "        }]," +
            "        \"bsonType\": \"string\"," +
            "        \"algorithm\": \"AEAD_AES_256_CBC_HMAC_SHA_512-Random\"" +
            "      }" +
            "    }," +
            "    \"creditCard\": {" +
            "      \"encrypt\": {" +
            "        \"keyId\": [{" +
            "          \"$binary\": {" +
            "            \"base64\": \"" + Base64.getEncoder().encodeToString(keyId.toString().getBytes()) + "\"," +
            "            \"subType\": \"04\"" +
            "          }" +
            "        }]," +
            "        \"bsonType\": \"string\"," +
            "        \"algorithm\": \"AEAD_AES_256_CBC_HMAC_SHA_512-Deterministic\"" +
            "      }" +
            "    }" +
            "  }," +
            "  \"bsonType\": \"object\"" +
            "}"
        );
        schemaMap.put("hospital.patients", schema);

        // 4. 创建自动加密设置
        AutoEncryptionSettings autoEncryptionSettings = AutoEncryptionSettings.builder()
            .keyVaultNamespace("encryption.__keyVault")
            .kmsProviders(Map.of("local", Map.of("key", localKey)))
            .schemaMap(schemaMap)
            .build();

        // 5. 构建MongoClient设置
        MongoClientSettings settings = MongoClientSettings.builder()
            .applyConnectionString(new ConnectionString(mongoUri))
            .autoEncryptionSettings(autoEncryptionSettings)
            .build();

        return MongoClients.create(settings);
    }

    @Bean
    public MongoTemplate mongoTemplate() {
        return new MongoTemplate(mongoClient(), "hospital");
    }

    private byte[] generateLocalKey() {
        try {
            KeyGenerator keyGenerator = KeyGenerator.getInstance("AES");
            keyGenerator.init(256, new SecureRandom());
            return keyGenerator.generateKey().getEncoded();
        } catch (Exception e) {
            throw new RuntimeException("Failed to generate encryption key", e);
        }
    }
}

import org.springframework.core.convert.converter.Converter;
import org.springframework.data.convert.ReadingConverter;
import org.springframework.data.convert.WritingConverter;
import org.springframework.data.mongodb.core.convert.MongoCustomConversions;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.bson.BsonBinary;
import org.bson.BsonString;
import org.bson.types.Binary;
import javax.crypto.Cipher;
import javax.crypto.spec.SecretKeySpec;
import java.nio.charset.StandardCharsets;
import java.util.Arrays;
import java.util.Base64;

@Configuration
public class EncryptionConverters {

    // 简单示例：使用AES加密（实际项目用FLE）
    @Bean
    public MongoCustomConversions customConversions() {
        return new MongoCustomConversions(Arrays.asList(
            new EncryptStringConverter(),
            new DecryptStringConverter()
        ));
    }

    @WritingConverter
    static class EncryptStringConverter implements Converter<EncryptedString, String> {
        private static final String ALGORITHM = "AES/ECB/PKCS5Padding";
        private final byte[] key = "your-32-byte-secret-key".getBytes();

        @Override
        public String convert(EncryptedString source) {
            try {
                Cipher cipher = Cipher.getInstance(ALGORITHM);
                cipher.init(Cipher.ENCRYPT_MODE, new SecretKeySpec(key, "AES"));
                byte[] encrypted = cipher.doFinal(source.getValue().getBytes());
                return Base64.getEncoder().encodeToString(encrypted);
            } catch (Exception e) {
                throw new RuntimeException("Encryption failed", e);
            }
        }
    }

    @ReadingConverter
    static class DecryptStringConverter implements Converter<String, EncryptedString> {
        private static final String ALGORITHM = "AES/ECB/PKCS5Padding";
        private final byte[] key = "your-32-byte-secret-key".getBytes();

        @Override
        public EncryptedString convert(String source) {
            try {
                Cipher cipher = Cipher.getInstance(ALGORITHM);
                cipher.init(Cipher.DECRYPT_MODE, new SecretKeySpec(key, "AES"));
                byte[] decrypted = cipher.doFinal(Base64.getDecoder().decode(source));
                return new EncryptedString(new String(decrypted, StandardCharsets.UTF_8));
            } catch (Exception e) {
                throw new RuntimeException("Decryption failed", e);
            }
        }
    }

    public static class EncryptedString {
        private final String value;

        public EncryptedString(String value) { this.value = value; }
        public String getValue() { return value; }
    }
}

@Repository
public interface PatientRepository extends MongoRepository<Patient, String> {
    // 支持对加密字段的等值查询（仅确定性加密字段）
    List<Patient> findByCreditCard(String creditCard);
}

@Service
@RequiredArgsConstructor
public class PatientService {
    private final PatientRepository patientRepository;

    public Patient createPatient(Patient patient) {
        // 保存时自动加密
        return patientRepository.save(patient);
    }

    public Patient getPatientBySsn(String ssn) {
        // 注意：随机加密的ssn字段不能直接查询
        // 需要先查询其他条件，然后在结果中过滤
        return patientRepository.findAll().stream()
            .filter(p -> decryptField(p.getSsn()).equals(ssn))
            .findFirst()
            .orElse(null);
    }

    public List<Patient> findByCreditCard(String cardNumber) {
        // 确定性加密字段可直接查询
        return patientRepository.findByCreditCard(cardNumber);
    }
}

五、重要注意事项
1. 版本要求

MongoDB 企业版 4.2+​ 或 Atlas 集群

Java Driver 4.0+（支持 FLE 2.0）

Spring Data MongoDB 3.0+

2. 密钥管理（生产环境）
yaml
复制
# application.yml
mongodb:
  uri: mongodb://localhost:27017/hospital
encryption:
  kms:
    type: aws
    access-key: ${AWS_ACCESS_KEY}
    secret-key: ${AWS_SECRET_KEY}
    region: us-east-1
    key-arn: arn:aws:kms:us-east-1:123456789012:key/your-key-id
3. 性能影响

加密开销：加密/解密增加 5-15% CPU 使用

查询限制：

随机加密：无法查询，只能全表扫描后解密比较

确定性加密：支持等值查询，但安全性略低

索引：加密字段可建索引，但索引存储的是密文

4. 安全最佳实践

分离密钥：使用外部 KMS（如 AWS KMS），不要硬编码密钥

最小权限：应用使用专用数据库用户，仅有数据读写权限

审计日志：开启 MongoDB 审计日志

字段选择：只加密真正敏感的字段，避免过度加密
