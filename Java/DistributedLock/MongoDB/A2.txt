package com.example.distributedlock.config;

import org.springframework.context.annotation.Configuration;
import org.springframework.data.mongodb.core.MongoTemplate;
import org.springframework.data.mongodb.core.index.IndexOperations;
import org.springframework.data.mongodb.core.index.IndexResolver;
import org.springframework.data.mongodb.core.index.MongoPersistentEntityIndexResolver;
import javax.annotation.PostConstruct;

@Configuration
public class MongoConfig {

    private final MongoTemplate mongoTemplate;

    public MongoConfig(MongoTemplate mongoTemplate) {
        this.mongoTemplate = mongoTemplate;
    }

    @PostConstruct
    public void initIndexes() {
        // 自动创建实体类定义的索引
        IndexOperations indexOps = mongoTemplate.indexOps("distributed_locks");
        IndexResolver resolver = new MongoPersistentEntityIndexResolver(
            mongoTemplate.getConverter().getMappingContext()
        );
        resolver.resolveIndexFor(LockDocument.class).forEach(indexOps::ensureIndex);
    }
}

// MongoDB锁存储实现
package com.example.distributedlock.lock;

import lombok.RequiredArgsConstructor;
import org.springframework.data.mongodb.core.FindAndModifyOptions;
import org.springframework.data.mongodb.core.MongoTemplate;
import org.springframework.data.mongodb.core.query.Criteria;
import org.springframework.data.mongodb.core.query.Query;
import org.springframework.data.mongodb.core.query.Update;
import org.springframework.stereotype.Component;
import java.time.LocalDateTime;
import java.util.Optional;

@Component
@RequiredArgsConstructor
public class MongoLockStore implements LockStore {

    private final MongoTemplate mongoTemplate;

    @Override
    public Optional<LockDocument> tryAcquire(String lockKey, String lockValue,
                                             String owner, long expireAfterMillis) {
        LocalDateTime now = LocalDateTime.now();
        LocalDateTime expireAt = now.plusNanos(expireAfterMillis * 1_000_000L);

        // 原子操作：获取锁
        Query query = new Query();
        Criteria criteria = new Criteria().andOperator(
            Criteria.where("_id").is(lockKey),
            new Criteria().orOperator(
                Criteria.where("expireAt").lt(now),  // 锁已过期
                Criteria.where("owner").is(owner)     // 自己是持有者（可重入）
            )
        );
        query.addCriteria(criteria);

        Update update = new Update();
        update.set("lockValue", lockValue)
              .set("owner", owner)
              .set("expireAt", expireAt)
              .set("updatedAt", now)
              .inc("lockCount", 1)  // 重入计数+1
              .setOnInsert("createdAt", now)
              .setOnInsert("version", 1);

        FindAndModifyOptions options = FindAndModifyOptions.options()
                .returnNew(true)  // 返回更新后的文档
                .upsert(true);     // 不存在则创建

        try {
            LockDocument lock = mongoTemplate.findAndModify(
                query, update, options, LockDocument.class
            );

            // 关键：验证锁是否真的被当前客户端获得
            if (lock != null && owner.equals(lock.getOwner())
                && lockValue.equals(lock.getLockValue())) {
                return Optional.of(lock);
            }
            return Optional.empty();
        } catch (Exception e) {
            // 可能发生唯一键冲突（并发创建）
            return Optional.empty();
        }
    }

    @Override
    public boolean release(String lockKey, String lockValue, String owner) {
        Query query = new Query(Criteria.where("_id").is(lockKey)
                .and("lockValue").is(lockValue)
                .and("owner").is(owner));

        // 先获取当前锁信息
        LockDocument current = mongoTemplate.findOne(query, LockDocument.class);
        if (current == null) {
            return false;  // 锁不存在或不属于自己
        }

        if (current.getLockCount() > 1) {
            // 重入锁，减少计数
            Update update = new Update();
            update.inc("lockCount", -1)
                  .set("updatedAt", LocalDateTime.now());
            mongoTemplate.updateFirst(query, update, LockDocument.class);
            return true;
        } else {
            // 完全释放锁
            Update update = new Update();
            update.set("owner", null)
                  .set("lockValue", null)
                  .set("expireAt", LocalDateTime.now())  // 立即过期
                  .set("updatedAt", LocalDateTime.now())
                  .set("lockCount", 0);
            return mongoTemplate.updateFirst(query, update, LockDocument.class)
                    .getModifiedCount() > 0;
        }
    }

    @Override
    public boolean renew(String lockKey, String lockValue, long expireAfterMillis) {
        Query query = new Query(Criteria.where("_id").is(lockKey)
                .and("lockValue").is(lockValue));

        Update update = new Update();
        update.set("expireAt", LocalDateTime.now()
                .plusNanos(expireAfterMillis * 1_000_000L))
              .set("updatedAt", LocalDateTime.now());

        return mongoTemplate.updateFirst(query, update, LockDocument.class)
                .getModifiedCount() > 0;
    }

    @Override
    public boolean isHeldBy(String lockKey, String lockValue) {
        Query query = new Query(Criteria.where("_id").is(lockKey)
                .and("lockValue").is(lockValue)
                .and("expireAt").gt(LocalDateTime.now()));
        return mongoTemplate.exists(query, LockDocument.class);
    }
}

// 分布式锁核心类
package com.example.distributedlock.lock;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;
import java.time.Duration;
import java.time.LocalDateTime;
import java.util.Optional;
import java.util.UUID;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.locks.Condition;
import java.util.concurrent.locks.Lock;

@Slf4j
@Component
@RequiredArgsConstructor
public class MongoDistributedLock implements Lock {

    private final MongoLockStore lockStore;

    private final ThreadLocal<LockContext> lockContext = new ThreadLocal<>();

    // 锁上下文
    private static class LockContext {
        String lockKey;
        String lockValue;  // 唯一标识，用于验证锁归属
        String owner;      // 持有者标识
        int holdCount;     // 当前线程持有计数

        LockContext(String lockKey, String lockValue, String owner) {
            this.lockKey = lockKey;
            this.lockValue = lockValue;
            this.owner = owner;
            this.holdCount = 1;
        }
    }

    // 获取锁（可重入）
    public boolean tryLock(LockOptions options) {
        String lockValue = UUID.randomUUID().toString();
        String owner = getOwnerIdentifier();

        LocalDateTime endTime = LocalDateTime.now().plus(options.getWaitTime());

        while (LocalDateTime.now().isBefore(endTime)) {
            try {
                Optional<LockDocument> lockDoc = lockStore.tryAcquire(
                    options.getLockKey(),
                    lockValue,
                    owner,
                    options.getExpireTime().toMillis()
                );

                if (lockDoc.isPresent()) {
                    // 成功获取锁
                    lockContext.set(new LockContext(
                        options.getLockKey(), lockValue, owner
                    ));

                    // 启动续期线程（如果锁过期时间较长）
                    if (options.getExpireTime().getSeconds() > 10) {
                        startRenewThread(options.getLockKey(), lockValue,
                                       options.getExpireTime().toMillis());
                    }

                    log.debug("成功获取分布式锁: {}, lockValue: {}",
                             options.getLockKey(), lockValue);
                    return true;
                }

                // 等待后重试
                TimeUnit.MILLISECONDS.sleep(options.getRetryInterval().toMillis());

            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                throw new LockException("获取锁被中断", e);
            }
        }

        log.warn("获取分布式锁超时: {}", options.getLockKey());
        return false;
    }

    // 释放锁
    public void unlock() {
        LockContext context = lockContext.get();
        if (context == null) {
            throw new IllegalStateException("未持有锁");
        }

        try {
            boolean released = lockStore.release(
                context.lockKey,
                context.lockValue,
                context.owner
            );

            if (released) {
                log.debug("成功释放分布式锁: {}, lockValue: {}",
                         context.lockKey, context.lockValue);
            } else {
                log.warn("释放锁失败，可能已过期或被他人占用: {}", context.lockKey);
            }
        } finally {
            lockContext.remove();
        }
    }

    // 启动续期线程
    private void startRenewThread(String lockKey, String lockValue, long expireMillis) {
        Thread renewThread = new Thread(() -> {
            try {
                long renewInterval = expireMillis / 3;  // 在过期前1/3时间续期

                while (!Thread.currentThread().isInterrupted()) {
                    TimeUnit.MILLISECONDS.sleep(renewInterval);

                    boolean renewed = lockStore.renew(lockKey, lockValue, expireMillis);
                    if (!renewed) {
                        log.warn("锁续期失败: {}", lockKey);
                        break;
                    }
                    log.debug("锁续期成功: {}", lockKey);
                }
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                log.debug("锁续期线程被中断: {}", lockKey);
            }
        });

        renewThread.setDaemon(true);
        renewThread.setName("lock-renew-" + lockKey);
        renewThread.start();
    }

    // 获取当前持有者标识
    private String getOwnerIdentifier() {
        // 组合：IP + 进程ID + 线程ID
        String ip = getLocalIp();
        long pid = ProcessHandle.current().pid();
        long threadId = Thread.currentThread().getId();
        return String.format("%s-%d-%d", ip, pid, threadId);
    }

    private String getLocalIp() {
        try {
            return java.net.InetAddress.getLocalHost().getHostAddress();
        } catch (Exception e) {
            return "unknown";
        }
    }

    // 实现java.util.concurrent.locks.Lock接口
    @Override
    public void lock() {
        throw new UnsupportedOperationException("请使用tryLock(LockOptions)");
    }

    @Override
    public void lockInterruptibly() throws InterruptedException {
        throw new UnsupportedOperationException("请使用tryLock(LockOptions)");
    }

    @Override
    public boolean tryLock() {
        return tryLock(LockOptions.defaults());
    }

    @Override
    public boolean tryLock(long time, TimeUnit unit) throws InterruptedException {
        LockOptions options = LockOptions.defaults();
        options.setWaitTime(Duration.ofMillis(unit.toMillis(time)));
        return tryLock(options);
    }

    @Override
    public Condition newCondition() {
        throw new UnsupportedOperationException("不支持Condition");
    }
}

// 锁切面
package com.example.distributedlock.lock;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.aspectj.lang.ProceedingJoinPoint;
import org.aspectj.lang.annotation.Around;
import org.aspectj.lang.annotation.Aspect;
import org.aspectj.lang.reflect.MethodSignature;
import org.springframework.core.LocalVariableTableParameterNameDiscoverer;
import org.springframework.expression.EvaluationContext;
import org.springframework.expression.Expression;
import org.springframework.expression.ExpressionParser;
import org.springframework.expression.spel.standard.SpelExpressionParser;
import org.springframework.expression.spel.support.StandardEvaluationContext;
import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;
import java.lang.reflect.Method;
import java.time.Duration;

@Slf4j
@Aspect
@Component
@RequiredArgsConstructor
public class DistributedLockAspect {

    private final MongoDistributedLock distributedLock;
    private final ExpressionParser parser = new SpelExpressionParser();

    @Around("@annotation(lockAnnotation)")
    public Object around(ProceedingJoinPoint joinPoint, DistributedLock lockAnnotation)
            throws Throwable {

        String lockKey = resolveKey(lockAnnotation, joinPoint);

        LockOptions options = LockOptions.builder()
                .lockKey(lockKey)
                .expireTime(Duration.of(lockAnnotation.expire(),
                           lockAnnotation.timeUnit().toChronoUnit()))
                .waitTime(Duration.of(lockAnnotation.waitTime(),
                           lockAnnotation.timeUnit().toChronoUnit()))
                .retryInterval(Duration.ofMillis(100))
                .reentrant(true)
                .build();

        boolean acquired = false;
        try {
            acquired = distributedLock.tryLock(options);
            if (!acquired) {
                throw new LockException(lockAnnotation.message());
            }
            return joinPoint.proceed();
        } finally {
            if (acquired) {
                try {
                    distributedLock.unlock();
                } catch (Exception e) {
                    log.error("释放锁异常", e);
                }
            }
        }
    }

    private String resolveKey(DistributedLock lockAnnotation,
                             ProceedingJoinPoint joinPoint) {
        String key = lockAnnotation.key();
        String prefix = lockAnnotation.keyPrefix();

        if (key.contains("#")) {
            // 解析SpEL表达式
            MethodSignature signature = (MethodSignature) joinPoint.getSignature();
            Method method = signature.getMethod();
            Object[] args = joinPoint.getArgs();

            EvaluationContext context = new StandardEvaluationContext();
            String[] paramNames = new LocalVariableTableParameterNameDiscoverer()
                    .getParameterNames(method);

            if (paramNames != null) {
                for (int i = 0; i < paramNames.length; i++) {
                    context.setVariable(paramNames[i], args[i]);
                }
            }

            Expression expression = parser.parseExpression(key);
            key = expression.getValue(context, String.class);
        }

        return StringUtils.hasText(prefix) ? prefix + ":" + key : key;
    }
}

package com.example.distributedlock.service;

import com.example.distributedlock.lock.DistributedLock;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.util.UUID;

@Slf4j
@Service
public class OrderService {

    // 方案1：使用注解（推荐）
    @DistributedLock(
        key = "'order:create:' + #userId",
        keyPrefix = "business",
        expire = 10,
        waitTime = 5,
        message = "订单创建中，请勿重复提交"
    )
    @Transactional
    public String createOrder(Long userId, String productId) {
        // 业务逻辑
        String orderNo = generateOrderNo();
        log.info("用户{}创建订单{}成功", userId, orderNo);

        // 模拟业务处理
        try {
            Thread.sleep(2000);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        return orderNo;
    }

    // 方案2：编程式使用
    public String createOrderProgrammatic(Long userId, String productId) {
        String lockKey = "order:create:" + userId;
        String lockValue = UUID.randomUUID().toString();

        try {
            // 尝试获取锁
            boolean acquired = lockStore.tryAcquire(lockKey, lockValue,
                getOwnerIdentifier(), 10000).isPresent();

            if (!acquired) {
                throw new RuntimeException("系统繁忙，请稍后重试");
            }

            // 执行业务
            return doCreateOrder(userId, productId);

        } finally {
            // 释放锁
            lockStore.release(lockKey, lockValue, getOwnerIdentifier());
        }
    }

    // 方案3：手动控制锁
    public String createOrderManual(Long userId, String productId) {
        // 手动创建锁选项
        LockOptions options = LockOptions.builder()
                .lockKey("order:create:" + userId)
                .expireTime(Duration.ofSeconds(10))
                .waitTime(Duration.ofSeconds(5))
                .reentrant(true)
                .build();

        try {
            if (!distributedLock.tryLock(options)) {
                throw new RuntimeException("获取锁失败");
            }

            return doCreateOrder(userId, productId);

        } finally {
            distributedLock.unlock();
        }
    }

    private String doCreateOrder(Long userId, String productId) {
        // 实际业务逻辑
        return "ORDER-" + System.currentTimeMillis();
    }

    private String generateOrderNo() {
        return "NO-" + UUID.randomUUID().toString().substring(0, 8);
    }

    private String getOwnerIdentifier() {
        return Thread.currentThread().getName();
    }
}

package com.example.distributedlock;

import com.example.distributedlock.service.OrderService;
import org.junit.jupiter.api.Test;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.boot.test.mock.mockito.SpyBean;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.atomic.AtomicInteger;
import static org.junit.jupiter.api.Assertions.*;

@SpringBootTest
class DistributedLockTest {

    @SpyBean
    private OrderService orderService;

    @Test
    void testConcurrentOrderCreation() throws InterruptedException {
        int threadCount = 10;
        ExecutorService executor = Executors.newFixedThreadPool(threadCount);
        CountDownLatch latch = new CountDownLatch(threadCount);

        AtomicInteger successCount = new AtomicInteger(0);
        AtomicInteger failCount = new AtomicInteger(0);

        // 10个线程同时为同一用户创建订单
        Long userId = 12345L;
        String productId = "PROD_001";

        for (int i = 0; i < threadCount; i++) {
            executor.submit(() -> {
                try {
                    String orderNo = orderService.createOrder(userId, productId);
                    System.out.println("创建订单成功: " + orderNo);
                    successCount.incrementAndGet();
                } catch (Exception e) {
                    System.out.println("创建订单失败: " + e.getMessage());
                    failCount.incrementAndGet();
                } finally {
                    latch.countDown();
                }
            });
        }

        latch.await();
        executor.shutdown();

        System.out.println("成功: " + successCount.get() + ", 失败: " + failCount.get());

        // 只有一个线程应该成功
        assertEquals(1, successCount.get());
        assertEquals(threadCount - 1, failCount.get());
    }
}

package com.example.distributedlock.service;

import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor

    // 每30秒清理过期锁
    @Scheduled(fixedDelay = 30000)
    public void cleanExpiredLocks() {
        Query query = new Query();
        query.addCriteria(Criteria.where("expireAt").lt(LocalDateTime.now()));

        long count = mongoTemplate.remove(query, LockDocument.class).getDeletedCount();
        if (count > 0) {
            log.warn("清理了{}个过期锁", count);
        }
    }

    // 获取当前所有锁
    public List<LockDocument> getAllLocks() {
        return mongoTemplate.findAll(LockDocument.class);
    }

    // 强制释放锁（管理员用）
    public boolean forceRelease(String lockKey) {
        Query query = new Query(Criteria.where("_id").is(lockKey));
        return mongoTemplate.remove(query, LockDocument.class).getDeletedCount() > 0;
    }
}
🎯 设计要点总结

原子性保证：使用 findAndModify确保检查和设置的原子性

可重入支持：通过锁计数支持同一线程重入

自动续期：后台线程自动续期，防止业务未完成锁过期

TTL自动清理：MongoDB自动清理过期锁，避免堆积

锁归属验证：通过 lockValue确保只有锁的持有者能释放

优雅降级：锁获取失败有明确异常和重试机制

监控完善：提供监控接口和管理功能

⚠️ 生产环境建议

集群配置：MongoDB使用副本集保证高可用

连接池：合理配置MongoDB连接池大小

超时设置：根据业务设置合理的锁超时时间

熔断降级：在锁服务不可用时要有降级策略

日志记录：记录锁的获取、释放、续期等关键操作

告警监控：监控锁等待时间、获取失败率等指标

这个实现提供了企业级的分布式锁解决方案，可以直接集成到SpringBoot项目中使用。
