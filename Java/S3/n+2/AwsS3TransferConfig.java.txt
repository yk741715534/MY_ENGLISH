package com.example.s3scanner.config;

import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Primary;
import software.amazon.awssdk.auth.credentials.AwsCredentialsProvider;
import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.awscore.defaultsmode.DefaultsMode;
import software.amazon.awssdk.core.async.AsyncRequestBody;
import software.amazon.awssdk.core.async.AsyncResponseTransformer;
import software.amazon.awssdk.http.async.SdkAsyncHttpClient;
import software.amazon.awssdk.http.nio.netty.NettyNioAsyncHttpClient;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.s3.S3AsyncClient;
import software.amazon.awssdk.services.s3.S3Configuration;
import software.amazon.awssdk.transfer.s3.*;
import software.amazon.awssdk.transfer.s3.model.*;
import software.amazon.awssdk.transfer.s3.progress.LoggingTransferListener;
import java.net.URI;
import java.time.Duration;
import java.util.concurrent.CompletableFuture;
import java.util.function.Consumer;

@Configuration
@Slf4j
public class AwsS3TransferConfig {
    
    // ========== 基础配置 ==========
    @Value("${aws.region:us-east-1}")
    private String region;
    
    @Value("${aws.s3.endpoint:}")
    private String s3Endpoint;
    
    @Value("${aws.s3.path-style-access-enabled:false}")
    private boolean pathStyleAccessEnabled;
    
    // ========== HTTP客户端配置 ==========
    @Value("${aws.http-client.max-concurrency:100}")
    private int maxConcurrency;
    
    @Value("${aws.http-client.connection-timeout:10s}")
    private Duration connectionTimeout;
    
    @Value("${aws.http-client.read-timeout:30s}")
    private Duration readTimeout;
    
    @Value("${aws.http-client.write-timeout:30s}")
    private Duration writeTimeout;
    
    // ========== 传输管理器配置 ==========
    @Value("${aws.transfer-manager.upload.multipart-threshold:8MB}")
    private Long multipartUploadThreshold;
    
    @Value("${aws.transfer-manager.upload.multipart-part-size:8MB}")
    private Long multipartPartSize;
    
    @Value("${aws.transfer-manager.upload.max-concurrency:10}")
    private int uploadMaxConcurrency;
    
    @Value("${aws.transfer-manager.download.max-concurrency:10}")
    private int downloadMaxConcurrency;
    
    @Value("${aws.transfer-manager.upload.directory.max-concurrency:5}")
    private int uploadDirectoryMaxConcurrency;
    
    @Value("${aws.transfer-manager.download.directory.max-concurrency:5}")
    private int downloadDirectoryMaxConcurrency;
    
    @Value("${aws.transfer-manager.enable-resume:true}")
    private boolean enableResume;
    
    @Value("${aws.transfer-manager.enable-trailing-checksum:true}")
    private boolean enableTrailingChecksum;
    
    // ========== 重试配置 ==========
    @Value("${aws.transfer-manager.retry.max-attempts:3}")
    private int maxRetryAttempts;
    
    @Value("${aws.transfer-manager.retry.timeout:5m}")
    private Duration transferTimeout;
    
    /**
     * S3异步客户端
     */
    @Bean
    public S3AsyncClient s3AsyncClient(AwsCredentialsProvider credentialsProvider,
                                      SdkAsyncHttpClient httpClient) {
        S3AsyncClient.Builder builder = S3AsyncClient.builder()
            .region(Region.of(region))
            .credentialsProvider(credentialsProvider)
            .httpClient(httpClient)
            .serviceConfiguration(s3Configuration())
            .defaultsMode(DefaultsMode.STANDARD);
        
        if (s3Endpoint != null && !s3Endpoint.isEmpty()) {
            builder.endpointOverride(URI.create(s3Endpoint));
            log.info("配置S3端点: {}", s3Endpoint);
        }
        
        S3AsyncClient client = builder.build();
        log.info("S3AsyncClient初始化完成，区域: {}", region);
        return client;
    }
    
    /**
     * 异步HTTP客户端
     */
    @Bean
    public SdkAsyncHttpClient sdkAsyncHttpClient() {
        return NettyNioAsyncHttpClient.builder()
            .maxConcurrency(maxConcurrency)
            .connectionAcquisitionTimeout(connectionTimeout)
            .connectionTimeout(connectionTimeout)
            .readTimeout(readTimeout)
            .writeTimeout(writeTimeout)
            .tcpKeepAlive(true)
            .maxPendingConnectionAcquires(1000)
            .build();
    }
    
    /**
     * S3服务配置
     */
    private Consumer<S3Configuration.Builder> s3Configuration() {
        return builder -> builder
            .pathStyleAccessEnabled(pathStyleAccessEnabled)
            .chunkedEncodingEnabled(true)
            .checksumValidationEnabled(true);
    }
    
    /**
     * 主TransferManager（用于大文件传输）
     */
    @Bean
    @Primary
    public S3TransferManager s3TransferManager(S3AsyncClient s3AsyncClient) {
        S3TransferManagerOverrideConfiguration overrideConfig = 
            S3TransferManagerOverrideConfiguration.builder()
                .uploadDirectoryConfiguration(uploadDirectoryConfiguration())
                .downloadDirectoryConfiguration(downloadDirectoryConfiguration())
                .build();
        
        S3TransferManager transferManager = S3TransferManager.builder()
            .s3Client(s3AsyncClient)
            .transferConfiguration(builder -> builder
                .uploadDirectoryOverrideConfiguration(overrideConfig.uploadDirectoryConfiguration())
                .downloadDirectoryOverrideConfiguration(overrideConfig.downloadDirectoryConfiguration())
            )
            .build();
        
        log.info("S3TransferManager初始化完成，上传并发: {}，下载并发: {}", 
            uploadMaxConcurrency, downloadMaxConcurrency);
        
        return transferManager;
    }
    
    /**
     * 批量TransferManager（用于目录操作）
     */
    @Bean(name = "batchTransferManager")
    public S3TransferManager batchS3TransferManager(S3AsyncClient s3AsyncClient) {
        return S3TransferManager.builder()
            .s3Client(s3AsyncClient)
            .transferConfiguration(builder -> builder
                .uploadDirectoryOverrideConfiguration(UploadDirectoryOverrideConfiguration.builder()
                    .maxConcurrency(uploadDirectoryMaxConcurrency)
                    .followSymbolicLinks(false)
                    .recursive(true)
                    .build())
                .downloadDirectoryOverrideConfiguration(DownloadDirectoryOverrideConfiguration.builder()
                    .maxConcurrency(downloadDirectoryMaxConcurrency)
                    .downloadFileRequestTransformer(request -> request
                        .overrideConfiguration(b -> b
                            .apiCallTimeout(Duration.ofMinutes(30))
                        )
                    )
                    .build())
            )
            .build();
    }
    
    /**
     * 高性能TransferManager（用于高并发场景）
     */
    @Bean(name = "highPerfTransferManager")
    public S3TransferManager highPerfS3TransferManager(S3AsyncClient s3AsyncClient) {
        return S3TransferManager.builder()
            .s3Client(s3AsyncClient)
            .transferConfiguration(builder -> builder
                .uploadDirectoryOverrideConfiguration(UploadDirectoryOverrideConfiguration.builder()
                    .maxConcurrency(20)  // 更高并发
                    .uploadFileRequestTransformer(req -> req
                        .overrideConfiguration(o -> o
                            .apiCallTimeout(Duration.ofMinutes(5))
                            .apiCallAttemptTimeout(Duration.ofMinutes(2))
                        )
                        .multipartUploadEnabled(true)
                    )
                    .build())
                .downloadDirectoryOverrideConfiguration(DownloadDirectoryOverrideConfiguration.builder()
                    .maxConcurrency(20)  // 更高并发
                    .downloadFileRequestTransformer(req -> req
                        .overrideConfiguration(o -> o
                            .apiCallTimeout(Duration.ofMinutes(5))
                            .apiCallAttemptTimeout(Duration.ofMinutes(2))
                        )
                    )
                    .build())
            )
            .build();
    }
    
    /**
     * 上传目录配置
     */
    private UploadDirectoryOverrideConfiguration uploadDirectoryConfiguration() {
        return UploadDirectoryOverrideConfiguration.builder()
            .maxConcurrency(uploadDirectoryMaxConcurrency)
            .followSymbolicLinks(false)
            .recursive(true)
            .uploadFileRequestTransformer(this::configureUploadRequest)
            .build();
    }
    
    /**
     * 下载目录配置
     */
    private DownloadDirectoryOverrideConfiguration downloadDirectoryConfiguration() {
        return DownloadDirectoryOverrideConfiguration.builder()
            .maxConcurrency(downloadDirectoryMaxConcurrency)
            .downloadFileRequestTransformer(this::configureDownloadRequest)
            .build();
    }
    
    /**
     * 配置上传请求
     */
    private UploadFileRequest configureUploadRequest(UploadFileRequest request) {
        UploadFileRequest.Builder builder = request.toBuilder();
        
        if (enableResume) {
            builder.resumableFileUploadEnabled(true);
        }
        
        if (enableTrailingChecksum) {
            builder.checksumValidationEnabled(true);
        }
        
        builder.overrideConfiguration(o -> o
            .apiCallTimeout(transferTimeout)
            .apiCallAttemptTimeout(Duration.ofMinutes(1))
        );
        
        // 添加进度监听器
        if (log.isDebugEnabled()) {
            builder.addTransferListener(LoggingTransferListener.create());
        }
        
        return builder.build();
    }
    
    /**
     * 配置下载请求
     */
    private DownloadFileRequest configureDownloadRequest(DownloadFileRequest request) {
        DownloadFileRequest.Builder builder = request.toBuilder();
        
        if (enableResume) {
            builder.resumableFileDownloadEnabled(true);
        }
        
        builder.overrideConfiguration(o -> o
            .apiCallTimeout(transferTimeout)
            .apiCallAttemptTimeout(Duration.ofMinutes(1))
        );
        
        return builder.build();
    }
    
    /**
     * AWS凭证提供者
     */
    @Bean
    public AwsCredentialsProvider awsCredentialsProvider() {
        return DefaultCredentialsProvider.builder().build();
    }
}
