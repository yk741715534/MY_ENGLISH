package com.example.s3scanner.config;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import software.amazon.awssdk.auth.credentials.AwsCredentialsProvider;
import software.amazon.awssdk.core.async.AsyncRequestBody;
import software.amazon.awssdk.core.async.AsyncResponseTransformer;
import software.amazon.awssdk.services.s3.S3AsyncClient;
import software.amazon.awssdk.transfer.s3.S3TransferManager;
import software.amazon.awssdk.transfer.s3.UploadDirectoryOverrideConfiguration;
import software.amazon.awssdk.transfer.s3.DownloadDirectoryOverrideConfiguration;
import software.amazon.awssdk.transfer.s3.FileDownload;
import software.amazon.awssdk.transfer.s3.FileUpload;
import software.amazon.awssdk.transfer.s3.S3TransferManagerOverrideConfiguration;
import software.amazon.awssdk.transfer.s3.model.*;
import java.nio.file.Path;
import java.time.Duration;
import java.util.concurrent.CompletableFuture;

@Configuration
public class S3TransferManagerConfig {
    
    @Value("${aws.transfer-manager.upload.multipart-threshold:8MB}")
    private Long multipartUploadThreshold;
    
    @Value("${aws.transfer-manager.upload.multipart-part-size:8MB}")
    private Long multipartPartSize;
    
    @Value("${aws.transfer-manager.upload.max-concurrency:10}")
    private int uploadMaxConcurrency;
    
    @Value("${aws.transfer-manager.download.max-concurrency:10}")
    private int downloadMaxConcurrency;
    
    @Value("${aws.transfer-manager.upload.directory.max-concurrency:5}")
    private int uploadDirectoryMaxConcurrency;
    
    @Value("${aws.transfer-manager.download.directory.max-concurrency:5}")
    private int downloadDirectoryMaxConcurrency;
    
    /**
     * 方式1：使用默认配置（最简单）
     */
    @Bean
    public S3TransferManager defaultS3TransferManager(S3AsyncClient s3AsyncClient) {
        return S3TransferManager.builder()
            .s3Client(s3AsyncClient)
            .build();
    }
    
    /**
     * 方式2：自定义配置（推荐用于生产环境）
     */
    @Bean(name = "customS3TransferManager")
    public S3TransferManager customS3TransferManager(S3AsyncClient s3AsyncClient) {
        S3TransferManagerOverrideConfiguration overrideConfiguration = 
            S3TransferManagerOverrideConfiguration.builder()
                .uploadDirectoryConfiguration(uploadDirectoryConfig())
                .downloadDirectoryConfiguration(downloadDirectoryConfig())
                .build();
        
        return S3TransferManager.builder()
            .s3Client(s3AsyncClient)
            .transferConfiguration(builder -> builder
                .uploadDirectoryOverrideConfiguration(overrideConfiguration.uploadDirectoryConfiguration())
                .downloadDirectoryOverrideConfiguration(overrideConfiguration.downloadDirectoryConfiguration())
            )
            .build();
    }
    
    /**
     * 上传目录配置
     */
    private UploadDirectoryOverrideConfiguration uploadDirectoryConfig() {
        return UploadDirectoryOverrideConfiguration.builder()
            .maxConcurrency(uploadDirectoryMaxConcurrency)
            .followSymbolicLinks(false)
            .recursive(true)
            .build();
    }
    
    /**
     * 下载目录配置
     */
    private DownloadDirectoryOverrideConfiguration downloadDirectoryConfig() {
        return DownloadDirectoryOverrideConfiguration.builder()
            .maxConcurrency(downloadDirectoryMaxConcurrency)
            .downloadFileRequestTransformer(request -> request
                .overrideConfiguration(b -> b
                    .apiCallTimeout(Duration.ofMinutes(30))
                )
            )
            .build();
    }
    
    /**
     * 方式3：高级配置（针对大文件优化）
     */
    @Bean(name = "advancedS3TransferManager")
    public S3TransferManager advancedS3TransferManager(S3AsyncClient s3AsyncClient) {
        return S3TransferManager.builder()
            .s3Client(s3AsyncClient)
            .transferConfiguration(b -> b
                .uploadDirectoryOverrideConfiguration(UploadDirectoryOverrideConfiguration.builder()
                    .maxConcurrency(uploadMaxConcurrency)
                    .uploadFileRequestTransformer(req -> req
                        .overrideConfiguration(o -> o
                            .apiCallTimeout(Duration.ofMinutes(10))
                        )
                    )
                    .build())
                .downloadDirectoryOverrideConfiguration(DownloadDirectoryOverrideConfiguration.builder()
                    .maxConcurrency(downloadMaxConcurrency)
                    .downloadFileRequestTransformer(req -> req
                        .overrideConfiguration(o -> o
                            .apiCallTimeout(Duration.ofMinutes(10))
                        )
                    )
                    .build())
            )
            .build();
    }
}
