package com.example.s3scanner;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import software.amazon.awssdk.transfer.s3.S3TransferManager;
import java.io.File;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.concurrent.CompletableFuture;
import static org.junit.jupiter.api.Assertions.*;

@SpringBootTest
class S3TransferManagerTest {
    
    @Autowired
    private S3TransferManager s3TransferManager;
    
    @Test
    void testTransferManagerBean() {
        assertNotNull(s3TransferManager, "S3TransferManager Bean应该被注入");
    }
    
    @Test
    void testFileUploadDownload() throws Exception {
        // 创建测试文件
        Path tempFile = Files.createTempFile("test-upload", ".txt");
        Files.write(tempFile, "Hello S3 Transfer Manager!".getBytes());
        
        String bucket = "test-bucket";
        String key = "test/" + tempFile.getFileName();
        
        try {
            // 上传文件
            CompletableFuture<Void> uploadFuture = uploadFile(bucket, key, tempFile);
            uploadFuture.get(); // 等待完成
            
            // 下载文件
            Path downloadPath = Files.createTempFile("test-download", ".txt");
            CompletableFuture<Void> downloadFuture = downloadFile(bucket, key, downloadPath);
            downloadFuture.get(); // 等待完成
            
            // 验证
            String content = Files.readString(downloadPath);
            assertEquals("Hello S3 Transfer Manager!", content);
            
        } finally {
            // 清理
            Files.deleteIfExists(tempFile);
        }
    }
}
