// AwsConfig.java
package com.example.s3scanner.config;

import software.amazon.awssdk.auth.credentials.DefaultCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.s3.S3AsyncClient;
import software.amazon.awssdk.services.s3.S3AsyncClientBuilder;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Primary;
import java.time.Duration;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

@Configuration
public class AwsConfig {
    
    @Value("${spring.cloud.aws.region.static}")
    private String region;
    
    @Bean
    @Primary
    public S3AsyncClient s3AsyncClient() {
        // 创建独立的线程池用于S3异步操作
        ExecutorService executorService = Executors.newFixedThreadPool(20, r -> {
            Thread thread = new Thread(r, "s3-async-" + r.hashCode());
            thread.setDaemon(true);
            return thread;
        });
        
        S3AsyncClientBuilder builder = S3AsyncClient.builder()
            .region(Region.of(region))
            .credentialsProvider(DefaultCredentialsProvider.create());
        
        // 配置连接池
        return builder
            .httpConfiguration(h -> h
                .maxConnections(100)
                .connectionTimeout(Duration.ofSeconds(10))
                .connectionAcquisitionTimeout(Duration.ofSeconds(5))
                .connectionTimeToLive(Duration.ofMinutes(5))
            )
            .overrideConfiguration(o -> o
                .apiCallTimeout(Duration.ofSeconds(30))
                .apiCallAttemptTimeout(Duration.ofSeconds(10))
            )
            .asyncConfiguration(a -> a
                .advancedOption(AdvancedAsyncClientOption.FUTURE_COMPLETION_EXECUTOR, 
                    executorService)
            )
            .build();
    }
}
