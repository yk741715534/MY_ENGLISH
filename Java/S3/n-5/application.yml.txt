# 应用配置
spring:
  application:
    name: s3-realtime-scanner

  # MongoDB配置
  data:
    mongodb:
      uri: ${MONGODB_URI:mongodb://localhost:27017/s3-scanner}
      auto-index-creation: true

  # AWS配置
  cloud:
    aws:
      region:
        static: ${AWS_REGION:us-east-1}
      credentials:
        access-key: ${AWS_ACCESS_KEY_ID:}
        secret-key: ${AWS_SECRET_ACCESS_KEY:}

  # 线程池配置
  task:
    execution:
      pool:
        core-size: 10
        max-size: 50
        queue-capacity: 1000
        keep-alive: 60s
      thread-name-prefix: "async-task-"

    scheduling:
      pool:
        size: 5
      thread-name-prefix: "scheduler-"

# 应用实例配置
app:
  instance:
    id: ${HOSTNAME:${spring.application.name}-${random.uuid}}
    concurrency-limit: 20
    heartbeat-interval: 30

  # 扫描配置
  scanner:
    enabled: true
    interval-ms: 3000
    initial-delay-ms: 10000
    max-scan-time-ms: 2500
    incremental-scan: true
    etag-cache-size: 10000
    etag-cache-ttl: 3600

  # S3配置
  s3:
    bucket: ${S3_BUCKET:my-bucket}
    scan-prefixes: uploads/,pending/
    page-size: 100

  # 处理配置
  processor:
    download-dir: /tmp/s3-downloads
    max-concurrent-downloads: 5
    download-timeout-ms: 30000
    process-timeout-ms: 60000
    max-retries: 3
    retry-delay-ms: 1000

  # 文件移动配置
  mover:
    enabled: true
    strategy: ARCHIVE_AND_BACKUP
    archive-prefix: archive/
    backup-prefix: backup/
    failed-prefix: failed/
    add-timestamp: true
    compress-large-files: true
    compress-threshold-mb: 10

  # 分布式锁配置
  lock:
    coordination-ttl-ms: 10000
    process-ttl-ms: 30000
    cleanup-ttl-ms: 60000
    retry-interval-ms: 100
    max-retries: 3

# 监控配置
management:
  endpoints:
    web:
      exposure:
        include: health,metrics,prometheus,info,scanner
  metrics:
    export:
      prometheus:
        enabled: true
  endpoint:
    health:
      show-details: always
      probes:
        enabled: true

# 日志配置
logging:
  level:
    com.example.s3scanner: INFO
    org.springframework.data.mongodb: WARN
  file:
    name: logs/s3-scanner.log
  logback:
    rollingpolicy:
      max-file-size: 10MB
      max-history: 30
