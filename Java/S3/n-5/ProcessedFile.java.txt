package com.example.s3scanner.model.domain;

import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;
import org.springframework.data.annotation.Id;
import org.springframework.data.annotation.Version;
import org.springframework.data.mongodb.core.index.CompoundIndex;
import org.springframework.data.mongodb.core.index.Indexed;
import org.springframework.data.mongodb.core.mapping.Document;
import org.springframework.data.mongodb.core.mapping.Field;
import java.util.Date;
import java.util.Map;

@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
@Document(collection = "processed_files")
@CompoundIndex(name = "bucket_key_version_idx", 
    def = "{'bucket': 1, 'key': 1, 'version': 1}", unique = true)
public class ProcessedFile {
    
    @Id
    private String id;  // bucket:key:version
    
    @Indexed
    private String bucket;
    
    @Indexed
    private String key;
    
    @Indexed
    private String prefix;
    
    @Indexed
    private ProcessingStatus status;
    
    @Indexed
    private String instanceId;
    
    private Date processStartTime;
    private Date processEndTime;
    
    private Long durationMs;
    private Long fileSize;
    private String etag;
    private String localPath;
    
    @Indexed
    private Integer version;
    
    private String errorMessage;
    private String stackTrace;
    
    @Indexed
    private String archivedKey;
    
    @Indexed
    private String failedKey;
    
    private Map<String, Object> processingResult;
    private Map<String, Object> metadata;
    
    @Indexed(expireAfterSeconds = 604800)  // 7天后自动删除
    private Date createdAt;
    
    @Field("_version")
    private Long docVersion = 1L;
    
    public enum ProcessingStatus {
        DOWNLOADING,
        PROCESSING,
        COMPLETED,
        FAILED,
        ARCHIVED
    }
}
