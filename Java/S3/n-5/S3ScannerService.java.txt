package com.example.s3scanner.service.scanner;

import com.example.s3scanner.model.domain.S3ObjectInfo;
import com.example.s3scanner.repository.S3ObjectInfoRepository;
import com.example.s3scanner.service.coordinator.DistributedCoordinator;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import software.amazon.awssdk.core.async.AsyncResponseTransformer;
import software.amazon.awssdk.services.s3.S3AsyncClient;
import software.amazon.awssdk.services.s3.model.*;
import java.time.Instant;
import java.util.*;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.stream.Collectors;

@Service
@Slf4j
@RequiredArgsConstructor
public class S3ScannerService {
    
    private final S3AsyncClient s3Client;
    private final S3ObjectInfoRepository objectRepository;
    private final DistributedCoordinator coordinator;
    
    @Value("${app.s3.bucket}")
    private String bucket;
    
    @Value("${app.scanner.incremental-scan}")
    private boolean incrementalScan;
    
    @Value("${app.scanner.page-size}")
    private int pageSize;
    
    @Value("${app.scanner.etag-cache-size}")
    private int etagCacheSize;
    
    @Value("${app.scanner.etag-cache-ttl}")
    private int etagCacheTtl;
    
    // ETag缓存：key -> (etag, timestamp)
    private final Map<String, CacheEntry> etagCache = new ConcurrentHashMap<>();
    
    // 最后扫描时间
    private volatile Instant lastFullScanTime = Instant.MIN;
    private final Map<String, Instant> lastPrefixScanTime = new ConcurrentHashMap<>();
    
    @Data
    @AllArgsConstructor
    private static class CacheEntry {
        private String etag;
        private long timestamp;
        
        public boolean isExpired(int ttlSeconds) {
            return (System.currentTimeMillis() - timestamp) > ttlSeconds * 1000L;
        }
    }
    
    /**
     * 扫描指定前缀
     */
    @Transactional
    public ScanResult scanPrefix(String prefix) {
        log.info("开始扫描前缀: {}", prefix);
        
        List<S3ObjectInfo> newObjects = new ArrayList<>();
        String continuationToken = null;
        int totalScanned = 0;
        int totalFound = 0;
        
        long startTime = System.currentTimeMillis();
        
        try {
            do {
                // 构建请求
                ListObjectsV2Request.Builder requestBuilder = ListObjectsV2Request.builder()
                    .bucket(bucket)
                    .prefix(prefix)
                    .maxKeys(pageSize);
                
                if (continuationToken != null) {
                    requestBuilder.continuationToken(continuationToken);
                }
                
                // 执行扫描
                ListObjectsV2Response response = s3Client.listObjectsV2(requestBuilder.build())
                    .get(10, TimeUnit.SECONDS);
                
                // 处理本页结果
                List<S3ObjectInfo> pageObjects = processPageObjects(response.contents(), prefix);
                newObjects.addAll(pageObjects);
                
                totalScanned += response.contents().size();
                totalFound += pageObjects.size();
                
                continuationToken = response.nextContinuationToken();
                
                // 如果扫描时间过长，分批处理
                if (System.currentTimeMillis() - startTime > 2000 && totalFound > 0) {
                    log.info("扫描超时，分批处理: 已找到 {} 个新对象", totalFound);
                    break;
                }
                
            } while (continuationToken != null);
            
            // 更新最后扫描时间
            lastPrefixScanTime.put(prefix, Instant.now());
            
            // 清理缓存
            cleanupCache();
            
            return ScanResult.builder()
                .prefix(prefix)
                .totalScanned(totalScanned)
                .newObjects(newObjects)
                .scanDurationMs(System.currentTimeMillis() - startTime)
                .success(true)
                .build();
            
        } catch (Exception e) {
            log.error("扫描前缀失败: {}", prefix, e);
            return ScanResult.builder()
                .prefix(prefix)
                .totalScanned(totalScanned)
                .newObjects(newObjects)
                .scanDurationMs(System.currentTimeMillis() - startTime)
                .success(false)
                .errorMessage(e.getMessage())
                .build();
        }
    }
    
    /**
     * 增量扫描
     */
    public ScanResult incrementalScan(String prefix) {
        if (!incrementalScan) {
            return scanPrefix(prefix);
        }
        
        // 获取最后修改时间作为起始点
        Instant lastScan = lastPrefixScanTime.getOrDefault(prefix, Instant.MIN);
        
        log.info("增量扫描前缀: {}，最后扫描时间: {}", prefix, lastScan);
        
        List<S3ObjectInfo> newObjects = new ArrayList<>();
        String continuationToken = null;
        int totalScanned = 0;
        int totalFound = 0;
        
        long startTime = System.currentTimeMillis();
        
        try {
            do {
                // 构建增量扫描请求
                ListObjectsV2Request.Builder requestBuilder = ListObjectsV2Request.builder()
                    .bucket(bucket)
                    .prefix(prefix)
                    .maxKeys(pageSize);
                
                if (continuationToken != null) {
                    requestBuilder.continuationToken(continuationToken);
                }
                
                ListObjectsV2Response response = s3Client.listObjectsV2(requestBuilder.build())
                    .get(10, TimeUnit.SECONDS);
                
                // 筛选新增或修改的文件
                List<S3Object> filtered = response.contents().stream()
                    .filter(obj -> {
                        // 跳过目录
                        if (obj.key().endsWith("/")) {
                            return false;
                        }
                        
                        // 检查最后修改时间
                        if (obj.lastModified() != null && 
                            obj.lastModified().isAfter(lastScan)) {
                            return true;
                        }
                        
                        // 检查ETag是否变化
                        return isObjectModified(obj.key(), obj.eTag());
                    })
                    .collect(Collectors.toList());
                
                List<S3ObjectInfo> pageObjects = processPageObjects(filtered, prefix);
                newObjects.addAll(pageObjects);
                
                totalScanned += response.contents().size();
                totalFound += pageObjects.size();
                
                continuationToken = response.nextContinuationToken();
                
                // 增量扫描时间控制
                if (System.currentTimeMillis() - startTime > 1000) {
                    break;
                }
                
            } while (continuationToken != null);
            
            // 更新最后扫描时间
            lastPrefixScanTime.put(prefix, Instant.now());
            
            return ScanResult.builder()
                .prefix(prefix)
                .totalScanned(totalScanned)
                .newObjects(newObjects)
                .scanDurationMs(System.currentTimeMillis() - startTime)
                .success(true)
                .build();
            
        } catch (Exception e) {
            log.error("增量扫描失败: {}", prefix, e);
            return ScanResult.builder()
                .prefix(prefix)
                .totalScanned(totalScanned)
                .newObjects(newObjects)
                .scanDurationMs(System.currentTimeMillis() - startTime)
                .success(false)
                .errorMessage(e.getMessage())
                .build();
        }
    }
    
    /**
     * 处理单页对象
     */
    private List<S3ObjectInfo> processPageObjects(List<S3Object> s3Objects, String prefix) {
        List<S3ObjectInfo> newObjects = new ArrayList<>();
        
        for (S3Object s3Object : s3Objects) {
            try {
                // 跳过目录
                if (s3Object.key().endsWith("/")) {
                    continue;
                }
                
                // 检查是否应该由本实例处理
                if (!coordinator.shouldProcess(bucket, s3Object.key())) {
                    continue;
                }
                
                // 检查是否已处理
                Optional<S3ObjectInfo> existing = objectRepository.findByBucketAndKey(
                    bucket, s3Object.key());
                
                if (existing.isPresent()) {
                    S3ObjectInfo existingObj = existing.get();
                    
                    // 检查文件是否被修改
                    if (existingObj.getEtag() != null && 
                        existingObj.getEtag().equals(s3Object.eTag())) {
                        continue;  // 文件未修改
                    }
                    
                    // 文件被修改，更新信息
                    existingObj.setEtag(s3Object.eTag());
                    existingObj.setSize(s3Object.size());
                    existingObj.setLastModified(Date.from(s3Object.lastModified()));
                    existingObj.setStatus(S3ObjectInfo.ProcessingStatus.PENDING);
                    existingObj.setRetryCount(0);
                    existingObj.setErrorMessage(null);
                    
                    objectRepository.save(existingObj);
                    newObjects.add(existingObj);
                    
                } else {
                    // 新文件
                    S3ObjectInfo newObj = S3ObjectInfo.builder()
                        .id(bucket + ":" + s3Object.key())
                        .bucket(bucket)
                        .key(s3Object.key())
                        .etag(s3Object.eTag())
                        .size(s3Object.size())
                        .lastModified(Date.from(s3Object.lastModified()))
                        .prefix(prefix)
                        .status(S3ObjectInfo.ProcessingStatus.PENDING)
                        .retryCount(0)
                        .createdAt(new Date())
                        .build();
                    
                    objectRepository.save(newObj);
                    newObjects.add(newObj);
                }
                
                // 更新缓存
                updateCache(s3Object.key(), s3Object.eTag());
                
            } catch (Exception e) {
                log.error("处理对象失败: {}", s3Object.key(), e);
            }
        }
        
        return newObjects;
    }
    
    /**
     * 检查对象是否被修改
     */
    private boolean isObjectModified(String key, String currentEtag) {
        CacheEntry cached = etagCache.get(key);
        if (cached == null) {
            return true;  // 不在缓存中，认为是新文件
        }
        
        if (cached.isExpired(etagCacheTtl)) {
            etagCache.remove(key);
            return true;
        }
        
        return !cached.getEtag().equals(currentEtag);
    }
    
    /**
     * 更新缓存
     */
    private void updateCache(String key, String etag) {
        etagCache.put(key, new CacheEntry(etag, System.currentTimeMillis()));
        
        // 如果缓存太大，清理最老的条目
        if (etagCache.size() > etagCacheSize) {
            cleanupCache();
        }
    }
    
    /**
     * 清理缓存
     */
    private void cleanupCache() {
        if (etagCache.size() <= etagCacheSize) {
            return;
        }
        
        // 按时间排序，移除最老的条目
        List<Map.Entry<String, CacheEntry>> entries = new ArrayList<>(etagCache.entrySet());
        entries.sort(Comparator.comparingLong(e -> e.getValue().getTimestamp()));
        
        int toRemove = entries.size() - etagCacheSize;
        for (int i = 0; i < toRemove && i < entries.size(); i++) {
            etagCache.remove(entries.get(i).getKey());
        }
    }
    
    @Data
    @Builder
    public static class ScanResult {
        private String prefix;
        private int totalScanned;
        private List<S3ObjectInfo> newObjects;
        private long scanDurationMs;
        private boolean success;
        private String errorMessage;
    }
}
