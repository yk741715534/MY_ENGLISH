package com.example.s3scanner.service.processor;

import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.util.DigestUtils;
import org.apache.commons.io.FilenameUtils;
import org.apache.tika.Tika;
import javax.imageio.ImageIO;
import java.awt.image.BufferedImage;
import java.io.*;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.*;
import java.util.zip.ZipEntry;
import java.util.zip.ZipInputStream;

@Service
@Slf4j
public class FileProcessor {
    
    private final Tika tika = new Tika();
    
    /**
     * 处理文件
     */
    public Map<String, Object> process(File file) throws IOException {
        Map<String, Object> result = new HashMap<>();
        long startTime = System.currentTimeMillis();
        
        try {
            // 1. 基础信息
            result.put("filename", file.getName());
            result.put("size", file.length());
            result.put("path", file.getAbsolutePath());
            result.put("lastModified", new Date(file.lastModified()));
            
            // 2. 计算MD5
            String md5 = calculateMD5(file);
            result.put("md5", md5);
            
            // 3. 检测MIME类型
            String mimeType = detectMimeType(file);
            result.put("mimeType", mimeType);
            result.put("extension", FilenameUtils.getExtension(file.getName()));
            
            // 4. 根据文件类型处理
            String fileType = mimeType.split("/")[0];
            
            switch (fileType) {
                case "text":
                    processTextFile(file, result);
                    break;
                case "image":
                    processImageFile(file, result);
                    break;
                case "application":
                    if (mimeType.contains("pdf")) {
                        processPdfFile(file, result);
                    } else if (mimeType.contains("zip") || mimeType.contains("compress")) {
                        processZipFile(file, result);
                    }
                    break;
                default:
                    result.put("processed", true);
                    result.put("message", "File processed, no specific handler");
            }
            
            // 5. 提取元数据
            Map<String, Object> metadata = extractMetadata(file, mimeType);
            result.put("metadata", metadata);
            
            // 6. 处理统计
            result.put("processTime", System.currentTimeMillis() - startTime);
            result.put("success", true);
            
        } catch (Exception e) {
            log.error("处理文件失败: {}", file.getName(), e);
            result.put("success", false);
            result.put("error", e.getMessage());
            result.put("processTime", System.currentTimeMillis() - startTime);
            
            throw new IOException("文件处理失败: " + file.getName(), e);
        }
        
        return result;
    }
    
    /**
     * 处理文本文件
     */
    private void processTextFile(File file, Map<String, Object> result) throws IOException {
        try (BufferedReader reader = new BufferedReader(new FileReader(file))) {
            int lineCount = 0;
            int wordCount = 0;
            int charCount = 0;
            String line;
            
            while ((line = reader.readLine()) != null) {
                lineCount++;
                charCount += line.length();
                wordCount += line.split("\\s+").length;
            }
            
            result.put("lineCount", lineCount);
            result.put("wordCount", wordCount);
            result.put("charCount", charCount);
            
            // 如果是小文件，读取前几行
            if (file.length() < 1024 * 1024) {  // 小于1MB
                List<String> previewLines = new ArrayList<>();
                try (BufferedReader previewReader = new BufferedReader(new FileReader(file))) {
                    for (int i = 0; i < 10; i++) {
                        String previewLine = previewReader.readLine();
                        if (previewLine == null) break;
                        previewLines.add(previewLine);
                    }
                }
                result.put("preview", previewLines);
            }
        }
    }
    
    /**
     * 处理图片文件
     */
    private void processImageFile(File file, Map<String, Object> result) throws IOException {
        BufferedImage image = ImageIO.read(file);
        if (image != null) {
            result.put("width", image.getWidth());
            result.put("height", image.getHeight());
            result.put("colorDepth", image.getColorModel().getPixelSize());
        }
    }
    
    /**
     * 处理PDF文件
     */
    private void processPdfFile(File file, Map<String, Object> result) {
        // 使用PDF解析库，这里简化处理
        result.put("pdfProcessed", true);
        result.put("message", "PDF file detected");
    }
    
    /**
     * 处理ZIP文件
     */
    private void processZipFile(File file, Map<String, Object> result) throws IOException {
        List<Map<String, Object>> zipEntries = new ArrayList<>();
        
        try (ZipInputStream zis = new ZipInputStream(new FileInputStream(file))) {
            ZipEntry entry;
            while ((entry = zis.getNextEntry()) != null) {
                Map<String, Object> entryInfo = new HashMap<>();
                entryInfo.put("name", entry.getName());
                entryInfo.put("size", entry.getSize());
                entryInfo.put("compressedSize", entry.getCompressedSize());
                entryInfo.put("isDirectory", entry.isDirectory());
                zipEntries.add(entryInfo);
                
                zis.closeEntry();
            }
        }
        
        result.put("zipEntries", zipEntries);
        result.put("entryCount", zipEntries.size());
    }
    
    /**
     * 计算MD5
     */
    private String calculateMD5(File file) throws IOException {
        try (InputStream is = new FileInputStream(file)) {
            byte[] digest = DigestUtils.md5Digest(is);
            return bytesToHex(digest);
        }
    }
    
    /**
     * 检测MIME类型
     */
    private String detectMimeType(File file) throws IOException {
        return tika.detect(file);
    }
    
    /**
     * 提取元数据
     */
    private Map<String, Object> extractMetadata(File file, String mimeType) {
        Map<String, Object> metadata = new HashMap<>();
        
        try {
            // 使用Tika提取元数据
            org.apache.tika.metadata.Metadata tikaMetadata = new org.apache.tika.metadata.Metadata();
            tika.parse(file, tikaMetadata);
            
            for (String name : tikaMetadata.names()) {
                String[] values = tikaMetadata.getValues(name);
                if (values.length == 1) {
                    metadata.put(name, values[0]);
                } else {
                    metadata.put(name, Arrays.asList(values));
                }
            }
            
        } catch (Exception e) {
            log.warn("提取元数据失败: {}", file.getName(), e);
        }
        
        return metadata;
    }
    
    /**
     * 字节数组转十六进制
     */
    private String bytesToHex(byte[] bytes) {
        StringBuilder sb = new StringBuilder();
        for (byte b : bytes) {
            sb.append(String.format("%02x", b));
        }
        return sb.toString();
    }
}
