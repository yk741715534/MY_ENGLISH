// EnhancedS3ObjectProcessor.java
@Service
@Slf4j
public class EnhancedS3ObjectProcessor {
    
    @Autowired
    private S3FileMover fileMover;
    
    @Autowired
    private DownloadService downloadService;
    
    @Autowired
    private ProcessedFileRepository processedFileRepository;
    
    @Value("${s3.move.enabled:true}")
    private boolean moveEnabled;
    
    @Value("${s3.archive.prefix:archive/}")
    private String archivePrefix;
    
    @Value("${s3.backup.enabled:false}")
    private boolean backupEnabled;
    
    /**
     * 处理并移动文件
     */
    public ProcessingResult processAndMove(String bucket, String key) {
        long startTime = System.currentTimeMillis();
        
        try {
            // 1. 检查是否已处理
            if (isAlreadyProcessed(bucket, key)) {
                return ProcessingResult.skipped("Already processed");
            }
            
            // 2. 下载文件
            File localFile = downloadService.downloadObjectWithResume(bucket, key);
            
            // 3. 处理文件
            Object processResult = processFileContent(localFile);
            
            // 4. 记录处理成功
            recordProcessingSuccess(bucket, key, processResult);
            
            // 5. 移动文件（如果启用）
            if (moveEnabled) {
                moveAfterProcessing(bucket, key, processResult);
            }
            
            // 6. 清理本地文件
            cleanupLocalFile(localFile);
            
            long duration = System.currentTimeMillis() - startTime;
            return ProcessingResult.success(processResult, duration);
            
        } catch (Exception e) {
            // 处理失败，移动到失败目录
            handleProcessingFailure(bucket, key, e);
            
            long duration = System.currentTimeMillis() - startTime;
            return ProcessingResult.failed(e.getMessage(), duration);
        }
    }
    
    /**
     * 处理成功后移动文件
     */
    private void moveAfterProcessing(String bucket, String key, Object processResult) {
        try {
            S3FileMover.MoveResult moveResult = null;
            
            if (backupEnabled) {
                // 先备份
                moveResult = fileMover.backupByDate(bucket, key);
                log.info("文件已备份: {} -> {}", key, moveResult.getTargetKey());
                
                // 再移动到归档
                String newKey = moveResult.getTargetKey();
                moveResult = fileMover.archiveProcessedFile(bucket, newKey, 
                    ProcessingResult.success(processResult, 0));
            } else {
                // 直接移动到归档
                moveResult = fileMover.archiveProcessedFile(bucket, key, 
                    ProcessingResult.success(processResult, 0));
            }
            
            log.info("文件已归档: {} -> {}", key, moveResult.getTargetKey());
            
        } catch (Exception e) {
            log.error("移动文件失败: {}", key, e);
            // 记录移动失败，但不影响处理结果
            recordMoveFailure(bucket, key, e.getMessage());
        }
    }
    
    /**
     * 处理失败时移动文件
     */
    private void handleProcessingFailure(String bucket, String key, Exception error) {
        try {
            S3FileMover.MoveResult moveResult = fileMover.moveFailedFile(
                bucket, key, error.getMessage());
            
            log.warn("处理失败，文件已移动到失败目录: {} -> {}", 
                key, moveResult.getTargetKey());
            
            // 记录失败状态
            recordProcessingFailure(bucket, key, error.getMessage(), 
                moveResult.getTargetKey());
            
        } catch (Exception e) {
            log.error("移动失败文件出错: {}", key, e);
        }
    }
    
    private boolean isAlreadyProcessed(String bucket, String key) {
        return processedFileRepository.findByBucketAndKeyAndStatus(
            bucket, key, ProcessedFile.ProcessingStatus.COMPLETED).isPresent();
    }
    
    private void recordProcessingSuccess(String bucket, String key, Object result) {
        ProcessedFile processed = ProcessedFile.builder()
            .id(bucket + ":" + key)
            .bucket(bucket)
            .key(key)
            .status(ProcessedFile.ProcessingStatus.COMPLETED)
            .processStartTime(new Date())
            .processEndTime(new Date())
            .result(result)
            .build();
        
        processedFileRepository.save(processed);
    }
    
    private void recordProcessingFailure(String bucket, String key, 
                                        String error, String failedPath) {
        ProcessedFile processed = ProcessedFile.builder()
            .id(bucket + ":" + key)
            .bucket(bucket)
            .key(key)
            .status(ProcessedFile.ProcessingStatus.FAILED)
            .processStartTime(new Date())
            .processEndTime(new Date())
            .errorMessage(error)
            .failedPath(failedPath)
            .retryCount(1)
            .build();
        
        processedFileRepository.save(processed);
    }
    
    private void recordMoveFailure(String bucket, String key, String error) {
        // 记录移动失败，但不改变处理状态
        log.error("记录移动失败: {}/{} - {}", bucket, key, error);
    }
    
    private void cleanupLocalFile(File file) {
        try {
            if (file.exists() && !file.delete()) {
                file.deleteOnExit();
            }
        } catch (Exception e) {
            log.warn("清理本地文件失败: {}", file.getPath(), e);
        }
    }
}
