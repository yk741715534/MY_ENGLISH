// EnhancedThreeSecondScanner.java
@Component
@Slf4j
public class EnhancedThreeSecondScanner {
    
    @Autowired
    private IncrementalS3Scanner incrementalScanner;
    
    @Autowired
    private EnhancedS3ObjectProcessor objectProcessor;
    
    @Autowired
    private InstanceRegistry instanceRegistry;
    
    @Value("${s3.move.strategy:ARCHIVE}")
    private String moveStrategy;
    
    /**
     * 增强的3秒扫描
     */
    @Scheduled(fixedDelay = 3000)
    public void enhancedScan() {
        if (isScanning()) {
            return;
        }
        
        setScanning(true);
        long startTime = System.currentTimeMillis();
        
        try {
            // 1. 获取待处理文件
            List<S3ObjectInfo> pendingFiles = incrementalScanner.incrementalScan();
            
            if (pendingFiles.isEmpty()) {
                return;
            }
            
            log.info("发现 {} 个待处理文件，开始处理并归档", pendingFiles.size());
            
            // 2. 并发处理
            ExecutorService executor = Executors.newVirtualThreadPerTaskExecutor();
            List<CompletableFuture<ProcessingResult>> futures = pendingFiles.stream()
                .map(obj -> CompletableFuture.supplyAsync(() -> 
                    processWithMove(obj.getBucket(), obj.getKey()), executor))
                .collect(Collectors.toList());
            
            // 3. 等待完成
            CompletableFuture.allOf(futures.toArray(new CompletableFuture[0]))
                .orTimeout(2500, TimeUnit.MILLISECONDS)
                .whenComplete((result, throwable) -> {
                    if (throwable != null) {
                        log.warn("处理超时，部分文件未完成");
                    }
                    executor.shutdown();
                });
            
            // 4. 统计结果
            List<ProcessingResult> results = futures.stream()
                .map(future -> {
                    try {
                        return future.get();
                    } catch (Exception e) {
                        return ProcessingResult.failed(e.getMessage(), 0);
                    }
                })
                .collect(Collectors.toList());
            
            long successCount = results.stream()
                .filter(ProcessingResult::isSuccess)
                .count();
            
            log.info("处理完成: 成功 {} 个，失败 {} 个", 
                successCount, results.size() - successCount);
            
        } finally {
            setScanning(false);
            long duration = System.currentTimeMillis() - startTime;
            log.debug("扫描处理耗时: {}ms", duration);
        }
    }
    
    private ProcessingResult processWithMove(String bucket, String key) {
        return objectProcessor.processAndMove(bucket, key);
    }
}
