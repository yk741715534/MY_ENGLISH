// MoveHistory.java
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
@Document(collection = "file_move_history")
public class MoveHistory {
    
    @Id
    private String id;
    
    @Indexed
    private String bucket;
    
    @Indexed
    private String sourceKey;
    
    @Indexed
    private String targetKey;
    
    @Indexed
    private String moveType;  // ARCHIVE, BACKUP, FAILED
    
    private Date movedAt;
    
    private Long fileSize;
    
    private String etag;
    
    private String status;  // SUCCESS, FAILED
    
    private String errorMessage;
    
    @Indexed(expireAfterSeconds = 2592000)  // 30天后自动清理
    private Date createdAt;
}

// ProcessingResult扩展
@Data
@Builder
public class EnhancedProcessingResult extends ProcessingResult {
    
    private String originalKey;
    private String archivedKey;
    private String backupKey;
    
    private Long originalSize;
    private Long archivedSize;
    
    private String moveStatus;
    private String moveMessage;
    
    public static EnhancedProcessingResult successWithMove(
        Object data, String originalKey, String archivedKey) {
        
        return EnhancedProcessingResult.builder()
            .success(true)
            .data(data)
            .originalKey(originalKey)
            .archivedKey(archivedKey)
            .moveStatus("MOVED")
            .moveMessage("文件已成功归档")
            .processedAt(new Date())
            .build();
    }
}
