@Component
@Slf4j
public class FileMoveStrategy {
    
    public enum MovePolicy {
        MOVE_TO_ARCHIVE,      // 移动到归档目录
        MOVE_TO_DATE_FOLDER,  // 按日期归档
        COPY_AND_MOVE,        // 先复制到备份再移动
        DELETE_AFTER_MOVE,    // 移动后删除原文件
        KEEP_ORIGINAL         // 保留原文件
    }
    
    @Data
    @Builder
    public static class MoveConfig {
        private String sourceBucket;
        private String targetBucket;  // 可为空，同bucket不同目录
        private String archivePrefix = "archive/";
        private String failedPrefix = "failed/";
        private String backupPrefix = "backup/";
        private MovePolicy policy = MovePolicy.MOVE_TO_ARCHIVE;
        private boolean compress = false;  // 是否压缩
        private String compressionFormat = "zip";
        private boolean addTimestamp = true;  // 添加时间戳
    }
}
