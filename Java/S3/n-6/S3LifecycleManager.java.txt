// S3LifecycleManager.java
@Component
@Slf4j
public class S3LifecycleManager {
    
    @Autowired
    private S3AsyncClient s3Client;
    
    @Value("${s3.lifecycle.enabled:true}")
    private boolean lifecycleEnabled;
    
    /**
     * 配置S3生命周期规则
     */
    public void configureLifecycleRules(String bucket) {
        if (!lifecycleEnabled) {
            return;
        }
        
        try {
            // 创建生命周期配置
            LifecycleConfiguration lifecycleConfig = LifecycleConfiguration.builder()
                .rules(
                    // 归档目录：30天后转为GLACIER
                    createTransitionRule("archive/", 30, TransitionStorageClass.GLACIER),
                    
                    // 失败目录：7天后删除
                    createExpirationRule("failed/", 7),
                    
                    // 备份目录：90天后删除
                    createExpirationRule("backup/", 90),
                    
                    // 临时目录：1天后删除
                    createExpirationRule("temp/", 1)
                )
                .build();
            
            // 应用配置
            s3Client.putBucketLifecycleConfiguration(
                PutBucketLifecycleConfigurationRequest.builder()
                    .bucket(bucket)
                    .lifecycleConfiguration(lifecycleConfig)
                    .build()
            ).join();
            
            log.info("S3生命周期规则配置成功");
            
        } catch (Exception e) {
            log.error("配置S3生命周期规则失败", e);
        }
    }
    
    private LifecycleRule createTransitionRule(String prefix, int days, 
                                              TransitionStorageClass storageClass) {
        return LifecycleRule.builder()
            .id("transition-" + prefix.replace("/", "-"))
            .filter(LifecycleRuleFilter.builder()
                .prefix(prefix)
                .build())
            .transitions(
                Transition.builder()
                    .days(days)
                    .storageClass(storageClass)
                    .build()
            )
            .status(ExpirationStatus.ENABLED)
            .build();
    }
    
    private LifecycleRule createExpirationRule(String prefix, int days) {
        return LifecycleRule.builder()
            .id("expire-" + prefix.replace("/", "-"))
            .filter(LifecycleRuleFilter.builder()
                .prefix(prefix)
                .build())
            .expiration(LifecycleExpiration.builder()
                .days(days)
                .build())
            .status(ExpirationStatus.ENABLED)
            .build();
    }
    
    /**
     * 清理过期文件
     */
    @Scheduled(cron = "0 0 2 * * ?")  // 每天凌晨2点执行
    public void cleanupExpiredFiles() {
        log.info("开始清理过期文件");
        
        // 清理本地临时文件
        cleanupLocalTempFiles();
        
        // 清理数据库中的过期记录
        cleanupExpiredRecords();
    }
    
    private void cleanupLocalTempFiles() {
        try {
            Path tempDir = Paths.get("/tmp/s3-downloads");
            if (Files.exists(tempDir)) {
                Files.walk(tempDir)
                    .sorted(Comparator.reverseOrder())
                    .map(Path::toFile)
                    .filter(file -> 
                        System.currentTimeMillis() - file.lastModified() > 24 * 60 * 60 * 1000)
                    .forEach(File::delete);
            }
        } catch (Exception e) {
            log.error("清理本地临时文件失败", e);
        }
    }
    
    private void cleanupExpiredRecords() {
        // 清理30天前的处理记录
        Date cutoff = new Date(System.currentTimeMillis() - 30L * 24 * 60 * 60 * 1000);
        
        Query query = new Query(Criteria.where("processEndTime").lt(cutoff));
        mongoTemplate.remove(query, ProcessedFile.class);
    }
}
