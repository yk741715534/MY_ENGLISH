// CompressArchiver.java
@Component
public class CompressArchiver {
    
    /**
     * 压缩后归档
     */
    public String compressAndArchive(String bucket, String key) throws IOException {
        // 1. 下载文件
        File original = downloadFile(bucket, key);
        
        // 2. 压缩
        File compressed = compressFile(original);
        
        // 3. 上传压缩文件
        String archiveKey = generateArchiveKey(key, "zip");
        uploadFile(bucket, archiveKey, compressed);
        
        // 4. 删除原始文件
        deleteS3File(bucket, key);
        
        // 5. 清理本地文件
        cleanupLocalFiles(original, compressed);
        
        return archiveKey;
    }
}
