package com.example.s3scanner.service.processor;

import com.example.s3scanner.model.dto.MoveRequest;
import com.example.s3scanner.model.dto.MoveResult;
import com.example.s3scanner.service.mover.S3FileMover;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import java.util.HashMap;
import java.util.Map;
import java.util.UUID;

@Service
@Slf4j
@RequiredArgsConstructor
public class FileProcessingService {

    private final S3FileMover s3FileMover;

    /**
     * 处理并归档文件
     */
    public MoveResult processAndArchive(String bucket, String key, Map<String, Object> processResult) {
        try {
            // 创建归档请求
            MoveRequest moveRequest = MoveRequest.builder()
                .bucket(bucket)
                .sourceKey(key)
                .moveType(MoveRequest.MoveType.ARCHIVE)
                .targetPrefix("archive/")
                .addTimestamp(true)
                .compress(true)
                .compressionLevel(6)
                .metadata(processResult)
                .addMetadata("processor", "FileProcessingService")
                .addMetadata("processTime", System.currentTimeMillis())
                .addUserMetadata("processed-by", "s3-scanner")
                .addTag("status", "processed")
                .addTag("archive", "true")
                .requestId(UUID.randomUUID().toString())
                .operator("system")
                .build();

            // 执行移动
            MoveResult moveResult = s3FileMover.move(moveRequest);

            if (moveResult.isSuccess()) {
                log.info("文件归档成功: {} -> {}",
                    moveResult.getSourcePath(), moveResult.getTargetPath());

                if (moveResult.isCompressed()) {
                    log.info("压缩比率: {:.2f}%，节省空间: {}",
                        moveResult.getCompressionPercentage(),
                        formatBytes(moveResult.getSpaceSaved()));
                }
            } else {
                log.error("文件归档失败: {}", moveResult.getErrorMessage());
            }

            return moveResult;

        } catch (Exception e) {
            log.error("归档文件失败: {}/{}", bucket, key, e);

            // 尝试将失败文件移动到失败目录
            return s3FileMover.moveFailedFile(bucket, key, e);
        }
    }

    private String formatBytes(long bytes) {
        if (bytes < 1024) {
            return bytes + " B";
        } else if (bytes < 1024 * 1024) {
            return String.format("%.2f KB", bytes / 1024.0);
        } else if (bytes < 1024 * 1024 * 1024) {
            return String.format("%.2f MB", bytes / (1024.0 * 1024.0));
        } else {
            return String.format("%.2f GB", bytes / (1024.0 * 1024.0 * 1024.0));
        }
    }
}
