package com.example.s3scanner.service.scanner;

import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import java.time.Duration;
import java.time.Instant;
import java.util.Optional;

@Service
@Slf4j
public class TimeSafeScanner {

    /**
     * 安全的时间比较方法
     */
    public boolean isScanTimeout(Instant startTime, long maxScanTimeMs) {
        if (startTime == null) {
            return false;  // 没有开始时间，不算超时
        }

        try {
            // 方法1：使用 Duration（推荐）
            Duration duration = Duration.between(startTime, Instant.now());
            return duration.toMillis() > maxScanTimeMs;

        } catch (ArithmeticException e) {
            // 如果时间差太大，Duration.between 会抛异常
            log.warn("时间差计算溢出: {}", e.getMessage());
            return true;  // 认为超时
        }
    }

    /**
     * 安全的增量扫描时间检查
     */
    public boolean shouldStopIncrementalScan(Instant lastScanTime,
                                           long maxScanTimeMs,
                                           String prefix) {
        if (lastScanTime == null || Instant.MIN.equals(lastScanTime)) {
            // 第一次扫描，不应该因为"超时"停止
            return false;
        }

        try {
            Duration sinceLastScan = Duration.between(lastScanTime, Instant.now());
            long elapsedMs = sinceLastScan.toMillis();

            if (elapsedMs > maxScanTimeMs) {
                log.warn("增量扫描超时: {}, 已耗时: {}ms, 限制: {}ms",
                    prefix, elapsedMs, maxScanTimeMs);
                return true;
            }

            return false;

        } catch (ArithmeticException e) {
            log.warn("时间计算溢出，提前结束扫描: {}", prefix, e);
            return true;
        } catch (Exception e) {
            log.error("检查扫描超时失败: {}", prefix, e);
            return false;  // 出错时不停止扫描
        }
    }
}



@Service
@Slf4j
@RequiredArgsConstructor
public class DistributedScanner {

    // ... 其他代码 ...

    /**
     * 增量扫描（修复版）
     */
    @Retryable(
        retryFor = {Exception.class},
        maxAttempts = 3,
        backoff = @Backoff(delay = 1000, multiplier = 2)
    )
    public ScanResult incrementalScan(String prefix, Instant lastScanTime) {
        List<S3ObjectInfo> newObjects = new CopyOnWriteArrayList<>();
        AtomicInteger totalScanned = new AtomicInteger(0);
        String continuationToken = null;

        // 扫描开始时间
        Instant scanStartTime = Instant.now();

        ExecutorService executor = Executors.newVirtualThreadPerTaskExecutor();

        try {
            do {
                // ✅ 修复：使用安全的时间检查
                if (isScanTimeout(scanStartTime, maxScanTimeMs, prefix)) {
                    log.warn("扫描超时，提前结束: {}", prefix);
                    break;
                }

                // ✅ 修复：检查增量扫描是否应该继续
                if (lastScanTime != null && !Instant.MIN.equals(lastScanTime)) {
                    Duration sinceLastScan = Duration.between(lastScanTime, Instant.now());
                    if (sinceLastScan.toMillis() > maxIncrementalScanWindowMs) {
                        log.info("增量扫描窗口过期: {}，距离上次扫描 {}ms",
                            prefix, sinceLastScan.toMillis());
                        break;
                    }
                }

                // 构建请求
                ListObjectsV2Request.Builder requestBuilder = ListObjectsV2Request.builder()
                    .bucket(bucket)
                    .prefix(prefix)
                    .maxKeys(pageSize);

                if (continuationToken != null) {
                    requestBuilder.continuationToken(continuationToken);
                }

                // 异步执行扫描
                CompletableFuture<ListObjectsV2Response> future = s3Client.listObjectsV2(
                    requestBuilder.build()
                );

                ListObjectsV2Response response = future.get(10, TimeUnit.SECONDS);

                // 处理本页结果
                List<CompletableFuture<Void>> processingFutures = response.contents().stream()
                    .filter(this::shouldProcess)
                    .map(s3Object -> CompletableFuture.runAsync(() -> {
                        processObject(s3Object, prefix, newObjects, lastScanTime);
                    }, executor))
                    .collect(Collectors.toList());

                // 等待本页处理完成
                CompletableFuture.allOf(processingFutures.toArray(new CompletableFuture[0]))
                    .get(5, TimeUnit.SECONDS);

                totalScanned.addAndGet(response.contents().size());
                continuationToken = response.nextContinuationToken();

                // 如果发现大量新文件，提前分批处理
                if (newObjects.size() >= 50) {
                    log.info("发现足够的新文件，提前结束扫描: {} 个", newObjects.size());
                    break;
                }

            } while (continuationToken != null);

            return ScanResult.builder()
                .prefix(prefix)
                .scannedObjects(totalScanned.get())
                .newObjects(new ArrayList<>(newObjects))
                .success(true)
                .incremental(true)
                .lastScanTime(lastScanTime)
                .scanTime(Instant.now())
                .build();

        } catch (Exception e) {
            log.error("增量扫描失败: {}", prefix, e);
            return ScanResult.builder()
                .prefix(prefix)
                .scannedObjects(totalScanned.get())
                .newObjects(new ArrayList<>(newObjects))
                .success(false)
                .errorMessage(e.getMessage())
                .build();
        } finally {
            executor.shutdown();
        }
    }

    /**
     * ✅ 修复：安全检查扫描超时
     */
    private boolean isScanTimeout(Instant startTime, long maxTimeMs, String prefix) {
        if (startTime == null) {
            return false;
        }

        try {
            Duration duration = Duration.between(startTime, Instant.now());
            long elapsedMs = duration.toMillis();

            if (elapsedMs > maxTimeMs) {
                log.debug("扫描超时检查: {} 已耗时 {}ms > 限制 {}ms",
                    prefix, elapsedMs, maxTimeMs);
                return true;
            }

            return false;

        } catch (ArithmeticException e) {
            // 处理溢出
            log.warn("扫描时间计算溢出: {}，认为超时", prefix, e);
            return true;
        } catch (Exception e) {
            log.error("检查扫描超时失败: {}", prefix, e);
            return false;  // 出错时不停止
        }
    }

    /**
     * ✅ 修复：安全的最后扫描时间处理
     */
    public CompletableFuture<ScanResult> scanPrefixDistributed(String prefix) {
        String lockKey = String.format("scan:%s:%s", bucket, prefix);
        Instant startTime = Instant.now();

        // 获取扫描锁
        if (!lockService.acquireLock(lockKey, "scan", scanIntervalMs + 1000)) {
            log.debug("无法获取扫描锁，跳过本次扫描: {}", prefix);
            return CompletableFuture.completedFuture(
                ScanResult.skipped(prefix, "Lock acquisition failed")
            );
        }

        try {
            // ✅ 修复：安全的最后扫描时间获取
            Instant lastScan = getLastScanTimeSafely(prefix);

            ScanResult result = incrementalScan(prefix, lastScan);

            // 更新最后扫描时间
            updateLastScanTime(prefix, Instant.now());

            // 记录指标
            long duration = Duration.between(startTime, Instant.now()).toMillis();
            metricsCollector.recordScan(prefix, result.getNewObjects().size(),
                result.getScannedObjects(), duration, result.isSuccess());

            log.debug("扫描完成: {}, 发现 {} 个新文件, 耗时: {}ms",
                prefix, result.getNewObjects().size(), duration);

            return CompletableFuture.completedFuture(result);

        } finally {
            lockService.releaseLock(lockKey, "scan");
        }
    }

    /**
     * ✅ 修复：安全获取最后扫描时间
     */
    private Instant getLastScanTimeSafely(String prefix) {
        Instant lastScan = lastScanTime.get(prefix);

        if (lastScan == null) {
            // 第一次扫描，使用合理的初始值
            return Instant.now().minusSeconds(scanIntervalMs / 1000);
        }

        // 防止使用 Instant.MIN
        if (Instant.MIN.equals(lastScan)) {
            return Instant.now().minusSeconds(scanIntervalMs / 1000);
        }

        return lastScan;
    }

    /**
     * ✅ 修复：安全更新最后扫描时间
     */
    private void updateLastScanTime(String prefix, Instant scanTime) {
        try {
            // 确保不是未来的时间
            Instant now = Instant.now();
            if (scanTime.isAfter(now)) {
                log.warn("扫描时间在将来，使用当前时间: {}", prefix);
                scanTime = now;
            }

            lastScanTime.put(prefix, scanTime);

        } catch (Exception e) {
            log.error("更新最后扫描时间失败: {}", prefix, e);
        }
    }

}




package com.example.s3scanner.util;

import lombok.experimental.UtilityClass;
import lombok.extern.slf4j.Slf4j;
import java.time.Duration;
import java.time.Instant;
import java.time.LocalDateTime;
import java.time.ZoneId;
import java.time.temporal.ChronoUnit;
import java.util.Date;
import java.util.concurrent.TimeUnit;

/**
 * 安全的时间计算工具类
 * 防止溢出和异常
 */
@UtilityClass
@Slf4j
public class TimeSafeUtils {

    /**
     * 安全计算时间差（毫秒）
     * 不会溢出，返回可能的最大值
     */
    public static long safeTimeDifferenceMs(Instant start, Instant end) {
        if (start == null || end == null) {
            return 0L;
        }

        try {
            // 确保 end 不早于 start
            if (end.isBefore(start)) {
                log.warn("结束时间早于开始时间，交换计算: start={}, end={}", start, end);
                Instant temp = start;
                start = end;
                end = temp;
            }

            // 使用 Duration 计算
            Duration duration = Duration.between(start, end);

            // 检查是否溢出
            if (duration.getSeconds() > Long.MAX_VALUE / 1000 - 1) {
                log.warn("时间差过大，可能溢出: {} 到 {}", start, end);
                return Long.MAX_VALUE;
            }

            return duration.toMillis();

        } catch (ArithmeticException e) {
            log.warn("时间差计算溢出: {} 到 {}", start, end, e);
            return Long.MAX_VALUE;
        } catch (Exception e) {
            log.error("计算时间差失败", e);
            return 0L;
        }
    }

    /**
     * 安全检查是否超时
     */
    public static boolean isTimeout(Instant startTime, long timeoutMs) {
        if (startTime == null || timeoutMs <= 0) {
            return false;
        }

        try {
            Instant now = Instant.now();

            // 处理未来的开始时间
            if (startTime.isAfter(now)) {
                log.warn("开始时间在未来: {}", startTime);
                return false;
            }

            Duration elapsed = Duration.between(startTime, now);

            // 检查是否可能溢出
            if (timeoutMs > 0 && elapsed.getSeconds() > Long.MAX_VALUE / 1000 - timeoutMs / 1000) {
                log.warn("超时检查可能溢出: elapsed={}s, timeout={}ms",
                    elapsed.getSeconds(), timeoutMs);
                return true;  // 认为超时
            }

            return elapsed.toMillis() > timeoutMs;

        } catch (ArithmeticException e) {
            log.warn("超时检查溢出", e);
            return true;  // 溢出时认为超时
        } catch (Exception e) {
            log.error("检查超时失败", e);
            return false;
        }
    }

    /**
     * 安全的未来时间计算
     */
    public static Instant safePlusMillis(Instant time, long millisToAdd) {
        if (time == null) {
            time = Instant.now();
        }

        try {
            if (millisToAdd > 0) {
                // 检查加法是否溢出
                if (millisToAdd > Long.MAX_VALUE - time.toEpochMilli()) {
                    log.warn("时间加法可能溢出: {} + {}ms", time, millisToAdd);
                    return Instant.MAX;
                }

                return time.plusMillis(millisToAdd);
            } else if (millisToAdd < 0) {
                // 检查减法是否下溢
                if (-millisToAdd > time.toEpochMilli() - Long.MIN_VALUE) {
                    log.warn("时间减法可能下溢: {} - {}ms", time, -millisToAdd);
                    return Instant.MIN;
                }

                return time.minusMillis(-millisToAdd);
            }

            return time;

        } catch (ArithmeticException e) {
            log.warn("时间计算溢出: {} + {}ms", time, millisToAdd, e);
            return millisToAdd > 0 ? Instant.MAX : Instant.MIN;
        }
    }

    /**
     * 安全的时间转换
     */
    public static Instant safeToInstant(Date date) {
        if (date == null) {
            return null;
        }

        try {
            return date.toInstant();
        } catch (Exception e) {
            log.warn("Date转换Instant失败", e);

            // 尝试其他方式
            try {
                return Instant.ofEpochMilli(date.getTime());
            } catch (Exception e2) {
                log.error("完全无法转换Date到Instant", e2);
                return Instant.now();  // 返回当前时间作为回退
            }
        }
    }

    /**
     * 安全的日期时间转换
     */
    public static LocalDateTime safeToLocalDateTime(Date date) {
        if (date == null) {
            return null;
        }

        try {
            return LocalDateTime.ofInstant(date.toInstant(), ZoneId.systemDefault());
        } catch (Exception e) {
            log.warn("转换Date到LocalDateTime失败", e);

            // 回退方案
            try {
                return LocalDateTime.now();
            } catch (Exception e2) {
                return LocalDateTime.of(1970, 1, 1, 0, 0);
            }
        }
    }

    /**
     * 获取合理的最早时间
     */
    public static Instant getReasonableEarliestTime() {
        // 不要使用 Instant.MIN，使用一个合理的最早时间
        // 比如：1年前
        return Instant.now().minus(365, ChronoUnit.DAYS);
    }

    /**
     * 获取合理的最后扫描时间
     */
    public static Instant getReasonableLastScanTime() {
        // 默认：5分钟前
        return Instant.now().minus(5, ChronoUnit.MINUTES);
    }

    /**
     * 格式化持续时间
     */
    public static String formatDuration(long millis) {
        if (millis < 0) {
            return "N/A";
        }

        if (millis < 1000) {
            return millis + "ms";
        }

        long seconds = TimeUnit.MILLISECONDS.toSeconds(millis);
        if (seconds < 60) {
            return String.format("%.1fs", millis / 1000.0);
        }

        long minutes = TimeUnit.MILLISECONDS.toMinutes(millis);
        if (minutes < 60) {
            long remainingSeconds = seconds - minutes * 60;
            return String.format("%dm %ds", minutes, remainingSeconds);
        }

        long hours = TimeUnit.MILLISECONDS.toHours(millis);
        long remainingMinutes = minutes - hours * 60;
        return String.format("%dh %dm", hours, remainingMinutes);
    }
