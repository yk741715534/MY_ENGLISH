@Component
public class AdaptiveTimeoutManager {

    private final Map<String, ScanHistory> scanHistories = new ConcurrentHashMap<>();

    /**
     * 根据历史调整超时时间
     */
    public long calculateAdaptiveTimeout(String prefix) {
        ScanHistory history = scanHistories.computeIfAbsent(
            prefix, k -> new ScanHistory()
        );

        // 基础配置
        long baseTimeout = 2500;  // 2.5秒

        // 根据历史性能调整
        if (history.getAverageDuration() > 2000) {
            // 历史扫描较慢，增加超时
            return baseTimeout + 1000;
        } else if (history.getAverageDuration() < 500) {
            // 历史扫描很快，减少超时
            return Math.max(1000, baseTimeout - 500);
        }

        return baseTimeout;
    }

    /**
     * 动态调整当前扫描的超时
     */
    public boolean shouldContinueScan(String prefix, Instant startTime,
                                     int scannedSoFar) {
        long elapsed = Duration.between(startTime, Instant.now()).toMillis();
        long adaptiveTimeout = calculateAdaptiveTimeout(prefix);

        // 已扫描数量很少但时间很长 -> 可能有问题
        if (scannedSoFar < 10 && elapsed > 1000) {
            log.warn("扫描性能异常: {}，{}ms 只扫描了 {} 个文件",
                prefix, elapsed, scannedSoFar);
            return false;
        }

        // 检查自适应超时
        return elapsed <= adaptiveTimeout;
    }
}

    }

6.1 分级超时处理

@Service
public class TimeoutHandler {

    /**
     * 分级超时处理策略
     */
    public void handleScanTimeout(String prefix, long elapsedMs,
                                 long maxScanTimeMs, ScanContext context) {

        double percentage = (double) elapsedMs / maxScanTimeMs * 100;

        if (percentage <= 80) {
            // 正常范围内
            return;
        }

        if (percentage <= 100) {
            // 接近超时，记录警告
            log.warn("扫描接近超时: {}，已用 {}ms ({}%)，限制 {}ms",
                prefix, elapsedMs, String.format("%.1f", percentage), maxScanTimeMs);
            return;
        }

        if (percentage <= 120) {
            // 轻微超时
            log.warn("扫描轻微超时: {}，超时 {}ms",
                prefix, elapsedMs - maxScanTimeMs);
            metricsCollector.recordMinorTimeout(prefix);
            return;
        }

        if (percentage <= 200) {
            // 中度超时
            log.error("扫描中度超时: {}，超时 {}ms",
                prefix, elapsedMs - maxScanTimeMs);
            metricsCollector.recordMajorTimeout(prefix);

            // 降低该前缀的扫描优先级
            adjustScanPriority(prefix, -1);
            return;
        }

        // 严重超时
        log.error("扫描严重超时: {}，超时 {}ms，立即中止",
            prefix, elapsedMs - maxScanTimeMs);
        metricsCollector.recordCriticalTimeout(prefix);

        // 1. 记录异常
        recordTimeoutException(prefix, elapsedMs, context);

        // 2. 发送告警
        sendTimeoutAlert(prefix, elapsedMs, maxScanTimeMs);

        // 3. 暂停该前缀的扫描
        suspendPrefixScan(prefix, 300000);  // 暂停5分钟

        // 4. 尝试恢复策略
        executeRecoveryStrategy(prefix);
    }
}

@Service
public class ScanRecoveryManager {

    private final Map<String, TimeoutRecoveryContext> recoveryContexts =
        new ConcurrentHashMap<>();

    /**
     * 执行超时恢复策略
     */
    public void executeRecoveryStrategy(String prefix) {
        TimeoutRecoveryContext context = recoveryContexts.computeIfAbsent(
            prefix, k -> new TimeoutRecoveryContext()
        );

        context.incrementTimeoutCount();

        // 根据超时次数采取不同策略
        switch (context.getTimeoutCount()) {
            case 1:
                // 第一次超时：增加扫描间隔
                adjustScanInterval(prefix, 1.5);
                log.info("第一次超时，增加 {} 的扫描间隔", prefix);
                break;

            case 2:
                // 第二次超时：减少页大小
                adjustPageSize(prefix, 0.5);
                log.info("第二次超时，减少 {} 的页大小", prefix);
                break;

            case 3:
                // 第三次超时：切换到增量扫描模式
                switchToIncrementalMode(prefix);
                log.warn("第三次超时，切换到增量扫描模式: {}", prefix);
                break;

            default:
                // 多次超时：暂停扫描
                suspendScanning(prefix, calculateSuspendTime(context));
                log.error("多次超时，暂停扫描 {} 分钟: {}",
                    calculateSuspendTime(context) / 60000, prefix);
        }
    }

    /**
     * 成功的扫描后重置恢复状态
     */
    public void recordSuccessfulScan(String prefix) {
        TimeoutRecoveryContext context = recoveryContexts.get(prefix);
        if (context != null) {
            // 连续成功扫描，逐渐恢复正常参数
            if (context.getConsecutiveSuccesses() >= 3) {
                resetToDefaultParameters(prefix);
                recoveryContexts.remove(prefix);
                log.info("恢复成功，重置 {} 的扫描参数", prefix);
            } else {
                context.incrementConsecutiveSuccesses();
            }
        }
    }
}

10. 常见问题解答
Q: 为什么设置为2.5秒而不是3秒？

A: 因为扫描间隔是3秒，需要留出0.5秒的处理和容错时间。

Q: 超时后应该怎么办？

A: 记录日志、收集指标、可能调整扫描策略，但不要无限重试。

Q: 如何确定最佳值？

A: 通过监控和分析：平均扫描时间、超时率、系统负载等。

Q: 可以动态调整吗？

A: 可以，建议实现自适应超时机制。

@Component
@Slf4j
public class ScanTimeCoordinator {

    @Value("${app.scanner.interval-ms:3000}")
    private long scanIntervalMs;

    @Value("${app.scanner.max-scan-time-ms:2500}")
    private long maxScanTimeMs;

    @Value("${app.scanner.min-idle-time-ms:200}")
    private long minIdleTimeMs;

    private final Map<String, ScanSchedule> schedules = new ConcurrentHashMap<>();

    @Data
    @Builder
    public static class ScanSchedule {
        private String prefix;
        private Instant lastScanStart;
        private Instant lastScanEnd;
        private long averageDuration;
        private int consecutiveSlowScans;
        private boolean suspended;
        private Instant suspendUntil;

        public boolean isEligibleForScan() {
            if (suspended) {
                return suspendUntil != null && Instant.now().isAfter(suspendUntil);
            }
            return true;
        }
    }

    @PostConstruct
    public void validateConfiguration() {
        // 配置验证
        if (maxScanTimeMs >= scanIntervalMs) {
            throw new IllegalStateException(String.format(
                "配置错误: maxScanTimeMs(%d) >= scanIntervalMs(%d)",
                maxScanTimeMs, scanIntervalMs
            ));
        }

        long idleTime = scanIntervalMs - maxScanTimeMs;
        if (idleTime < minIdleTimeMs) {
            log.warn("空闲时间不足: {}ms < 建议的 {}ms", idleTime, minIdleTimeMs);
        }

        log.info("扫描时间配置: 间隔={}ms, 最大扫描={}ms, 空闲={}ms",
            scanIntervalMs, maxScanTimeMs, idleTime);
    }

    /**
     * 计算下一次扫描时间
     */
    public boolean shouldScanNow(String prefix) {
        ScanSchedule schedule = schedules.get(prefix);

        if (schedule == null || schedule.getLastScanEnd() == null) {
            return true;  // 第一次扫描
        }

        // 计算距离上次扫描结束的时间
        Duration sinceLastEnd = Duration.between(schedule.getLastScanEnd(), Instant.now());
        long elapsedMs = sinceLastEnd.toMillis();

        // 必须满足：elapsedMs >= scanIntervalMs
        return elapsedMs >= scanIntervalMs;
    }

    /**
     * 计算剩余扫描时间
     */
    public long calculateRemainingScanTime(String prefix) {
        ScanSchedule schedule = schedules.get(prefix);

        if (schedule == null || schedule.getLastScanStart() == null) {
            return maxScanTimeMs;
        }

        Duration elapsed = Duration.between(schedule.getLastScanStart(), Instant.now());
        long elapsedMs = elapsed.toMillis();

        // 剩余时间 = 最大扫描时间 - 已用时间
        long remaining = maxScanTimeMs - elapsedMs;

        // 确保不为负
        return Math.max(0, remaining);
    }

    /**
     * 计算安全提前停止时间
     */
    public long calculateSafeStopTime(String prefix) {
        // 在超时前提前停止，留出处理时间
        long bufferTime = 200;  // 200ms缓冲区
        return maxScanTimeMs - bufferTime;
    }

    /**
     * 记录扫描开始
     */
    public void recordScanStart(String prefix) {
        ScanSchedule schedule = schedules.computeIfAbsent(
            prefix, k -> ScanSchedule.builder().prefix(prefix).build()
        );

        schedule.setLastScanStart(Instant.now());
        schedule.setSuspended(false);
        schedule.setSuspendUntil(null);
    }

    /**
     * 记录扫描结束
     */
    public void recordScanEnd(String prefix, long actualDuration) {
        ScanSchedule schedule = schedules.get(prefix);
        if (schedule == null) {
            return;
        }

        schedule.setLastScanEnd(Instant.now());

        // 更新平均耗时
        updateAverageDuration(schedule, actualDuration);

        // 检查性能
        checkScanPerformance(schedule, actualDuration);

        // 计算到下次扫描的时间
        calculateNextScanTime(schedule);
    }

    private void updateAverageDuration(ScanSchedule schedule, long actualDuration) {
        long currentAvg = schedule.getAverageDuration();

        if (currentAvg == 0) {
            schedule.setAverageDuration(actualDuration);
        } else {
            // 指数移动平均
            schedule.setAverageDuration((currentAvg * 7 + actualDuration * 3) / 10);
        }
    }

    private void checkScanPerformance(ScanSchedule schedule, long actualDuration) {
        double utilization = (double) actualDuration / maxScanTimeMs * 100;

        if (utilization > 90) {
            schedule.setConsecutiveSlowScans(schedule.getConsecutiveSlowScans() + 1);

            if (schedule.getConsecutiveSlowScans() >= 3) {

                    schedule.getPrefix(), schedule.getConsecutiveSlowScans());

            }
        } else {
            schedule.setConsecutiveSlowScans(0);

    }

@Component
@Slf4j
public class AdaptiveScheduler {

    private static final long MIN_SCAN_INTERVAL = 1000;  // 最小1秒
    private static final long MAX_SCAN_INTERVAL = 60000; // 最大60秒

    private static final long MIN_SCAN_TIME = 500;      // 最小0.5秒
    private static final long MAX_SCAN_TIME = 30000;    // 最大30秒

    /**
     * 动态计算最佳时间配置
     */
    public ScanTimeConfig calculateOptimalConfig(String prefix,
                                                SystemLoad load,
                                                ScanHistory history) {

        // 基础配置
        long baseInterval = 3000;  // 3秒
        long baseMaxTime = 2500;   // 2.5秒

        // 根据系统负载调整
        double loadFactor = calculateLoadFactor(load);

        // 根据历史性能调整
        double perfFactor = calculatePerformanceFactor(history);

        // 计算最终值
        long optimalInterval = (long) (baseInterval * loadFactor * perfFactor);
        long optimalMaxTime = (long) (baseMaxTime * perfFactor);

        // 确保关系：maxTime < interval
        optimalMaxTime = Math.min(optimalMaxTime, optimalInterval - 500);

        // 应用边界限制
        optimalInterval = clamp(optimalInterval, MIN_SCAN_INTERVAL, MAX_SCAN_INTERVAL);
        optimalMaxTime = clamp(optimalMaxTime, MIN_SCAN_TIME, MAX_SCAN_TIME);

        // 最终验证
        if (optimalMaxTime >= optimalInterval) {
            optimalMaxTime = optimalInterval - 500;  // 强制留出500ms
        }

        return ScanTimeConfig.builder()
            .scanIntervalMs(optimalInterval)
            .maxScanTimeMs(optimalMaxTime)
            .idleTimeMs(optimalInterval - optimalMaxTime)
            .utilizationRate((double) optimalMaxTime / optimalInterval)
            .build();
    }

    private double calculateLoadFactor(SystemLoad load) {
        double cpuUsage = load.getCpuUsage();
        double memoryUsage = load.getMemoryUsage();
        int threadCount = load.getActiveThreads();

        // 高负载时增加间隔
        if (cpuUsage > 0.8 || memoryUsage > 0.8 || threadCount > 50) {
            return 2.0;  // 间隔加倍
        }

        // 低负载时减少间隔
        if (cpuUsage < 0.3 && memoryUsage < 0.3 && threadCount < 10) {
            return 0.7;  // 间隔减少
        }

        return 1.0;  // 保持
    }

    private double calculatePerformanceFactor(ScanHistory history) {
        double timeoutRate = history.getTimeoutRate();
        long avgDuration = history.getAverageDuration();

        // 高超时率时增加间隔
        if (timeoutRate > 0.2) {  // 20%超时
            return 1.5;
        }

        // 长时间扫描时增加最大时间
        if (avgDuration > 2000) {  // 平均超过2秒
            return 1.2;
        }

        return 1.0;
    }
}

@Service
@Slf4j
@RequiredArgsConstructor
public class SmartScannerScheduler {

    private final ScanTimeCoordinator timeCoordinator;
    private final AdaptiveScheduler adaptiveScheduler;
    private final DistributedScanner scanner;

    private final ScheduledExecutorService scheduler =
        Executors.newScheduledThreadPool(1, r -> {
            Thread t = new Thread(r, "smart-scanner-scheduler");
            t.setDaemon(true);
            return t;
        });

    private final Map<String, ScanJob> activeJobs = new ConcurrentHashMap<>();

    @Data
    @Builder
    public static class ScanJob {
        private String prefix;
        private ScheduledFuture<?> future;
        private ScanTimeConfig config;
        private Instant lastExecution;
        private Instant nextExecution;
        private int consecutiveFailures;
    }

    @PostConstruct
    public void init() {
        // 启动所有扫描作业
        List<String> prefixes = getScanPrefixes();

        for (String prefix : prefixes) {
            schedulePrefixScan(prefix);
        }

        log.info("扫描调度器初始化完成，共 {} 个前缀", prefixes.size());
    }

    /**
     * 调度前缀扫描
     */
    public void schedulePrefixScan(String prefix) {
        // 计算初始配置
        ScanTimeConfig config = adaptiveScheduler.calculateOptimalConfig(
            prefix, getSystemLoad(), getScanHistory(prefix)
        );

        ScanJob job = ScanJob.builder()
            .prefix(prefix)
            .config(config)
            .build();

        // 计算初始延迟
        long initialDelay = calculateInitialDelay(prefix);

        // 创建调度任务
        ScheduledFuture<?> future = scheduler.scheduleAtFixedRate(
            () -> executeScanJob(job),
            initialDelay,
            config.getScanIntervalMs(),
            TimeUnit.MILLISECONDS
        );

        job.setFuture(future);
        activeJobs.put(prefix, job);

        log.info("调度扫描任务: {}, 间隔={}ms, 最大时间={}ms",
            prefix, config.getScanIntervalMs(), config.getMaxScanTimeMs());
    }

    /**
     * 执行扫描任务
     */
    private void executeScanJob(ScanJob job) {
        String prefix = job.getPrefix();

        try {
            job.setLastExecution(Instant.now());

            // 检查是否应该扫描
            if (!timeCoordinator.shouldScanNow(prefix)) {
                log.debug("跳过扫描: {}，未到间隔时间", prefix);
                return;
            }

            // 记录扫描开始
            timeCoordinator.recordScanStart(prefix);

            // 计算本次扫描的超时时间
            long timeout = job.getConfig().getMaxScanTimeMs();

            // 执行扫描（带超时控制）
            CompletableFuture<ScanResult> scanFuture = scanner.scanPrefixDistributed(prefix);

            ScanResult result = scanFuture.get(timeout, TimeUnit.MILLISECONDS);

            // 记录扫描结束
            timeCoordinator.recordScanEnd(prefix, result.getScanDurationMs());

            // 处理结果
            handleScanResult(prefix, result);

            // 成功后重置失败计数
            job.setConsecutiveFailures(0);

            // 动态调整配置
            adaptConfiguration(job, result);

        } catch (TimeoutException e) {
            // 扫描超时
            handleScanTimeout(job, e);

        } catch (Exception e) {
            // 其他错误
            handleScanError(job, e);

        } finally {
            // 计算下次执行时间
            job.setNextExecution(Instant.now().plusMillis(
                job.getConfig().getScanIntervalMs()
            ));
        }
    }

    /**
     * 处理扫描超时
     */
    private void handleScanTimeout(ScanJob job, TimeoutException e) {
        String prefix = job.getPrefix();

        job.setConsecutiveFailures(job.getConsecutiveFailures() + 1);

        log.error("扫描超时: {}, 连续失败 {} 次",
            prefix, job.getConsecutiveFailures());

        // 记录指标
        recordTimeoutMetric(prefix);

        // 根据连续失败次数采取行动
        if (job.getConsecutiveFailures() >= 3) {
            // 连续3次超时，暂停扫描
            pauseScanJob(job, 300000);  // 暂停5分钟

            // 发送告警
            sendAlert("ScanTimeout",
                String.format("前缀 %s 连续 %d 次扫描超时，已暂停",
                    prefix, job.getConsecutiveFailures()));
        } else if (job.getConsecutiveFailures() >= 2) {
            // 连续2次超时，增加间隔
            increaseScanInterval(job, 1.5);
        }
    }

    /**
     * 动态调整配置
     */
    private void adaptConfiguration(ScanJob job, ScanResult result) {
        String prefix = job.getPrefix();
        ScanTimeConfig currentConfig = job.getConfig();

        long actualDuration = result.getScanDurationMs();
        long maxTime = currentConfig.getMaxScanTimeMs();

        double utilization = (double) actualDuration / maxTime;

        // 如果利用率持续高于阈值，调整配置
        if (utilization > 0.8) {
            // 接近超时，需要调整
            if (actualDuration > maxTime * 0.9) {
                // 非常接近超时，增加最大时间
                long newMaxTime = (long) (maxTime * 1.2);
                long newInterval = currentConfig.getScanIntervalMs();

                // 确保新间隔足够大
                if (newMaxTime >= newInterval - 200) {
                    newInterval = (long) (newMaxTime * 1.5);
                }

                updateJobConfig(job, newInterval, newMaxTime);

            } else if (actualDuration < maxTime * 0.5) {
                // 有很多空闲时间，可以减少间隔
                long newInterval = Math.max(
                    MIN_SCAN_INTERVAL,
                    (long) (currentConfig.getScanIntervalMs() * 0.8)
                );

                updateJobConfig(job, newInterval, maxTime);
            }
        }
    }

    private void updateJobConfig(ScanJob job, long newInterval, long newMaxTime) {
        // 取消旧的调度
        job.getFuture().cancel(false);

        // 创建新的调度
        ScanTimeConfig newConfig = ScanTimeConfig.builder()
            .scanIntervalMs(newInterval)
            .maxScanTimeMs(newMaxTime)
            .build();

        job.setConfig(newConfig);

        ScheduledFuture<?> newFuture = scheduler.scheduleAtFixedRate(
            () -> executeScanJob(job),
            0,
            newInterval,
            TimeUnit.MILLISECONDS
        );

        job.setFuture(newFuture);

        log.info("更新扫描配置: {}, 间隔={}ms, 最大时间={}ms",
            job.getPrefix(), newInterval, newMaxTime);
    }
}

    private void checkTimeRelations(String prefix, ScanMetrics metric) {

        long maxScanTime = metric.getMaxScanTimeMs();
        long avgDuration = metric.getAverageDuration();

        // 1. 检查基本关系
        if (maxScanTime >= scanInterval) {
            sendCriticalAlert("TimeRelationViolation",
                String.format("前缀 %s: maxScanTime(%d) >= scanInterval(%d)",
                    prefix, maxScanTime, scanInterval));
        }

        // 2. 检查空闲时间
        long idleTime = scanInterval - avgDuration;
        long minRequiredIdle = 200;  // 至少200ms空闲

        if (idleTime < minRequiredIdle) {
            sendWarning("LowIdleTime",
                String.format("前缀 %s: 空闲时间仅 %dms < 要求 %dms",
                    prefix, idleTime, minRequiredIdle));
        }

        // 3. 检查利用率
        double utilization = (double) avgDuration / scanInterval * 100;

        if (utilization > 80) {
            sendWarning("HighUtilization",
                String.format("前缀 %s: 时间利用率 %.1f%%", prefix, utilization));
        }

        // 4. 检查缓冲
        double buffer = (double) (maxScanTime - avgDuration) / maxScanTime * 100;

        if (buffer < 10) {  // 缓冲小于10%
            sendWarning("LowBuffer",
                String.format("前缀 %s: 时间缓冲仅 %.1f%%", prefix, buffer));
        }
    }

public class AutoConfigGenerator {

        long scanInterval = Math.max(desiredFrequency, (long)(baseScanTime * 1.5));

        long maxScanTime = (long)(baseScanTime * 1.2);

            scanInterval = (long)(maxScanTime * 1.3);
        }

        return ScanConfig.builder()

            .maxScanTimeMs(maxScanTime)
            .idleTimeMs(scanInterval - maxScanTime)
            .utilization((double) maxScanTime / scanInterval)
            .build();

    private long calculateBaseScanTime(int fileCount, double networkLatency) {

        // 每页时间 = 网络延迟 + 处理时间
        long perPageTime = (long) (networkLatency + 50);  // 50ms处理时间

        return pages * perPageTime;

O = 系统开销 (调度、GC等)

稳定性条件:
1. M < T
2. A + O + B < T
3. B = T - (A + O) > 最小缓冲

推荐值:
O ≈ 100-200ms
B_min ≈ 200ms
所以: M ≈ T - 300ms

空闲率 = (T - M) / T
安全系数 = (M - A) / M

目标:

安全系数: >20%
