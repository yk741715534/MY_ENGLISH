package com.example.s3scanner.service.coordinator;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.domain.Sort;
import org.springframework.data.mongodb.core.FindAndModifyOptions;
import org.springframework.data.mongodb.core.MongoTemplate;
import org.springframework.data.mongodb.core.query.Criteria;
import org.springframework.data.mongodb.core.query.Query;
import org.springframework.data.mongodb.core.query.Update;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import javax.annotation.PostConstruct;
import java.net.InetAddress;
import java.time.Duration;
import java.time.Instant;
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicLong;
import java.util.stream.Collectors;

@Service
@Slf4j
@RequiredArgsConstructor
public class EnhancedInstanceRegistry {

    private final MongoTemplate mongoTemplate;

    // 集合名称
    private static final String INSTANCES_COLLECTION = "distributed_instances";
    private static final String INSTANCE_HISTORY_COLLECTION = "instance_history";

    // 配置
    private static final long HEARTBEAT_INTERVAL_MS = 30000;    // 30秒
    private static final long INSTANCE_TTL_MS = 90000;          // 90秒过期
    private static final long CLEANUP_THRESHOLD_MS = 180000;    // 180秒清理阈值

    // 当前实例信息
    private String instanceId;
    private String hostname;
    private String ipAddress;

    // 本地缓存
    private final Map<String, ClusterInstance> activeInstances = new ConcurrentHashMap<>();
    private final Map<String, ClusterInstance> allInstances = new ConcurrentHashMap<>();

    // 统计
    private final AtomicInteger currentLoad = new AtomicInteger(0);
    private final AtomicLong heartbeatCount = new AtomicLong(0);
    private int concurrencyLimit = 20;

    @PostConstruct
    public void init() {
        // 初始化实例信息
        this.instanceId = resolveInstanceId();
        this.hostname = getHostname();
        this.ipAddress = getIpAddress();

        log.info("实例注册中心初始化: ID={}, Host={}, IP={}",
            instanceId, hostname, ipAddress);

        // 1. 立即注册/更新心跳
        sendHeartbeatWithUpsert();

        // 2. 发现其他实例
        discoverAndRefreshInstances();

        // 3. 启动定时任务
        startScheduledTasks();

        log.info("实例注册中心初始化完成，当前活跃实例数: {}", activeInstances.size());
    }

    // ========== 1. 心跳更新（使用upsert） ==========

    /**
     * 使用upsert发送心跳
     */
    @Transactional
    public void sendHeartbeatWithUpsert() {
        try {
            Instant now = Instant.now();
            Instant expiresAt = now.plusMillis(INSTANCE_TTL_MS);

            // 构建查询条件
            Query query = new Query(Criteria.where("instanceId").is(instanceId));

            // 构建更新操作
            Update update = new Update();

            // 如果是新实例，设置插入字段
            boolean isNew = !existsInDatabase(instanceId);
            if (isNew) {
                update.setOnInsert("_id", UUID.randomUUID().toString());
                update.setOnInsert("instanceId", instanceId);
                update.setOnInsert("hostname", hostname);
                update.setOnInsert("ipAddress", ipAddress);
                update.setOnInsert("registeredAt", Date.from(now));
                update.setOnInsert("startedAt", Date.from(now));
                update.setOnInsert("version", 1);
                update.setOnInsert("restartCount", 0);
                update.setOnInsert("totalHeartbeatCount", 0L);
                update.setOnInsert("totalUptime", 0L);
                update.setOnInsert("capabilities", buildCapabilities());
                update.setOnInsert("metadata", buildMetadata());
                update.setOnInsert("zone", getZone());
                update.setOnInsert("region", getRegion());
                update.setOnInsert("podName", getPodName());
                update.setOnInsert("containerId", getContainerId());

                log.info("注册新实例: {}", instanceId);
            } else {
                // 如果是重启，增加重启计数
                ClusterInstance existing = getInstanceFromDatabase(instanceId);
                if (existing != null && isRecentRestart(existing)) {
                    update.inc("restartCount", 1);
                    update.set("lastRestartAt", Date.from(now));
                    log.info("实例重启: {}", instanceId);
                }
            }

            // 通用更新字段
            update.set("lastHeartbeat", Date.from(now));
            update.set("expiresAt", Date.from(expiresAt));
            update.set("status", "ACTIVE");
            update.set("currentLoad", currentLoad.get());
            update.set("concurrencyLimit", concurrencyLimit);
            update.set("hostname", hostname);
            update.set("ipAddress", ipAddress);
            update.inc("version", 1);
            update.inc("totalHeartbeatCount", 1);
            update.inc("heartbeatCount", 1);

            // 更新总运行时间
            update.set("totalUptime", calculateTotalUptime());

            // 执行upsert
            FindAndModifyOptions options = FindAndModifyOptions.options()
                .upsert(true)      // 关键：不存在时插入
                .returnNew(true);  // 返回新文档

            ClusterInstance updated = mongoTemplate.findAndModify(
                query, update, options, ClusterInstance.class, INSTANCES_COLLECTION
            );

            if (updated != null) {
                // 更新本地缓存
                activeInstances.put(instanceId, updated);
                allInstances.put(instanceId, updated);

                // 更新统计
                heartbeatCount.incrementAndGet();

                if (heartbeatCount.get() % 10 == 0) {  // 每10次心跳记录一次
                    log.debug("心跳发送成功: {}，心跳计数: {}", instanceId, heartbeatCount.get());
                }
            }

        } catch (Exception e) {
            log.error("发送心跳失败: {}", instanceId, e);
            // 可以考虑重试或降级处理
        }
    }

    /**
     * 检查实例是否在数据库中
     */
    private boolean existsInDatabase(String instanceId) {
        Query query = new Query(Criteria.where("instanceId").is(instanceId));
        return mongoTemplate.exists(query, INSTANCES_COLLECTION);
    }

    // ========== 2. 刷新实例并删除过期节点 ==========

    /**
     * 发现并刷新实例，删除过期节点
     */
    @Scheduled(fixedDelay = 15000)  // 每15秒执行一次
    @Transactional
    public void discoverAndRefreshInstances() {
        try {
            Instant now = Instant.now();
            Instant activeCutoff = now.minusMillis(INSTANCE_TTL_MS);
            Instant cleanupCutoff = now.minusMillis(CLEANUP_THRESHOLD_MS);

            // 1. 查询活跃实例
            Query activeQuery = new Query(Criteria.where("lastHeartbeat")
                .gte(Date.from(activeCutoff))
                .and("status").in("ACTIVE", "STARTING"));

            List<ClusterInstance> activeInstancesList = mongoTemplate.find(
                activeQuery, ClusterInstance.class, INSTANCES_COLLECTION
            );

            // 2. 查找需要清理的过期实例
            Query expiredQuery = new Query(Criteria.where("lastHeartbeat")
                .lt(Date.from(cleanupCutoff)));

            List<ClusterInstance> expiredInstances = mongoTemplate.find(
                expiredQuery, ClusterInstance.class, INSTANCES_COLLECTION
            );

            // 3. 归档并删除过期实例
            if (!expiredInstances.isEmpty()) {
                cleanupExpiredInstances(expiredInstances, now);
            }

            // 4. 更新本地缓存
            updateLocalCache(activeInstancesList);

            // 5. 记录统计
            logDiscoveryStats(activeInstancesList.size(), expiredInstances.size());

        } catch (Exception e) {
            log.error("发现和刷新实例失败", e);
        }
    }

    /**
     * 清理过期实例
     */
    @Transactional
    public void cleanupExpiredInstances(List<ClusterInstance> expiredInstances, Instant cleanupTime) {
        int deletedCount = 0;
        int archivedCount = 0;

        for (ClusterInstance instance : expiredInstances) {
            try {
                // 跳过当前实例（可能是网络问题导致心跳延迟）
                if (instanceId.equals(instance.getInstanceId())) {
                    log.warn("跳过清理当前实例: {}", instanceId);
                    continue;
                }

                // 1. 归档到历史表
                archiveInstanceToHistory(instance, cleanupTime);
                archivedCount++;

                // 2. 从主表删除
                Query deleteQuery = new Query(Criteria.where("instanceId").is(instance.getInstanceId()));
                mongoTemplate.remove(deleteQuery, INSTANCES_COLLECTION);
                deletedCount++;

                // 3. 从本地缓存中移除
                activeInstances.remove(instance.getInstanceId());
                allInstances.remove(instance.getInstanceId());

                // 4. 记录日志
                log.info("清理过期实例: {}，最后心跳: {}，状态: {}",
                    instance.getInstanceId(),
                    instance.getLastHeartbeat(),
                    instance.getStatus());

            } catch (Exception e) {
                log.error("清理实例失败: {}", instance.getInstanceId(), e);
            }
        }

        if (deletedCount > 0) {
            log.info("清理完成: 删除了 {} 个实例，归档了 {} 个实例到历史表",
                deletedCount, archivedCount);
        }
    }

    /**
     * 归档实例到历史表
     */
    private void archiveInstanceToHistory(ClusterInstance instance, Instant cleanupTime) {
        try {
            InstanceHistory history = InstanceHistory.builder()
                .originalId(instance.getId())
                .instanceId(instance.getInstanceId())
                .hostname(instance.getHostname())
                .ipAddress(instance.getIpAddress())
                .status(instance.getStatus())
                .registeredAt(instance.getRegisteredAt())
                .lastHeartbeat(instance.getLastHeartbeat())
                .startedAt(instance.getStartedAt())
                .terminatedAt(Date.from(cleanupTime))
                .totalUptime(calculateInstanceUptime(instance))
                .totalHeartbeatCount(instance.getTotalHeartbeatCount())
                .restartCount(instance.getRestartCount())
                .finalLoad(instance.getCurrentLoad())
                .capabilities(instance.getCapabilities())
                .metadata(instance.getMetadata())
                .cleanupReason("TTL_EXPIRED")
                .build();

            mongoTemplate.insert(history, INSTANCE_HISTORY_COLLECTION);

        } catch (Exception e) {
            log.error("归档实例失败: {}", instance.getInstanceId(), e);
        }
    }

    /**
     * 更新本地缓存
     */
    private synchronized void updateLocalCache(List<ClusterInstance> freshInstances) {
        // 清空旧缓存
        activeInstances.clear();
        allInstances.clear();

        // 添加新实例
        for (ClusterInstance instance : freshInstances) {
            activeInstances.put(instance.getInstanceId(), instance);
            allInstances.put(instance.getInstanceId(), instance);

            // 如果是新发现的实例（非当前实例）
            if (!instanceId.equals(instance.getInstanceId())) {
                log.debug("发现实例: {}，负载: {}/{}",
                    instance.getInstanceId(),
                    instance.getCurrentLoad(),
                    instance.getConcurrencyLimit());
            }
        }
    }

    // ========== 3. 查询和工具方法 ==========

    /**
     * 获取所有活跃实例
     */
    public List<ClusterInstance> getActiveInstances() {
        Instant cutoff = Instant.now().minusMillis(INSTANCE_TTL_MS);

        return activeInstances.values().stream()
            .filter(instance ->
                instance.getLastHeartbeat() != null &&
                instance.getLastHeartbeat().toInstant().isAfter(cutoff) &&
                "ACTIVE".equals(instance.getStatus())
            )
            .collect(Collectors.toList());
    }

    /**
     * 获取其他实例（排除自己）
     */
    public List<ClusterInstance> getOtherInstances() {
        return getActiveInstances().stream()
            .filter(instance -> !instanceId.equals(instance.getInstanceId()))
            .collect(Collectors.toList());
    }

    /**
     * 检查实例是否活跃
     */
    public boolean isInstanceActive(String checkInstanceId) {
        if (instanceId.equals(checkInstanceId)) {
            return true;  // 当前实例总是活跃
        }

        ClusterInstance instance = activeInstances.get(checkInstanceId);
        if (instance == null) {
            return false;
        }

        Instant cutoff = Instant.now().minusMillis(INSTANCE_TTL_MS);
        return instance.getLastHeartbeat() != null &&
               instance.getLastHeartbeat().toInstant().isAfter(cutoff) &&
               "ACTIVE".equals(instance.getStatus());
    }

    /**
     * 获取集群统计
     */
    public Map<String, Object> getClusterStats() {
        List<ClusterInstance> active = getActiveInstances();
        List<ClusterInstance> others = getOtherInstances();

        int totalLoad = active.stream()
            .mapToInt(ClusterInstance::getCurrentLoad)
            .sum();

        int totalCapacity = active.stream()
            .mapToInt(ClusterInstance::getConcurrencyLimit)
            .sum();

        double loadPercentage = totalCapacity > 0 ?
            (double) totalLoad / totalCapacity * 100 : 0;

        Map<String, Object> stats = new LinkedHashMap<>();
        stats.put("timestamp", Instant.now());
        stats.put("currentInstanceId", instanceId);
        stats.put("currentInstanceLoad", currentLoad.get());
        stats.put("totalInstances", active.size());
        stats.put("otherInstances", others.size());
        stats.put("totalLoad", totalLoad);
        stats.put("totalCapacity", totalCapacity);
        stats.put("loadPercentage", String.format("%.1f%%", loadPercentage));
        stats.put("heartbeatCount", heartbeatCount.get());
        stats.put("localCacheSize", activeInstances.size());

        // 实例详情
        List<Map<String, Object>> instancesDetail = active.stream()
            .map(this::instanceToMap)
            .collect(Collectors.toList());
        stats.put("instances", instancesDetail);

        return stats;
    }

    // ========== 4. 定时任务 ==========

    /**
     * 启动定时任务
     */
    private void startScheduledTasks() {
        // 心跳任务
        Timer heartbeatTimer = new Timer("instance-heartbeat", true);
        heartbeatTimer.scheduleAtFixedRate(new TimerTask() {
            @Override
            public void run() {
                sendHeartbeatWithUpsert();
            }
        }, HEARTBEAT_INTERVAL_MS, HEARTBEAT_INTERVAL_MS);

        // 定期清理历史数据（每天一次）
        Timer cleanupTimer = new Timer("history-cleanup", true);
        cleanupTimer.scheduleAtFixedRate(new TimerTask() {
            @Override
            public void run() {
                cleanupOldHistory();
            }
        }, 24 * 60 * 60 * 1000, 24 * 60 * 60 * 1000);  // 24小时

        log.info("定时任务启动完成");
    }

    /**
     * 清理旧的历史记录
     */
    @Scheduled(cron = "0 0 2 * * ?")  // 每天凌晨2点执行
    @Transactional
    public void cleanupOldHistory() {
        try {
            Instant cutoff = Instant.now().minusSeconds(30 * 24 * 60 * 60);  // 30天前

            Query query = new Query(Criteria.where("terminatedAt")
                .lt(Date.from(cutoff)));

            long deleted = mongoTemplate.remove(query, INSTANCE_HISTORY_COLLECTION)
                .getDeletedCount();

            if (deleted > 0) {
                log.info("清理了 {} 条旧的历史记录（30天前）", deleted);
            }

        } catch (Exception e) {
            log.error("清理历史记录失败", e);
        }
    }

    // ========== 5. 工具方法 ==========

    private String resolveInstanceId() {
        // 优先使用环境变量
        String podName = System.getenv("POD_NAME");
        if (podName != null && !podName.trim().isEmpty()) {
            return podName.trim();
        }

        // 回退方案
        try {
            String host = InetAddress.getLocalHost().getHostName();
            return host + "-" + UUID.randomUUID().toString().substring(0, 8);
        } catch (Exception e) {
            return "instance-" + UUID.randomUUID().toString();
        }
    }

    private String getHostname() {
        try {
            return InetAddress.getLocalHost().getHostName();
        } catch (Exception e) {
            return "unknown";
        }
    }

    private String getIpAddress() {
        try {
            return InetAddress.getLocalHost().getHostAddress();
        } catch (Exception e) {
            return "127.0.0.1";
        }
    }

    private boolean isRecentRestart(ClusterInstance instance) {
        if (instance == null || instance.getLastHeartbeat() == null) {
            return true;
        }

        Duration sinceLastHeartbeat = Duration.between(
            instance.getLastHeartbeat().toInstant(),
            Instant.now()
        );

        // 5分钟内有心跳，认为是重启
        return sinceLastHeartbeat.toMinutes() < 5;
    }

    private long calculateTotalUptime() {
        // 这里可以计算实例的总运行时间
        return 0L;  // 简化实现
    }

    private long calculateInstanceUptime(ClusterInstance instance) {
        if (instance.getStartedAt() == null || instance.getLastHeartbeat() == null) {
            return 0L;
        }

        return Duration.between(
            instance.getStartedAt().toInstant(),
            instance.getLastHeartbeat().toInstant()
        ).toMillis();
    }

    private void logDiscoveryStats(int activeCount, int expiredCount) {
        if (log.isDebugEnabled() || expiredCount > 0) {
            log.info("实例发现统计: 活跃 {} 个，过期 {} 个", activeCount, expiredCount);
        }
    }

    private Map<String, Object> instanceToMap(ClusterInstance instance) {
        Map<String, Object> map = new LinkedHashMap<>();
        map.put("instanceId", instance.getInstanceId());
        map.put("hostname", instance.getHostname());
        map.put("status", instance.getStatus());
        map.put("load", instance.getCurrentLoad());
        map.put("limit", instance.getConcurrencyLimit());
        map.put("lastHeartbeat", instance.getLastHeartbeat());
        map.put("isCurrent", instanceId.equals(instance.getInstanceId()));
        return map;
    }

        private Date startedAt;

    }
