package com.example.s3scanner.service.coordinator;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import java.util.UUID;
import static org.junit.jupiter.api.Assertions.*;

@SpringBootTest
class InstanceRegistryTest {
    
    @Autowired
    private InstanceRegistry instanceRegistry;
    
    @Test
    void testShouldProcess() {
        String instanceId = instanceRegistry.getInstanceId();
        String bucket = "test-bucket";
        
        // 生成测试key
        String key1 = "uploads/file1.txt";
        String key2 = "uploads/file2.txt";
        String key3 = "pending/file3.txt";
        String key4 = "archive/file4.txt";  // 可能不在扫描前缀中
        
        // 测试应该处理的key
        boolean result1 = instanceRegistry.shouldProcess(bucket, key1);
        boolean result2 = instanceRegistry.shouldProcess(bucket, key2);
        boolean result3 = instanceRegistry.shouldProcess(bucket, key3);
        boolean result4 = instanceRegistry.shouldProcess(bucket, key4);
        
        // 验证
        System.out.println("key1 shouldProcess: " + result1);
        System.out.println("key2 shouldProcess: " + result2);
        System.out.println("key3 shouldProcess: " + result3);
        System.out.println("key4 shouldProcess: " + result4);
        
        // 获取实例统计
        var stats = instanceRegistry.getInstanceStats();
        System.out.println("Instance stats: " + stats);
    }
    
    @Test
    void testLoadManagement() {
        // 测试负载管理
        int initialLoad = instanceRegistry.getCurrentLoad();
        
        instanceRegistry.incrementLoad();
        assertEquals(initialLoad + 1, instanceRegistry.getCurrentLoad());
        
        instanceRegistry.decrementLoad();
        assertEquals(initialLoad, instanceRegistry.getCurrentLoad());
        
        double loadPercentage = instanceRegistry.getLoadPercentage();
        assertTrue(loadPercentage >= 0 && loadPercentage <= 100);
    }
    
    @Test
    void testInstanceActivity() {
        String instanceId = instanceRegistry.getInstanceId();
        
        boolean isActive = instanceRegistry.isInstanceActive(instanceId);
        assertTrue(isActive, "当前实例应该是活跃的");
        
        var activeInstances = instanceRegistry.getActiveInstances();
        assertTrue(activeInstances.size() >= 1, "至少有一个活跃实例");
    }
}
