package com.example.s3scanner.service.coordinator;

import com.example.s3scanner.model.domain.InstanceInfo;
import com.example.s3scanner.repository.InstanceInfoRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.data.mongodb.core.MongoTemplate;
import org.springframework.data.mongodb.core.query.Criteria;
import org.springframework.data.mongodb.core.query.Query;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;
import javax.annotation.PostConstruct;
import java.net.InetAddress;
import java.net.UnknownHostException;
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.stream.Collectors;

@Service
@Slf4j
@RequiredArgsConstructor
public class InstanceRegistry {
    
    private final MongoTemplate mongoTemplate;
    private final InstanceInfoRepository instanceRepository;
    
    @Value("${app.instance.id}")
    private String instanceId;
    
    @Value("${app.instance.concurrency-limit:20}")
    private int concurrencyLimit;
    
    @Value("${app.instance.heartbeat-interval:30}")
    private int heartbeatInterval;
    
    @Value("${app.scanner.scan-prefixes:uploads/,pending/}")
    private List<String> scanPrefixes;
    
    // 一致性哈希环
    private final SortedMap<Integer, String> consistentHashRing = new TreeMap<>();
    private final Map<String, InstanceInfo> activeInstances = new ConcurrentHashMap<>();
    private final AtomicInteger currentLoad = new AtomicInteger(0);
    
    // 分区映射缓存
    private volatile String assignedPartition = null;
    private final Map<String, Set<String>> partitionKeyRanges = new ConcurrentHashMap<>();
    
    // 虚拟节点数
    private static final int VIRTUAL_NODES_PER_INSTANCE = 100;
    
    /**
     * 初始化实例注册
     */
    @PostConstruct
    public void init() {
        registerInstance();
        startHeartbeat();
        updateConsistentHashRing();
    }
    
    /**
     * 注册实例
     */
    private void registerInstance() {
        InstanceInfo instance = InstanceInfo.builder()
            .instanceId(instanceId)
            .hostname(getHostname())
            .ipAddress(getIpAddress())
            .status(InstanceInfo.InstanceStatus.STARTING)
            .startedAt(new Date())
            .concurrencyLimit(concurrencyLimit)
            .currentLoad(0)
            .totalProcessed(0L)
            .totalFailed(0L)
            .capabilities(buildCapabilities())
            .build();
        
        instanceRepository.save(instance);
        log.info("实例注册成功: {}", instanceId);
    }
    
    /**
     * 启动心跳
     */
    private void startHeartbeat() {
        Timer timer = new Timer("instance-heartbeat", true);
        timer.scheduleAtFixedRate(new TimerTask() {
            @Override
            public void run() {
                updateHeartbeat();
            }
        }, 0, heartbeatInterval * 1000L);
    }
    
    /**
     * 更新心跳
     */
    private void updateHeartbeat() {
        try {
            Date now = new Date();
            Date expiresAt = new Date(now.getTime() + (heartbeatInterval + 30) * 1000L);
            
            InstanceInfo instance = InstanceInfo.builder()
                .instanceId(instanceId)
                .hostname(getHostname())
                .ipAddress(getIpAddress())
                .lastHeartbeat(now)
                .expiresAt(expiresAt)
                .status(InstanceInfo.InstanceStatus.RUNNING)
                .concurrencyLimit(concurrencyLimit)
                .currentLoad(currentLoad.get())
                .totalProcessed(getTotalProcessed())
                .totalFailed(getTotalFailed())
                .startedAt(getStartTime())
                .capabilities(buildCapabilities())
                .build();
            
            instanceRepository.save(instance);
            
            // 更新活跃实例缓存
            activeInstances.put(instanceId, instance);
            
        } catch (Exception e) {
            log.error("更新心跳失败", e);
        }
    }
    
    /**
     * 更新一致性哈希环
     */
    public synchronized void updateConsistentHashRing() {
        try {
            // 获取活跃实例
            List<InstanceInfo> instances = getActiveInstances();
            
            // 清空哈希环
            consistentHashRing.clear();
            partitionKeyRanges.clear();
            
            // 为每个实例添加虚拟节点
            for (InstanceInfo instance : instances) {
                String instanceId = instance.getInstanceId();
                
                for (int i = 0; i < VIRTUAL_NODES_PER_INSTANCE; i++) {
                    String virtualNode = instanceId + "#" + i;
                    int hash = Math.abs(virtualNode.hashCode());
                    consistentHashRing.put(hash, instanceId);
                }
                
                // 记录实例的分区
                assignPartition(instance);
            }
            
            // 更新当前实例的分区
            updateAssignedPartition();
            
            log.info("一致性哈希环更新完成: {} 个实例，{} 个虚拟节点", 
                instances.size(), consistentHashRing.size());
            
        } catch (Exception e) {
            log.error("更新一致性哈希环失败", e);
        }
    }
    
    /**
     * 检查是否应该处理指定key
     * 核心方法：通过一致性哈希决定哪个实例处理哪个key
     */
    public boolean shouldProcess(String bucket, String key) {
        // 1. 检查当前实例是否活跃
        if (!isInstanceActive(instanceId)) {
            log.warn("当前实例不活跃: {}", instanceId);
            return false;
        }
        
        // 2. 检查当前负载
        if (currentLoad.get() >= concurrencyLimit) {
            log.debug("当前实例负载已满: {}/{}", currentLoad.get(), concurrencyLimit);
            return false;
        }
        
        // 3. 检查前缀是否应该处理
        if (!shouldProcessPrefix(key)) {
            return false;
        }
        
        // 4. 一致性哈希分配
        String assignedInstance = getAssignedInstance(bucket, key);
        boolean shouldProcess = instanceId.equals(assignedInstance);
        
        if (shouldProcess) {
            log.debug("应该处理: {}/{} -> 实例: {}", bucket, key, instanceId);
        } else {
            log.trace("不应该处理: {}/{} -> 分配给实例: {}", bucket, key, assignedInstance);
        }
        
        return shouldProcess;
    }
    
    /**
     * 获取应该处理指定key的实例
     */
    public String getAssignedInstance(String bucket, String key) {
        if (consistentHashRing.isEmpty()) {
            return instanceId;  // 如果没有其他实例，当前实例处理
        }
        
        // 生成资源标识
        String resource = bucket + ":" + key;
        int hash = getConsistentHash(resource);
        
        // 在哈希环中查找
        SortedMap<Integer, String> tailMap = consistentHashRing.tailMap(hash);
        int targetHash = tailMap.isEmpty() ? consistentHashRing.firstKey() : tailMap.firstKey();
        String assignedInstance = consistentHashRing.get(targetHash);
        
        return assignedInstance;
    }
    
    /**
     * 使用一致性哈希算法
     */
    private int getConsistentHash(String key) {
        // MurmurHash 或简单的Java hashCode
        int hash = key.hashCode();
        
        // 确保正数
        if (hash < 0) {
            hash = -hash;
        }
        
        // 取模确保在哈希环范围内
        if (!consistentHashRing.isEmpty()) {
            int maxHash = consistentHashRing.lastKey();
            hash = hash % (maxHash + 1);
        }
        
        return hash;
    }
    
    /**
     * 检查是否应该处理指定前缀
     */
    private boolean shouldProcessPrefix(String key) {
        if (scanPrefixes == null || scanPrefixes.isEmpty()) {
            return true;
        }
        
        for (String prefix : scanPrefixes) {
            if (key.startsWith(prefix)) {
                return true;
            }
        }
        
        return false;
    }
    
    /**
     * 分配分区
     */
    private void assignPartition(InstanceInfo instance) {
        if (activeInstances.isEmpty()) {
            return;
        }
        
        // 获取所有实例并排序
        List<String> instanceIds = activeInstances.values().stream()
            .map(InstanceInfo::getInstanceId)
            .sorted()
            .collect(Collectors.toList());
        
        int index = instanceIds.indexOf(instance.getInstanceId());
        int totalInstances = instanceIds.size();
        
        if (index >= 0 && totalInstances > 0) {
            String partition = String.format("partition-%d-of-%d", index, totalInstances);
            instance.setPartitionKey(partition);
            
            // 计算该分区负责的哈希范围
            calculatePartitionRanges(partition, index, totalInstances);
        }
    }
    
    /**
     * 计算分区哈希范围
     */
    private void calculatePartitionRanges(String partition, int partitionIndex, int totalPartitions) {
        if (totalPartitions <= 0) {
            return;
        }
        
        int hashRange = Integer.MAX_VALUE / totalPartitions;
        int startHash = partitionIndex * hashRange;
        int endHash = (partitionIndex + 1) * hashRange - 1;
        
        Set<String> ranges = new HashSet<>();
        ranges.add(String.format("%d-%d", startHash, endHash));
        partitionKeyRanges.put(partition, ranges);
        
        log.debug("分区 {} 负责哈希范围: {}-{}", partition, startHash, endHash);
    }
    
    /**
     * 更新分配的分区
     */
    private void updateAssignedPartition() {
        InstanceInfo instance = activeInstances.get(instanceId);
        if (instance != null && instance.getPartitionKey() != null) {
            assignedPartition = instance.getPartitionKey();
            log.info("当前实例分配分区: {}", assignedPartition);
        }
    }
    
    /**
     * 增加当前负载
     */
    public void incrementLoad() {
        currentLoad.incrementAndGet();
        log.debug("增加负载: 当前 {}/{}", currentLoad.get(), concurrencyLimit);
    }
    
    /**
     * 减少当前负载
     */
    public void decrementLoad() {
        if (currentLoad.get() > 0) {
            currentLoad.decrementAndGet();
            log.debug("减少负载: 当前 {}/{}", currentLoad.get(), concurrencyLimit);
        }
    }
    
    /**
     * 获取当前负载
     */
    public int getCurrentLoad() {
        return currentLoad.get();
    }
    
    /**
     * 获取负载百分比
     */
    public double getLoadPercentage() {
        if (concurrencyLimit <= 0) {
            return 0;
        }
        return (double) currentLoad.get() / concurrencyLimit * 100;
    }
    
    /**
     * 检查实例是否活跃
     */
    public boolean isInstanceActive(String instanceId) {
        InstanceInfo instance = activeInstances.get(instanceId);
        if (instance == null) {
            return false;
        }
        
        Date now = new Date();
        return instance.getExpiresAt() != null && 
               instance.getExpiresAt().after(now) &&
               instance.getStatus() == InstanceInfo.InstanceStatus.RUNNING;
    }
    
    /**
     * 获取活跃实例列表
     */
    public List<InstanceInfo> getActiveInstances() {
        Date cutoff = new Date(System.currentTimeMillis() - (heartbeatInterval + 30) * 1000L);
        
        return activeInstances.values().stream()
            .filter(instance -> 
                instance.getLastHeartbeat() != null &&
                instance.getLastHeartbeat().after(cutoff) &&
                instance.getStatus() == InstanceInfo.InstanceStatus.RUNNING
            )
            .collect(Collectors.toList());
    }
    
    /**
     * 获取所有实例
     */
    public List<InstanceInfo> getAllInstances() {
        return new ArrayList<>(activeInstances.values());
    }
    
    /**
     * 获取实例统计
     */
    public Map<String, Object> getInstanceStats() {
        Map<String, Object> stats = new HashMap<>();
        
        List<InstanceInfo> active = getActiveInstances();
        stats.put("totalInstances", activeInstances.size());
        stats.put("activeInstances", active.size());
        stats.put("currentInstanceId", instanceId);
        stats.put("currentLoad", currentLoad.get());
        stats.put("concurrencyLimit", concurrencyLimit);
        stats.put("loadPercentage", getLoadPercentage());
        stats.put("assignedPartition", assignedPartition);
        stats.put("consistentHashRingSize", consistentHashRing.size());
        stats.put("virtualNodesPerInstance", VIRTUAL_NODES_PER_INSTANCE);
        
        // 实例详情
        List<Map<String, Object>> instancesDetail = active.stream()
            .map(instance -> {
                Map<String, Object> detail = new HashMap<>();
                detail.put("instanceId", instance.getInstanceId());
                detail.put("hostname", instance.getHostname());
                detail.put("status", instance.getStatus().name());
                detail.put("load", instance.getCurrentLoad());
                detail.put("limit", instance.getConcurrencyLimit());
                detail.put("lastHeartbeat", instance.getLastHeartbeat());
                detail.put("partition", instance.getPartitionKey());
                detail.put("totalProcessed", instance.getTotalProcessed());
                detail.put("totalFailed", instance.getTotalFailed());
                return detail;
            })
            .collect(Collectors.toList());
        
        stats.put("instances", instancesDetail);
        
        return stats;
    }
    
    /**
     * 清理过期实例
     */
    @Scheduled(fixedDelay = 60000)  // 每分钟清理一次
    public void cleanupExpiredInstances() {
        try {
            Date cutoff = new Date(System.currentTimeMillis() - (heartbeatInterval + 60) * 1000L);
            
            int count = 0;
            for (Map.Entry<String, InstanceInfo> entry : activeInstances.entrySet()) {
                InstanceInfo instance = entry.getValue();
                
                if (instance.getLastHeartbeat() != null && 
                    instance.getLastHeartbeat().before(cutoff)) {
                    
                    // 从数据库删除
                    instanceRepository.deleteById(instance.getInstanceId());
                    
                    // 从缓存删除
                    activeInstances.remove(entry.getKey());
                    count++;
                    
                    log.warn("清理过期实例: {}", instance.getInstanceId());
                }
            }
            
            if (count > 0) {
                // 重新计算哈希环
                updateConsistentHashRing();
                log.info("清理了 {} 个过期实例，重新计算哈希环", count);
            }
            
        } catch (Exception e) {
            log.error("清理过期实例失败", e);
        }
    }
    
    /**
     * 更新一致性哈希环（定时任务）
     */
    @Scheduled(fixedDelay = 30000)  // 每30秒更新一次
    public void scheduledUpdateHashRing() {
        updateConsistentHashRing();
    }
    
    // ========== 私有辅助方法 ==========
    
    /**
     * 构建实例能力
     */
    private Map<String, Object> buildCapabilities() {
        Map<String, Object> capabilities = new HashMap<>();
        capabilities.put("s3.scan", true);
        capabilities.put("s3.process", true);
        capabilities.put("s3.move", true);
        capabilities.put("s3.download", true);
        capabilities.put("s3.upload", true);
        capabilities.put("distributed", true);
        capabilities.put("heartbeat", true);
        return capabilities;
    }
    
    /**
     * 获取主机名
     */
    private String getHostname() {
        try {
            return InetAddress.getLocalHost().getHostName();
        } catch (UnknownHostException e) {
            return "unknown";
        }
    }
    
    /**
     * 获取IP地址
     */
    private String getIpAddress() {
        try {
            return InetAddress.getLocalHost().getHostAddress();
        } catch (UnknownHostException e) {
            return "127.0.0.1";
        }
    }
    
    /**
     * 获取总处理数
     */
    private Long getTotalProcessed() {
        // 这里可以从数据库或缓存获取实际值
        return 0L;
    }
    
    /**
     * 获取总失败数
     */
    private Long getTotalFailed() {
        // 这里可以从数据库或缓存获取实际值
        return 0L;
    }
    
    /**
     * 获取启动时间
     */
    private Date getStartTime() {
        InstanceInfo instance = activeInstances.get(instanceId);
        return instance != null ? instance.getStartedAt() : new Date();
    }
}
