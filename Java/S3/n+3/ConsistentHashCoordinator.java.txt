package com.example.s3scanner.service.coordinator;

import lombok.Data;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ConcurrentSkipListMap;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.stream.Collectors;

/**
 * 一致性哈希协调器
 */
@Component
@Slf4j
public class ConsistentHashCoordinator {
    
    // 一致性哈希环
    private final ConcurrentSkipListMap<Integer, String> hashRing = new ConcurrentSkipListMap<>();
    
    // 实例映射
    private final Map<String, InstanceNode> instances = new ConcurrentHashMap<>();
    
    // 虚拟节点数
    private static final int VIRTUAL_NODES_PER_INSTANCE = 1000;
    
    // 哈希算法
    private final MessageDigest md5Digest;
    
    public ConsistentHashCoordinator() throws NoSuchAlgorithmException {
        this.md5Digest = MessageDigest.getInstance("MD5");
    }
    
    /**
     * 添加实例到哈希环
     */
    public synchronized void addInstance(String instanceId, InstanceNode node) {
        if (instances.containsKey(instanceId)) {
            log.warn("实例已存在: {}", instanceId);
            return;
        }
        
        instances.put(instanceId, node);
        
        // 添加虚拟节点
        for (int i = 0; i < VIRTUAL_NODES_PER_INSTANCE; i++) {
            String virtualNode = instanceId + "#vnode#" + i;
            int hash = hash(virtualNode);
            hashRing.put(hash, instanceId);
        }
        
        log.info("实例 {} 添加到哈希环，添加了 {} 个虚拟节点", 
            instanceId, VIRTUAL_NODES_PER_INSTANCE);
    }
    
    /**
     * 移除实例
     */
    public synchronized void removeInstance(String instanceId) {
        if (!instances.containsKey(instanceId)) {
            log.warn("实例不存在: {}", instanceId);
            return;
        }
        
        // 移除所有虚拟节点
        for (int i = 0; i < VIRTUAL_NODES_PER_INSTANCE; i++) {
            String virtualNode = instanceId + "#vnode#" + i;
            int hash = hash(virtualNode);
            hashRing.remove(hash);
        }
        
        instances.remove(instanceId);
        log.info("实例 {} 从哈希环移除", instanceId);
    }
    
    /**
     * 获取应该处理指定key的实例
     */
    public String getAssignedInstance(String key) {
        if (hashRing.isEmpty()) {
            throw new IllegalStateException("哈希环为空");
        }
        
        int hash = hash(key);
        
        // 在哈希环中顺时针查找
        Map.Entry<Integer, String> entry = hashRing.ceilingEntry(hash);
        
        if (entry == null) {
            // 如果没有找到，回到环的开始
            entry = hashRing.firstEntry();
        }
        
        return entry.getValue();
    }
    
    /**
     * 检查当前实例是否应该处理指定key
     */
    public boolean shouldProcess(String currentInstanceId, String key) {
        String assignedInstance = getAssignedInstance(key);
        return currentInstanceId.equals(assignedInstance);
    }
    
    /**
     * 获取实例负责的key范围
     */
    public List<Range> getInstanceRanges(String instanceId) {
        if (!instances.containsKey(instanceId)) {
            return Collections.emptyList();
        }
        
        List<Range> ranges = new ArrayList<>();
        
        // 找到实例的所有虚拟节点
        List<Integer> virtualNodeHashes = new ArrayList<>();
        for (int i = 0; i < VIRTUAL_NODES_PER_INSTANCE; i++) {
            String virtualNode = instanceId + "#vnode#" + i;
            virtualNodeHashes.add(hash(virtualNode));
        }
        
        // 排序虚拟节点
        Collections.sort(virtualNodeHashes);
        
        // 计算每个虚拟节点负责的范围
        for (int i = 0; i < virtualNodeHashes.size(); i++) {
            int startHash = virtualNodeHashes.get(i);
            int endHash = (i < virtualNodeHashes.size() - 1) ? 
                virtualNodeHashes.get(i + 1) - 1 : Integer.MAX_VALUE;
            
            ranges.add(new Range(startHash, endHash));
        }
        
        return ranges;
    }
    
    /**
     * 获取负载均衡的分配
     */
    public Map<String, Double> getLoadDistribution() {
        Map<String, AtomicInteger> instanceCounts = new HashMap<>();
        
        // 统计每个实例的虚拟节点数
        for (String instanceId : hashRing.values()) {
            instanceCounts
                .computeIfAbsent(instanceId, k -> new AtomicInteger(0))
                .incrementAndGet();
        }
        
        // 计算百分比
        int totalNodes = hashRing.size();
        Map<String, Double> distribution = new HashMap<>();
        
        for (Map.Entry<String, AtomicInteger> entry : instanceCounts.entrySet()) {
            double percentage = (double) entry.getValue().get() / totalNodes * 100;
            distribution.put(entry.getKey(), percentage);
        }
        
        return distribution;
    }
    
    /**
     * 重新平衡哈希环
     */
    public synchronized void rebalance() {
        log.info("开始重新平衡哈希环...");
        
        // 保存当前实例
        Map<String, InstanceNode> currentInstances = new HashMap<>(instances);
        
        // 清空哈希环
        hashRing.clear();
        instances.clear();
        
        // 重新添加所有实例
        for (Map.Entry<String, InstanceNode> entry : currentInstances.entrySet()) {
            addInstance(entry.getKey(), entry.getValue());
        }
        
        log.info("哈希环重新平衡完成");
    }
    
    /**
     * 哈希函数
     */
    private int hash(String key) {
        try {
            byte[] digest = md5Digest.digest(key.getBytes(StandardCharsets.UTF_8));
            
            // 取前4个字节作为哈希值
            int hash = 0;
            for (int i = 0; i < 4; i++) {
                hash = (hash << 8) | (digest[i] & 0xFF);
            }
            
            // 确保为正数
            hash = hash & 0x7FFFFFFF;
            
            return hash;
            
        } catch (Exception e) {
            // 回退到简单的hashCode
            return Math.abs(key.hashCode());
        }
    }
    
    /**
     * 获取哈希环统计
     */
    public Map<String, Object> getHashRingStats() {
        Map<String, Object> stats = new HashMap<>();
        
        stats.put("totalInstances", instances.size());
        stats.put("totalVirtualNodes", hashRing.size());
        stats.put("virtualNodesPerInstance", VIRTUAL_NODES_PER_INSTANCE);
        
        // 负载分布
        Map<String, Double> distribution = getLoadDistribution();
        stats.put("loadDistribution", distribution);
        
        // 实例详情
        List<Map<String, Object>> instancesDetail = instances.entrySet().stream()
            .map(entry -> {
                Map<String, Object> detail = new HashMap<>();
                detail.put("instanceId", entry.getKey());
                detail.put("node", entry.getValue());
                
                // 计算该实例的虚拟节点数
                long virtualNodeCount = hashRing.values().stream()
                    .filter(v -> v.equals(entry.getKey()))
                    .count();
                detail.put("virtualNodeCount", virtualNodeCount);
                
                return detail;
            })
            .collect(Collectors.toList());
        
        stats.put("instances", instancesDetail);
        
        return stats;
    }
    
    @Data
    public static class InstanceNode {
        private String instanceId;
        private String hostname;
        private String ipAddress;
        private int concurrencyLimit;
        private int currentLoad;
        private boolean active = true;
        private Date lastHeartbeat;
        private Map<String, Object> metadata = new HashMap<>();
        
        public double getLoadPercentage() {
            if (concurrencyLimit <= 0) {
                return 0;
            }
            return (double) currentLoad / concurrencyLimit * 100;
        }
        
        public boolean isOverloaded() {
            return currentLoad >= concurrencyLimit;
        }
    }
    
    @Data
    public static class Range {
        private final int start;
        private final int end;
        
        public boolean contains(int hash) {
            return hash >= start && hash <= end;
        }
        
        public int size() {
            return end - start + 1;
        }
    }
}
