import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import software.amazon.awssdk.auth.credentials.AwsBasicCredentials;
import software.amazon.awssdk.auth.credentials.StaticCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.s3.S3Client;
import java.net.URI;

@Configuration
public class S3Config {

    @Value("${aws.region:us-east-1}")
    private String region;

    @Value("${aws.accessKeyId:}")
    private String accessKey;

    @Value("${aws.secretKey:}")
    private String secretKey;

    @Value("${aws.s3.endpoint:}")
    private String endpoint;

    @Bean
    public S3Client s3Client() {
        S3Client.Builder builder = S3Client.builder()
            .region(Region.of(region));

        // 如果有自定义endpoint（如MinIO）
        if (endpoint != null && !endpoint.isEmpty()) {
            builder.endpointOverride(URI.create(endpoint));
        }

        // 如果使用access key/secret
        if (accessKey != null && !accessKey.isEmpty() &&
            secretKey != null && !secretKey.isEmpty()) {
            builder.credentialsProvider(StaticCredentialsProvider.create(
                AwsBasicCredentials.create(accessKey, secretKey)
            ));
        }

        return builder.build();
    }
}

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import software.amazon.awssdk.services.s3.S3Client;
import software.amazon.awssdk.services.s3.model.*;
import java.util.*;
import java.util.stream.Collectors;

@Slf4j
@Service
@RequiredArgsConstructor
public class S3MetadataService {

    private final S3Client s3Client;

    /**
     * 移动文件并修改元数据
     */
    public void moveFileWithModifiedMetadata(
            String sourceKey,
            String destinationKey,
            String bucket,
            MetadataUpdateStrategy strategy) {

        // 1. 获取原始元数据
        Map<String, String> originalMetadata = getObjectMetadata(bucket, sourceKey);
        log.info("原始元数据: {}", originalMetadata);

        // 2. 根据策略生成新元数据
        Map<String, String> newMetadata = strategy.apply(originalMetadata);
        log.info("新元数据: {}", newMetadata);

        // 3. 复制文件到新位置，并设置新元数据
        copyWithMetadata(bucket, sourceKey, destinationKey, newMetadata);

        // 4. 删除原文件
        deleteObject(bucket, sourceKey);

        log.info("文件移动完成: {} -> {} (bucket: {})",
            sourceKey, destinationKey, bucket);
    }

    /**
     * 获取对象元数据
     */
    public Map<String, String> getObjectMetadata(String bucket, String key) {
        try {
            HeadObjectResponse response = s3Client.headObject(
                HeadObjectRequest.builder()
                    .bucket(bucket)
                    .key(key)
                    .build()
            );

            // 返回所有元数据（包括系统元数据）
            Map<String, String> metadata = new HashMap<>();

            // 用户自定义元数据
            metadata.putAll(response.metadata());

            // 系统元数据
            metadata.put("content-type", response.contentType());
            metadata.put("content-length", String.valueOf(response.contentLength()));
            metadata.put("last-modified", response.lastModified().toString());
            metadata.put("etag", response.eTag());

            if (response.contentEncoding() != null) {
                metadata.put("content-encoding", response.contentEncoding());
            }
            if (response.contentDisposition() != null) {
                metadata.put("content-disposition", response.contentDisposition());
            }
            if (response.cacheControl() != null) {
                metadata.put("cache-control", response.cacheControl());
            }

            return metadata;
        } catch (S3Exception e) {
            log.error("获取元数据失败: {}/{}", bucket, key, e);
            throw new RuntimeException("无法获取文件元数据", e);
        }
    }

    /**
     * 复制文件并设置新元数据
     */
    private void copyWithMetadata(String bucket, String sourceKey,
                                  String destinationKey, Map<String, String> newMetadata) {
        try {
            // 构建元数据映射
            Map<String, String> metadataForUpload = new HashMap<>(newMetadata);

            // 准备复制请求
            CopyObjectRequest.Builder requestBuilder = CopyObjectRequest.builder()
                .sourceBucket(bucket)
                .sourceKey(sourceKey)
                .destinationBucket(bucket)
                .destinationKey(destinationKey)
                .metadata(metadataForUpload)
                .metadataDirective(MetadataDirective.REPLACE);

            // 如果有特定的Content-Type，也设置
            if (metadataForUpload.containsKey("content-type")) {
                requestBuilder.contentType(metadataForUpload.get("content-type"));
            }

            // 执行复制
            s3Client.copyObject(requestBuilder.build());

            log.info("文件复制完成: {} -> {} (已更新元数据)", sourceKey, destinationKey);
        } catch (S3Exception e) {
            log.error("复制文件失败: {}/{} -> {}", bucket, sourceKey, destinationKey, e);
            throw new RuntimeException("文件复制失败", e);
        }
    }

    /**
     * 删除原文件
     */
    private void deleteObject(String bucket, String key) {
        try {
            s3Client.deleteObject(
                DeleteObjectRequest.builder()
                    .bucket(bucket)
                    .key(key)
                    .build()
            );
            log.info("已删除原文件: {}/{}", bucket, key);
        } catch (S3Exception e) {
            log.error("删除文件失败: {}/{}", bucket, key, e);
            // 这里可以选择是否抛出异常，根据业务需求决定
        }
    }

    /**
     * 批量移动文件
     */
    public void batchMoveWithMetadataUpdate(
            String sourcePrefix,
            String destinationPrefix,
            String bucket,
            MetadataUpdateStrategy strategy) {

        ListObjectsV2Request listRequest = ListObjectsV2Request.builder()
            .bucket(bucket)
            .prefix(sourcePrefix)
            .build();

        ListObjectsV2Response listResponse = s3Client.listObjectsV2(listRequest);

        for (S3Object s3Object : listResponse.contents()) {
            String sourceKey = s3Object.key();
            String fileName = sourceKey.substring(sourceKey.lastIndexOf('/') + 1);
            String destinationKey = destinationPrefix + "/" + fileName;

            try {
                moveFileWithModifiedMetadata(sourceKey, destinationKey, bucket, strategy);
            } catch (Exception e) {
                log.error("移动文件失败: {} -> {}", sourceKey, destinationKey, e);
                // 可以记录失败，继续处理其他文件
            }
        }
    }
}

import org.springframework.stereotype.Component;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.HashMap;
import java.util.Map;
import java.util.UUID;

@Component
public class MetadataStrategies {

    /**
     * 策略1: 添加时间戳和处理器信息
     */
    public static final MetadataUpdateStrategy ADD_TIMESTAMP_STRATEGY =
        original -> {
            Map<String, String> newMetadata = new HashMap<>(original);
            newMetadata.put("processed-time", LocalDateTime.now()
                .format(DateTimeFormatter.ISO_LOCAL_DATE_TIME));
            newMetadata.put("processor", "s3-metadata-processor");
            newMetadata.put("processed-id", UUID.randomUUID().toString());
            return newMetadata;
        };

    /**
     * 策略2: 修改Content-Type
     */
    public static MetadataUpdateStrategy changeContentType(String newContentType) {
        return original -> {
            Map<String, String> newMetadata = new HashMap<>(original);
            newMetadata.put("content-type", newContentType);
            return newMetadata;
        };
    }

    /**
     * 策略3: 添加分类标签
     */
    public static MetadataUpdateStrategy addClassificationTag(String category) {
        return original -> {
            Map<String, String> newMetadata = new HashMap<>(original);
            newMetadata.put("category", category);
            newMetadata.put("classification-date",
                LocalDateTime.now().format(DateTimeFormatter.ISO_LOCAL_DATE));
            return newMetadata;
        };
    }

    /**
     * 策略4: 保留部分元数据，添加新字段
     */
    public static MetadataUpdateStrategy selectiveUpdateStrategy() {
        return original -> {
            Map<String, String> newMetadata = new HashMap<>();

            // 保留特定元数据
            if (original.containsKey("original-owner")) {
                newMetadata.put("owner", original.get("original-owner"));
            }
            if (original.containsKey("content-type")) {
                newMetadata.put("original-content-type", original.get("content-type"));
            }

            // 添加新元数据
            newMetadata.put("migrated", "true");
            newMetadata.put("migration-time",
                LocalDateTime.now().format(DateTimeFormatter.ISO_LOCAL_DATE_TIME));
            newMetadata.put("new-content-type", "application/octet-stream");

            return newMetadata;
        };
    }
}

import lombok.RequiredArgsConstructor;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import java.util.Map;

@RestController
@RequestMapping("/api/s3")
@RequiredArgsConstructor
public class S3Controller {

    private final S3MetadataService s3MetadataService;

    /**
     * 移动单个文件并修改元数据
     */
    @PostMapping("/move-with-metadata")
    public ResponseEntity<Map<String, Object>> moveFileWithMetadata(
            @RequestParam String bucket,
            @RequestParam String sourceKey,
            @RequestParam String destinationKey,
            @RequestBody(required = false) Map<String, String> newMetadata) {

        MetadataUpdateStrategy strategy = original -> {
            // 如果提供了新元数据，完全替换
            if (newMetadata != null && !newMetadata.isEmpty()) {
                return new HashMap<>(newMetadata);
            }
            // 否则添加默认字段
            Map<String, String> updated = new HashMap<>(original);
            updated.put("processed", "true");
            updated.put("processed-at", java.time.LocalDateTime.now().toString());
            return updated;
        };

        s3MetadataService.moveFileWithModifiedMetadata(
            sourceKey, destinationKey, bucket, strategy);

        return ResponseEntity.ok(Map.of(
            "status", "success",
            "message", "文件移动完成",
            "source", sourceKey,
            "destination", destinationKey
        ));
    }

    /**
     * 批量移动文件
     */
    @PostMapping("/batch-move")
    public ResponseEntity<Map<String, Object>> batchMove(
            @RequestParam String bucket,
            @RequestParam String sourcePrefix,
            @RequestParam String destinationPrefix,
            @RequestParam String category) {

        s3MetadataService.batchMoveWithMetadataUpdate(
            sourcePrefix,
            destinationPrefix,
            bucket,
            MetadataStrategies.addClassificationTag(category));

        return ResponseEntity.ok(Map.of(
            "status", "success",
            "message", "批量移动任务已提交",
            "sourcePrefix", sourcePrefix,
            "destinationPrefix", destinationPrefix
        ));
    }
}

5. 关键要点

MetadataDirective.REPLACE: 必须设置此参数来更新元数据

元数据读取: 使用 headObject()获取原文件元数据

无数据移动: 复制操作是服务器端的，不经过你的应用服务器

原子性: 复制成功后才会删除原文件

系统元数据: Content-Type等也需要在复制时显式设置
