package com.example.s3scanner.health;

import lombok.RequiredArgsConstructor;
import org.springframework.boot.actuate.health.Health;
import org.springframework.boot.actuate.health.HealthIndicator;
import org.springframework.stereotype.Component;
import software.amazon.awssdk.auth.credentials.AwsCredentials;
import software.amazon.awssdk.auth.credentials.AwsCredentialsProvider;
import software.amazon.awssdk.services.s3.S3Client;
import software.amazon.awssdk.services.s3.model.ListBucketsRequest;
import software.amazon.awssdk.services.s3.model.ListBucketsResponse;

import java.time.Duration;
import java.util.Map;

@Component
@RequiredArgsConstructor
public class AwsHealthIndicator implements HealthIndicator {
    
    private final S3Client s3Client;
    private final AwsCredentialsProvider credentialsProvider;
    
    @Override
    public Health health() {
        Map<String, Object> details = new java.util.HashMap<>();
        
        try {
            // 检查凭证
            AwsCredentials credentials = credentialsProvider.resolveCredentials();
            details.put("credentials", "AVAILABLE");
            details.put("accessKeyId", mask(credentials.accessKeyId()));
            
            // 检查 S3 连接
            long startTime = System.currentTimeMillis();
            ListBucketsResponse response = s3Client.listBuckets(ListBucketsRequest.builder().build());
            long duration = System.currentTimeMillis() - startTime;
            
            details.put("s3Connection", "HEALTHY");
            details.put("responseTime", duration + "ms");
            details.put("bucketCount", response.buckets().size());
            details.put("region", s3Client.serviceClientConfiguration().region().id());
            
            return Health.up()
                .withDetails(details)
                .build();
            
        } catch (Exception e) {
            details.put("error", e.getMessage());
            return Health.down()
                .withDetails(details)
                .withException(e)
                .build();
        }
    }
    
    private String mask(String input) {
        if (input == null || input.length() <= 4) {
            return "***";
        }
        return input.substring(0, 4) + "***";
    }
}
