package com.example.s3scanner.service;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import software.amazon.awssdk.auth.credentials.AwsCredentials;
import software.amazon.awssdk.auth.credentials.AwsCredentialsProvider;
import software.amazon.awssdk.services.s3.S3AsyncClient;
import software.amazon.awssdk.services.s3.model.ListObjectsV2Request;
import software.amazon.awssdk.services.s3.model.ListObjectsV2Response;

import java.util.concurrent.CompletableFuture;

@Service
@Slf4j
@RequiredArgsConstructor
public class S3Service {
    
    private final S3AsyncClient s3AsyncClient;
    private final AwsCredentialsProvider credentialsProvider;
    
    @Value("${aws.region}")
    private String region;
    
    /**
     * 测试凭证和连接
     */
    public void testConnection() {
        try {
            // 验证凭证
            AwsCredentials credentials = credentialsProvider.resolveCredentials();
            log.info("AWS 凭证验证成功, Access Key: {}", 
                credentials.accessKeyId().substring(0, 4) + "***");
            
            // 测试 S3 连接
            ListObjectsV2Request request = ListObjectsV2Request.builder()
                .bucket("test-bucket")
                .maxKeys(1)
                .build();
            
            CompletableFuture<ListObjectsV2Response> future = s3AsyncClient.listObjectsV2(request);
            future.thenAccept(response -> {
                log.info("S3 连接测试成功, 区域: {}", region);
            }).exceptionally(e -> {
                log.error("S3 连接测试失败", e);
                return null;
            });
            
        } catch (Exception e) {
            log.error("AWS 凭证验证失败", e);
            throw new RuntimeException("AWS 凭证验证失败", e);
        }
    }
    
    /**
     * 获取当前凭证信息
     */
    public CredentialsInfo getCredentialsInfo() {
        AwsCredentials credentials = credentialsProvider.resolveCredentials();
        
        return CredentialsInfo.builder()
            .accessKeyId(maskString(credentials.accessKeyId()))
            .accountId(getAccountId(credentials))
            .region(region)
            .providerType(getProviderType())
            .build();
    }
    
    private String maskString(String input) {
        if (input == null || input.length() <= 4) {
            return "***";
        }
        return input.substring(0, 4) + "***" + input.substring(input.length() - 4);
    }
    
    private String getAccountId(AwsCredentials credentials) {
        // 从凭证中提取账户ID（如果可能）
        // 注意：这需要额外的权限和API调用
        return "Not Available";
    }
    
    private String getProviderType() {
        if (credentialsProvider instanceof StaticCredentialsProvider) {
            return "Static";
        } else if (credentialsProvider instanceof EnvironmentVariableCredentialsProvider) {
            return "Environment";
        } else if (credentialsProvider instanceof InstanceProfileCredentialsProvider) {
            return "EC2 Instance";
        } else if (credentialsProvider instanceof ContainerCredentialsProvider) {
            return "Container";
        } else if (credentialsProvider instanceof ProfileCredentialsProvider) {
            return "Profile";
        } else {
            return "Unknown";
        }
    }
    
    @lombok.Data
    @lombok.Builder
    public static class CredentialsInfo {
        private String accessKeyId;
        private String accountId;
        private String region;
        private String providerType;
    }
}
