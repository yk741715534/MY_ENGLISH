package com.example.s3scanner.config;

import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.autoconfigure.condition.ConditionalOnMissingBean;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import software.amazon.awssdk.auth.credentials.*;
import software.amazon.awssdk.awscore.client.builder.AwsClientBuilder;
import software.amazon.awssdk.core.client.config.ClientOverrideConfiguration;
import software.amazon.awssdk.core.retry.RetryPolicy;
import software.amazon.awssdk.core.retry.backoff.BackoffStrategy;
import software.amazon.awssdk.core.retry.backoff.EqualJitterBackoffStrategy;
import software.amazon.awssdk.core.retry.conditions.RetryCondition;
import software.amazon.awssdk.http.async.SdkAsyncHttpClient;
import software.amazon.awssdk.http.nio.netty.Http2Configuration;
import software.amazon.awssdk.http.nio.netty.NettyNioAsyncHttpClient;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.s3.S3AsyncClient;
import software.amazon.awssdk.services.s3.S3Client;
import software.amazon.awssdk.transfer.s3.S3TransferManager;
import software.amazon.awssdk.transfer.s3.S3TransferManagerOverrideConfiguration;
import software.amazon.awssdk.utils.StringUtils;

import java.net.URI;
import java.time.Duration;
import java.util.function.Consumer;

@Configuration
@Slf4j
public class AwsSdkConfig {
    
    // ========== AWS 凭证配置 ==========
    @Value("${aws.credentials.access-key-id:}")
    private String accessKeyId;
    
    @Value("${aws.credentials.secret-access-key:}")
    private String secretAccessKey;
    
    @Value("${aws.credentials.profile-file:~/.aws/credentials}")
    private String profileFile;
    
    @Value("${aws.credentials.profile-name:default}")
    private String profileName;
    
    // ========== 区域配置 ==========
    @Value("${aws.region:us-east-1}")
    private String region;
    
    // ========== S3 端点配置 ==========
    @Value("${aws.s3.endpoint:}")
    private String s3Endpoint;
    
    @Value("${aws.s3.path-style-access-enabled:false}")
    private boolean pathStyleAccessEnabled;
    
    @Value("${aws.s3.acceleration-mode-enabled:false}")
    private boolean accelerationModeEnabled;
    
    // ========== HTTP 客户端配置 ==========
    @Value("${aws.http-client.max-concurrency:100}")
    private int maxConcurrency;
    
    @Value("${aws.http-client.connection-timeout:10s}")
    private Duration connectionTimeout;
    
    @Value("${aws.http-client.read-timeout:30s}")
    private Duration readTimeout;
    
    @Value("${aws.http-client.write-timeout:30s}")
    private Duration writeTimeout;
    
    @Value("${aws.http-client.tcp-keep-alive:true}")
    private boolean tcpKeepAlive;
    
    @Value("${aws.http-client.max-pending-connection-acquires:1000}")
    private int maxPendingConnectionAcquires;
    
    // ========== 重试配置 ==========
    @Value("${aws.retry.max-attempts:3}")
    private int maxRetryAttempts;
    
    @Value("${aws.retry.base-delay:100ms}")
    private Duration baseDelay;
    
    @Value("${aws.retry.max-backoff-time:10s}")
    private Duration maxBackoffTime;
    
    // ========== 传输管理器配置 ==========
    @Value("${aws.transfer-manager.upload.multipart-threshold:8MB}")
    private Long multipartUploadThreshold;
    
    @Value("${aws.transfer-manager.upload.multipart-part-size:8MB}")
    private Long multipartPartSize;
    
    @Value("${aws.transfer-manager.upload.max-concurrency:10}")
    private int uploadMaxConcurrency;
    
    @Value("${aws.transfer-manager.download.max-concurrency:10}")
    private int downloadMaxConcurrency;
    
    /**
     * AWS 凭证提供者
     * 优先级: 1.配置文件凭证 2.环境变量 3.实例配置文件 4.默认凭证链
     */
    @Bean
    public AwsCredentialsProvider awsCredentialsProvider() {
        // 如果配置了 access-key-id 和 secret-access-key，使用配置的凭证
        if (StringUtils.isNotBlank(accessKeyId) && StringUtils.isNotBlank(secretAccessKey)) {
            log.info("使用配置文件中的 AWS 凭证");
            return StaticCredentialsProvider.create(
                AwsBasicCredentials.create(accessKeyId, secretAccessKey)
            );
        }
        
        // 否则使用默认凭证链
        log.info("使用 AWS 默认凭证链");
        return DefaultCredentialsProvider.builder()
            .profileFile(() -> profileFile)
            .profileName(profileName)
            .build();
    }
    
    /**
     * 异步 HTTP 客户端
     */
    @Bean
    public SdkAsyncHttpClient sdkAsyncHttpClient() {
        return NettyNioAsyncHttpClient.builder()
            .maxConcurrency(maxConcurrency)
            .connectionAcquisitionTimeout(connectionTimeout)
            .connectionTimeout(connectionTimeout)
            .readTimeout(readTimeout)
            .writeTimeout(writeTimeout)
            .tcpKeepAlive(tcpKeepAlive)
            .maxPendingConnectionAcquires(maxPendingConnectionAcquires)
            .http2Configuration(Http2Configuration.builder()
                .maxStreams(100L)
                .initialWindowSize(1_048_576)  // 1MB
                .build())
            .build();
    }
    
    /**
     * 重试策略
     */
    @Bean
    public RetryPolicy retryPolicy() {
        BackoffStrategy backoffStrategy = EqualJitterBackoffStrategy.builder()
            .baseDelay(baseDelay)
            .maxBackoffTime(maxBackoffTime)
            .build();
        
        return RetryPolicy.builder()
            .numRetries(maxRetryAttempts)
            .retryCondition(RetryCondition.defaultRetryCondition())
            .backoffStrategy(backoffStrategy)
            .throttlingBackoffStrategy(backoffStrategy)
            .build();
    }
    
    /**
     * 客户端覆盖配置
     */
    @Bean
    public ClientOverrideConfiguration clientOverrideConfiguration(RetryPolicy retryPolicy) {
        return ClientOverrideConfiguration.builder()
            .retryPolicy(retryPolicy)
            .apiCallTimeout(Duration.ofMinutes(5))
            .apiCallAttemptTimeout(Duration.ofSeconds(30))
            .build();
    }
    
    /**
     * S3 异步客户端
     */
    @Bean
    public S3AsyncClient s3AsyncClient(AwsCredentialsProvider credentialsProvider,
                                      SdkAsyncHttpClient httpClient,
                                      ClientOverrideConfiguration clientConfig) {
        S3AsyncClient.Builder builder = S3AsyncClient.builder()
            .region(Region.of(region))
            .credentialsProvider(credentialsProvider)
            .httpClient(httpClient)
            .overrideConfiguration(clientConfig)
            .serviceConfiguration(s3ServiceConfiguration());
        
        // 配置端点覆盖（用于测试或私有S3）
        if (StringUtils.isNotBlank(s3Endpoint)) {
            builder.endpointOverride(URI.create(s3Endpoint));
            log.info("配置 S3 端点: {}", s3Endpoint);
        }
        
        S3AsyncClient client = builder.build();
        log.info("S3AsyncClient 初始化完成，区域: {}", region);
        return client;
    }
    
    /**
     * S3 同步客户端
     */
    @Bean
    public S3Client s3Client(AwsCredentialsProvider credentialsProvider,
                           ClientOverrideConfiguration clientConfig) {
        S3Client.Builder builder = S3Client.builder()
            .region(Region.of(region))
            .credentialsProvider(credentialsProvider)
            .overrideConfiguration(clientConfig)
            .serviceConfiguration(s3ServiceConfiguration());
        
        if (StringUtils.isNotBlank(s3Endpoint)) {
            builder.endpointOverride(URI.create(s3Endpoint));
        }
        
        S3Client client = builder.build();
        log.info("S3Client 初始化完成，区域: {}", region);
        return client;
    }
    
    /**
     * S3 服务配置
     */
    private Consumer<software.amazon.awssdk.services.s3.S3Configuration.Builder> s3ServiceConfiguration() {
        return builder -> builder
            .pathStyleAccessEnabled(pathStyleAccessEnabled)
            .accelerateModeEnabled(accelerationModeEnabled)
            .chunkedEncodingEnabled(true)
            .checksumValidationEnabled(true);
    }
    
    /**
     * S3 传输管理器
     */
    @Bean
    public S3TransferManager s3TransferManager(S3AsyncClient s3AsyncClient) {
        S3TransferManagerOverrideConfiguration overrideConfig = 
            S3TransferManagerOverrideConfiguration.builder()
                .uploadDirectoryConfiguration(builder -> builder
                    .maxConcurrency(uploadMaxConcurrency)
                    .multipartUploadThreshold(multipartUploadThreshold)
                    .multipartPartSize(multipartPartSize)
                    .build())
                .downloadDirectoryConfiguration(builder -> builder
                    .maxConcurrency(downloadMaxConcurrency)
                    .build())
                .build();
        
        return S3TransferManager.builder()
            .s3Client(s3AsyncClient)
            .transferConfiguration(builder -> builder
                .uploadDirectoryOverrideConfiguration(overrideConfig.uploadDirectoryConfiguration())
                .downloadDirectoryOverrideConfiguration(overrideConfig.downloadDirectoryConfiguration())
            )
            .build();
    }
    
    /**
     * 可选的：AWS SDK 指标发布者
     */
    @Bean
    @ConditionalOnProperty(name = "aws.metrics.enabled", havingValue = "true")
    public software.amazon.awssdk.metrics.MetricPublisher s3MetricPublisher() {
        return new software.amazon.awssdk.metrics.LoggingMetricPublisher();
    }
    
    /**
     * 可选的：本地开发凭证提供者
     */
    @Bean
    @ConditionalOnProperty(name = "aws.credentials.type", havingValue = "local")
    @ConditionalOnMissingBean
    public AwsCredentialsProvider localCredentialsProvider() {
        log.warn("使用本地开发凭证，不应用于生产环境");
        
        // 本地开发使用固定凭证
        return StaticCredentialsProvider.create(
            AwsBasicCredentials.create("test-access-key", "test-secret-key")
        );
    }
    
    /**
     * 可选的：从环境变量获取凭证
     */
    @Bean
    @ConditionalOnProperty(name = "aws.credentials.source", havingValue = "environment")
    public AwsCredentialsProvider environmentCredentialsProvider() {
        log.info("从环境变量获取 AWS 凭证");
        
        return EnvironmentVariableCredentialsProvider.create();
    }
    
    /**
     * 可选的：从系统属性获取凭证
     */
    @Bean
    @ConditionalOnProperty(name = "aws.credentials.source", havingValue = "system")
    public AwsCredentialsProvider systemPropertyCredentialsProvider() {
        log.info("从系统属性获取 AWS 凭证");
        
        return SystemPropertyCredentialsProvider.create();
    }
    
    /**
     * 可选的：从容器角色获取凭证（ECS/EKS）
     */
    @Bean
    @ConditionalOnProperty(name = "aws.credentials.source", havingValue = "container")
    public AwsCredentialsProvider containerCredentialsProvider() {
        log.info("从容器角色获取 AWS 凭证");
        
        return ContainerCredentialsProvider.builder().build();
    }
    
    /**
     * 可选的：从实例元数据获取凭证（EC2）
     */
    @Bean
    @ConditionalOnProperty(name = "aws.credentials.source", havingValue = "instance")
    public AwsCredentialsProvider instanceCredentialsProvider() {
        log.info("从实例元数据获取 AWS 凭证");
        
        return InstanceProfileCredentialsProvider.builder().build();
    }
}
