package com.example.s3scanner.config;

import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.core.env.Environment;
import software.amazon.awssdk.auth.credentials.*;
import software.amazon.awssdk.profiles.ProfileFile;

import java.util.Arrays;
import java.util.List;
import java.util.Optional;

@Configuration
@Slf4j
public class AdvancedAwsCredentialsConfig {
    
    private final Environment environment;
    
    public AdvancedAwsCredentialsConfig(Environment environment) {
        this.environment = environment;
    }
    
    @Bean
    public AwsCredentialsProviderChain awsCredentialsProviderChain() {
        List<AwsCredentialsProvider> providers = Arrays.asList(
            // 1. 系统属性凭证
            systemPropertyCredentialsProvider(),
            
            // 2. 环境变量凭证
            environmentVariableCredentialsProvider(),
            
            // 3. 配置文件凭证
            profileFileCredentialsProvider(),
            
            // 4. Web Identity Token 凭证 (EKS/OIDC)
            webIdentityTokenFileCredentialsProvider(),
            
            // 5. 容器凭证 (ECS)
            containerCredentialsProvider(),
            
            // 6. 实例元数据凭证 (EC2)
            instanceProfileCredentialsProvider(),
            
            // 7. 默认凭证链
            defaultCredentialsProvider()
        );
        
        return AwsCredentialsProviderChain.of(providers.toArray(new AwsCredentialsProvider[0]));
    }
    
    private AwsCredentialsProvider systemPropertyCredentialsProvider() {
        return SystemPropertyCredentialsProvider.create();
    }
    
    private AwsCredentialsProvider environmentVariableCredentialsProvider() {
        return EnvironmentVariableCredentialsProvider.create();
    }
    
    private AwsCredentialsProvider profileFileCredentialsProvider() {
        String profileName = environment.getProperty("aws.credentials.profile-name", "default");
        
        return ProfileCredentialsProvider.builder()
            .profileFile(ProfileFile::defaultProfileFile)
            .profileName(profileName)
            .build();
    }
    
    private AwsCredentialsProvider webIdentityTokenFileCredentialsProvider() {
        return WebIdentityTokenFileCredentialsProvider.builder().build();
    }
    
    private AwsCredentialsProvider containerCredentialsProvider() {
        return ContainerCredentialsProvider.builder().build();
    }
    
    private AwsCredentialsProvider instanceProfileCredentialsProvider() {
        return InstanceProfileCredentialsProvider.builder().build();
    }
    
    private AwsCredentialsProvider defaultCredentialsProvider() {
        return DefaultCredentialsProvider.create();
    }
    
    /**
     * 安全的凭证获取
     */
    public AwsCredentials getAwsCredentials(AwsCredentialsProvider provider) {
        try {
            return provider.resolveCredentials();
        } catch (Exception e) {
            log.error("获取 AWS 凭证失败", e);
            throw new RuntimeException("无法获取 AWS 凭证", e);
        }
    }
}
