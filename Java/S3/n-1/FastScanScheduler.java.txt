// FastScanScheduler.java
package com.example.s3scanner.scheduler;

import com.example.s3scanner.model.dto.ScanResult;
import com.example.s3scanner.service.processor.S3ProcessorService;
import com.example.s3scanner.service.scanner.DistributedScanner;
import com.example.s3scanner.service.monitor.MetricsCollector;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;
import org.springframework.util.StopWatch;

import java.time.LocalDateTime;
import java.util.List;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;

@Component
@Slf4j
@RequiredArgsConstructor
public class FastScanScheduler {
    
    private final DistributedScanner distributedScanner;
    private final S3ProcessorService processorService;
    private final MetricsCollector metricsCollector;
    
    @Value("${app.scanner.interval-ms}")
    private long scanIntervalMs;
    
    @Value("${app.scanner.initial-delay-ms}")
    private long initialDelayMs;
    
    @Value("${app.scanner.max-scan-time-ms}")
    private long maxScanTimeMs;
    
    @Value("${app.scanner.scan-prefixes}")
    private List<String> scanPrefixes;
    
    private final ExecutorService scanExecutor = Executors.newVirtualThreadPerTaskExecutor();
    private volatile boolean scanning = false;
    private volatile LocalDateTime lastScanTime = LocalDateTime.now();
    
    /**
     * 3秒快速扫描
     */
    @Scheduled(fixedDelayString = "${app.scanner.interval-ms}", 
              initialDelayString = "${app.scanner.initial-delay-ms}")
    public void fastScan() {
        if (scanning) {
            log.debug("上次扫描还未完成，跳过本次");
            return;
        }
        
        scanning = true;
        StopWatch stopWatch = new StopWatch("fast-scan");
        stopWatch.start();
        
        try {
            // 并行扫描所有前缀
            List<CompletableFuture<ScanResult>> scanFutures = scanPrefixes.stream()
                .map(prefix -> CompletableFuture.supplyAsync(() -> 
                    distributedScanner.scanPrefixDistributed(prefix).join(), scanExecutor))
                .toList();
            
            // 等待所有扫描完成
            CompletableFuture.allOf(scanFutures.toArray(new CompletableFuture[0]))
                .orTimeout(maxScanTimeMs, TimeUnit.MILLISECONDS)
                .thenRun(() -> {
                    // 处理扫描结果
                    processScanResults(scanFutures);
                })
                .join();
            
        } catch (Exception e) {
            log.error("快速扫描异常", e);
            metricsCollector.recordSchedulerError("fastScan", e);
        } finally {
            scanning = false;
            stopWatch.stop();
            lastScanTime = LocalDateTime.now();
            
            long duration = stopWatch.getTotalTimeMillis();
            if (duration > scanIntervalMs) {
                log.warn("扫描耗时 {}ms 超过扫描间隔 {}ms", duration, scanIntervalMs);
            }
            
            metricsCollector.recordSchedulerExecution("fastScan", duration, true);
        }
    }
    
    /**
     * 处理扫描结果
     */
    private void processScanResults(List<CompletableFuture<ScanResult>> scanFutures) {
        AtomicInteger totalNewObjects = new AtomicInteger(0);
        
        scanFutures.forEach(future -> {
            try {
                ScanResult scanResult = future.get(5, TimeUnit.SECONDS);
                
                if (scanResult.isSuccess() && !scanResult.getNewObjects().isEmpty()) {
                    int newObjectsCount = scanResult.getNewObjects().size();
                    totalNewObjects.addAndGet(newObjectsCount);
                    
                    log.info("扫描前缀 {} 发现 {} 个新文件", 
                        scanResult.getPrefix(), newObjectsCount);
                    
                    // 批量处理新文件
                    processorService.batchProcessObjects(scanResult.getNewObjects())
                        .thenAccept(results -> {
                            int successCount = (int) results.stream()
                                .filter(r -> r != null && r.isSuccess())
                                .count();
                            
                            log.info("批量处理完成: 成功 {}/{} 个文件", 
                                successCount, newObjectsCount);
                        })
                        .exceptionally(e -> {
                            log.error("批量处理异常: {}", scanResult.getPrefix(), e);
                            return null;
                        });
                }
                
            } catch (Exception e) {
                log.error("处理扫描结果异常", e);
            }
        });
        
        if (totalNewObjects.get() > 0) {
            log.info("本次扫描共发现 {} 个新文件", totalNewObjects.get());
        }
    }
    
    /**
     * 健康检查扫描
     */
    @Scheduled(fixedDelay = 300000)  // 每5分钟
    public void healthCheckScan() {
        try {
            log.info("执行健康检查扫描...");
            
            // 检查扫描器状态
            boolean isHealthy = isScannerHealthy();
            
            if (!isHealthy) {
                log.warn("扫描器健康检查失败");
                metricsCollector.recordHealthCheckFailure("scanner");
            } else {
                log.info("扫描器健康检查通过");
            }
            
        } catch (Exception e) {
            log.error("健康检查扫描异常", e);
        }
    }
    
    /**
     * 检查扫描器健康状态
     */
    private boolean isScannerHealthy() {
        // 检查最后扫描时间
        long timeSinceLastScan = java.time.Duration.between(
            lastScanTime, LocalDateTime.now()
        ).toMillis();
        
        if (timeSinceLastScan > scanIntervalMs * 3) {
            log.warn("扫描器可能已停止，最后扫描时间: {}，已过去 {}ms", 
                lastScanTime, timeSinceLastScan);
            return false;
        }
        
        // 检查扫描任务是否卡住
        if (scanning && timeSinceLastScan > maxScanTimeMs * 2) {
            log.warn("扫描任务可能卡住，已运行超过 {}ms", maxScanTimeMs * 2);
            return false;
        }
        
        return true;
    }
    
    /**
     * 优雅关闭
     */
    public void shutdown() {
        scanExecutor.shutdown();
        try {
            if (!scanExecutor.awaitTermination(10, TimeUnit.SECONDS)) {
                scanExecutor.shutdownNow();
            }
        } catch (InterruptedException e) {
            scanExecutor.shutdownNow();
            Thread.currentThread().interrupt();
        }
    }
}
