// DistributedScanner.java
package com.example.s3scanner.service.scanner;

import com.example.s3scanner.model.domain.S3ObjectInfo;
import com.example.s3scanner.model.dto.ScanResult;
import com.example.s3scanner.service.coordinator.DistributedLockService;
import com.example.s3scanner.service.coordinator.InstanceRegistry;
import com.example.s3scanner.service.monitor.MetricsCollector;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.cache.annotation.CacheEvict;
import org.springframework.cache.annotation.Cacheable;
import org.springframework.retry.annotation.Backoff;
import org.springframework.retry.annotation.Retryable;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;
import software.amazon.awssdk.services.s3.S3AsyncClient;
import software.amazon.awssdk.services.s3.model.*;

import java.time.Duration;
import java.time.Instant;
import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.stream.Collectors;

@Service
@Slf4j
@RequiredArgsConstructor
public class DistributedScanner {
    
    private final S3AsyncClient s3Client;
    private final DistributedLockService lockService;
    private final InstanceRegistry instanceRegistry;
    private final MetricsCollector metricsCollector;
    
    @Value("${app.scanner.interval-ms}")
    private long scanIntervalMs;
    
    @Value("${app.scanner.max-scan-time-ms}")
    private long maxScanTimeMs;
    
    @Value("${app.s3.page-size}")
    private int pageSize;
    
    @Value("${app.scanner.scan-prefixes}")
    private List<String> scanPrefixes;
    
    @Value("${app.s3.bucket}")
    private String bucket;
    
    private final Map<String, Instant> lastScanTime = new ConcurrentHashMap<>();
    private final Map<String, String> etagCache = new ConcurrentHashMap<>();
    private final ScheduledExecutorService scheduler = Executors.newScheduledThreadPool(5);
    
    /**
     * 启动分布式扫描
     */
    public void startDistributedScanning() {
        log.info("启动分布式扫描，扫描间隔: {}ms", scanIntervalMs);
        
        for (String prefix : scanPrefixes) {
            scheduler.scheduleAtFixedRate(() -> {
                try {
                    scanPrefixDistributed(prefix);
                } catch (Exception e) {
                    log.error("扫描前缀失败: {}", prefix, e);
                    metricsCollector.recordScanError(prefix);
                }
            }, 0, scanIntervalMs, TimeUnit.MILLISECONDS);
        }
    }
    
    /**
     * 分布式扫描前缀
     */
    @Async
    public CompletableFuture<ScanResult> scanPrefixDistributed(String prefix) {
        String lockKey = String.format("scan:%s:%s", bucket, prefix);
        long startTime = System.currentTimeMillis();
        
        // 获取扫描锁
        if (!lockService.acquireLock(lockKey, "scan", scanIntervalMs + 1000)) {
            log.debug("无法获取扫描锁，跳过本次扫描: {}", prefix);
            return CompletableFuture.completedFuture(
                ScanResult.skipped(prefix, "Lock acquisition failed")
            );
        }
        
        try {
            Instant lastScan = lastScanTime.getOrDefault(prefix, Instant.MIN);
            ScanResult result = incrementalScan(prefix, lastScan);
            
            // 更新最后扫描时间
            lastScanTime.put(prefix, Instant.now());
            
            // 记录指标
            long duration = System.currentTimeMillis() - startTime;
            metricsCollector.recordScan(prefix, result.getNewObjects().size(), 
                result.getScannedObjects(), duration, result.isSuccess());
            
            log.debug("扫描完成: {}, 发现 {} 个新文件, 耗时: {}ms", 
                prefix, result.getNewObjects().size(), duration);
            
            return CompletableFuture.completedFuture(result);
            
        } finally {
            lockService.releaseLock(lockKey, "scan");
        }
    }
    
    /**
     * 增量扫描
     */
    @Retryable(
        retryFor = {Exception.class},
        maxAttempts = 3,
        backoff = @Backoff(delay = 1000, multiplier = 2)
    )
    public ScanResult incrementalScan(String prefix, Instant lastScanTime) {
        List<S3ObjectInfo> newObjects = new CopyOnWriteArrayList<>();
        AtomicInteger totalScanned = new AtomicInteger(0);
        String continuationToken = null;
        
        ExecutorService executor = Executors.newVirtualThreadPerTaskExecutor();
        
        try {
            do {
                if (System.currentTimeMillis() - lastScanTime.toEpochMilli() > maxScanTimeMs) {
                    log.warn("扫描超时，提前结束: {}", prefix);
                    break;
                }
                
                // 构建请求
                ListObjectsV2Request.Builder requestBuilder = ListObjectsV2Request.builder()
                    .bucket(bucket)
                    .prefix(prefix)
                    .maxKeys(pageSize);
                
                if (continuationToken != null) {
                    requestBuilder.continuationToken(continuationToken);
                }
                
                // 异步执行扫描
                CompletableFuture<ListObjectsV2Response> future = s3Client.listObjectsV2(
                    requestBuilder.build()
                );
                
                ListObjectsV2Response response = future.get(10, TimeUnit.SECONDS);
                
                // 处理本页结果
                List<CompletableFuture<Void>> processingFutures = response.contents().stream()
                    .filter(this::shouldProcess)
                    .map(s3Object -> CompletableFuture.runAsync(() -> {
                        processObject(s3Object, prefix, newObjects, lastScanTime);
                    }, executor))
                    .collect(Collectors.toList());
                
                // 等待本页处理完成
                CompletableFuture.allOf(processingFutures.toArray(new CompletableFuture[0]))
                    .get(5, TimeUnit.SECONDS);
                
                totalScanned.addAndGet(response.contents().size());
                continuationToken = response.nextContinuationToken();
                
                // 如果发现大量新文件，提前分批处理
                if (newObjects.size() >= 50) {
                    log.info("发现足够的新文件，提前结束扫描: {} 个", newObjects.size());
                    break;
                }
                
            } while (continuationToken != null);
            
            return ScanResult.builder()
                .prefix(prefix)
                .scannedObjects(totalScanned.get())
                .newObjects(new ArrayList<>(newObjects))
                .success(true)
                .incremental(true)
                .lastScanTime(lastScanTime)
                .scanTime(Instant.now())
                .build();
            
        } catch (Exception e) {
            log.error("增量扫描失败: {}", prefix, e);
            return ScanResult.builder()
                .prefix(prefix)
                .scannedObjects(totalScanned.get())
                .newObjects(new ArrayList<>(newObjects))
                .success(false)
                .errorMessage(e.getMessage())
                .build();
        } finally {
            executor.shutdown();
        }
    }
    
    /**
     * 处理单个对象
     */
    private void processObject(S3Object s3Object, String prefix, 
                              List<S3ObjectInfo> newObjects, Instant lastScanTime) {
        try {
            // 跳过目录
            if (s3Object.key().endsWith("/")) {
                return;
            }
            
            // 检查是否应该由本实例处理
            if (!instanceRegistry.shouldProcess(bucket, s3Object.key())) {
                return;
            }
            
            // 检查文件是否被修改
            String cachedEtag = etagCache.get(s3Object.key());
            String currentEtag = s3Object.eTag();
            
            boolean isModified = cachedEtag == null || !cachedEtag.equals(currentEtag);
            boolean isNew = s3Object.lastModified().isAfter(lastScanTime);
            
            if (isModified || isNew) {
                // 创建对象信息
                S3ObjectInfo objectInfo = S3ObjectInfo.builder()
                    .id(String.format("%s:%s", bucket, s3Object.key()))
                    .bucket(bucket)
                    .key(s3Object.key())
                    .etag(currentEtag)
                    .size(s3Object.size())
                    .lastModified(Date.from(s3Object.lastModified()))
                    .prefix(prefix)
                    .status(S3ObjectInfo.ProcessingStatus.PENDING)
                    .createdAt(new Date())
                    .metadata(new HashMap<>())
                    .build();
                
                newObjects.add(objectInfo);
                
                // 更新缓存
                etagCache.put(s3Object.key(), currentEtag);
                
                // 清理过期的缓存
                if (etagCache.size() > 10000) {
                    cleanupEtagCache();
                }
            }
            
        } catch (Exception e) {
            log.warn("处理对象失败: {}", s3Object.key(), e);
        }
    }
    
    /**
     * 检查是否应该处理
     */
    private boolean shouldProcess(S3Object s3Object) {
        // 跳过目录
        if (s3Object.key().endsWith("/")) {
            return false;
        }
        
        // 检查文件大小（可选过滤）
        if (s3Object.size() > 10L * 1024 * 1024 * 1024) {  // 10GB
            log.warn("文件过大，跳过: {} ({} bytes)", s3Object.key(), s3Object.size());
            return false;
        }
        
        return true;
    }
    
    /**
     * 清理ETag缓存
     */
    @CacheEvict(value = "etagCache", allEntries = true)
    public void cleanupEtagCache() {
        // 清理最老的50%缓存
        int targetSize = etagCache.size() / 2;
        if (targetSize > 0) {
            List<Map.Entry<String, String>> entries = new ArrayList<>(etagCache.entrySet());
            entries.sort(Comparator.comparing(Map.Entry::getKey));
            
            for (int i = 0; i < targetSize && i < entries.size(); i++) {
                etagCache.remove(entries.get(i).getKey());
            }
        }
    }
    
    /**
     * 获取ETag缓存
     */
    @Cacheable(value = "etagCache", key = "#key")
    public String getCachedEtag(String key) {
        return etagCache.get(key);
    }
    
    /**
     * 停止扫描
     */
    public void stopScanning() {
        scheduler.shutdown();
        try {
            if (!scheduler.awaitTermination(5, TimeUnit.SECONDS)) {
                scheduler.shutdownNow();
            }
        } catch (InterruptedException e) {
            scheduler.shutdownNow();
            Thread.currentThread().interrupt();
        }
    }
}
