# 应用配置
spring:
  application:
    name: s3-realtime-scanner

  # MongoDB配置
  data:
    mongodb:
      uri: ${MONGODB_URI:mongodb://localhost:27017/s3-scanner}
      auto-index-creation: true

  # 缓存配置
  cache:
    type: ehcache
    ehcache:
      config: classpath:ehcache.xml

  # 线程池配置
  task:
    execution:
      pool:
        core-size: 20
        max-size: 100
        queue-capacity: 1000
        keep-alive: 60s
      thread-name-prefix: "async-task-"

    scheduling:
      pool:
        size: 10
      thread-name-prefix: "scheduler-"

# 应用实例配置
app:
  instance:
    id: ${HOSTNAME:${spring.application.name}-${random.uuid}}
    concurrency-limit: 50
    heartbeat-interval: 30
    max-processing-time: 300000  # 5分钟

  # 扫描配置
  scanner:
    enabled: true
    interval-ms: 3000
    initial-delay-ms: 10000
    max-scan-time-ms: 2500
    incremental-scan: true
    etag-cache-size: 10000
    etag-cache-ttl: 3600
    scan-prefixes: uploads/,pending/,incoming/
    max-scan-depth: 1000

  # S3配置
  s3:
    bucket: ${S3_BUCKET:my-bucket}
    page-size: 100
    max-concurrent-scans: 5

  # 处理配置
  processor:
    download-dir: /tmp/s3-downloads
    max-concurrent-downloads: 10
    download-timeout-ms: 30000
    process-timeout-ms: 60000
    max-retries: 3
    retry-delay-ms: 1000
    retry-multiplier: 2.0
    max-retry-delay-ms: 10000

  # 文件移动配置
  mover:
    enabled: true
    strategy: ARCHIVE_AND_BACKUP
    archive-prefix: archive/
    backup-prefix: backup/
    failed-prefix: failed/
    temp-prefix: temp/
    add-timestamp: true
    compress-large-files: true
    compress-threshold-mb: 10
    compression-level: 6
    verify-move: true
    verify-retry-count: 3
    keep-source: false

  # 分布式锁配置
  lock:
    coordination-ttl-ms: 10000
    process-ttl-ms: 30000
    cleanup-ttl-ms: 60000
    retry-interval-ms: 100
    max-retries: 3
    lock-collection: distributed_locks

  # 监控配置
  monitor:
    metrics-enabled: true
    health-check-interval: 30000
    alert-enabled: true
    alert-thresholds:
      error-rate: 5.0
      processing-time: 10000
      queue-size: 1000

# AWS配置
aws:
  region: ${AWS_REGION:us-east-1}
  credentials:
    access-key-id: ${AWS_ACCESS_KEY_ID:}
    secret-access-key: ${AWS_SECRET_ACCESS_KEY:}
    profile-file: ${AWS_PROFILE_FILE:~/.aws/credentials}
    profile-name: ${AWS_PROFILE:default}

  s3:
    endpoint: ${AWS_S3_ENDPOINT:}
    path-style-access-enabled: ${AWS_S3_PATH_STYLE_ACCESS:false}
    acceleration-mode-enabled: ${AWS_S3_ACCELERATION_MODE:false}

  http-client:
    max-concurrency: 100
    connection-timeout: 10s
    read-timeout: 30s
    write-timeout: 30s
    tcp-keep-alive: true
    max-pending-connection-acquires: 1000

  retry:
    max-attempts: 3
    base-delay: 100ms
    max-backoff-time: 10s

  transfer-manager:
    upload:
      multipart-threshold: 8MB
      multipart-part-size: 8MB
      max-concurrency: 10
    download:
      max-concurrency: 10

# 监控配置
management:
  endpoints:
    web:
      exposure:
        include: health,metrics,prometheus,info,loggers,threaddump,heapdump
  endpoint:
    health:
      show-details: always
      probes:
        enabled: true
  metrics:
    export:
      prometheus:
        enabled: true
  tracing:
    sampling:
      probability: 1.0

# 日志配置
logging:
  level:
    com.example.s3scanner: INFO
    org.springframework.data.mongodb: WARN
    software.amazon.awssdk: WARN
  file:
    name: logs/s3-scanner.log
  logback:
    rollingpolicy:
      max-file-size: 10MB
      max-history: 30
  pattern:
    console: "%d{yyyy-MM-dd HH:mm:ss} [%thread] %-5level %logger{36} - %msg%n"
    file: "%d{yyyy-MM-dd HH:mm:ss} [%thread] %-5level %logger{36} - %msg%n"
