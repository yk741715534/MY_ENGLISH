// MetricsCollector.java
package com.example.s3scanner.service.monitor;

import io.micrometer.core.instrument.*;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;
import javax.annotation.PostConstruct;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.atomic.AtomicInteger;

@Component
@Slf4j
@RequiredArgsConstructor
public class MetricsCollector {
    
    private final MeterRegistry meterRegistry;
    
    // 自定义指标存储
    private final Map<String, AtomicLong> customCounters = new ConcurrentHashMap<>();
    private final Map<String, AtomicLong> customGauges = new ConcurrentHashMap<>();
    private final Map<String, Timer> customTimers = new ConcurrentHashMap<>();
    
    @PostConstruct
    public void init() {
        // 初始化自定义指标
        initCustomMetrics();
    }
    
    /**
     * 初始化自定义指标
     */
    private void initCustomMetrics() {
        // 扫描相关指标
        registerCounter("s3.scan.total", "总扫描次数");
        registerCounter("s3.scan.success", "成功扫描次数");
        registerCounter("s3.scan.failed", "失败扫描次数");
        registerCounter("s3.scan.objects.total", "扫描对象总数");
        registerCounter("s3.scan.objects.new", "发现新对象数");
        
        // 处理相关指标
        registerCounter("s3.process.total", "总处理次数");
        registerCounter("s3.process.success", "成功处理次数");
        registerCounter("s3.process.failed", "失败处理次数");
        registerCounter("s3.process.retry", "重试次数");
        
        // 移动相关指标
        registerCounter("s3.move.total", "总移动次数");
        registerCounter("s3.move.success", "成功移动次数");
        registerCounter("s3.move.failed", "失败移动次数");
        
        // 锁相关指标
        registerGauge("s3.lock.active", "活跃锁数量");
        registerGauge("s3.lock.waiting", "等待锁数量");
        
        // 队列相关指标
        registerGauge("s3.queue.pending", "待处理队列大小");
        registerGauge("s3.queue.processing", "处理中队列大小");
        
        // 性能指标
        registerTimer("s3.scan.duration", "扫描耗时");
        registerTimer("s3.process.duration", "处理耗时");
        registerTimer("s3.move.duration", "移动耗时");
        registerTimer("s3.lock.acquisition.duration", "锁获取耗时");
    }
    
    /**
     * 记录扫描
     */
    public void recordScan(String prefix, int newObjects, int scannedObjects, 
                          long durationMs, boolean success) {
        // 记录计数器
        incrementCounter("s3.scan.total");
        incrementCounter("s3.scan.objects.total", scannedObjects);
        incrementCounter("s3.scan.objects.new", newObjects);
        
        if (success) {
            incrementCounter("s3.scan.success");
        } else {
            incrementCounter("s3.scan.failed");
        }
        
        // 记录耗时
        recordTimer("s3.scan.duration", durationMs);
        
        // 记录标签化指标
        meterRegistry.counter("s3.scan.prefix", 
            "prefix", prefix,
            "success", String.valueOf(success)
        ).increment();
        
        meterRegistry.counter("s3.objects.scanned.prefix",
            "prefix", prefix
        ).increment(scannedObjects);
        
        // 记录性能指标
        if (durationMs > 0) {
            double objectsPerSecond = (double) scannedObjects / (durationMs / 1000.0);
            meterRegistry.gauge("s3.scan.rate", 
                Tags.of("prefix", prefix), 
                objectsPerSecond);
        }
        
        log.debug("记录扫描指标: prefix={}, new={}, scanned={}, duration={}ms, success={}",
            prefix, newObjects, scannedObjects, durationMs, success);
    }
    
    /**
     * 记录处理成功
     */
    public void recordProcessingSuccess(String bucket, String key, 
                                       long durationMs, int dataSize) {
        incrementCounter("s3.process.total");
        incrementCounter("s3.process.success");
        
        recordTimer("s3.process.duration", durationMs);
        
        meterRegistry.counter("s3.process.bucket",
            "bucket", bucket
        ).increment();
        
        if (durationMs > 0) {
            meterRegistry.gauge("s3.process.throughput",
                Tags.of("bucket", bucket),
                (double) dataSize / (durationMs / 1000.0)
            );
        }
        
        // 记录处理延迟
        long now = System.currentTimeMillis();
        // 这里可以添加文件创建时间的逻辑来计算延迟
        
        log.debug("记录处理成功: {}/{}, duration={}ms, dataSize={}",
            bucket, key, durationMs, dataSize);
    }
    
    /**
     * 记录处理失败
     */
    public void recordProcessingFailure(String bucket, String key, 
                                       long durationMs, String error) {
        incrementCounter("s3.process.total");
        incrementCounter("s3.process.failed");
        
        recordTimer("s3.process.duration", durationMs);
        
        meterRegistry.counter("s3.process.error",
            "bucket", bucket,
            "error", extractErrorType(error)
        ).increment();
        
        log.debug("记录处理失败: {}/{}, duration={}ms, error={}",
            bucket, key, durationMs, error);
    }
    
    /**
     * 记录移动
     */
    public void recordMove(String bucket, String key, String moveType, 
                          long durationMs, boolean success, long fileSize) {
        incrementCounter("s3.move.total");
        
        if (success) {
            incrementCounter("s3.move.success");
        } else {
            incrementCounter("s3.move.failed");
        }
        
        recordTimer("s3.move.duration", durationMs);
        
        meterRegistry.counter("s3.move.operation",
            "bucket", bucket,
            "type", moveType,
            "success", String.valueOf(success)
        ).increment();
        
        if (success && fileSize > 0 && durationMs > 0) {
            double transferRate = (double) fileSize / (durationMs / 1000.0) / 1024 / 1024;  // MB/s
            meterRegistry.gauge("s3.move.rate",
                Tags.of("bucket", bucket, "type", moveType),
                transferRate
            );
        }
        
        log.debug("记录移动: {}/{}, type={}, duration={}ms, success={}, size={}",
            bucket, key, moveType, durationMs, success, fileSize);
    }
    
    /**
     * 记录锁获取
     */
    public void recordLockAcquisition(String lockKey, String resource, 
                                     long durationMs, boolean success) {
        recordTimer("s3.lock.acquisition.duration", durationMs);
        
        meterRegistry.counter("s3.lock.acquisition",
            "resource", resource,
            "success", String.valueOf(success)
        ).increment();
        
        // 更新活跃锁数量
        if (success) {
            updateGauge("s3.lock.active", 1);
        }
        
        log.debug("记录锁获取: {}, resource={}, duration={}ms, success={}",
            lockKey, resource, durationMs, success);
    }
    
    /**
     * 记录锁释放
     */
    public void recordLockRelease(String lockKey, String resource, boolean success) {
        meterRegistry.counter("s3.lock.release",
            "resource", resource,
            "success", String.valueOf(success)
        ).increment();
        
        // 更新活跃锁数量
        if (success) {
            updateGauge("s3.lock.active", -1);
        }
        
        log.debug("记录锁释放: {}, resource={}, success={}", lockKey, resource, success);
    }
    
    /**
     * 记录扫描错误
     */
    public void recordScanError(String prefix) {
        meterRegistry.counter("s3.scan.error",
            "prefix", prefix
        ).increment();
        
        log.debug("记录扫描错误: prefix={}", prefix);
    }
    
    /**
     * 记录调度器执行
     */
    public void recordSchedulerExecution(String schedulerName, long durationMs, boolean success) {
        meterRegistry.counter("s3.scheduler.execution",
            "scheduler", schedulerName,
            "success", String.valueOf(success)
        ).increment();
        
        meterRegistry.timer("s3.scheduler.duration",
            "scheduler", schedulerName
        ).record(durationMs, TimeUnit.MILLISECONDS);
        
        log.debug("记录调度器执行: {}, duration={}ms, success={}",
            schedulerName, durationMs, success);
    }
    
    /**
     * 记录调度器错误
     */
    public void recordSchedulerError(String schedulerName, Exception e) {
        meterRegistry.counter("s3.scheduler.error",
            "scheduler", schedulerName,
            "error", e.getClass().getSimpleName()
        ).increment();
        
        log.debug("记录调度器错误: {}, error={}", schedulerName, e.getMessage());
    }
    
    /**
     * 记录健康检查失败
     */
    public void recordHealthCheckFailure(String component) {
        meterRegistry.counter("s3.health.check.failure",
            "component", component
        ).increment();
        
        log.debug("记录健康检查失败: component={}", component);
    }
    
    /**
     * 注册计数器
     */
    private void registerCounter(String name, String description) {
        Counter.builder(name)
            .description(description)
            .register(meterRegistry);
        
        customCounters.put(name, new AtomicLong(0));
    }
    
    /**
     * 注册仪表
     */
    private void registerGauge(String name, String description) {
        AtomicLong gauge = new AtomicLong(0);
        
        Gauge.builder(name, gauge, AtomicLong::get)
            .description(description)
            .register(meterRegistry);
        
        customGauges.put(name, gauge);
    }
    
    /**
     * 注册计时器
     */
    private void registerTimer(String name, String description) {
        Timer timer = Timer.builder(name)
            .description(description)
            .register(meterRegistry);
        
        customTimers.put(name, timer);
    }
    
    /**
     * 增加计数器
     */
    private void incrementCounter(String name) {
        incrementCounter(name, 1);
    }
    
    private void incrementCounter(String name, long amount) {
        AtomicLong counter = customCounters.get(name);
        if (counter != null) {
            counter.addAndGet(amount);
        }
        
        meterRegistry.counter(name).increment(amount);
    }
    
    /**
     * 更新仪表
     */
    private void updateGauge(String name, long delta) {
        AtomicLong gauge = customGauges.get(name);
        if (gauge != null) {
            gauge.addAndGet(delta);
        }
    }
    
    /**
     * 记录计时器
     */
    private void recordTimer(String name, long durationMs) {
        Timer timer = customTimers.get(name);
        if (timer != null) {
            timer.record(durationMs, TimeUnit.MILLISECONDS);
        }
    }
    
    /**
     * 提取错误类型
     */
    private String extractErrorType(String error) {
        if (error == null) {
            return "Unknown";
        }
        
        if (error.contains("Timeout")) {
            return "Timeout";
        } else if (error.contains("Connection")) {
            return "Connection";
        } else if (error.contains("Access Denied")) {
            return "AccessDenied";
        } else if (error.contains("NoSuchKey")) {
            return "NotFound";
        } else if (error.contains("Throttling")) {
            return "Throttling";
        } else {
            return "Other";
        }
    }
    
    /**
     * 获取指标快照
     */
    public Map<String, Object> getMetricsSnapshot() {
        Map<String, Object> snapshot = new java.util.HashMap<>();
        
        // 收集计数器
        Map<String, Long> counters = new java.util.HashMap<>();
        customCounters.forEach((name, counter) -> 
            counters.put(name, counter.get()));
        
        // 收集仪表
        Map<String, Long> gauges = new java.util.HashMap<>();
        customGauges.forEach((name, gauge) -> 
            gauges.put(name, gauge.get()));
        
        snapshot.put("counters", counters);
        snapshot.put("gauges", gauges);
        snapshot.put("timestamp", System.currentTimeMillis());
        
        return snapshot;
    }
}
