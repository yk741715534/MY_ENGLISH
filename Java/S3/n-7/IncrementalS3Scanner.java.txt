// IncrementalS3Scanner.java
@Service
@Slf4j
public class IncrementalS3Scanner {
    
    @Autowired
    private S3AsyncClient s3Client;
    
    @Autowired
    private InstanceRegistry instanceRegistry;
    
    @Value("${s3.scan.bucket}")
    private String bucket;
    
    @Value("${s3.scan.prefix}")
    private String scanPrefix;
    
    private volatile String lastContinuationToken = null;
    private volatile Date lastScanTime = new Date(0);
    private final Map<String, String> fileCache = new ConcurrentHashMap<>();  // key -> etag缓存
    
    /**
     * 增量扫描 - 只扫描新增/修改的文件
     */
    public List<S3ObjectInfo> incrementalScan() {
        List<S3ObjectInfo> newOrModified = new ArrayList<>();
        String continuationToken = null;
        
        do {
            try {
                // 构建请求，从上次中断处继续
                ListObjectsV2Request.Builder requestBuilder = ListObjectsV2Request.builder()
                    .bucket(bucket)
                    .prefix(scanPrefix)
                    .maxKeys(100);  // 小批量获取
                
                if (continuationToken != null) {
                    requestBuilder.continuationToken(continuationToken);
                }
                
                ListObjectsV2Response response = s3Client.listObjectsV2(requestBuilder.build())
                    .get(5, TimeUnit.SECONDS);  // 5秒超时
                
                // 处理本批结果
                for (S3Object s3Object : response.contents()) {
                    if (shouldProcess(s3Object)) {
                        newOrModified.add(convertToObjectInfo(s3Object));
                    }
                }
                
                continuationToken = response.nextContinuationToken();
                
                // 如果数量太多，分批处理
                if (newOrModified.size() >= 50) {
                    break;
                }
                
            } catch (Exception e) {
                log.error("增量扫描失败", e);
                break;
            }
            
        } while (continuationToken != null);
        
        lastScanTime = new Date();
        return newOrModified;
    }
    
    private boolean shouldProcess(S3Object s3Object) {
        String key = s3Object.key();
        String currentEtag = s3Object.eTag();
        
        // 跳过目录
        if (key.endsWith("/")) {
            return false;
        }
        
        // 检查缓存
        String cachedEtag = fileCache.get(key);
        if (cachedEtag != null && cachedEtag.equals(currentEtag)) {
            return false;  // 文件未变化
        }
        
        // 更新缓存
        fileCache.put(key, currentEtag);
        
        // 检查最后修改时间（可选）
        Instant lastModified = s3Object.lastModified();
        if (lastModified != null && lastModified.isBefore(lastScanTime.toInstant())) {
            return false;  // 在上次扫描后未修改
        }
        
        return true;
    }
    
    /**
     * 定时清理缓存
     */
    @Scheduled(fixedDelay = 300000)  // 每5分钟清理一次
    public void cleanupCache() {
        // 清理一小时未访问的缓存
        // 实际生产中可以更精细管理
        if (fileCache.size() > 10000) {
            fileCache.clear();
        }
    }
}
