// ScannerHealthEndpoint.java
@Component
@Endpoint(id = "scanner")
public class ScannerHealthEndpoint {
    
    @Autowired
    private ThreeSecondScanner scanner;
    
    @Autowired
    private ScanPerformanceMonitor monitor;
    
    @ReadOperation
    public Map<String, Object> health() {
        Map<String, Object> health = new LinkedHashMap<>();
        health.put("status", "UP");
        health.put("lastScanDuration", scanner.getLastScanDuration());
        health.put("scanning", scanner.isScanning());
        health.put("timestamp", new Date());
        return health;
    }
}
