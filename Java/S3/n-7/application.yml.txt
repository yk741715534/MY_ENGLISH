# 3秒扫描配置
s3:
  scan:
    bucket: ${S3_BUCKET:my-bucket}
    prefix: uploads/
    enabled: true

  # 性能优化配置
  performance:
    max-concurrent-downloads: 5
    download-timeout-ms: 5000
    process-timeout-ms: 10000

  # 内存缓存配置
  cache:
    file-etag-size: 10000
    file-etag-ttl: 3600  # 1小时

  # MongoDB锁配置
  lock:
    coordination-ttl: 10000  # 协调锁10秒
    process-ttl: 30000       # 处理锁30秒
    retry-interval: 100      # 重试间隔100ms

# 线程池配置
async:
  executor:
    core-pool-size: 5
    max-pool-size: 20
    queue-capacity: 1000
    keep-alive-seconds: 60
    thread-name-prefix: "s3-process-"

# 监控配置
management:
  metrics:
    export:
      prometheus:
        enabled: true
  endpoint:
    health:
      show-details: always
  web:
    exposure:
      include: health,prometheus,metrics
