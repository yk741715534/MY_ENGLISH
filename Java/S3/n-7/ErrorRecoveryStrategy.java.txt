// ErrorRecoveryStrategy.java
@Component
public class ErrorRecoveryStrategy {
    
    private final Map<String, Integer> errorCounts = new ConcurrentHashMap<>();
    private final Map<String, Long> backoffUntil = new ConcurrentHashMap<>();
    
    /**
     * 记录错误并计算退避时间
     */
    public boolean shouldRetry(String key, Exception error) {
        String errorKey = getErrorKey(key, error);
        
        int count = errorCounts.getOrDefault(errorKey, 0) + 1;
        errorCounts.put(errorKey, count);
        
        if (count > 3) {  // 超过3次错误
            long backoffTime = calculateBackoffTime(count);
            backoffUntil.put(errorKey, System.currentTimeMillis() + backoffTime);
            return false;
        }
        
        return true;
    }
    
    private long calculateBackoffTime(int errorCount) {
        // 指数退避：1, 2, 4, 8, 16, 32, 64秒...
        return (long) Math.pow(2, errorCount - 1) * 1000;
    }
    
    private String getErrorKey(String key, Exception error) {
        return key + ":" + error.getClass().getSimpleName();
    }
    
    /**
     * 清理过期的错误记录
     */
    @Scheduled(fixedRate = 60000)  // 每分钟清理
    public void cleanupErrorRecords() {
        long now = System.currentTimeMillis();
        backoffUntil.entrySet().removeIf(entry -> entry.getValue() < now);
    }
}
