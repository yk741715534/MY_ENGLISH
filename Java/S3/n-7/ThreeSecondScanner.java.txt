// ThreeSecondScanner.java
@Component
@Slf4j
public class ThreeSecondScanner {
    
    @Autowired
    private IncrementalS3Scanner incrementalScanner;
    
    @Autowired
    private LeaseBasedCoordinator coordinator;
    
    @Autowired
    private S3ObjectProcessor objectProcessor;
    
    @Autowired
    private InstanceRegistry instanceRegistry;
    
    @Value("${app.instance.id}")
    private String instanceId;
    
    private volatile boolean scanning = false;
    private volatile long lastScanDuration = 0;
    
    /**
     * 3秒定时扫描
     */
    @Scheduled(fixedDelay = 3000)  // 3秒一次
    public void scanEveryThreeSeconds() {
        if (scanning) {
            log.debug("上次扫描还未完成，跳过本次");
            return;
        }
        
        if (lastScanDuration > 2000) {  // 上次扫描超过2秒
            log.warn("上次扫描耗时较长: {}ms，考虑调整间隔", lastScanDuration);
        }
        
        long startTime = System.currentTimeMillis();
        scanning = true;
        
        try {
            // 1. 获取协调锁
            if (!coordinator.acquireCoordinationLock()) {
                log.debug("未获取到协调锁，跳过本次扫描");
                return;
            }
            
            // 2. 执行增量扫描
            List<S3ObjectInfo> objectsToProcess = incrementalScanner.incrementalScan();
            
            if (!objectsToProcess.isEmpty()) {
                log.info("发现 {} 个待处理文件", objectsToProcess.size());
                
                // 3. 并发处理
                processConcurrently(objectsToProcess);
            } else {
                log.debug("没有发现新文件");
            }
            
        } catch (Exception e) {
            log.error("扫描处理失败", e);
        } finally {
            scanning = false;
            lastScanDuration = System.currentTimeMillis() - startTime;
        }
    }
    
    private void processConcurrently(List<S3ObjectInfo> objects) {
        // 使用虚拟线程（Java 21+）或线程池
        ExecutorService executor = Executors.newVirtualThreadPerTaskExecutor();  // Java 21+
        
        List<CompletableFuture<Void>> futures = objects.stream()
            .map(obj -> CompletableFuture.runAsync(() -> {
                try {
                    objectProcessor.processWithMutex(obj.getBucket(), obj.getKey());
                } catch (Exception e) {
                    log.error("处理文件失败: {}", obj.getKey(), e);
                }
            }, executor))
            .collect(Collectors.toList());
        
        // 等待所有处理完成，但不超过2.5秒
        try {
            CompletableFuture.allOf(futures.toArray(new CompletableFuture[0]))
                .get(2500, TimeUnit.MILLISECONDS);
        } catch (TimeoutException e) {
            log.warn("处理超时，部分文件将在下次扫描处理");
        } catch (Exception e) {
            log.error("并发处理异常", e);
        } finally {
            executor.shutdown();
        }
    }
}
