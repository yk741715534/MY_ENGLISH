public class LockRecoveryService {

    @Scheduled(fixedDelay = 30000)
    public void recoverOrphanedLocks() {
        // 1. 查找"僵尸锁"（锁还在，但实例已宕机）
        List<LockInfo> orphanedLocks = findOrphanedLocks();

        for (LockInfo lock : orphanedLocks) {
            // 2. 检查关联的任务是否完成
            String taskId = extractTaskId(lock.getLockKey());
            TaskStatus status = taskService.getTaskStatus(taskId);

            if (status == TaskStatus.COMPLETED) {
                // 3. 清理已完成任务的锁
                lockService.forceReleaseLock(lock.getLockKey(), "recovery-completed");
            } else if (status == TaskStatus.FAILED) {
                // 4. 清理失败任务的锁
                lockService.forceReleaseLock(lock.getLockKey(), "recovery-failed");
            } else {
                // 5. 任务状态未知，需要人工干预
                sendAlert("Orphaned lock found: " + lock.getLockKey());
            }
        }
    }
}

@Service
public class FileProcessingService {

    private final DistributedLockService lockService;
    private final Map<String, Timer> renewalTimers = new ConcurrentHashMap<>();

    /**
     * 处理大文件
     */
    public void processLargeFile(String bucket, String key) {
        String lockKey = "process:" + bucket + ":" + key;
        String owner = getInstanceId();

        long startTime = System.currentTimeMillis();

        try {
            // 1. 获取锁
            if (!lockService.acquireLock(lockKey, owner, "file-process", 30000)) {
                log.warn("获取锁失败: {}", lockKey);
                return;
            }

            log.info("开始处理文件: {}, 获取锁成功", key);

            // 2. 启动续期
            startLockRenewal(lockKey, owner, 30000);

            // 3. 下载文件
            File localFile = downloadFile(bucket, key);
            long downloadTime = System.currentTimeMillis() - startTime;

            // 4. 处理文件
            processFileContent(localFile);
            long processTime = System.currentTimeMillis() - startTime - downloadTime;

            // 5. 上传结果
            uploadResult(bucket, key, getResult());
            long totalTime = System.currentTimeMillis() - startTime;

            log.info("文件处理完成: {}, 总耗时: {}ms", key, totalTime);

        } catch (Exception e) {
            log.error("处理文件失败: {}", key, e);

        } finally {
            // 6. 停止续期
            stopLockRenewal(lockKey);

            // 7. 释放锁
            lockService.releaseLock(lockKey, owner);
        }
    }

    /**
     * 启动锁续期
     */
    private void startLockRenewal(String lockKey, String owner, long ttlMs) {
        Timer timer = new Timer("lock-renewal-" + lockKey, true);

        // 在TTL的2/3时续期
        long renewalDelay = ttlMs * 2 / 3;

        timer.schedule(new TimerTask() {
            @Override
            public void run() {
                try {
                    boolean renewed = lockService.renewLock(lockKey, owner, ttlMs);

                    if (renewed) {
                        log.debug("锁续期成功: {}", lockKey);
                    } else {
                        log.warn("锁续期失败: {}", lockKey);
                        this.cancel();  // 停止续期
                    }

                } catch (Exception e) {
                    log.error("锁续期异常: {}", lockKey, e);
                }
            }
        }, renewalDelay, renewalDelay);  // 定期续期

        renewalTimers.put(lockKey, timer);
    }

    /**
     * 停止锁续期
     */
    private void stopLockRenewal(String lockKey) {
        Timer timer = renewalTimers.remove(lockKey);
        if (timer != null) {
            timer.cancel();
        }
    }
}

        String lockKey = "migration:" + migrationId;

        try {
            // 启动续期（每分钟续期一次）

                try {
                    lockService.renewLock(lockKey, owner, 300000);

                    log.error("迁移锁续期失败", e);
                }

            // 执行长时间迁移
            executeLongRunningMigration(migrationId);

        } finally {
            // 停止续期
            if (renewalFuture != null) {
                renewalFuture.cancel(false);
            }

            // 释放锁
            lockService.releaseLock(lockKey, owner);
        }
    }

                }
            }, 30, 30, TimeUnit.MINUTES);

            // 执行批量作业
            executeLongBatchJob();

        } finally {
            renewalExecutor.shutdown();
            lockService.releaseLock(lockKey, owner);
        }

@Component
public class SmartLockRenewalStrategy {

    private final Map<String, RenewalContext> renewalContexts = new ConcurrentHashMap<>();

    @Data
    @Builder
    public static class RenewalContext {
        private String lockKey;
        private String owner;
        private long originalTtl;
        private int renewalCount;
        private long lastRenewalTime;
        private double renewalBackoffFactor;
        private ScheduledFuture<?> renewalFuture;
    }

    /**
     * 智能启动续期
     */
    public void startSmartRenewal(String lockKey, String owner, long ttlMs) {
        RenewalContext context = RenewalContext.builder()
            .lockKey(lockKey)
            .owner(owner)
            .originalTtl(ttlMs)
            .renewalCount(0)
            .renewalBackoffFactor(1.0)
            .build();

        renewalContexts.put(lockKey, context);

        // 计算首次续期时间
        long firstRenewalDelay = calculateFirstRenewalDelay(ttlMs);

        ScheduledFuture<?> future = scheduler.schedule(() -> {
            renewWithStrategy(context);
        }, firstRenewalDelay, TimeUnit.MILLISECONDS);

        context.setRenewalFuture(future);
    }

    /**
     * 带策略的续期
     */
    private void renewWithStrategy(RenewalContext context) {
        try {
            // 尝试续期
            boolean renewed = lockService.renewLock(
                context.getLockKey(),
                context.getOwner(),
                context.getOriginalTtl()
            );

            if (renewed) {
                context.setRenewalCount(context.getRenewalCount() + 1);
                context.setLastRenewalTime(System.currentTimeMillis());

                // 计算下次续期时间（指数退避）
                long nextDelay = calculateNextRenewalDelay(context);

                if (nextDelay > 0) {
                    ScheduledFuture<?> future = scheduler.schedule(() -> {
                        renewWithStrategy(context);
                    }, nextDelay, TimeUnit.MILLISECONDS);

                    context.setRenewalFuture(future);
                } else {
                    // 达到最大续期次数
                    log.warn("达到最大续期次数: {}", context.getLockKey());
                }

            } else {
                log.error("锁续期失败: {}", context.getLockKey());
            }

        } catch (Exception e) {
            log.error("续期异常: {}", context.getLockKey(), e);
        }
    }

    /**
     * 计算首次续期延迟
     */
    private long calculateFirstRenewalDelay(long ttlMs) {
        // 在TTL的50%-80%之间随机续期
        double minFactor = 0.5;
        double maxFactor = 0.8;
        double factor = minFactor + Math.random() * (maxFactor - minFactor);

        return (long) (ttlMs * factor);
    }

    /**
     * 计算下次续期延迟
     */
    private long calculateNextRenewalDelay(RenewalContext context) {
        int maxRenewals = 10;  // 最大续期次数
        if (context.getRenewalCount() >= maxRenewals) {
            return -1;
        }

        // 指数退避
        double backoffFactor = Math.pow(1.5, context.getRenewalCount());
        context.setRenewalBackoffFactor(backoffFactor);

        long baseDelay = context.getOriginalTtl() / 2;
        return (long) (baseDelay * backoffFactor);
    }
}

@Component
public class AdaptiveTTLStrategy {

    /**
     * 根据任务类型自适应TTL
     */
    public long calculateAdaptiveTtl(String taskType, Map<String, Object> taskContext) {
        switch (taskType) {
            case "FILE_UPLOAD":
                long fileSize = (Long) taskContext.getOrDefault("fileSize", 0L);
                return calculateUploadTtl(fileSize);

            case "DATABASE_MIGRATION":
                int recordCount = (Integer) taskContext.getOrDefault("recordCount", 0);
                return calculateMigrationTtl(recordCount);

            case "REPORT_GENERATION":
                String reportType = (String) taskContext.getOrDefault("reportType", "daily");
                return calculateReportTtl(reportType);

            default:
                return 30000;  // 默认30秒
        }
    }

    private long calculateUploadTtl(long fileSize) {
        // 估计上传时间：假设10MB/s
        double uploadTimeSeconds = fileSize / (10.0 * 1024 * 1024);

        // 加上缓冲时间
        double totalTimeSeconds = uploadTimeSeconds * 2 + 30;

        return (long) (totalTimeSeconds * 1000);
    }

    /**
     * 动态续期
     */
    public long calculateRenewalTtl(String lockKey, long currentTtl,
                                   int renewalCount, long taskProgress) {
        // 如果是首次续期，使用原TTL
        if (renewalCount == 0) {
            return currentTtl;
        }

        // 根据任务进度调整TTL
        if (taskProgress > 0.8) {  // 接近完成
            return currentTtl / 2;  // 减少TTL
        } else if (taskProgress < 0.3) {  // 刚开始
            return currentTtl * 2;  // 增加TTL
        }

        return currentTtl;
    }
}

            if (status == TaskStatus.UNKNOWN || status == TaskStatus.STALLED) {
                sendAlert("锁长时间持有: " + lock.getLockKey() +
                          ", 持有者: " + lock.getOwner() +
                          ", 持续时间: " + calculateDuration(lock.getAcquiredAt()));
            }
        }
    }

    /**
     * 检查续期成功率
     */
    @Scheduled(fixedDelay = 300000)  // 5分钟
    public void monitorRenewalSuccessRate() {
        Map<String, RenewalStats> stats = renewalService.getRenewalStats();

        for (Map.Entry<String, RenewalStats> entry : stats.entrySet()) {
            RenewalStats stat = entry.getValue();
            double successRate = stat.getSuccessCount() * 100.0 / stat.getTotalCount();

            if (successRate < 95) {
                sendAlert("锁续期成功率低: " + entry.getKey() +
                          ", 成功率: " + String.format("%.1f%%", successRate));
            }
        }
    }
}

public void renewWithRetry(String lockKey, String owner, long ttlMs) {
    int maxRetries = 3;

    for (int i = 0; i < maxRetries; i++) {
        try {
            if (lockService.renewLock(lockKey, owner, ttlMs)) {
                return;  // 成功
            }

            // 重试前等待
            Thread.sleep(1000 * (i + 1));

        } catch (Exception e) {
            log.warn("锁续期重试 {}/{} 失败", i + 1, maxRetries, e);

            if (i == maxRetries - 1) {
                // 最后一次失败，执行回退
                handleRenewalFailure(lockKey, owner);
            }
        }
    }

总结

分布式锁续期的主要目的：

防止锁过早过期：确保长时间任务执行期间锁保持有效

支持长时间运行任务：允许任务运行时间超过初始TTL

提高系统容错性：实例崩溃时，锁会自动过期，避免死锁

资源优化：不需要为所有任务设置很长的TTL

动态调整：可根据任务进度调整锁的持续时间

关键设计原则：

在锁过期前足够早的时间续期

实现指数退避的续期策略

监控续期成功率和锁健康状态

实现优雅的失败处理

避免过度续期导致的性能问题

在你的S3扫描系统中，续期特别重要，因为：

文件处理时间不可预测

大文件可能需要几分钟处理

网络问题可能导致延迟

批量处理需要长时间保持锁

没有续期机制，你的系统可能会遇到重复处理、数据不一致和竞争条件等问题。
