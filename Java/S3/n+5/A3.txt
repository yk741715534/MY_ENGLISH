ownerId 的设计选择

ownerId 应该是每个实例一个唯一ID，而不是每次获取UUID。​ 让我详细解释原因和最佳实践。

1. 两种方式的对比
1.1 每次生成UUID（不推荐）

// 2. 无法正确释放锁（因为释放时需要相同ownerId）
// 3. 无法跟踪是哪个实例持有了锁

// 推荐的方式
@Component
public class InstanceIdentity {

    private final String instanceId;

    public InstanceIdentity() {
        // 启动时生成固定ID
        this.instanceId = generateInstanceId();
    }

    public String getInstanceId() {
        return instanceId;
    }

    private String generateInstanceId() {
        // 组合多种信息生成唯一ID
        StringBuilder sb = new StringBuilder();

        // 1. 主机名
        try {
            String hostname = InetAddress.getLocalHost().getHostName();
            sb.append(hostname).append("-");
        } catch (Exception e) {
            sb.append("unknown-host-");
        }

        // 2. IP地址最后一段
        try {
            String ip = InetAddress.getLocalHost().getHostAddress();
            String[] parts = ip.split("\\.");
            if (parts.length == 4) {
                sb.append(parts[3]).append("-");
            }
        } catch (Exception e) {
            // 忽略
        }

        // 3. 进程ID
        long pid = ProcessHandle.current().pid();
        sb.append(pid).append("-");

        // 4. 启动时间戳
        sb.append(System.currentTimeMillis()).append("-");

        // 5. 随机数防止冲突
        sb.append(UUID.randomUUID().toString().substring(0, 8));

        return sb.toString();
    }
}

import lombok.Data;
import lombok.extern.slf4j.Slf4j;

@Component
@Slf4j
public class InstanceIdentityManager {

    // 实例元数据
    @Data
    public static class InstanceMetadata {
        private final String instanceId;
        private final String hostname;
        private final String ipAddress;
        private final long pid;
        private final long startupTime;
        private final int port;
        private final String zone;
        private final String region;
        private final Map<String, Object> capabilities = new HashMap<>();
        private final AtomicLong lockCounter = new AtomicLong(0);
        private final AtomicLong processedCounter = new AtomicLong(0);

        public void incrementLockCount() {
            lockCounter.incrementAndGet();
        }

        public void incrementProcessedCount() {
            processedCounter.incrementAndGet();
        }

        public Map<String, Object> toMap() {
            Map<String, Object> map = new HashMap<>();
            map.put("instanceId", instanceId);
            map.put("hostname", hostname);
            map.put("ipAddress", ipAddress);
            map.put("pid", pid);
            map.put("startupTime", startupTime);
            map.put("port", port);
            map.put("zone", zone);
            map.put("region", region);
            map.put("totalLocksAcquired", lockCounter.get());
            map.put("totalProcessed", processedCounter.get());
            map.put("capabilities", capabilities);
            return map;
        }
    }

    private InstanceMetadata metadata;

    @Value("${server.port:8080}")
    private int serverPort;

    @Value("${app.instance.zone:default-zone}")
    private String zone;

    @Value("${app.instance.region:default-region}")
    private String region;

    /**
     * 初始化实例标识
     */
    @PostConstruct
    public void init() {
        this.metadata = createInstanceMetadata();
        log.info("实例初始化完成: {}", metadata.getInstanceId());
        log.info("实例详情: {}", metadata.toMap());
    }

    /**
     * 创建实例元数据
     */
    private InstanceMetadata createInstanceMetadata() {
        try {
            // 获取主机信息
            InetAddress localHost = InetAddress.getLocalHost();
            String hostname = localHost.getHostName();
            String ipAddress = localHost.getHostAddress();

            // 进程ID
            long pid = ProcessHandle.current().pid();

            // 启动时间
            long startupTime = System.currentTimeMillis();

            // 生成实例ID
            String instanceId = generateInstanceId(hostname, ipAddress, pid, startupTime);

            // 能力集
            Map<String, Object> capabilities = new HashMap<>();
            capabilities.put("s3.scan", true);
            capabilities.put("s3.process", true);
            capabilities.put("s3.move", true);
            capabilities.put("distributed.lock", true);
            capabilities.put("heartbeat", true);

            return new InstanceMetadata(
                instanceId,
                hostname,
                ipAddress,
                pid,
                startupTime,
                serverPort,
                zone,
                region
            );

        } catch (UnknownHostException e) {
            log.error("无法获取主机信息", e);

            // 回退方案
            return new InstanceMetadata(
                "unknown-" + UUID.randomUUID().toString(),
                "unknown",
                "127.0.0.1",
                -1,
                System.currentTimeMillis(),
                serverPort,
                zone,
                region
            );
        }
    }

    /**
     * 生成实例ID
     */
    private String generateInstanceId(String hostname, String ipAddress,
                                     long pid, long startupTime) {
        // 格式: {hostname}-{ip-suffix}-{pid}-{timestamp}-{random}
        StringBuilder sb = new StringBuilder();

        // 1. 主机名（清理特殊字符）
        String safeHostname = hostname.replaceAll("[^a-zA-Z0-9-]", "-");
        sb.append(safeHostname).append("-");

        // 2. IP最后一段
        String[] ipParts = ipAddress.split("\\.");
        if (ipParts.length == 4) {
            sb.append(ipParts[3]).append("-");
        } else {
            sb.append("0-");
        }

        // 3. 进程ID
        sb.append(pid).append("-");

        // 4. 启动时间戳（简短格式）
        String timestamp = Long.toString(startupTime, 36);  // 36进制缩短
        sb.append(timestamp.substring(Math.max(0, timestamp.length() - 6))).append("-");

        // 5. 随机后缀（防止冲突）
        String randomSuffix = UUID.randomUUID().toString()
            .replaceAll("-", "")
            .substring(0, 6)
            .toLowerCase();
        sb.append(randomSuffix);

        return sb.toString().toLowerCase();
    }

    /**
     * 获取实例ID
     */
    public String getInstanceId() {
        return metadata.getInstanceId();
    }

    /**
     * 获取完整元数据
     */
    public InstanceMetadata getMetadata() {
        return metadata;
    }

    /**
     * 获取实例统计
     */
    public Map<String, Object> getInstanceStats() {
        return metadata.toMap();
    }

    /**
     * 注册实例能力
     */
    public void registerCapability(String capability, Object value) {
        metadata.getCapabilities().put(capability, value);
    }

    /**
     * 增加锁计数
     */
    public void incrementLockCount() {
        metadata.incrementLockCount();
    }

    /**
     * 增加处理计数
     */
    public void incrementProcessedCount() {
        metadata.incrementProcessedCount();
    }
}

@Service
@RequiredArgsConstructor
public class DistributedLockService {

    private final InstanceIdentityManager identityManager;
    private final MongoTemplate mongoTemplate;

    /**
     * 获取锁
     */
    public boolean acquireLock(String lockKey, long ttlMs, String resource) {
        String instanceId = identityManager.getInstanceId();

        try {
            boolean acquired = doAcquireLock(lockKey, instanceId, ttlMs, resource);

            if (acquired) {
                identityManager.incrementLockCount();
                log.info("实例 {} 获取锁成功: {}", instanceId, lockKey);
            }

            return acquired;

        } catch (Exception e) {
            log.error("获取锁失败: {}", lockKey, e);
            return false;
        }
    }

    /**
     * 释放锁
     */
    public boolean releaseLock(String lockKey) {
        String instanceId = identityManager.getInstanceId();
        return doReleaseLock(lockKey, instanceId);
    }

    /**
     * 续期锁
     */
    public boolean renewLock(String lockKey, long additionalTtlMs) {
        String instanceId = identityManager.getInstanceId();
        return doRenewLock(lockKey, instanceId, additionalTtlMs);
    }

    /**
     * 获取当前实例持有的锁
     */
    public List<LockInfo> getCurrentInstanceLocks() {
        String instanceId = identityManager.getInstanceId();

        Query query = new Query(Criteria.where("owner").is(instanceId)
            .and("status").is("ACTIVE")
            .and("expiresAt").gt(new Date()));

        return mongoTemplate.find(query, LockInfo.class, "distributed_locks");
    }
}

@Component
public class EnvironmentAwareInstanceIdGenerator {

    public String generateInstanceId() {
        // 1. 检查Kubernetes环境
        String podName = System.getenv("POD_NAME");
        if (podName != null && !podName.isEmpty()) {
            return "k8s-" + podName;
        }

        // 2. 检查Docker环境
        String containerId = System.getenv("HOSTNAME");
        if (containerId != null && !containerId.isEmpty() &&
            containerId.matches("[a-f0-9]{12}")) {
            return "docker-" + containerId;
        }

        // 3. 检查ECS环境
        String ecsTaskArn = System.getenv("ECS_CONTAINER_METADATA_URI");
        if (ecsTaskArn != null) {
            return "ecs-" + extractEcsId(ecsTaskArn);
        }

        // 4. 传统环境
        return generateTraditionalInstanceId();
    }

    private String extractEcsId(String metadataUri) {
        try {
            // 从元数据URI提取ID
            java.net.URI uri = new java.net.URI(metadataUri);
            String path = uri.getPath();
            String[] parts = path.split("/");
            if (parts.length >= 3) {
                return parts[2];  // task ID
            }
        } catch (Exception e) {
            // 忽略
        }
        return UUID.randomUUID().toString().substring(0, 8);
    }
}

        result.put("owner", lock.getOwner());
        result.put("ownerInfo", instance.orElse(null));

        return result;

总结

在分布式锁系统中，ownerId应该是每个实例的固定唯一ID，而不是每次随机生成UUID。

核心原因：

锁的完整性：确保只有获取锁的实例能释放它

可观测性：便于监控和调试

正确性：支持锁续期和健康检查

安全性：防止锁被误释放

     */
    private InstanceMetadata collectInstanceMetadata() {
        String instanceId = generateInstanceId();

        return InstanceMetadata.builder()
            .instanceId(instanceId)
            .hostname(getBestHostname())
            .containerHostname(getContainerHostname())
            .containerIp(getContainerIp())
            .podIp(getPodIp())
            .nodeIp(getNodeIp())
            .podName(getPodName())
            .namespace(getNamespace())
            .containerId(getContainerId())
            .isContainer(isRunningInContainer())
            .isKubernetes(isRunningInKubernetes())
            .labels(collectLabels())
            .build();
    }

    /**
     * 生成实例ID
     */
    private String generateInstanceId() {
        // 1. 使用配置的ID
        if (configuredInstanceId != null && !configuredInstanceId.trim().isEmpty()) {
            return configuredInstanceId.trim();
        }

        // 2. 尝试获取Pod名称
        String podName = getPodName();
        if (podName != null && !podName.isEmpty()) {
            return podName;
        }

        // 3. 尝试获取容器ID
        String containerId = getContainerId();
        if (containerId != null && !containerId.isEmpty()) {
            return "container-" + containerId;
        }

        // 4. 回退到综合ID
        return generateFallbackId();
    }

    /**
     * 获取最佳主机名
     */
    public String getBestHostname() {
        // 优先级
        String[] candidates = {
            getPodName(),                    // 1. Pod名称
            getContainerHostname(),          // 2. 容器主机名
            getEnvHostname(),               // 3. HOSTNAME环境变量
            getInetHostname(),              // 4. InetAddress.getLocalHost()
            generateFallbackId()            // 5. 回退
        };

        for (String candidate : candidates) {
            if (candidate != null && !candidate.trim().isEmpty()) {
                return candidate.trim();
            }
        }

        return "unknown-host";
    }

    /**
     * 获取容器内主机名
     */
    public String getContainerHostname() {
        try {
            return InetAddress.getLocalHost().getHostName();
        } catch (UnknownHostException e) {
            return null;
        }
    }

    /**
     * 获取容器内IP
     */
    public String getContainerIp() {
        try {
            return InetAddress.getLocalHost().getHostAddress();
        } catch (UnknownHostException e) {
            return null;
        }
    }

    /**
     * 获取Pod IP (k8s特有)
     */
    public String getPodIp() {
        // Kubernetes设置的环境变量
        return System.getenv("POD_IP");
    }

    /**
     * 获取Node IP (k8s特有)
     */
    public String getNodeIp() {
        return System.getenv("NODE_IP");
    }

    /**
     * 获取Pod名称
     */
    public String getPodName() {
        // Kubernetes设置的环境变量
        return System.getenv("POD_NAME");
    }

    /**
     * 获取命名空间
     */
    public String getNamespace() {
        return System.getenv("POD_NAMESPACE");
    }

    /**
     * 获取容器ID
     */
    public String getContainerId() {
        // Docker/Kubernetes容器ID
        String hostname = System.getenv("HOSTNAME");
        if (hostname != null && hostname.matches("[a-f0-9]{12}")) {
            return hostname;  // Docker容器ID通常是12位十六进制
        }

        // 尝试从/proc文件系统获取
        if (isRunningInContainer()) {
            try {
                String hostnameContent = java.nio.file.Files.readString(
                    java.nio.file.Paths.get("/etc/hostname")
                ).trim();
                if (hostnameContent.matches("[a-f0-9]{12}")) {
                    return hostnameContent;
                }
            } catch (Exception e) {
                // 忽略
            }
        }

        return null;
    }

    /**
     * 检查是否在容器中运行
     */
    public boolean isRunningInContainer() {
        // 方法1: 检查/.dockerenv文件
        if (new java.io.File("/.dockerenv").exists()) {
            return true;
        }

        // 方法2: 检查/proc/1/cgroup
        try {
            String cgroup = java.nio.file.Files.readString(
                java.nio.file.Paths.get("/proc/1/cgroup")
            );
            return cgroup.contains("docker") || cgroup.contains("kubepods");
        } catch (Exception e) {
            // 忽略
        }

        // 方法3: 检查环境变量
        if (System.getenv("KUBERNETES_SERVICE_HOST") != null) {
            return true;
        }

        if (System.getenv("container") != null) {
            return true;
        }

        return false;
    }

    /**
     * 检查是否在Kubernetes中运行
     */
    public boolean isRunningInKubernetes() {
        // Kubernetes会设置特定的环境变量
        if (System.getenv("KUBERNETES_SERVICE_HOST") != null) {
            return true;
        }

        // 检查service account token
        if (new java.io.File("/var/run/secrets/kubernetes.io/serviceaccount/token").exists()) {
            return true;
        }

        // 检查Pod相关的环境变量
        if (System.getenv("POD_NAME") != null ||
            System.getenv("POD_IP") != null ||
            System.getenv("POD_NAMESPACE") != null) {
            return true;
        }

        return false;
    }

    /**
     * 从环境变量获取主机名
     */
    private String getEnvHostname() {
        return System.getenv("HOSTNAME");
    }

    /**
     * 从InetAddress获取主机名
     */
    private String getInetHostname() {
        try {
            return InetAddress.getLocalHost().getHostName();
        } catch (UnknownHostException e) {
            return null;
        }
    }

    /**
     * 收集标签
     */
    private Map<String, String> collectLabels() {
        Map<String, String> labels = new HashMap<>();

        // 应用标签
        labels.put("app", System.getenv("APP_NAME"));
        labels.put("version", System.getenv("APP_VERSION"));
        labels.put("environment", System.getenv("ENVIRONMENT"));

        // Kubernetes标签
        if (isRunningInKubernetes()) {
            labels.put("kubernetes.io/hostname", System.getenv("NODE_NAME"));
            labels.put("kubernetes.io/namespace", getNamespace());

            // 尝试从Downward API获取更多标签
            try {
                // 读取Pod标签文件
                String podLabelsPath = "/etc/podinfo/labels";
                if (new java.io.File(podLabelsPath).exists()) {
                    String labelsContent = java.nio.file.Files.readString(
                        java.nio.file.Paths.get(podLabelsPath)
                    );

                    for (String line : labelsContent.split("\n")) {
                        String[] parts = line.split("=", 2);
                        if (parts.length == 2) {
                            String key = parts[0].trim();
                            String value = parts[1].trim().replace("\"", "");
                            if (!key.isEmpty() && !value.isEmpty()) {
                                labels.put(key, value);
                            }
                        }
                    }
                }
            } catch (Exception e) {
                // 忽略
            }
        }

        return labels;
    }

    /**
     * 生成回退ID
     */
    private String generateFallbackId() {
        try {
            // 组合多种信息
            String timestamp = Long.toString(System.currentTimeMillis(), 36);
            String random = UUID.randomUUID().toString().substring(0, 6);

            String hostPart = "unknown";
            try {
                hostPart = InetAddress.getLocalHost().getHostName();
            } catch (Exception e) {
                // 忽略
            }

            return String.format("%s-%s-%s",
                hostPart.replaceAll("[^a-zA-Z0-9]", "-"),
                timestamp.substring(Math.max(0, timestamp.length() - 6)),
                random
            ).toLowerCase();

        } catch (Exception e) {
            return "instance-" + UUID.randomUUID().toString();
        }
    }

    /**
     * 获取实例ID
     */
    public String getInstanceId() {
        return metadata.getInstanceId();
    }

    /**
     * 获取完整元数据
     */
    public InstanceMetadata getMetadata() {
        return metadata;
    }

    /**
     * 获取网络信息摘要
     */
    public Map<String, Object> getNetworkSummary() {
        Map<String, Object> summary = new HashMap<>();

        try {
            // 获取所有网络接口
            Enumeration<NetworkInterface> interfaces = NetworkInterface.getNetworkInterfaces();

            List<Map<String, Object>> interfaceList = new ArrayList<>();

            while (interfaces.hasMoreElements()) {
                NetworkInterface iface = interfaces.nextElement();

                if (iface.isUp() && !iface.isLoopback() && !iface.isVirtual()) {
                    Map<String, Object> ifaceInfo = new HashMap<>();
                    ifaceInfo.put("name", iface.getName());
                    ifaceInfo.put("displayName", iface.getDisplayName());

                    List<String> addresses = new ArrayList<>();
                    Enumeration<InetAddress> inetAddresses = iface.getInetAddresses();

                    while (inetAddresses.hasMoreElements()) {
                        InetAddress addr = inetAddresses.nextElement();
                        if (!addr.isLoopbackAddress()) {
                            addresses.add(addr.getHostAddress());
                        }
                    }

                    ifaceInfo.put("addresses", addresses);
                    interfaceList.add(ifaceInfo);
                }
            }

            summary.put("networkInterfaces", interfaceList);

        } catch (SocketException e) {
            summary.put("error", e.getMessage());
        }

        return summary;
    }
}

@Component
public class S3ScannerInstanceIdentity {

    @Getter
    private final String instanceId;

    public S3ScannerInstanceIdentity() {
        this.instanceId = resolveInstanceId();
    }

    private String resolveInstanceId() {
        // 1. 从环境变量获取（K8s/Docker设置）
        String podName = System.getenv("POD_NAME");
        if (podName != null && !podName.isEmpty()) {
            return "k8s-pod-" + podName;
        }

        String hostname = System.getenv("HOSTNAME");
        if (hostname != null && !hostname.isEmpty()) {
            if (hostname.matches("[a-f0-9]{12}")) {
                return "docker-" + hostname;  // Docker容器ID
            }
            return "container-" + hostname;
        }

        // 2. 从InetAddress获取（容器内信息）
        try {
            String containerHostname = InetAddress.getLocalHost().getHostName();
            if (isLikelyContainerId(containerHostname)) {
                return "container-" + containerHostname;
            }
        } catch (Exception e) {
            // 忽略
        }

        // 3. 生成唯一ID
        return "instance-" + UUID.randomUUID().toString();
    }

    private boolean isLikelyContainerId(String hostname) {
        // Docker容器ID通常是12位十六进制
        return hostname != null && hostname.matches("[a-f0-9]{12}");
    }
}
重要结论

InetAddress.getLocalHost()在容器中获取的是容器内部的信息，不是宿主机信息。

正确做法：

在Kubernetes中，使用环境变量：POD_NAME, POD_IP, NODE_NAME

在Docker中，使用环境变量：HOSTNAME

回退方案：InetAddress.getLocalHost()+ 唯一后缀

在你的分布式锁系统中，实例ID应该是：

在K8s中：使用Pod名称

在Docker中：使用容器ID

在物理机中：使用主机名+其他标识
