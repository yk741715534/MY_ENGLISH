package com.example.s3scanner.service.coordinator;

import lombok.Getter;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.context.event.ApplicationReadyEvent;
import org.springframework.context.ApplicationListener;
import org.springframework.core.annotation.Order;
import org.springframework.stereotype.Component;
import javax.annotation.PostConstruct;
import java.net.InetAddress;
import java.net.UnknownHostException;
import java.util.*;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.atomic.AtomicReference;

/**
 * 实例标识管理器
 * 在应用初始化时生成唯一ID
 */
@Component
@Slf4j
public class InstanceIdentityManager implements ApplicationListener<ApplicationReadyEvent> {

    // 核心：应用启动时生成并保持不变的实例ID
    @Getter
    private final AtomicReference<String> instanceId = new AtomicReference<>();

    // 实例元数据
    @Getter
    private final Map<String, Object> metadata = new HashMap<>();

    // 统计
    private final AtomicLong lockAcquiredCount = new AtomicLong(0);
    private final AtomicLong lockReleasedCount = new AtomicLong(0);
    private final AtomicLong tasksProcessedCount = new AtomicLong(0);

    // 配置
    @Value("${app.instance.id:#{null}}")
    private String configuredInstanceId;

    @Value("${app.instance.prefix:s3-scanner}")
    private String instancePrefix;

    @Value("${app.instance.cluster:default}")
    private String clusterName;

    @Value("${app.instance.zone:default-zone}")
    private String zone;

    @Value("${server.port:8080}")
    private int serverPort;

    /**
     * 初始化阶段：生成实例ID
     */
    @PostConstruct
    @Order(1)  // 确保在其他Bean之前初始化
    public void init() {
        log.info("开始初始化实例标识...");

        // 生成实例ID
        String id = generateOrGetInstanceId();
        instanceId.set(id);

        // 收集元数据
        collectMetadata();

        log.info("实例标识初始化完成: {}", id);
        log.info("实例元数据: {}", metadata);
    }

    /**
     * 应用启动完成后：注册实例
     */
    @Override
    public void onApplicationEvent(ApplicationReadyEvent event) {
        log.info("应用启动完成，实例ID: {}", instanceId.get());

        // 注册到注册中心
        registerInstance();

        // 输出实例信息
        printInstanceInfo();
    }

    /**
     * 生成或获取实例ID
     */
    private String generateOrGetInstanceId() {
        // 1. 使用配置的ID（最高优先级）
        if (configuredInstanceId != null && !configuredInstanceId.trim().isEmpty()) {
            log.info("使用配置的实例ID: {}", configuredInstanceId);
            return configuredInstanceId.trim();
        }

        // 2. 生成唯一ID
        return generateUniqueInstanceId();
    }

    /**
     * 生成唯一实例ID
     */
    private String generateUniqueInstanceId() {
        StringBuilder idBuilder = new StringBuilder();

        // 前缀
        idBuilder.append(instancePrefix).append("-");

        // 环境信息
        idBuilder.append(getEnvironmentIdentifier()).append("-");

        // 时间戳（确保同一环境多次启动不同）
        String timestamp = Long.toString(System.currentTimeMillis(), 36);
        idBuilder.append(timestamp.substring(Math.max(0, timestamp.length() - 8))).append("-");

        // 随机后缀（防止冲突）
        String randomSuffix = UUID.randomUUID().toString()
            .replaceAll("-", "")
            .substring(0, 6)
            .toLowerCase();
        idBuilder.append(randomSuffix);

        String generatedId = idBuilder.toString().toLowerCase();
        log.info("生成实例ID: {}", generatedId);

        return generatedId;
    }

    /**
     * 获取环境标识符
     */
    private String getEnvironmentIdentifier() {
        // 尝试多种方式获取环境标识

        // 1. Kubernetes Pod
        String podName = System.getenv("POD_NAME");
        if (podName != null && !podName.isEmpty()) {
            return extractPodIdentifier(podName);
        }

        // 2. Docker容器
        String hostname = System.getenv("HOSTNAME");
        if (hostname != null && !hostname.isEmpty()) {
            if (hostname.matches("[a-f0-9]{12}")) {
                return "docker-" + hostname.substring(0, 8);
            }
            return "container-" + hostname.replaceAll("[^a-zA-Z0-9]", "-");
        }

        // 3. 主机名
        try {
            String localHostname = InetAddress.getLocalHost().getHostName();
            if (localHostname != null && !localHostname.isEmpty()) {
                return "host-" + localHostname.replaceAll("[^a-zA-Z0-9]", "-");
            }
        } catch (UnknownHostException e) {
            // 忽略
        }

        // 4. 回退
        return "env-" + UUID.randomUUID().toString().substring(0, 4);
    }

    /**
     * 提取Pod标识符
     */
    private String extractPodIdentifier(String podName) {
        // Pod名称格式通常为: {deployment-name}-{pod-hash}
        String[] parts = podName.split("-");
        if (parts.length >= 2) {
            // 取最后两部分
            return parts[parts.length - 2] + "-" + parts[parts.length - 1];
        }
        return podName;
    }

    /**
     * 收集元数据
     */
    private void collectMetadata() {
        metadata.clear();

        // 基本信息
        metadata.put("instanceId", instanceId.get());
        metadata.put("cluster", clusterName);
        metadata.put("zone", zone);
        metadata.put("startupTime", new Date());
        metadata.put("serverPort", serverPort);

        // 环境信息
        metadata.put("podName", System.getenv("POD_NAME"));
        metadata.put("podIp", System.getenv("POD_IP"));
        metadata.put("nodeName", System.getenv("NODE_NAME"));
        metadata.put("namespace", System.getenv("POD_NAMESPACE"));
        metadata.put("hostname", getHostname());
        metadata.put("ipAddress", getIpAddress());

        // 运行时信息
        metadata.put("javaVersion", System.getProperty("java.version"));
        metadata.put("pid", getProcessId());

        // 应用信息
        metadata.put("appName", "s3-realtime-scanner");
        metadata.put("version", getClass().getPackage().getImplementationVersion());

        // 网络信息
        metadata.put("networkInterfaces", getNetworkInterfaces());
    }

    /**
     * 注册实例
     */
    private void registerInstance() {
        try {
            // 这里可以注册到服务发现、数据库等
            Map<String, Object> registration = new HashMap<>();
            registration.put("instanceId", instanceId.get());
            registration.put("metadata", metadata);
            registration.put("status", "RUNNING");
            registration.put("registeredAt", new Date());

            log.info("实例注册信息: {}", registration);

        } catch (Exception e) {
            log.error("实例注册失败", e);
        }
    }

    /**
     * 获取主机名
     */
    private String getHostname() {
        try {
            return InetAddress.getLocalHost().getHostName();
        } catch (UnknownHostException e) {
            return "unknown";
        }
    }

    /**
     * 获取IP地址
     */
    private String getIpAddress() {
        try {
            return InetAddress.getLocalHost().getHostAddress();
        } catch (UnknownHostException e) {
            return "127.0.0.1";
        }
    }

    /**
     * 获取进程ID
     */
    private long getProcessId() {
        try {
            return ProcessHandle.current().pid();
        } catch (Exception e) {
            return -1;
        }
    }

    /**
     * 获取网络接口信息
     */
    private List<Map<String, Object>> getNetworkInterfaces() {
        List<Map<String, Object>> interfaces = new ArrayList<>();

        try {
            Enumeration<java.net.NetworkInterface> nets = java.net.NetworkInterface.getNetworkInterfaces();

            while (nets.hasMoreElements()) {
                java.net.NetworkInterface netInterface = nets.nextElement();

                if (netInterface.isUp() && !netInterface.isLoopback() && !netInterface.isVirtual()) {
                    Map<String, Object> ifaceInfo = new HashMap<>();
                    ifaceInfo.put("name", netInterface.getName());
                    ifaceInfo.put("displayName", netInterface.getDisplayName());

                    List<String> addresses = new ArrayList<>();
                    Enumeration<InetAddress> inetAddresses = netInterface.getInetAddresses();

                    while (inetAddresses.hasMoreElements()) {
                        InetAddress addr = inetAddresses.nextElement();
                        if (!addr.isLoopbackAddress() && !addr.isLinkLocalAddress()) {
                            addresses.add(addr.getHostAddress());
                        }
                    }

                    if (!addresses.isEmpty()) {
                        ifaceInfo.put("addresses", addresses);
                        interfaces.add(ifaceInfo);
                    }
                }
            }
        } catch (Exception e) {
            // 忽略
        }

        return interfaces;
    }

    /**
     * 增加锁获取计数
     */
    public void incrementLockAcquiredCount() {
        lockAcquiredCount.incrementAndGet();
    }

    /**
     * 增加锁释放计数
     */
    public void incrementLockReleasedCount() {
        lockReleasedCount.incrementAndGet();
    }

    /**
     * 增加任务处理计数
     */
    public void incrementTasksProcessedCount() {
        tasksProcessedCount.incrementAndGet();
    }

    /**
     * 获取实例统计
     */
    public Map<String, Object> getInstanceStats() {
        Map<String, Object> stats = new HashMap<>(metadata);

        stats.put("lockAcquiredCount", lockAcquiredCount.get());
        stats.put("lockReleasedCount", lockReleasedCount.get());
        stats.put("tasksProcessedCount", tasksProcessedCount.get());
        stats.put("uptime", System.currentTimeMillis() - (Long) metadata.get("startupTime"));
        stats.put("currentTime", new Date());

        return stats;
    }

    /**
     * 打印实例信息
     */
    private void printInstanceInfo() {
        log.info("========================================");
        log.info("实例启动完成");
        log.info("实例ID: {}", instanceId.get());
        log.info("集群: {}", clusterName);
        log.info("区域: {}", zone);
        log.info("主机名: {}", metadata.get("hostname"));
        log.info("IP地址: {}", metadata.get("ipAddress"));
        log.info("启动时间: {}", metadata.get("startupTime"));
        log.info("========================================");
    }

    /**
     * 优雅关闭
     */
    public void shutdown() {
        log.info("实例关闭，ID: {}", instanceId.get());

        // 这里可以执行注销逻辑
        Map<String, Object> shutdownInfo = new HashMap<>();
        shutdownInfo.put("instanceId", instanceId.get());
        shutdownInfo.put("shutdownTime", new Date());
        shutdownInfo.put("totalLocksAcquired", lockAcquiredCount.get());
        shutdownInfo.put("totalTasksProcessed", tasksProcessedCount.get());

        log.info("实例关闭统计: {}", shutdownInfo);
    }
}

@Service
@RequiredArgsConstructor
public class DistributedLockService {

    private final InstanceIdentityManager identityManager;
    private final MongoTemplate mongoTemplate;

    /**
     * 获取锁
     */
    public boolean acquireLock(String lockKey, String resource, long ttlMs) {
        // 使用初始化时生成的实例ID
        String ownerId = identityManager.getInstanceId();

        try {
            boolean acquired = doAcquireLock(lockKey, ownerId, resource, ttlMs);

            if (acquired) {
                identityManager.incrementLockAcquiredCount();
                log.debug("实例 {} 获取锁成功: {}", ownerId, lockKey);
            }

            return acquired;

        } catch (Exception e) {
            log.error("获取锁失败: {}", lockKey, e);
            return false;
        }
    }

    /**
     * 释放锁
     */
    public boolean releaseLock(String lockKey) {
        String ownerId = identityManager.getInstanceId();
        boolean released = doReleaseLock(lockKey, ownerId);

        if (released) {
            identityManager.incrementLockReleasedCount();
        }

        return released;
    }

    /**
     * 获取当前实例持有的所有锁
     */
    public List<LockInfo> getCurrentInstanceLocks() {
        String ownerId = identityManager.getInstanceId();

        Query query = new Query(Criteria.where("owner").is(ownerId)
            .and("status").is("ACTIVE")
            .and("expiresAt").gt(new Date()));

        return mongoTemplate.find(query, LockInfo.class, "distributed_locks");
    }
}

        for (String lockKey : lockResults.keySet()) {
            if (lockResults.get(lockKey)) {
                lockService.releaseLock(lockKey);
            }
        }
    }

@Component
public class SimpleInstanceId {

    @Getter
    private final String instanceId;

    public SimpleInstanceId() {
        // 应用启动时立即生成
        this.instanceId = generateId();
    }

    private String generateId() {
        // 1. 尝试从环境变量获取
        String podName = System.getenv("POD_NAME");
        if (podName != null) {
            return podName;
        }

        // 2. 组合生成
        String hostname = getHostname();
        String timestamp = Long.toString(System.currentTimeMillis(), 36);
        String random = UUID.randomUUID().toString().substring(0, 6);

        return String.format("s3-%s-%s-%s",
            hostname,
            timestamp.substring(Math.max(0, timestamp.length() - 6)),
            random
        ).toLowerCase();
    }

    private String getHostname() {
        try {
            return InetAddress.getLocalHost().getHostName();
        } catch (Exception e) {
            return "unknown";
        }
    }
}
结论

是的，ownerId应该在应用初始化时生成唯一ID，这是分布式系统的最佳实践。

核心原则：

一次性生成：应用启动时生成，运行期间不变

环境感知：结合环境信息（K8s/Docker/物理机）

唯一性保证：通过组合多种信息确保集群内唯一

可配置性：支持通过配置覆盖自动生成的ID

可观测性：便于监控、调试和故障排查

在你的S3扫描系统中，这样做可以：

确保分布式锁的正确性

简化调试和监控

提高系统可靠性

支持多环境部署
