    private Map<String, Object> extractedMetadata = new HashMap<>();
    
    /**
     * 处理统计信息
     */
    @Builder.Default
    private Map<String, Object> statistics = new HashMap<>();

    // ========== 枚举 ==========
    
    /**
     * 处理状态枚举
     */
    public enum ProcessingStatus {
        /** 待处理 */
        PENDING,
        /** 下载中 */
        DOWNLOADING,
        /** 处理中 */
        PROCESSING,
        /** 移动中 */
        MOVING,
        /** 完成 */
        COMPLETED,
        /** 失败 */
        FAILED,
        /** 跳过 */
        SKIPPED,
        /** 锁定 */
        LOCKED,
        /** 超时 */
        TIMEOUT,
        /** 重试中 */
        RETRYING
    }
    
    /**
     * 处理阶段枚举
     */
    public enum ProcessingPhase {
        /** 初始化 */
        INITIALIZATION,
        /** 验证 */
        VALIDATION,
        /** 下载 */
        DOWNLOAD,
        /** 处理 */
        PROCESSING,
        /** 移动 */
        MOVING,
        /** 清理 */
        CLEANUP,
        /** 完成 */
        COMPLETION
    }
    
    // ========== 静态工厂方法 ==========
    
    /**
     * 创建成功结果
     */
    public static ProcessingResult success(String bucket, String key, Object data) {
        LocalDateTime endTime = LocalDateTime.now();
        
        return ProcessingResult.builder()
            .resultId(generateResultId())
            .bucket(bucket)
            .key(key)
            .objectId(bucket + ":" + key)
            .status(ProcessingStatus.COMPLETED)
            .success(true)
            .phase(ProcessingPhase.COMPLETION)
            .progress(100)
            .startTime(endTime.minusSeconds(1))
            .endTime(endTime)
            .durationMs(1000L)
            .data(data)
            .instanceId(getInstanceId())
            .hostname(getHostname())
            .threadId(getThreadId())
            .build();
    }
    
    /**
     * 创建成功结果（带处理时间）
     */
    public static ProcessingResult success(String bucket, String key, Object data, 
                                          LocalDateTime startTime, LocalDateTime endTime) {
        long duration = Duration.between(startTime, endTime).toMillis();
        
        return ProcessingResult.builder()
            .resultId(generateResultId())
            .bucket(bucket)
            .key(key)
            .objectId(bucket + ":" + key)
            .status(ProcessingStatus.COMPLETED)
            .success(true)
            .phase(ProcessingPhase.COMPLETION)
            .progress(100)
            .startTime(startTime)
            .endTime(endTime)
            .durationMs(duration)
            .data(data)
            .instanceId(getInstanceId())
            .hostname(getHostname())
            .threadId(getThreadId())
            .build();
    }
    
    /**
     * 创建成功结果（带移动结果）
     */
    public static ProcessingResult successWithMove(String bucket, String key, 
                                                  Object data, MoveResult moveResult) {
        ProcessingResult result = success(bucket, key, data);
        result.setMoved(true);
        result.setMoveResult(moveResult);
        result.setMoveSuccess(moveResult.isSuccess());
        result.setMoveType(moveResult.getMoveType() != null ? 
            moveResult.getMoveType().name() : "ARCHIVE");
        
        if (moveResult.isCompressed()) {
            result.setCompressed(true);
            result.setCompressionFormat(moveResult.getCompressionFormat() != null ? 
                moveResult.getCompressionFormat().name() : "ZIP");
            result.setOriginalSize(moveResult.getOriginalFileSize());
            result.setCompressedSize(moveResult.getCompressedFileSize());
            result.setCompressionRatio(moveResult.getCompressionRatio());
        }
        
        return result;
    }
    
    /**
     * 创建失败结果
     */
    public static ProcessingResult failure(String bucket, String key, Exception error) {
        LocalDateTime endTime = LocalDateTime.now();
        
        return ProcessingResult.builder()
            .resultId(generateResultId())
            .bucket(bucket)
            .key(key)
            .objectId(bucket + ":" + key)
            .status(ProcessingStatus.FAILED)
            .success(false)
            .phase(ProcessingPhase.PROCESSING)
            .progress(0)
            .startTime(endTime.minusSeconds(1))
            .endTime(endTime)
            .durationMs(1000L)
            .errorMessage(error.getMessage())
            .errorCode(getErrorCode(error))
            .exceptionClass(error.getClass().getName())
            .stackTrace(getStackTrace(error))
            .instanceId(getInstanceId())
            .hostname(getHostname())
            .threadId(getThreadId())
            .build();
    }
    
    /**
     * 创建失败结果（带详细信息和重试）
     */
    public static ProcessingResult failure(String bucket, String key, Exception error, 
                                          int retryCount, int maxRetries) {
        ProcessingResult result = failure(bucket, key, error);
        result.setRetryCount(retryCount);
        result.setMaxRetries(maxRetries);
        result.setRetryable(retryCount < maxRetries);
        result.setStatus(retryCount < maxRetries ? 
            ProcessingStatus.RETRYING : ProcessingStatus.FAILED);
        return result;
    }
    
    /**
     * 创建跳过结果
     */
    public static ProcessingResult skipped(String bucket, String key, String reason) {
        return ProcessingResult.builder()
            .resultId(generateResultId())
            .bucket(bucket)
            .key(key)
            .objectId(bucket + ":" + key)
            .status(ProcessingStatus.SKIPPED)
            .success(true)
            .skipped(true)
            .skipReason(reason)
            .phase(ProcessingPhase.VALIDATION)
            .progress(100)
            .startTime(LocalDateTime.now())
            .endTime(LocalDateTime.now())
            .durationMs(0L)
            .instanceId(getInstanceId())
            .hostname(getHostname())
            .threadId(getThreadId())
            .build();
    }
    
    /**
     * 创建锁定结果
     */
    public static ProcessingResult locked(String bucket, String key, String lockHolder) {
        return ProcessingResult.builder()
            .resultId(generateResultId())
            .bucket(bucket)
            .key(key)
            .objectId(bucket + ":" + key)
            .status(ProcessingStatus.LOCKED)
            .success(false)
            .locked(true)
            .lockHolder(lockHolder)
            .phase(ProcessingPhase.VALIDATION)
            .progress(0)
            .startTime(LocalDateTime.now())
            .endTime(LocalDateTime.now())
            .durationMs(0L)
            .instanceId(getInstanceId())
            .hostname(getHostname())
            .threadId(getThreadId())
            .build();
    }
    
    /**
     * 创建处理中结果
     */
    public static ProcessingResult inProgress(String bucket, String key, ProcessingPhase phase, 
                                             int progress) {
        return ProcessingResult.builder()
            .resultId(generateResultId())
            .bucket(bucket)
            .key(key)
            .objectId(bucket + ":" + key)
            .status(ProcessingStatus.PROCESSING)
            .success(false)
            .phase(phase)
            .progress(progress)
            .startTime(LocalDateTime.now())
            .instanceId(getInstanceId())
            .hostname(getHostname())
            .threadId(getThreadId())
            .build();
    }
    
    /**
     * 创建下载中结果
     */
    public static ProcessingResult downloading(String bucket, String key, int progress) {
        return inProgress(bucket, key, ProcessingPhase.DOWNLOAD, progress)
            .toBuilder()
            .status(ProcessingStatus.DOWNLOADING)
            .build();
    }
    
    /**
     * 创建移动中结果
     */
    public static ProcessingResult moving(String bucket, String key, String moveType) {
        return inProgress(bucket, key, ProcessingPhase.MOVING, 50)
            .toBuilder()
            .status(ProcessingStatus.MOVING)
            .moveType(moveType)
            .build();
    }
    
    // ========== 工具方法 ==========
    
    /**
     * 获取完整路径
     */
    public String getFullPath() {
        return String.format("s3://%s/%s", bucket, key);
    }
    
    /**
     * 获取处理状态描述
     */
    public String getStatusDescription() {
        if (skipped) {
            return String.format("已跳过: %s", skipReason);
        }
        
        if (locked) {
            return String.format("被锁定: %s 正在处理", lockHolder);
        }
        
        if (success) {
            if (moved && moveSuccess != null && moveSuccess) {
                return String.format("处理成功并已移动到: %s", 
                    moveResult != null ? moveResult.getTargetPath() : "未知");
            }
            return "处理成功";
        } else {
            if (retryable && retryCount < maxRetries) {
                return String.format("处理失败，第 %d/%d 次重试: %s", 
                    retryCount + 1, maxRetries, errorMessage);
            }
            return String.format("处理失败: %s", errorMessage);
        }
    }
    
    /**
     * 获取处理耗时（毫秒）
     */
    public Long getProcessingDurationMs() {
        if (startTime == null || endTime == null) {
            return null;
        }
        return Duration.between(startTime, endTime).toMillis();
    }
    
    /**
     * 获取格式化耗时
     */
    public String getFormattedDuration() {
        Long duration = getProcessingDurationMs();
        if (duration == null) {
            return "N/A";
        }
        
        if (duration < 1000) {
            return String.format("%dms", duration);
        } else if (duration < 60000) {
            return String.format("%.1fs", duration / 1000.0);
        } else {
            long minutes = duration / 60000;
            long seconds = (duration % 60000) / 1000;
            return String.format("%dm %ds", minutes, seconds);
        }
    }
    
    /**
     * 获取格式化文件大小
     */
    public String getFormattedFileSize() {
        if (fileSize == null) {
            return "N/A";
        }
        
        if (fileSize < 1024) {
            return String.format("%d B", fileSize);
        } else if (fileSize < 1024 * 1024) {
            return String.format("%.2f KB", fileSize / 1024.0);
        } else if (fileSize < 1024 * 1024 * 1024) {
            return String.format("%.2f MB", fileSize / (1024.0 * 1024.0));
        } else {
            return String.format("%.2f GB", fileSize / (1024.0 * 1024.0 * 1024.0));
        }
    }
    
    /**
     * 获取压缩节省的空间
     */
    public Long getSpaceSaved() {
        if (compressed && originalSize != null && compressedSize != null) {
            return originalSize - compressedSize;
        }
        return null;
    }
    
    /**
     * 获取压缩率百分比
     */
    public Double getCompressionPercentage() {
        if (compressionRatio != null) {
            return (1 - compressionRatio) * 100;
        }
        return null;
    }
    
    /**
     * 是否应该重试
     */
    public boolean shouldRetry() {
        return retryable && retryCount < maxRetries;
    }
    
    /**
     * 记录阶段耗时
     */
    public void recordPhaseDuration(ProcessingPhase phase, long durationMs) {
        phaseDurations.put(phase.name(), durationMs);
    }
    
    /**
     * 获取阶段耗时
     */
    public Long getPhaseDuration(ProcessingPhase phase) {
        return phaseDurations.get(phase.name());
    }

    /**
     * 转换为Map（用于日志或存储）
     */
    public Map<String, Object> toMap() {
        Map<String, Object> map = new HashMap<>();
        map.put("resultId", resultId);
        map.put("bucket", bucket);
        map.put("key", key);
        map.put("objectId", objectId);
        map.put("status", status != null ? status.name() : null);
        map.put("success", success);
        map.put("skipped", skipped);
        map.put("skipReason", skipReason);
        map.put("locked", locked);
        map.put("lockHolder", lockHolder);
        map.put("progress", progress);
        map.put("retryCount", retryCount);
        map.put("maxRetries", maxRetries);
        map.put("retryable", retryable);
        map.put("startTime", startTime);
        map.put("endTime", endTime);
        map.put("durationMs", getProcessingDurationMs());
        map.put("fileSize", fileSize);
        map.put("etag", etag);
        map.put("md5", md5);
        map.put("contentType", contentType);
        map.put("compressed", compressed);
        map.put("moved", moved);
        map.put("moveSuccess", moveSuccess);
        map.put("moveType", moveType);
        map.put("errorMessage", errorMessage);
        map.put("errorCode", errorCode);
        map.put("instanceId", instanceId);
        map.put("businessId", businessId);
        map.put("phaseDurations", phaseDurations);
        map.put("metadata", metadata);
        map.put("extractedMetadata", extractedMetadata);
        map.put("statistics", statistics);
        map.put("warnings", warnings != null ? warnings.size() : 0);
        return map;
    }
    
    // ========== 私有工具方法 ==========
    
    private static String generateResultId() {
        return UUID.randomUUID().toString();
    }
    
    private static String getInstanceId() {
        return System.getenv("HOSTNAME");
    }
    
    private static String getHostname() {
        try {
            return java.net.InetAddress.getLocalHost().getHostName();
        } catch (Exception e) {
            return "unknown";
        }
    }
    
    private static String getThreadId() {
        return String.valueOf(Thread.currentThread().getId());
    }
    
    private static String getErrorCode(Exception e) {
        if (e instanceof software.amazon.awssdk.services.s3.model.S3Exception) {
            return ((software.amazon.awssdk.services.s3.model.S3Exception) e)
                .awsErrorDetails().errorCode();
        }
        return "UNKNOWN";
    }
    
    private static String getStackTrace(Exception e) {
        StringBuilder sb = new StringBuilder();
        sb.append(e.getClass().getName()).append(": ").append(e.getMessage()).append("\n");
        for (StackTraceElement element : e.getStackTrace()) {
            sb.append("\tat ").append(element.toString()).append("\n");
        }
        return sb.toString();
    }
    
    // ========== Builder 扩展 ==========
    
    public static class ProcessingResultBuilder {
        public ProcessingResultBuilder completeWithDuration() {
            if (startTime != null && endTime != null) {
                this.durationMs(Duration.between(startTime, endTime).toMillis());
            }
            return this;
        }
        
        public ProcessingResultBuilder calculateCompressionRatio() {
            if (originalSize != null && originalSize > 0 && compressedSize != null) {
                this.compressionRatio((double) compressedSize / originalSize);
            }
            return this;
        }
        
        public ProcessingResultBuilder withPhase(ProcessingPhase phase) {
            this.phase(phase);
            this.step(phase.name());
            return this;
        }
    }
