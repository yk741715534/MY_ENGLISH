package com.example.s3scanner.service.processor.dto;

import lombok.*;
import java.time.LocalDateTime;
import java.util.*;
import java.util.concurrent.atomic.AtomicLong;

/**
 * 批量处理结果
 */
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class BatchProcessingResult {
    
    /**
     * 批量处理ID
     */
    private String batchId;
    
    /**
     * 批量处理开始时间
     */
    private LocalDateTime batchStartTime;
    
    /**
     * 批量处理结束时间
     */
    private LocalDateTime batchEndTime;
    
    /**
     * 批量处理耗时（毫秒）
     */
    private Long batchDurationMs;
    
    /**
     * 处理前缀/目录
     */
    private String prefix;
    
    /**
     * 成功的结果列表
     */
    @Builder.Default
    private List<ProcessingResult> successes = new ArrayList<>();
    
    /**
     * 失败的结果列表
     */
    @Builder.Default
    private List<ProcessingResult> failures = new ArrayList<>();
    
    /**
     * 跳过的结果列表
     */
    @Builder.Default
    private List<ProcessingResult> skips = new ArrayList<>();
    
    /**
     * 锁定的结果列表
     */
    @Builder.Default
    private List<ProcessingResult> lockeds = new ArrayList<>();
    
    /**
     * 处理中的结果列表
     */
    @Builder.Default
    private List<ProcessingResult> processings = new ArrayList<>();
    
    /**
     * 统计信息
     */
    private Statistics statistics;
    
    /**
     * 处理实例ID
     */
    private String instanceId;
    
    /**
     * 警告信息
     */
    @Builder.Default
    private List<String> warnings = new ArrayList<>();
    
    /**
     * 错误摘要
     */
    @Builder.Default
    private Map<String, Integer> errorSummary = new HashMap<>();
    
    /**
     * 添加处理结果
     */
    public void addResult(ProcessingResult result) {
        switch (result.getStatus()) {
            case COMPLETED:
                successes.add(result);
                break;
            case FAILED:
            case TIMEOUT:
                failures.add(result);
                recordError(result.getErrorMessage());
                break;
            case SKIPPED:
                skips.add(result);
                break;
            case LOCKED:
                lockeds.add(result);
                break;
            case PENDING:
            case DOWNLOADING:
            case PROCESSING:
            case MOVING:
            case RETRYING:
                processings.add(result);
                break;
        }
    }
    
    /**
     * 记录错误
     */
    private void recordError(String errorMessage) {
        if (errorMessage == null) {
            return;
        }
        
        String errorKey = extractErrorKey(errorMessage);
        errorSummary.put(errorKey, errorSummary.getOrDefault(errorKey, 0) + 1);
    }
    
    /**
     * 计算统计信息
     */
    public Statistics calculateStatistics() {
        AtomicLong totalBytes = new AtomicLong(0);
        AtomicLong totalDuration = new AtomicLong(0);
        AtomicLong compressedBytesSaved = new AtomicLong(0);
        
        successes.forEach(result -> {
            if (result.getFileSize() != null) {
                totalBytes.addAndGet(result.getFileSize());
            }
            if (result.getDurationMs() != null) {
                totalDuration.addAndGet(result.getDurationMs());
            }
            if (result.getSpaceSaved() != null) {
                compressedBytesSaved.addAndGet(result.getSpaceSaved());
            }
        });
        
        statistics = Statistics.builder()
            .totalObjects(successes.size() + failures.size() + skips.size() + 
                         lockeds.size() + processings.size())
            .successfulObjects(successes.size())
            .failedObjects(failures.size())
            .skippedObjects(skips.size())
            .lockedObjects(lockeds.size())
            .processingObjects(processings.size())
            .successRate(calculateSuccessRate())
            .failureRate(calculateFailureRate())
            .skipRate(calculateSkipRate())
            .totalBytesProcessed(totalBytes.get())
            .averageDurationPerObject(calculateAverageDuration())
            .totalDurationMs(batchDurationMs != null ? batchDurationMs : 0)
            .compressedBytesSaved(compressedBytesSaved.get())
            .errorCounts(errorSummary)
            .build();
        
        return statistics;
    }
    
    /**
     * 计算成功率
     */
    public double calculateSuccessRate() {
        int total = successes.size() + failures.size() + skips.size();
        if (total == 0) {
            return 0;
        }
        return (double) successes.size() / total * 100;
    }
    
    /**
     * 计算失败率
     */
    public double calculateFailureRate() {
        int total = successes.size() + failures.size() + skips.size();
        if (total == 0) {
            return 0;
        }
        return (double) failures.size() / total * 100;
    }
    
    /**
     * 计算跳过率
     */
    public double calculateSkipRate() {
        int total = successes.size() + failures.size() + skips.size();
        if (total == 0) {
            return 0;
        }
        return (double) skips.size() / total * 100;
    }
    
    /**
     * 计算平均处理时间
     */
    public double calculateAverageDuration() {
        if (successes.isEmpty()) {
            return 0;
        }
        return successes.stream()
            .filter(result -> result.getDurationMs() != null)
            .mapToLong(ProcessingResult::getDurationMs)
            .average()
            .orElse(0);
    }
    
    /**
     * 提取错误关键字
     */
    private String extractErrorKey(String errorMessage) {
        if (errorMessage == null || errorMessage.isEmpty()) {
            return "Unknown";
        }
        
        // 提取错误关键字
        if (errorMessage.contains("Connection")) {
            return "Connection Error";
        } else if (errorMessage.contains("Timeout")) {
            return "Timeout";
        } else if (errorMessage.contains("Access Denied")) {
            return "Access Denied";
        } else if (errorMessage.contains("NoSuchKey")) {
            return "File Not Found";
        } else if (errorMessage.contains("Lock")) {
            return "Lock Error";
        } else {
            return "Other";
        }
    }
    
    /**
     * 添加警告
     */
    public void addWarning(String warning) {
        warnings.add(warning);
    }
    
    /**
     * 获取摘要信息
     */
    public String getSummary() {
        calculateStatistics();
        
        return String.format(
            "批量处理完成: 总共 %d 个文件, 成功 %d 个, 失败 %d 个, 跳过 %d 个, 锁定 %d 个, 处理中 %d 个, " +
            "成功率: %.1f%%, 总耗时: %s, 平均耗时: %.1fms",
            statistics.getTotalObjects(),
            statistics.getSuccessfulObjects(),
            statistics.getFailedObjects(),
            statistics.getSkippedObjects(),
            statistics.getLockedObjects(),
            statistics.getProcessingObjects(),
            statistics.getSuccessRate(),
            formatDuration(statistics.getTotalDurationMs()),
            statistics.getAverageDurationPerObject()
        );
    }
    
    /**
     * 获取错误摘要
     */
    public String getErrorSummary() {
        if (errorSummary.isEmpty()) {
            return "无错误";
        }
        
        StringBuilder sb = new StringBuilder("错误统计:\n");
        errorSummary.entrySet().stream()
            .sorted(Map.Entry.<String, Integer>comparingByValue().reversed())
            .forEach(entry -> sb.append(String.format("  %s: %d 次\n", entry.getKey(), entry.getValue())));
        
        return sb.toString();
    }
    
    private String formatDuration(long durationMs) {
        if (durationMs < 1000) {
            return String.format("%dms", durationMs);
        } else if (durationMs < 60000) {
            return String.format("%.1fs", durationMs / 1000.0);
        } else {
            long minutes = durationMs / 60000;
            long seconds = (durationMs % 60000) / 1000;
            return String.format("%dm %ds", minutes, seconds);
        }
    }
    
    /**
     * 转换为简洁的Map
     */
    public Map<String, Object> toSummaryMap() {
        calculateStatistics();
        
        Map<String, Object> summary = new HashMap<>();
        summary.put("batchId", batchId);
        summary.put("prefix", prefix);
        summary.put("instanceId", instanceId);
        summary.put("batchStartTime", batchStartTime);
        summary.put("batchEndTime", batchEndTime);
        summary.put("batchDurationMs", batchDurationMs);
        summary.put("statistics", statistics);
        summary.put("warningCount", warnings.size());
        summary.put("errorSummary", errorSummary);
        
        return summary;
    }
    
    /**
     * 统计信息
     */
    @Data
    @Builder
    public static class Statistics {
        private int totalObjects;
        private int successfulObjects;
        private int failedObjects;
        private int skippedObjects;
        private int lockedObjects;
        private int processingObjects;
        private double successRate;
        private double failureRate;
        private double skipRate;
        private long totalBytesProcessed;
        private double averageDurationPerObject;
        private long totalDurationMs;
        private long compressedBytesSaved;
        private Map<String, Integer> errorCounts;
    }
}
