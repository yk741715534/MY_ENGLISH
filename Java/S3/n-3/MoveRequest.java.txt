package com.example.s3scanner.service.mover.dto;

import lombok.*;
import software.amazon.awssdk.services.s3.model.StorageClass;
import software.amazon.awssdk.services.s3.model.Tag;
import java.util.*;
import java.time.LocalDateTime;

/**
 * 文件移动请求DTO
 */
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
@EqualsAndHashCode
public class MoveRequest {
    
    // ========== 必填字段 ==========
    
    /**
     * 源Bucket名称
     */
    @NonNull
    private String bucket;
    
    /**
     * 源文件Key
     */
    @NonNull
    private String sourceKey;
    
    /**
     * 移动类型
     */
    @NonNull
    private MoveType moveType;
    
    // ========== 目标配置 ==========
    
    /**
     * 目标Bucket（为空表示同Bucket移动）
     */
    private String targetBucket;
    
    /**
     * 目标文件Key（为空则自动生成）
     */
    private String targetKey;
    
    /**
     * 目标前缀（用于自动生成目标Key）
     */
    private String targetPrefix;
    
    /**
     * 目标存储类型
     */
    @Builder.Default
    private StorageClass targetStorageClass = StorageClass.STANDARD;
    
    // ========== 移动配置 ==========
    
    /**
     * 是否在文件名中添加时间戳
     */
    @Builder.Default
    private boolean addTimestamp = true;
    
    /**
     * 时间戳格式
     */
    @Builder.Default
    private String timestampFormat = "yyyyMMdd_HHmmss";
    
    /**
     * 是否压缩文件
     */
    @Builder.Default
    private boolean compress = false;
    
    /**
     * 压缩格式
     */
    @Builder.Default
    private CompressionFormat compressionFormat = CompressionFormat.ZIP;
    
    /**
     * 压缩级别 (0-9, 0=不压缩, 9=最高压缩)
     */
    @Builder.Default
    private int compressionLevel = 6;
    
    /**
     * 是否保留源文件
     */
    @Builder.Default
    private boolean keepSource = false;
    
    /**
     * 是否验证移动
     */
    @Builder.Default
    private boolean verifyMove = true;
    
    /**
     * 验证重试次数
     */
    @Builder.Default
    private int verifyRetryCount = 3;
    
    // ========== 元数据和标签 ==========
    
    /**
     * 元数据
     */
    @Builder.Default
    private Map<String, Object> metadata = new HashMap<>();
    
    /**
     * 用户自定义元数据
     */
    @Builder.Default
    private Map<String, String> userMetadata = new HashMap<>();
    
    /**
     * S3标签
     */
    @Builder.Default
    private List<Tag> tags = new ArrayList<>();
    
    /**
     * 处理结果（用于归档时记录处理信息）
     */
    private Object processingResult;
    
    // ========== 错误处理 ==========
    
    /**
     * 错误信息（用于移动到失败目录时记录）
     */
    private String errorMessage;
    
    /**
     * 错误堆栈
     */
    private String errorStackTrace;
    
    /**
     * 重试配置
     */
    private RetryConfig retryConfig;
    
    // ========== 审计信息 ==========
    
    /**
     * 请求ID（用于追踪）
     */
    private String requestId;
    
    /**
     * 操作用户
     */
    private String operator;
    
    /**
     * 操作时间
     */
    @Builder.Default
    private LocalDateTime operationTime = LocalDateTime.now();
    
    /**
     * 业务ID（用于关联业务数据）
     */
    private String businessId;
    
    // ========== 枚举和内部类 ==========
    
    /**
     * 移动类型枚举
     */
    public enum MoveType {
        /** 归档 */
        ARCHIVE,
        /** 备份 */
        BACKUP,
        /** 移动到失败目录 */
        FAILED,
        /** 移动到临时目录 */
        TEMP,
        /** 移动到处理中目录 */
        PROCESSING,
        /** 移动到完成目录 */
        COMPLETED,
        /** 重命名 */
        RENAME,
        /** 跨区域复制 */
        CROSS_REGION
    }
    
    /**
     * 压缩格式枚举
     */
    public enum CompressionFormat {
        ZIP,
        GZIP,
        BZIP2
    }
    
    /**
     * 重试配置
     */
    @Data
    @Builder
    @NoArgsConstructor
    @AllArgsConstructor
    public static class RetryConfig {
        /**
         * 最大重试次数
         */
        @Builder.Default
        private int maxAttempts = 3;
        
        /**
         * 初始延迟（毫秒）
         */
        @Builder.Default
        private long initialDelayMs = 1000;
        
        /**
         * 延迟乘数
         */
        @Builder.Default
        private double multiplier = 2.0;
        
        /**
         * 最大延迟（毫秒）
         */
        @Builder.Default
        private long maxDelayMs = 10000;
        
        /**
         * 是否启用指数退避
         */
        @Builder.Default
        private boolean exponentialBackoff = true;
    }
    
    // ========== 工具方法 ==========
    
    /**
     * 获取目标Bucket，如果为空则使用源Bucket
     */
    public String getEffectiveTargetBucket() {
        return targetBucket != null ? targetBucket : bucket;
    }
    
    /**
     * 是否需要跨Bucket移动
     */
    public boolean isCrossBucketMove() {
        return targetBucket != null && !targetBucket.equals(bucket);
    }
    
    /**
     * 是否应该压缩文件
     */
    public boolean shouldCompress() {
        return compress && compressionFormat != null;
    }
    
    /**
     * 获取重试配置
     */
    public RetryConfig getEffectiveRetryConfig() {
        return retryConfig != null ? retryConfig : RetryConfig.builder().build();
    }
    
    /**
     * 添加元数据
     */
    public void addMetadata(String key, Object value) {
        if (metadata == null) {
            metadata = new HashMap<>();
        }
        metadata.put(key, value);
    }
    
    /**
     * 添加用户元数据
     */
    public void addUserMetadata(String key, String value) {
        if (userMetadata == null) {
            userMetadata = new HashMap<>();
        }
        userMetadata.put(key, value);
    }
    
    /**
     * 添加标签
     */
    public void addTag(String key, String value) {
        if (tags == null) {
            tags = new ArrayList<>();
        }
        tags.add(Tag.builder().key(key).value(value).build());
    }
    
    /**
     * 创建构建器
     */
    public static MoveRequestBuilder builder(String bucket, String sourceKey, MoveType moveType) {
        return new MoveRequestBuilder()
            .bucket(bucket)
            .sourceKey(sourceKey)
            .moveType(moveType);
    }
    
    /**
     * 创建归档请求
     */
    public static MoveRequest createArchiveRequest(String bucket, String sourceKey) {
        return MoveRequest.builder()
            .bucket(bucket)
            .sourceKey(sourceKey)
            .moveType(MoveType.ARCHIVE)
            .targetPrefix("archive/")
            .addTimestamp(true)
            .build();
    }
    
    /**
     * 创建备份请求
     */
    public static MoveRequest createBackupRequest(String bucket, String sourceKey) {
        return MoveRequest.builder()
            .bucket(bucket)
            .sourceKey(sourceKey)
            .moveType(MoveType.BACKUP)
            .targetPrefix("backup/")
            .addTimestamp(true)
            .build();
    }
    
    /**
     * 创建失败请求
     */
    public static MoveRequest createFailedRequest(String bucket, String sourceKey, Exception error) {
        return MoveRequest.builder()
            .bucket(bucket)
            .sourceKey(sourceKey)
            .moveType(MoveType.FAILED)
            .targetPrefix("failed/")
            .addTimestamp(true)
            .errorMessage(error.getMessage())
            .errorStackTrace(getStackTrace(error))
            .build();
    }
    
    private static String getStackTrace(Exception e) {
        StringBuilder sb = new StringBuilder();
        for (StackTraceElement element : e.getStackTrace()) {
            sb.append(element.toString()).append("\n");
        }
        return sb.toString();
    }
}
