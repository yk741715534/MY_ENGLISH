package com.example.s3scanner.model.dto;

import com.example.s3scanner.model.domain.S3ObjectInfo;
import lombok.*;
import java.time.Instant;
import java.util.*;
import java.util.concurrent.atomic.AtomicInteger;

/**
 * 扫描结果DTO
 */
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
@EqualsAndHashCode
public class ScanResult {
    
    // ========== 扫描基本信息 ==========
    
    /**
     * 扫描ID
     */
    private String scanId;
    
    /**
     * 扫描的前缀
     */
    private String prefix;
    
    /**
     * 扫描的Bucket
     */
    private String bucket;
    
    /**
     * 扫描开始时间
     */
    private Instant scanStartTime;
    
    /**
     * 扫描结束时间
     */
    private Instant scanEndTime;
    
    /**
     * 扫描耗时（毫秒）
     */
    private Long scanDurationMs;
    
    // ========== 扫描统计 ==========
    
    /**
     * 总扫描对象数
     */
    private int scannedObjects;
    
    /**
     * 新增对象数
     */
    private int newObjects;
    
    /**
     * 跳过对象数
     */
    private int skippedObjects;
    
    /**
     * 修改对象数
     */
    private int modifiedObjects;
    
    /**
     * 错误对象数
     */
    private int errorObjects;
    
    // ========== 对象列表 ==========
    
    /**
     * 新增对象列表
     */
    @Builder.Default
    private List<S3ObjectInfo> newObjectsList = new ArrayList<>();
    
    /**
     * 跳过对象列表
     */
    @Builder.Default
    private List<SkippedObject> skippedObjectsList = new ArrayList<>();
    
    /**
     * 修改对象列表
     */
    @Builder.Default
    private List<ModifiedObject> modifiedObjectsList = new ArrayList<>();
    
    /**
     * 错误对象列表
     */
    @Builder.Default
    private List<ErrorObject> errorObjectsList = new ArrayList<>();
    
    // ========== 扫描状态 ==========
    
    /**
     * 扫描是否成功
     */
    private boolean success;
    
    /**
     * 是否为增量扫描
     */
    @Builder.Default
    private boolean incremental = true;
    
    /**
     * 上次扫描时间
     */
    private Instant lastScanTime;
    
    /**
     * 扫描状态
     */
    private ScanStatus status;
    
    /**
     * 错误信息
     */
    private String errorMessage;
    
    /**
     * 错误堆栈
     */
    private String errorStackTrace;
    
    // ========== 分页信息 ==========
    
    /**
     * 分页Token
     */
    private String continuationToken;
    
    /**
     * 是否还有更多结果
     */
    @Builder.Default
    private boolean hasMore = false;
    
    /**
     * 分页大小
     */
    private int pageSize;
    
    // ========== 性能指标 ==========
    
    /**
     * 扫描速率（对象/秒）
     */
    private Double scanRate;
    
    /**
     * 平均对象大小
     */
    private Double averageObjectSize;
    
    /**
     * 总数据大小
     */
    private Long totalDataSize;
    
    // ========== 系统信息 ==========
    
    /**
     * 扫描实例ID
     */
    private String instanceId;
    
    /**
     * 扫描线程ID
     */
    private String threadId;
    
    /**
     * 内存使用（字节）
     */
    private Long memoryUsed;
    
    // ========== 扩展信息 ==========
    
    /**
     * 扫描配置
     */
    @Builder.Default
    private Map<String, Object> scanConfig = new HashMap<>();
    
    /**
     * 扫描元数据
     */
    @Builder.Default
    private Map<String, Object> metadata = new HashMap<>();
    
    /**
     * 扫描统计详情
     */
    @Builder.Default
    private Map<String, Object> statistics = new HashMap<>();
    
    /**
     * 警告信息
     */
    @Builder.Default
    private List<String> warnings = new ArrayList<>();
    
    // ========== 枚举 ==========
    
    /**
     * 扫描状态枚举
     */
    public enum ScanStatus {
        PENDING,
        SCANNING,
        COMPLETED,
        FAILED,
        PARTIAL,
        CANCELLED,
        TIMEOUT
    }
    
    // ========== 内部类 ==========
    
    /**
     * 跳过对象
     */
    @Data
    @Builder
    @NoArgsConstructor
    @AllArgsConstructor
    public static class SkippedObject {
        private String key;
        private String reason;
        private String etag;
        private Instant lastModified;
        private Long size;
        
        public static SkippedObject alreadyProcessed(String key, String etag) {
            return SkippedObject.builder()
                .key(key)
                .reason("ALREADY_PROCESSED")
                .etag(etag)
                .build();
        }
        
        public static SkippedObject locked(String key, String lockHolder) {
            return SkippedObject.builder()
                .key(key)
                .reason("LOCKED")
                .metadata(Map.of("lockHolder", lockHolder))
                .build();
        }
        
        public static SkippedObject tooLarge(String key, long size) {
            return SkippedObject.builder()
                .key(key)
                .reason("TOO_LARGE")
                .size(size)
                .build();
        }
        
        private Map<String, Object> metadata = new HashMap<>();
    }
    
    /**
     * 修改对象
     */
    @Data
    @Builder
    @NoArgsConstructor
    @AllArgsConstructor
    public static class ModifiedObject {
        private String key;
        private String oldEtag;
        private String newEtag;
        private Instant oldLastModified;
        private Instant newLastModified;
        private Long oldSize;
        private Long newSize;
        private Double changePercentage;
    }
    
    /**
     * 错误对象
     */
    @Data
    @Builder
    @NoArgsConstructor
    @AllArgsConstructor
    public static class ErrorObject {
        private String key;
        private String error;
        private String errorCode;
        private String exceptionClass;
        private String stackTrace;
        private Instant errorTime;
        
        public static ErrorObject fromException(String key, Exception e) {
            return ErrorObject.builder()
                .key(key)
                .error(e.getMessage())
                .errorCode(extractErrorCode(e))
                .exceptionClass(e.getClass().getName())
                .stackTrace(getStackTrace(e))
                .errorTime(Instant.now())
                .build();
        }
        
        private static String extractErrorCode(Exception e) {
            if (e instanceof software.amazon.awssdk.services.s3.model.S3Exception) {
                return ((software.amazon.awssdk.services.s3.model.S3Exception) e)
                    .awsErrorDetails().errorCode();
            }
            return "UNKNOWN";
        }
        
        private static String getStackTrace(Exception e) {
            StringBuilder sb = new StringBuilder();
            sb.append(e.getClass().getName()).append(": ").append(e.getMessage()).append("\n");
            for (StackTraceElement element : e.getStackTrace()) {
                sb.append("\tat ").append(element.toString()).append("\n");
            }
            return sb.toString();
        }
    }
    
    // ========== 静态工厂方法 ==========
    
    /**
     * 创建成功结果
     */
    public static ScanResult success(String prefix, String bucket, 
                                    List<S3ObjectInfo> newObjects, 
                                    int scannedObjects) {
        Instant endTime = Instant.now();
        
        return ScanResult.builder()
            .scanId(generateScanId())
            .prefix(prefix)
            .bucket(bucket)
            .scanStartTime(endTime.minusSeconds(1))
            .scanEndTime(endTime)
            .scanDurationMs(1000L)
            .scannedObjects(scannedObjects)
            .newObjects(newObjects.size())
            .newObjectsList(newObjects)
            .success(true)
            .status(ScanStatus.COMPLETED)
            .instanceId(getInstanceId())
            .threadId(getThreadId())
            .build();
    }
    
    /**
     * 创建增量扫描结果
     */
    public static ScanResult incrementalSuccess(String prefix, String bucket,
                                              List<S3ObjectInfo> newObjects,
                                              int scannedObjects,
                                              Instant lastScanTime) {
        ScanResult result = success(prefix, bucket, newObjects, scannedObjects);
        result.setIncremental(true);
        result.setLastScanTime(lastScanTime);
        return result;
    }
    
    /**
     * 创建失败结果
     */
    public static ScanResult failure(String prefix, String bucket, Exception error) {
        Instant endTime = Instant.now();
        
        return ScanResult.builder()
            .scanId(generateScanId())
            .prefix(prefix)
            .bucket(bucket)
            .scanStartTime(endTime.minusSeconds(1))
            .scanEndTime(endTime)
            .scanDurationMs(1000L)
            .success(false)
            .status(ScanStatus.FAILED)
            .errorMessage(error.getMessage())
            .errorStackTrace(getStackTrace(error))
            .instanceId(getInstanceId())
            .threadId(getThreadId())
            .build();
    }
    
    /**
     * 创建跳过结果
     */
    public static ScanResult skipped(String prefix, String bucket, String reason) {
        Instant endTime = Instant.now();
        
        return ScanResult.builder()
            .scanId(generateScanId())
            .prefix(prefix)
            .bucket(bucket)
            .scanStartTime(endTime)
            .scanEndTime(endTime)
            .scanDurationMs(0L)
            .success(true)
            .status(ScanStatus.COMPLETED)
            .skippedObjects(1)
            .addSkippedObject(SkippedObject.builder()
                .key(prefix)
                .reason(reason)
                .build())
            .instanceId(getInstanceId())
            .threadId(getThreadId())
            .build();
    }
    
    /**
     * 创建部分结果
     */
    public static ScanResult partial(String prefix, String bucket,
                                    List<S3ObjectInfo> newObjects,
                                    int scannedObjects,
                                    int errorObjects) {
        ScanResult result = success(prefix, bucket, newObjects, scannedObjects);
        result.setStatus(ScanStatus.PARTIAL);
        result.setErrorObjects(errorObjects);
        result.addWarning("部分扫描失败");
        return result;
    }
    
    // ========== 工具方法 ==========
    
    /**
     * 计算扫描速率
     */
    public void calculateScanRate() {
        if (scanDurationMs != null && scanDurationMs > 0) {
            this.scanRate = (double) scannedObjects / (scanDurationMs / 1000.0);
        }
    }
    
    /**
     * 计算平均对象大小
     */
    public void calculateObjectSizeStats() {
        if (!newObjectsList.isEmpty()) {
            long totalSize = 0L;
            int count = 0;
            
            for (S3ObjectInfo obj : newObjectsList) {
                if (obj.getSize() != null) {
                    totalSize += obj.getSize();
                    count++;
                }
            }
            
            if (count > 0) {
                this.averageObjectSize = (double) totalSize / count;
                this.totalDataSize = totalSize;
            }
        }
    }
    
    /**
     * 添加新增对象
     */
    public void addNewObject(S3ObjectInfo object) {
        newObjectsList.add(object);
        newObjects++;
        scannedObjects++;
    }
    
    /**
     * 添加跳过对象
     */
    public void addSkippedObject(SkippedObject object) {
        skippedObjectsList.add(object);
        skippedObjects++;
        scannedObjects++;
    }
    
    /**
     * 添加修改对象
     */
    public void addModifiedObject(ModifiedObject object) {
        modifiedObjectsList.add(object);
        modifiedObjects++;
        scannedObjects++;
    }
    
    /**
     * 添加错误对象
     */
    public void addErrorObject(ErrorObject object) {
        errorObjectsList.add(object);
        errorObjects++;
        scannedObjects++;
    }
    
    /**
     * 添加错误对象
     */
    public void addErrorObject(String key, Exception e) {
        addErrorObject(ErrorObject.fromException(key, e));
    }
    
    /**
     * 添加警告
     */
    public void addWarning(String warning) {
        warnings.add(warning);
    }
    
    /**
     * 添加元数据
     */
    public void addMetadata(String key, Object value) {
        metadata.put(key, value);
    }
    
    /**
     * 添加统计信息
     */
    public void addStatistic(String key, Object value) {
        statistics.put(key, value);
    }
    
    /**
     * 获取成功处理的百分比
     */
    public double getSuccessPercentage() {
        if (scannedObjects == 0) {
            return 0;
        }
        int successful = scannedObjects - errorObjects;
        return (double) successful / scannedObjects * 100;
    }
    
    /**
     * 获取新对象百分比
     */
    public double getNewObjectPercentage() {
        if (scannedObjects == 0) {
            return 0;
        }
        return (double) newObjects / scannedObjects * 100;
    }
    
    /**
     * 获取跳过对象百分比
     */
    public double getSkippedObjectPercentage() {
        if (scannedObjects == 0) {
            return 0;
        }
        return (double) skippedObjects / scannedObjects * 100;
    }
    
    /**
     * 获取错误率
     */
    public double getErrorRate() {
        if (scannedObjects == 0) {
            return 0;
        }
        return (double) errorObjects / scannedObjects * 100;
    }
    
    /**
     * 获取格式化耗时
     */
    public String getFormattedDuration() {
        if (scanDurationMs == null) {
            return "N/A";
        }
        
        if (scanDurationMs < 1000) {
            return String.format("%dms", scanDurationMs);
        } else if (scanDurationMs < 60000) {
            return String.format("%.1fs", scanDurationMs / 1000.0);
        } else {
            long minutes = scanDurationMs / 60000;
            long seconds = (scanDurationMs % 60000) / 1000;
            return String.format("%dm %ds", minutes, seconds);
        }
    }
    
    /**
     * 获取扫描摘要
     */
    public String getSummary() {
        StringBuilder sb = new StringBuilder();
        sb.append("扫描摘要: ");
        
        if (success) {
            sb.append(String.format("成功扫描 %s，", prefix));
            sb.append(String.format("总共 %d 个对象，", scannedObjects));
            sb.append(String.format("新增 %d 个，", newObjects));
            sb.append(String.format("跳过 %d 个，", skippedObjects));
            sb.append(String.format("修改 %d 个，", modifiedObjects));
            sb.append(String.format("错误 %d 个，", errorObjects));
            sb.append(String.format("耗时 %s", getFormattedDuration()));
            
            if (scanRate != null) {
                sb.append(String.format("，速率 %.1f 对象/秒", scanRate));
            }
        } else {
            sb.append(String.format("扫描失败: %s，", prefix));
            sb.append(String.format("错误: %s", errorMessage));
        }
        
        return sb.toString();
    }
    
    /**
     * 转换为Map
     */
    public Map<String, Object> toMap() {
        Map<String, Object> map = new HashMap<>();
        
        map.put("scanId", scanId);
        map.put("prefix", prefix);
        map.put("bucket", bucket);
        map.put("scanStartTime", scanStartTime);
        map.put("scanEndTime", scanEndTime);
        map.put("scanDurationMs", scanDurationMs);
        map.put("scannedObjects", scannedObjects);
        map.put("newObjects", newObjects);
        map.put("skippedObjects", skippedObjects);
        map.put("modifiedObjects", modifiedObjects);
        map.put("errorObjects", errorObjects);
        map.put("success", success);
        map.put("incremental", incremental);
        map.put("status", status != null ? status.name() : null);
        map.put("errorMessage", errorMessage);
        map.put("scanRate", scanRate);
        map.put("totalDataSize", totalDataSize);
        map.put("averageObjectSize", averageObjectSize);
        map.put("instanceId", instanceId);
        map.put("successPercentage", getSuccessPercentage());
        map.put("newObjectPercentage", getNewObjectPercentage());
        map.put("errorRate", getErrorRate());
        map.put("hasMore", hasMore);
        map.put("continuationToken", continuationToken);
        map.put("warningsCount", warnings.size());
        
        return map;
    }
    
    /**
     * 转换为简洁Map
     */
    public Map<String, Object> toSimpleMap() {
        Map<String, Object> map = new HashMap<>();
        
        map.put("scanId", scanId);
        map.put("prefix", prefix);
        map.put("scannedObjects", scannedObjects);
        map.put("newObjects", newObjects);
        map.put("success", success);
        map.put("scanDurationMs", scanDurationMs);
        map.put("status", status != null ? status.name() : null);
        
        return map;
    }
    
    /**
     * 获取跳过的原因统计
     */
    public Map<String, Integer> getSkipReasons() {
        Map<String, Integer> reasons = new HashMap<>();
        
        for (SkippedObject obj : skippedObjectsList) {
            String reason = obj.getReason();
            reasons.put(reason, reasons.getOrDefault(reason, 0) + 1);
        }
        
        return reasons;
    }
    
    /**
     * 获取错误类型统计
     */
    public Map<String, Integer> getErrorTypes() {
        Map<String, Integer> types = new HashMap<>();
        
        for (ErrorObject obj : errorObjectsList) {
            String type = obj.getErrorCode();
            types.put(type, types.getOrDefault(type, 0) + 1);
        }
        
        return types;
    }
    
    /**
     * 是否有警告
     */
    public boolean hasWarnings() {
        return !warnings.isEmpty();
    }
    
    /**
     * 是否有错误
     */
    public boolean hasErrors() {
        return !errorObjectsList.isEmpty();
    }
    
    /**
     * 是否需要重试
     */
    public boolean shouldRetry() {
        return status == ScanStatus.FAILED || status == ScanStatus.TIMEOUT;
    }
    
    /**
     * 合并两个扫描结果
     */
    public void merge(ScanResult other) {
        if (other == null) {
            return;
        }
        
        // 合并计数
        this.scannedObjects += other.scannedObjects;
        this.newObjects += other.newObjects;
        this.skippedObjects += other.skippedObjects;
        this.modifiedObjects += other.modifiedObjects;
        this.errorObjects += other.errorObjects;
        
        // 合并列表
        this.newObjectsList.addAll(other.newObjectsList);
        this.skippedObjectsList.addAll(other.skippedObjectsList);
        this.modifiedObjectsList.addAll(other.modifiedObjectsList);
        this.errorObjectsList.addAll(other.errorObjectsList);
        this.warnings.addAll(other.warnings);
        
        // 合并元数据
        if (other.metadata != null) {
            this.metadata.putAll(other.metadata);
        }
        
        // 合并统计
        if (other.statistics != null) {
            this.statistics.putAll(other.statistics);
        }
        
        // 更新耗时
        if (other.scanEndTime != null && 
            (this.scanEndTime == null || other.scanEndTime.isAfter(this.scanEndTime))) {
            this.scanEndTime = other.scanEndTime;
        }
        
        if (other.scanDurationMs != null) {
            this.scanDurationMs = (this.scanDurationMs == null ? 0 : this.scanDurationMs) + 
                                 other.scanDurationMs;
        }
        
        // 重新计算统计
        calculateScanRate();
        calculateObjectSizeStats();
    }
    
    // ========== 私有工具方法 ==========
    
    private static String generateScanId() {
        return "scan-" + UUID.randomUUID().toString();
    }
    
    private static String getInstanceId() {
        return System.getenv().getOrDefault("HOSTNAME", 
            System.getenv().getOrDefault("POD_NAME", "unknown"));
    }
    
    private static String getThreadId() {
        return String.valueOf(Thread.currentThread().getId());
    }
    
    private static String getStackTrace(Exception e) {
        StringBuilder sb = new StringBuilder();
        sb.append(e.getClass().getName()).append(": ").append(e.getMessage()).append("\n");
        for (StackTraceElement element : e.getStackTrace()) {
            sb.append("\tat ").append(element.toString()).append("\n");
        }
        return sb.toString();
    }
    
    // ========== Builder 扩展 ==========
    
    public static class ScanResultBuilder {
        public ScanResultBuilder durationFromStart() {
            if (scanStartTime != null && scanEndTime != null) {
                this.scanDurationMs(java.time.Duration.between(scanStartTime, scanEndTime).toMillis());
            }
            return this;
        }
        
        public ScanResultBuilder calculateMetrics() {
            this.scanDurationMs(java.time.Duration.between(scanStartTime, scanEndTime).toMillis());
            return this;
        }
    }
}
