    /**
     * 记录扫描
     */
    public void recordScan(String prefix, int newObjects, int scannedObjects, 
                          long durationMs, boolean success) {
        if (!metricsEnabled) return;
        
        try {
            // 基本计数器
            incrementCounter("s3.scan.total");
            incrementCounter("s3.scan.objects.total", scannedObjects);
            incrementCounter("s3.scan.objects.new", newObjects);
            
            if (success) {
                incrementCounter("s3.scan.success");
            } else {
                incrementCounter("s3.scan.failed");
            }
            
            // 耗时
            recordTimer("s3.scan.duration", durationMs);
            
            // 标签化指标
            Counter.builder("s3.scan.operation")
                .description("扫描操作")
                .tag("prefix", prefix)
                .tag("success", String.valueOf(success))
                .register(meterRegistry)
                .increment();
            
            Counter.builder("s3.scan.objects")
                .description("扫描对象数")
                .tag("prefix", prefix)
                .tag("type", "total")
                .register(meterRegistry)
                .increment(scannedObjects);
            
            Counter.builder("s3.scan.objects")
                .description("扫描对象数")
                .tag("prefix", prefix)
                .tag("type", "new")
                .register(meterRegistry)
                .increment(newObjects);
            
            // 速率
            if (durationMs > 0) {
                double objectsPerSecond = (double) scannedObjects / (durationMs / 1000.0);
                Gauge.builder("s3.scan.rate", objectsPerSecond, Double::doubleValue)
                    .description("扫描速率")
                    .tag("prefix", prefix)
                    .register(meterRegistry);
            }
            
            // 分布摘要
            DistributionSummary.builder("s3.scan.objects.per.operation")
                .description("每次扫描对象数")
                .tag("prefix", prefix)
                .register(meterRegistry)
                .record(scannedObjects);
            
            log.debug("记录扫描指标: prefix={}, new={}, scanned={}, duration={}ms, success={}",
                prefix, newObjects, scannedObjects, durationMs, success);
            
        } catch (Exception e) {
            log.error("记录扫描指标失败", e);
        }
    }
    
    /**
     * 记录处理成功
     */
    public void recordProcessingSuccess(String bucket, String key, 
                                       long durationMs, int dataSize) {
        if (!metricsEnabled) return;
        
        try {
            incrementCounter("s3.process.total");
            incrementCounter("s3.process.success");
            
            recordTimer("s3.process.duration", durationMs);
            
            // 标签化指标
            Counter.builder("s3.process.operation")
                .description("处理操作")
                .tag("bucket", bucket)
                .tag("success", "true")
                .register(meterRegistry)
                .increment();
            
            // 吞吐量
            if (durationMs > 0 && dataSize > 0) {
                double throughput = (double) dataSize / (durationMs / 1000.0) / 1024; // KB/s
                Gauge.builder("s3.process.throughput.instant", throughput, Double::doubleValue)
                    .description("处理吞吐量")
                    .tag("bucket", bucket)
                    .register(meterRegistry);
            }
            
            // 延迟
            recordTimer("s3.latency.process", durationMs);
            
            log.debug("记录处理成功: {}/{}, duration={}ms, dataSize={}",
                bucket, key, durationMs, dataSize);
            
        } catch (Exception e) {
            log.error("记录处理成功指标失败", e);
        }
    }

    /**
     * 记录移动
     */
    public void recordMove(String bucket, String key, String moveType, 
                          long durationMs, boolean success, long fileSize) {
        if (!metricsEnabled) return;
        
        try {
            incrementCounter("s3.move.total");
            
            if (success) {
                incrementCounter("s3.move.success");
                incrementCounter("s3.move.by.type." + moveType.toLowerCase());
            } else {
                incrementCounter("s3.move.failed");
            }
            
            recordTimer("s3.move.duration", durationMs);
            
            // 标签化指标
            Counter.builder("s3.move.operation")
                .description("移动操作")
                .tag("bucket", bucket)
                .tag("type", moveType)
                .tag("success", String.valueOf(success))
                .register(meterRegistry)
                .increment();
            
            // 传输速率
            if (success && fileSize > 0 && durationMs > 0) {
                double transferRate = (double) fileSize / (durationMs / 1000.0) / 1024 / 1024; // MB/s
                Gauge.builder("s3.move.transfer.rate", transferRate, Double::doubleValue)
                    .description("移动传输速率")
                    .tag("bucket", bucket)
                    .tag("type", moveType)
                    .register(meterRegistry);
            }
            
            // 文件大小分布
            if (fileSize > 0) {
                DistributionSummary.builder("s3.move.file.size.distribution")
                    .description("移动文件大小分布")
                    .tag("bucket", bucket)
                    .tag("type", moveType)
                    .register(meterRegistry)
                    .record(fileSize);
            }
            
            log.debug("记录移动: {}/{}, type={}, duration={}ms, success={}, size={}",
                bucket, key, moveType, durationMs, success, fileSize);
            
        } catch (Exception e) {
            log.error("记录移动指标失败", e);
        }
    }

    /**
     * 开始计时
     */
    public String startTimer(String timerName) {
        if (!metricsEnabled) return null;
        
        try {
            String timerId = UUID.randomUUID().toString();
            Timer.Sample sample = Timer.start(meterRegistry);
            activeTimers.put(timerId, sample);
            return timerId;
            
        } catch (Exception e) {
            log.error("开始计时失败", e);
            return null;
        }
    }
    
    /**
     * 停止计时
     */
    public void stopTimer(String timerId, String timerName, String... tags) {
        if (!metricsEnabled || timerId == null) return;
        
        try {
            Timer.Sample sample = activeTimers.remove(timerId);
            if (sample != null) {
                Timer timer = Timer.builder(timerName)
                    .description("自定义计时器")
                    .tags(tags)
                    .register(meterRegistry);
                
                sample.stop(timer);
            }
            
        } catch (Exception e) {
            log.error("停止计时失败", e);
        }
    }

    /**
     * 更新队列大小
     */
    public void updateQueueSize(String queueName, int size) {
        if (!metricsEnabled) return;
        
        try {
            AtomicLong gauge = customGauges.get(queueName);
            if (gauge == null) {
                gauge = new AtomicLong(0);
                Gauge.builder(queueName, gauge, AtomicLong::get)
                    .description("队列大小")
                    .register(meterRegistry);
                customGauges.put(queueName, gauge);
            }
            
            gauge.set(size);
            
        } catch (Exception e) {
            log.error("更新队列大小指标失败", e);
        }
    }
    
    /**
     * 获取指标快照
     */
    public Map<String, Object> getMetricsSnapshot() {
        if (!metricsEnabled) {
            return Collections.singletonMap("metricsEnabled", false);
        }
        
        // 检查缓存
        if (Duration.between(lastCacheUpdate, Instant.now()).compareTo(CACHE_TTL) < 0) {
            return metricsCache;
        }
        
        try {
            Map<String, Object> snapshot = new LinkedHashMap<>();
            snapshot.put("timestamp", Instant.now().toString());
            snapshot.put("metricsEnabled", true);
            
            // 收集计数器
            Map<String, Long> counterValues = new HashMap<>();
            customCounters.forEach((name, counter) -> 
                counterValues.put(name, counter.longValue()));
            snapshot.put("counters", counterValues);
            
            // 收集仪表
            Map<String, Long> gaugeValues = new HashMap<>();
            customGauges.forEach((name, gauge) -> 
                gaugeValues.put(name, gauge.get()));
            snapshot.put("gauges", gaugeValues);
            
            // 收集计时器统计
            Map<String, Object> timerStats = new HashMap<>();
            timers.forEach((name, timer) -> {
                Map<String, Object> stats = new HashMap<>();
                stats.put("count", timer.count());
                stats.put("totalTime", timer.totalTime(TimeUnit.MILLISECONDS));
                stats.put("mean", timer.mean(TimeUnit.MILLISECONDS));
                stats.put("max", timer.max(TimeUnit.MILLISECONDS));
                timerStats.put(name, stats);
            });
            snapshot.put("timers", timerStats);
            
            // 收集分布摘要统计
            Map<String, Object> summaryStats = new HashMap<>();
            summaries.forEach((name, summary) -> {
                Map<String, Object> stats = new HashMap<>();
                stats.put("count", summary.count());
                stats.put("total", summary.totalAmount());
                stats.put("mean", summary.mean());
                stats.put("max", summary.max());
                summaryStats.put(name, stats);
            });
            snapshot.put("summaries", summaryStats);
            
            // 系统指标
            Runtime runtime = Runtime.getRuntime();
            snapshot.put("system", Map.of(
                "memoryUsed", runtime.totalMemory() - runtime.freeMemory(),
                "memoryMax", runtime.maxMemory(),
                "threadCount", Thread.activeCount(),
                "uptime", ManagementFactory.getRuntimeMXBean().getUptime()
            ));
            
            // 更新缓存
            metricsCache.clear();
            metricsCache.putAll(snapshot);
            lastCacheUpdate = Instant.now();
            
            return snapshot;
            
        } catch (Exception e) {
            log.error("获取指标快照失败", e);
            return Collections.singletonMap("error", e.getMessage());
        }
    }
    
    /**
     * 重置所有指标
     */
    public void resetMetrics() {
        if (!metricsEnabled) return;
        
        try {
            // 重置计数器
            customCounters.values().forEach(LongAdder::reset);
            
            // 重置仪表
            customGauges.values().forEach(gauge -> gauge.set(0));
            
            // 清空缓存
            metricsCache.clear();
            activeTimers.clear();
            
            log.info("指标已重置");
            
        } catch (Exception e) {
            log.error("重置指标失败", e);
        }
    }
    
    /**
     * 注册所有指标
     */
    private void registerAllMetrics() {
        // 系统指标
        new MeterBinder() {
            @Override
            public void bindTo(MeterRegistry registry) {
                // JVM 内存
                Gauge.builder("jvm.memory.used", Runtime.getRuntime(), 
                    r -> r.totalMemory() - r.freeMemory())
                    .description("JVM内存使用量")
                    .register(registry);
                
                Gauge.builder("jvm.memory.max", Runtime.getRuntime(), 
                    Runtime::maxMemory)
                    .description("JVM最大内存")
                    .register(registry);
                
                // 线程
                Gauge.builder("jvm.threads.live", Thread::activeCount)
                    .description("JVM活动线程数")
                    .register(registry);
            }
        }.bindTo(meterRegistry);
    }
    
    /**
     * 注册计数器
     */
    private void registerCounter(String name, String description) {
        Counter counter = Counter.builder(name)
            .description(description)
            .register(meterRegistry);
        
        counters.put(name, counter);
        customCounters.put(name, new LongAdder());
    }
    
    /**
     * 注册仪表
     */
    private void registerGauge(String name, String description) {
        AtomicLong gauge = new AtomicLong(0);
        
        Gauge.builder(name, gauge, AtomicLong::get)
            .description(description)
            .register(meterRegistry);
        
        gauges.put(name, null); // 存储引用
        customGauges.put(name, gauge);
    }
    
    /**
     * 注册计时器
     */
    private void registerTimer(String name, String description) {
        Timer timer = Timer.builder(name)
            .description(description)
            .register(meterRegistry);
        
        timers.put(name, timer);
    }
    
    /**
     * 注册分布摘要
     */
    private void registerDistributionSummary(String name, String description) {
        DistributionSummary summary = DistributionSummary.builder(name)
            .description(description)
            .register(meterRegistry);
        
        summaries.put(name, summary);
    }
    
    /**
     * 增加计数器
     */
    private void incrementCounter(String name) {
        incrementCounter(name, 1);
    }
    
    private void incrementCounter(String name, long amount) {
        Counter counter = counters.get(name);
        if (counter != null) {
            counter.increment(amount);
        }
        
        LongAdder customCounter = customCounters.get(name);
        if (customCounter != null) {
            customCounter.add(amount);
        }
    }
    
    /**
     * 更新仪表
     */
    private void updateGauge(String name, long delta) {
        AtomicLong gauge = customGauges.get(name);
        if (gauge != null) {
            gauge.addAndGet(delta);
        }
    }
    
    /**
     * 记录计时器
     */
    private void recordTimer(String name, long durationMs) {
        Timer timer = timers.get(name);
        if (timer != null) {
            timer.record(durationMs, TimeUnit.MILLISECONDS);
        }
    }
    
    /**
     * 提取错误类型
     */
    private String extractErrorType(String error) {
        if (error == null) {
            return "Unknown";
        }
        
        if (error.contains("Timeout")) {
            return "Timeout";
        } else if (error.contains("Connection")) {
            return "Connection";
        } else if (error.contains("Access Denied") || error.contains("AccessDenied")) {
            return "AccessDenied";
        } else if (error.contains("NoSuchKey")) {
            return "NotFound";
        } else if (error.contains("Throttling")) {
            return "Throttling";
        } else if (error.contains("BucketAlreadyExists")) {
            return "BucketExists";
        } else if (error.contains("NoSuchBucket")) {
            return "BucketNotFound";
        } else if (error.contains("InvalidAccessKeyId") || error.contains("Signature")) {
            return "AuthError";
        } else if (error.contains("SlowDown")) {
            return "Throttling";
        } else {
            return "Other";
        }
    }
