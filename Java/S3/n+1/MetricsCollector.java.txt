    @PostConstruct
    public void init() {
        if (!metricsEnabled) {
            log.warn("指标收集已禁用");
            return;
        }
        
        // 初始化所有指标
        initMetrics();
        log.info("指标收集器初始化完成");
    }
    
    @Override
    public void bindTo(MeterRegistry registry) {
        // 绑定所有指标
        registerAllMetrics();
    }

    /**
     * 记录扫描
     */
    public void recordScan(String prefix, int newObjects, int scannedObjects, 
                          long durationMs, boolean success) {
        if (!metricsEnabled) return;
        
        try {
            // 基本计数器
            incrementCounter("s3.scan.total");
            incrementCounter("s3.scan.objects.total", scannedObjects);
            incrementCounter("s3.scan.objects.new", newObjects);
            
            if (success) {
                incrementCounter("s3.scan.success");
            } else {
                incrementCounter("s3.scan.failed");
            }
            
            // 耗时
            recordTimer("s3.scan.duration", durationMs);
            
            // 标签化指标
            Counter.builder("s3.scan.operation")
                .description("扫描操作")
                .tag("prefix", prefix)
                .tag("success", String.valueOf(success))
                .register(meterRegistry)
                .increment();
            
            Counter.builder("s3.scan.objects")
                .description("扫描对象数")
                .tag("prefix", prefix)
                .tag("type", "total")
                .register(meterRegistry)
                .increment(scannedObjects);
            
            Counter.builder("s3.scan.objects")
                .description("扫描对象数")
                .tag("prefix", prefix)
                .tag("type", "new")
                .register(meterRegistry)
                .increment(newObjects);
            
            // 速率
            if (durationMs > 0) {
                double objectsPerSecond = (double) scannedObjects / (durationMs / 1000.0);
                Gauge.builder("s3.scan.rate", objectsPerSecond, Double::doubleValue)
                    .description("扫描速率")
                    .tag("prefix", prefix)
                    .register(meterRegistry);
            }

            DistributionSummary.builder("s3.scan.objects.per.operation")
                .description("每次扫描对象
