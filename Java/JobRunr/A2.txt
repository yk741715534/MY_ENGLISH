package com.example.s3processor.config;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import software.amazon.awssdk.auth.credentials.AwsBasicCredentials;
import software.amazon.awssdk.auth.credentials.StaticCredentialsProvider;
import software.amazon.awssdk.regions.Region;
import software.amazon.awssdk.services.s3.S3Client;
import software.amazon.awssdk.services.s3.S3AsyncClient;

import java.net.URI;

@Configuration
public class S3Config {

    @Value("${aws.s3.region:us-east-1}")
    private String region;

    @Value("${aws.s3.access-key:}")
    private String accessKey;

    @Value("${aws.s3.secret-key:}")
    private String secretKey;

    @Value("${aws.s3.endpoint:#{null}}")
    private String endpoint;

    @Bean
    public S3Client s3Client() {
        S3Client.Builder builder = S3Client.builder()
            .region(Region.of(region));

        // 如果有凭证，使用凭证
        if (accessKey != null && !accessKey.isEmpty() &&
            secretKey != null && !secretKey.isEmpty()) {
            AwsBasicCredentials credentials = AwsBasicCredentials.create(accessKey, secretKey);
            builder.credentialsProvider(StaticCredentialsProvider.create(credentials));
        }

        // 如果是本地开发或MinIO，设置endpoint
        if (endpoint != null && !endpoint.isEmpty()) {
            builder.endpointOverride(URI.create(endpoint))
                   .forcePathStyle(true);
        }

        return builder.build();
    }

    @Bean
    public S3AsyncClient s3AsyncClient() {
        S3AsyncClient.Builder builder = S3AsyncClient.builder()
            .region(Region.of(region));

        if (accessKey != null && !accessKey.isEmpty() &&
            secretKey != null && !secretKey.isEmpty()) {
            AwsBasicCredentials credentials = AwsBasicCredentials.create(accessKey, secretKey);
            builder.credentialsProvider(StaticCredentialsProvider.create(credentials));
        }

        if (endpoint != null && !endpoint.isEmpty()) {
            builder.endpointOverride(URI.create(endpoint))
                   .forcePathStyle(true);
        }

        return builder.build();
    }
}

    @Value("${org.jobrunr.background-job-server.worker-count:10}")
    private int workerCount;

    @Bean
    public StorageProvider storageProvider(MongoClient mongoClient,
                                          @Value("${spring.data.mongodb.database:s3processor}")

        return new MongoDBStorageProvider(mongoClient, databaseName);

        return JobRunr.configure()
            .useStorageProvider(storageProvider)
            .useBackgroundJobServer(workerCount)
            .useDashboard(dashboardPort)

            .initialize();
    }

    @Bean
    public MongoTemplate mongoTemplate(MongoClient mongoClient,
                                      @Value("${spring.data.mongodb.database:s3processor}")
                                      String databaseName) {
        return new MongoTemplate(mongoClient, databaseName);
    }

package com.example.s3processor.config;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.scheduling.concurrent.ThreadPoolTaskExecutor;

import java.util.concurrent.Executor;
import java.util.concurrent.ThreadPoolExecutor;

@Configuration
public class AppConfig {

    @Value("${processing.workers:5}")
    private int processingWorkers;

    @Value("${processing.timeout-seconds:300}")
    private int timeoutSeconds;

    @Bean("fileProcessingExecutor")
    public Executor fileProcessingExecutor() {
        ThreadPoolTaskExecutor executor = new ThreadPoolTaskExecutor();
        executor.setCorePoolSize(processingWorkers);
        executor.setMaxPoolSize(processingWorkers * 2);
        executor.setQueueCapacity(100);
        executor.setThreadNamePrefix("file-processor-");
        executor.setRejectedExecutionHandler(new ThreadPoolExecutor.CallerRunsPolicy());
        executor.setKeepAliveSeconds(60);
        executor.initialize();
        return executor;
    }

    @Bean
    public S3DirectoryConfig s3DirectoryConfig(
            @Value("${aws.s3.directories.source:uploads/}") String source,
            @Value("${aws.s3.directories.processing:processing/}") String processing,
            @Value("${aws.s3.directories.backup:backup/}") String backup,
            @Value("${aws.s3.directories.processed:processed/}") String processed,
            @Value("${aws.s3.directories.failed:failed/}") String failed,
            @Value("${aws.s3.directories.retry:retry/}") String retry) {
        return new S3DirectoryConfig(source, processing, backup, processed, failed, retry);
    }

    public record S3DirectoryConfig(
        String source,
        String processing,
        String backup,
        String processed,
        String failed,
        String retry
    ) {}
}

package com.example.s3processor.model;

import lombok.Data;
import lombok.Builder;
import lombok.NoArgsConstructor;
import lombok.AllArgsConstructor;
import software.amazon.awssdk.services.s3.model.S3Object;

import java.time.Instant;

@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class S3FileInfo {
    private String key;               // S3 key
    private String bucket;            // 桶名
    private String fileName;          // 文件名
    private long size;                // 文件大小
    private Instant lastModified;     // 最后修改时间
    private String contentType;       // 内容类型
    private String etag;              // ETag
    private String storageClass;      // 存储类型

    public static S3FileInfo fromS3Object(S3Object s3Object, String bucket) {
        String key = s3Object.key();
        String fileName = key.substring(key.lastIndexOf('/') + 1);

        return S3FileInfo.builder()
            .key(key)
            .bucket(bucket)
            .fileName(fileName)
            .size(s3Object.size())
            .lastModified(s3Object.lastModified())
            .etag(s3Object.eTag())
            .storageClass(s3Object.storageClassAsString())
            .build();
    }

    public boolean isRecentFile(int minutesThreshold) {
        if (lastModified == null) return true;
        Instant threshold = Instant.now().minusSeconds(minutesThreshold * 60L);
        return lastModified.isAfter(threshold);
    }
}

package com.example.s3processor.service;

import com.example.s3processor.config.AppConfig.S3DirectoryConfig;
import com.example.s3processor.model.S3FileInfo;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import software.amazon.awssdk.core.ResponseBytes;
import software.amazon.awssdk.core.async.AsyncRequestBody;
import software.amazon.awssdk.core.sync.RequestBody;
import software.amazon.awssdk.services.s3.S3AsyncClient;
import software.amazon.awssdk.services.s3.S3Client;
import software.amazon.awssdk.services.s3.model.*;
import software.amazon.awssdk.services.s3.waiters.S3Waiter;

import java.io.ByteArrayInputStream;
import java.io.InputStream;
import java.nio.file.Path;
import java.time.Instant;
import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.CompletableFuture;
import java.util.stream.Collectors;

@Slf4j
@Service
public class S3FileService {

    @Autowired
    private S3Client s3Client;

    @Autowired
    private S3AsyncClient s3AsyncClient;

    @Autowired
    private S3DirectoryConfig directoryConfig;

    @Value("${aws.s3.bucket}")
    private String bucketName;

    @Value("${aws.s3.polling.file-age-minutes:5}")
    private int fileAgeMinutes;

    /**
     * 列出指定目录下的新文件
     */
    public List<S3FileInfo> listNewFiles(String prefix) {
        return listNewFiles(prefix, fileAgeMinutes);
    }

    /**
     * 列出指定目录下、指定时间内的新文件
     */
    public List<S3FileInfo> listNewFiles(String prefix, int minutesThreshold) {
        List<S3FileInfo> newFiles = new ArrayList<>();
        String continuationToken = null;

        do {
            ListObjectsV2Request request = ListObjectsV2Request.builder()
                .bucket(bucketName)
                .prefix(prefix)
                .continuationToken(continuationToken)
                .build();

            ListObjectsV2Response response = s3Client.listObjectsV2(request);

            // 过滤新文件
            List<S3FileInfo> batchFiles = response.contents().stream()
                .filter(s3Object -> !s3Object.key().endsWith("/"))  // 排除目录
                .filter(s3Object -> isNewFile(s3Object, minutesThreshold))
                .map(s3Object -> S3FileInfo.fromS3Object(s3Object, bucketName))
                .collect(Collectors.toList());

            newFiles.addAll(batchFiles);
            continuationToken = response.nextContinuationToken();

        } while (continuationToken != null && newFiles.size() < 1000); // 限制数量

        log.debug("在目录 {} 中找到 {} 个新文件", prefix, newFiles.size());
        return newFiles;
    }

    /**
     * 检查是否是新的文件
     */
    private boolean isNewFile(S3Object s3Object, int minutesThreshold) {
        if (s3Object.lastModified() == null) return true;

        Instant threshold = Instant.now().minusSeconds(minutesThreshold * 60L);
        return s3Object.lastModified().isBefore(threshold);
    }

    /**
     * 下载文件
     */
    public byte[] downloadFile(String key) {
        try {
            GetObjectRequest request = GetObjectRequest.builder()
                .bucket(bucketName)
                .key(key)
                .build();

            ResponseBytes<GetObjectResponse> responseBytes = s3Client.getObjectAsBytes(request);
            return responseBytes.asByteArray();

        } catch (S3Exception e) {
            log.error("下载文件失败: {}", key, e);
            throw new RuntimeException("下载文件失败: " + key, e);
        }
    }

    /**
     * 异步下载文件
     */
    public CompletableFuture<byte[]> downloadFileAsync(String key) {
        GetObjectRequest request = GetObjectRequest.builder()
            .bucket(bucketName)
            .key(key)
            .build();

        return s3AsyncClient.getObject(request, new S3AsyncResponseTransformer())
            .thenApply(responseBytes -> {
                try {
                    return responseBytes.asByteArray();
                } catch (Exception e) {
                    throw new RuntimeException("下载文件失败: " + key, e);
                }
            });
    }

    /**
     * 上传文件
     */
    public void uploadFile(String key, byte[] content, String contentType) {
        try {
            PutObjectRequest request = PutObjectRequest.builder()
                .bucket(bucketName)
                .key(key)
                .contentType(contentType)
                .contentLength((long) content.length)
                .build();

            s3Client.putObject(request, RequestBody.fromBytes(content));
            log.debug("文件上传成功: {}", key);

        } catch (S3Exception e) {
            log.error("上传文件失败: {}", key, e);
            throw new RuntimeException("上传文件失败: " + key, e);
        }
    }

    /**
     * 复制文件
     */
    public void copyFile(String sourceKey, String destinationKey) {
        try {
            CopyObjectRequest request = CopyObjectRequest.builder()
                .sourceBucket(bucketName)
                .sourceKey(sourceKey)
                .destinationBucket(bucketName)
                .destinationKey(destinationKey)
                .build();

            s3Client.copyObject(request);
            log.debug("文件复制成功: {} -> {}", sourceKey, destinationKey);

        } catch (S3Exception e) {
            log.error("复制文件失败: {} -> {}", sourceKey, destinationKey, e);
            throw new RuntimeException("复制文件失败: " + sourceKey + " -> " + destinationKey, e);
        }
    }

    /**
     * 移动文件（复制+删除）
     */
    public void moveFile(String sourceKey, String destinationKey) {
        copyFile(sourceKey, destinationKey);
        deleteFile(sourceKey);
        log.info("文件移动成功: {} -> {}", sourceKey, destinationKey);
    }

    /**
     * 删除文件
     */
    public void deleteFile(String key) {
        try {
            DeleteObjectRequest request = DeleteObjectRequest.builder()
                .bucket(bucketName)
                .key(key)
                .build();

            s3Client.deleteObject(request);
            log.debug("文件删除成功: {}", key);

        } catch (S3Exception e) {
            log.error("删除文件失败: {}", key, e);
            throw new RuntimeException("删除文件失败: " + key, e);
        }
    }

    /**
     * 批量删除文件
     */
    public void deleteFiles(List<String> keys) {
        if (keys.isEmpty()) return;

        try {
            List<ObjectIdentifier> objects = keys.stream()
                .map(key -> ObjectIdentifier.builder().key(key).build())
                .collect(Collectors.toList());

            Delete delete = Delete.builder().objects(objects).build();
            DeleteObjectsRequest request = DeleteObjectsRequest.builder()
                .bucket(bucketName)
                .delete(delete)
                .build();

            s3Client.deleteObjects(request);
            log.debug("批量删除成功，数量: {}", keys.size());

        } catch (S3Exception e) {
            log.error("批量删除失败", e);
            throw new RuntimeException("批量删除失败", e);
        }
    }

    /**
     * 检查文件是否存在
     */
    public boolean fileExists(String key) {
        try {
            HeadObjectRequest request = HeadObjectRequest.builder()
                .bucket(bucketName)
                .key(key)
                .build();

            s3Client.headObject(request);
            return true;

        } catch (NoSuchKeyException e) {
            return false;
        } catch (S3Exception e) {
            log.error("检查文件是否存在失败: {}", key, e);
            throw new RuntimeException("检查文件失败: " + key, e);
        }
    }

    /**
     * 获取文件元数据
     */
    public HeadObjectResponse getFileMetadata(String key) {
        try {
            HeadObjectRequest request = HeadObjectRequest.builder()
                .bucket(bucketName)
                .key(key)
                .build();

            return s3Client.headObject(request);

        } catch (S3Exception e) {

            throw new RuntimeException("获取文件元数据失败: " + key, e);
        }
    }

            // 这里可以使用MessageDigest计算MD5

            return "md5_" + content.length; // 简化返回
        } catch (Exception e) {
            log.error("计算MD5失败", e);
            return null;
        }
    }

    public String backupFile(String sourceKey) {

     * 使用 @Job 注解让 JobRunr 管理重试
     */
    @Job(
        name = "处理文件: %0",
        retries = 3,
        labels = {"s3-file-processing"}
    )
    @Transactional
    public ProcessedFile processFile(String fileKey) {
        log.info("开始处理文件: {}", fileKey);

        ProcessedFile processedFile = null;
        LocalDateTime startTime = LocalDateTime.now();
        String backupKey = null;

        try {
            // 1. 检查文件是否已处理过
            processedFile = processedFileRepository.findByFileKey(fileKey).orElse(null);
            if (processedFile != null && processedFile.getStatus() == FileStatus.SUCCESS) {
                log.info("文件已处理过，跳过: {}", fileKey);
                return processedFile;
            }

            // 2. 记录开始处理
            if (processedFile == null) {
                processedFile = createProcessedFileRecord(fileKey, startTime);
            } else {
                updateProcessedFileRecord(processedFile, startTime, FileStatus.PROCESSING);
            }

            // 3. 备份文件
            backupKey = s3FileService.backupFile(fileKey);
            processedFile.setBackupKey(backupKey);
            processedFile.setStatus(FileStatus.PROCESSING);
            processedFileRepository.save(processedFile);

            // 4. 下载文件
            byte[] fileContent = s3FileService.downloadFile(fileKey);

            // 5. 验证文件
            validateFile(fileKey, fileContent, processedFile);

            // 6. 执行业务处理逻辑
            Map<String, Object> result = executeBusinessLogic(fileKey, fileContent, processedFile);

            // 7. 移动文件到已处理目录
            String processedKey = directoryConfig.processed() +
                                 fileKey.substring(fileKey.lastIndexOf('/') + 1);
            s3FileService.moveFile(fileKey, processedKey);

            // 8. 更新处理结果
            updateProcessedFileSuccess(processedFile, processedKey, result, startTime);

            log.info("文件处理成功: {}", fileKey);
            return processedFile;

        } catch (Exception e) {
            log.error("文件处理失败: {}", fileKey, e);

            if (processedFile != null) {
                updateProcessedFileFailure(processedFile, e.getMessage(), startTime);
            }

            // 如果处理失败，移动文件到失败目录
            try {
                String failedKey = directoryConfig.failed() +
                                  fileKey.substring(fileKey.lastIndexOf('/') + 1);
                s3FileService.moveFile(fileKey, failedKey);
            } catch (Exception moveEx) {
                log.error("移动失败文件到失败目录异常: {}", fileKey, moveEx);
            }

            throw e; // 抛出异常让 JobRunr 重试
        }
    }

import java.util.List;

@Slf4j
@Service
public class FileMonitorService {

    @Autowired
    private S3FileService s3FileService;

    @Autowired
    private FileProcessorService fileProcessorService;

    @Autowired
    private S3DirectoryConfig directoryConfig;

    @Value("${aws.s3.polling.enabled:true}")
    private boolean pollingEnabled;

    @Value("${aws.s3.polling.batch-size:100}")
    private int batchSize;

    /**
     * 定时轮询S3目录，发现新文件并处理
     * 使用 @Recurring 注解，确保在集群中只有一个实例执行
     */
    @Recurring(
        id = "s3-file-monitor",
        cron = "${aws.s3.polling.cron:*/30 * * * * *}",
        zoneId = "${user.timezone:Asia/Shanghai}"
    )
    public void monitorAndProcessFiles(JobContext jobContext) {
        if (!pollingEnabled) {
            log.debug("轮询监控已禁用");
            return;
        }

        log.info("开始轮询S3目录: {}", directoryConfig.source());

        try {
            // 1. 发现新文件
            List<S3FileInfo> newFiles = s3FileService.listNewFiles(directoryConfig.source());

            if (newFiles.isEmpty()) {
                log.info("没有发现新文件");
                jobContext.saveMetadata("newFiles", 0);
                return;
            }

            log.info("发现 {} 个新文件", newFiles.size());
            jobContext.saveMetadata("newFiles", newFiles.size());

            // 2. 限制处理数量
            List<S3FileInfo> filesToProcess = newFiles.stream()
                .limit(batchSize)
                .collect(Collectors.toList());

            // 3. 创建进度条
            JobDashboardProgressBar progressBar = jobContext.progressBar(filesToProcess.size());

            // 4. 批量处理文件
            AtomicInteger processedCount = new AtomicInteger(0);

            for (S3FileInfo file : filesToProcess) {
                try {
                    String fileKey = file.getKey();
                    log.debug("调度处理文件: {}", fileKey);

                    // 异步处理文件
                    fileProcessorService.processFileAsync(fileKey);

                    // 更新进度
                    processedCount.incrementAndGet();
                    progressBar.setValue(processedCount.get());

                } catch (Exception e) {
                    log.error("调度处理文件失败: {}", file.getKey(), e);
                }
            }

            // 5. 记录统计信息
            jobContext.saveMetadata("processedCount", processedCount.get());
            jobContext.saveMetadata("skippedCount", newFiles.size() - processedCount.get());

            log.info("文件轮询处理完成，已调度 {} 个文件", processedCount.get());

        } catch (Exception e) {
            log.error("轮询监控异常", e);
            jobContext.saveMetadata("error", e.getMessage());
            throw e; // 抛出异常让JobRunr记录失败
        }
    }

    /**
     * 手动触发文件处理
     */
    public int triggerManualProcessing() {
        log.info("手动触发文件处理");

        List<S3FileInfo> newFiles = s3FileService.listNewFiles(directoryConfig.source());

        if (newFiles.isEmpty()) {
            log.info("没有发现新文件");
            return 0;
        }

        int processedCount = 0;
        for (S3FileInfo file : newFiles) {
            try {
                fileProcessorService.processFileAsync(file.getKey());
                processedCount++;
            } catch (Exception e) {
                log.error("处理文件失败: {}", file.getKey(), e);
            }
        }

        log.info("手动触发处理完成，已调度 {} 个文件", processedCount);
        return processedCount;
    }

@Slf4j
@Service
public class FileRetryService {

    @Autowired
    private S3FileService s3FileService;

    @Autowired
    private FileProcessorService fileProcessorService;

    @Autowired
    private ProcessedFileRepository processedFileRepository;

    @Autowired
    private S3DirectoryConfig directoryConfig;

    /**
     * 重试单个失败文件
     */
    @Job(
        name = "重试文件: %0",
        retries = 3,
        labels = {"file-retry"}
    )
    public ProcessedFile retryFailedFile(String fileName, FileRetryRequest request) {
        log.info("开始重试文件: {}", fileName);

        String sourceDirectory = request.getSourceDirectory();
        if (!StringUtils.hasText(sourceDirectory)) {
            sourceDirectory = directoryConfig.failed();
        }

        String sourceKey = sourceDirectory + fileName;

        // 检查文件是否存在
        if (!s3FileService.fileExists(sourceKey)) {
            throw new RuntimeException("文件不存在: " + sourceKey);
        }

        // 检查是否已处理过
        if (!request.getForceRetry()) {
            boolean alreadyProcessed = processedFileRepository.existsByFileNameAndOriginalPathAndStatusIn(
                fileName, sourceKey,
                List.of(FileStatus.SUCCESS, FileStatus.PROCESSING)
            );

            if (alreadyProcessed) {
                log.info("文件已处理过，跳过重试: {}", fileName);
                return null;
            }
        }

        // 从源目录复制到重试目录
        String retryKey = directoryConfig.retry() + fileName;
        s3FileService.copyFile(sourceKey, retryKey);

        // 处理重试目录中的文件
        return fileProcessorService.processFile(retryKey);
    }

    /**
     * 从备份目录恢复并重试文件
     */
    @Job(
        name = "从备份恢复文件: %0",
        retries = 2,
        labels = {"file-restore"}
    )
    public ProcessedFile retryFromBackup(String fileName) {
        log.info("从备份恢复文件: {}", fileName);

        // 查找备份文件
        String backupKey = findBackupFile(fileName);
        if (backupKey == null) {
            throw new RuntimeException("未找到备份文件: " + fileName);
        }

        // 从备份复制到重试目录
        String retryKey = directoryConfig.retry() + fileName;
        s3FileService.copyFile(backupKey, retryKey);

        // 处理文件
        return fileProcessorService.processFile(retryKey);
    }

    /**
     * 批量重试文件
     */
    @Job(
        name = "批量重试文件",
        retries = 2,
        labels = {"batch-retry"}
    )
    public List<ProcessedFile> batchRetryFiles(FileRetryRequest request) {
        log.info("批量重试文件，数量: {}",
            request.getFileNames() != null ? request.getFileNames().size() : 0);

        List<ProcessedFile> results = new ArrayList<>();

        if (request.getFileNames() != null && !request.getFileNames().isEmpty()) {
            for (String fileName : request.getFileNames()) {
                try {
                    FileRetryRequest singleRequest = FileRetryRequest.singleFile(
                        fileName, request.getSourceDirectory()
                    );
                    singleRequest.setMaxRetries(request.getMaxRetries());
                    singleRequest.setForceRetry(request.getForceRetry());

                    ProcessedFile result = retryFailedFile(fileName, singleRequest);
                    if (result != null) {
                        results.add(result);
                    }
                } catch (Exception e) {
                    log.error("重试文件失败: {}", fileName, e);
                }
            }
        }

        return results;
    }

    /**
     * 重试所有失败文件
     */
    @Job(
        name = "重试所有失败文件",
        labels = {"retry-all-failed"}
    )
    public List<ProcessedFile> retryAllFailedFiles() {
        log.info("重试所有失败文件");

        // 查询所有失败的文件
        List<ProcessedFile> failedFiles = processedFileRepository.findByStatus(FileStatus.FAILED);

        List<ProcessedFile> results = new ArrayList<>();
        for (ProcessedFile file : failedFiles) {
            try {
                // 检查文件是否还在失败目录
                String failedKey = directoryConfig.failed() + file.getFileName();
                if (s3FileService.fileExists(failedKey)) {
                    FileRetryRequest request = FileRetryRequest.singleFile(
                        file.getFileName(), directoryConfig.failed()
                    );

                    ProcessedFile result = retryFailedFile(file.getFileName(), request);
                    if (result != null) {
                        results.add(result);
                    }
                }
            } catch (Exception e) {
                log.error("重试失败文件异常: {}", file.getFileName(), e);
            }
        }

        log.info("重试所有失败文件完成，成功: {}，总数: {}", results.size(), failedFiles.size());
        return results;
    }

    /**
     * 根据文件模式重试
     */
    public List<ProcessedFile> retryFilesByPattern(String filePattern, String sourceDirectory) {
        log.info("根据模式重试文件: {}", filePattern);

        // 查询匹配的文件
        List<S3FileInfo> matchedFiles = s3FileService.listNewFiles(sourceDirectory, 0).stream()
            .filter(file -> matchesPattern(file.getFileName(), filePattern))
            .collect(Collectors.toList());

        List<ProcessedFile> results = new ArrayList<>();
        for (S3FileInfo file : matchedFiles) {
            try {
                FileRetryRequest request = FileRetryRequest.singleFile(
                    file.getFileName(), sourceDirectory
                );

                ProcessedFile result = retryFailedFile(file.getFileName(), request);
                if (result != null) {
                    results.add(result);
                }
            } catch (Exception e) {
                log.error("模式重试文件失败: {}", file.getFileName(), e);
            }
        }

        return results;
    }

    /**
     * 查找备份文件
     */
    private String findBackupFile(String fileName) {
        // 在备份目录中查找包含文件名的文件
        List<S3FileInfo> backupFiles = s3FileService.listNewFiles(directoryConfig.backup(), 0);

        return backupFiles.stream()
            .filter(file -> file.getFileName().contains(fileName))
            .map(S3FileInfo::getKey)
            .findFirst()
            .orElse(null);
    }

    /**
     * 匹配文件模式
     */
    private boolean matchesPattern(String fileName, String pattern) {
        if (pattern == null || pattern.equals("*")) {
            return true;
        }

        if (pattern.startsWith("*") && pattern.endsWith("*")) {
            String middle = pattern.substring(1, pattern.length() - 1);
            return fileName.contains(middle);
        } else if (pattern.startsWith("*")) {
            String suffix = pattern.substring(1);
            return fileName.endsWith(suffix);
        } else if (pattern.endsWith("*")) {
            String prefix = pattern.substring(0, pattern.length() - 1);
            return fileName.startsWith(prefix);
        } else {
            return fileName.equals(pattern);
        }
    }

    /**
     * 准备文件重试（从备份复制到重试目录）
     */
    public String prepareFileForRetry(String fileName, String sourceDirectory) {
        String sourceKey = sourceDirectory + fileName;
        String retryKey = directoryConfig.retry() + fileName + "_" + System.currentTimeMillis();

        if (s3FileService.fileExists(sourceKey)) {
            s3FileService.copyFile(sourceKey, retryKey);
            return retryKey;
        } else {
            throw new RuntimeException("源文件不存在: " + sourceKey);
        }
    }

package com.example.s3processor.service;

import software.amazon.awssdk.core.async.AsyncResponseTransformer;
import software.amazon.awssdk.core.async.SdkPublisher;
import software.amazon.awssdk.services.s3.model.GetObjectResponse;
import software.amazon.awssdk.utils.IoUtils;

import java.nio.ByteBuffer;
import java.nio.file.Path;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.atomic.AtomicLong;

public class S3AsyncResponseTransformer
    implements AsyncResponseTransformer<GetObjectResponse, software.amazon.awssdk.core.ResponseBytes<GetObjectResponse>> {

    private final CompletableFuture<software.amazon.awssdk.core.ResponseBytes<GetObjectResponse>> future =
        new CompletableFuture<>();

    private software.amazon.awssdk.core.ResponseBytes.Builder<GetObjectResponse> responseBuilder;
    private GetObjectResponse response;
    private final ByteBufferOutputStream byteBufferOutputStream = new ByteBufferOutputStream();

    @Override
    public CompletableFuture<software.amazon.awssdk.core.ResponseBytes<GetObjectResponse>> prepare() {
        responseBuilder = software.amazon.awssdk.core.ResponseBytes.builder();
        return future;
    }

    @Override
    public void onResponse(GetObjectResponse response) {
        this.response = response;
        responseBuilder.response(response);
    }

    @Override
    public void onStream(SdkPublisher<ByteBuffer> publisher) {
        publisher.subscribe(new Subscriber());
    }

    @Override
    public void exceptionOccurred(Throwable error) {
        future.completeExceptionally(error);
    }

    private class Subscriber implements org.reactivestreams.Subscriber<ByteBuffer> {
        private org.reactivestreams.Subscription subscription;
        private final AtomicLong totalBytesRead = new AtomicLong();

        @Override
        public void onSubscribe(org.reactivestreams.Subscription subscription) {
            this.subscription = subscription;
            subscription.request(Long.MAX_VALUE);
        }

        @Override
        public void onNext(ByteBuffer byteBuffer) {
            try {
                byteBufferOutputStream.write(byteBuffer);
                totalBytesRead.addAndGet(byteBuffer.remaining());
            } catch (Exception e) {
                onError(e);
            }
        }

        @Override
        public void onError(Throwable t) {
            future.completeExceptionally(t);
        }

        @Override
        public void onComplete() {
            try {
                responseBuilder.bytes(byteBufferOutputStream.toByteArray());
                future.complete(responseBuilder.build());
            } catch (Exception e) {
                future.completeExceptionally(e);
            }
        }
    }

    private static class ByteBufferOutputStream {
        private final java.io.ByteArrayOutputStream outputStream = new java.io.ByteArrayOutputStream();

        public void write(ByteBuffer byteBuffer) throws java.io.IOException {
            byte[] bytes = new byte[byteBuffer.remaining()];
            byteBuffer.get(bytes);
            outputStream.write(bytes);
        }

        public byte[] toByteArray() {
            return outputStream.toByteArray();
        }
    }
}

            return s3Client.getObject(request);

        } catch (S3Exception e) {
            log.error("流式下载文件失败: {}", key, e);
            throw new RuntimeException("流式下载文件失败: " + key, e);
        }
    }

    /**
     * 获取文件URL（预签名URL）
     */
    public String generatePresignedUrl(String key, int expirationMinutes) {
        try {
            software.amazon.awssdk.services.s3.presigner.S3Presigner presigner =
                software.amazon.awssdk.services.s3.presigner.S3Presigner.builder()
                    .region(software.amazon.awssdk.regions.Region.of(region))
                    .build();

            GetObjectRequest getObjectRequest = GetObjectRequest.builder()
                .bucket(bucketName)
                .key(key)
                .build();

            software.amazon.awssdk.services.s3.presigner.model.GetObjectPresignRequest presignRequest =
                software.amazon.awssdk.services.s3.presigner.model.GetObjectPresignRequest.builder()
                    .signatureDuration(java.time.Duration.ofMinutes(expirationMinutes))
                    .getObjectRequest(getObjectRequest)
                    .build();

            return presigner.presignGetObject(presignRequest).url().toString();

        } catch (Exception e) {
            log.error("生成预签名URL失败: {}", key, e);
            return null;
        }
    }

    /**
     * 批量复制文件
     */
    public void batchCopyFiles(Map<String, String> sourceToDestinationMap) {
        for (Map.Entry<String, String> entry : sourceToDestinationMap.entrySet()) {
            try {
                copyFile(entry.getKey(), entry.getValue());
            } catch (Exception e) {
                log.error("批量复制文件失败: {} -> {}", entry.getKey(), entry.getValue(), e);
            }
        }
    }

    /**
     * 获取目录下的所有文件
     */
    public List<S3FileInfo> listAllFiles(String prefix) {
        List<S3FileInfo> allFiles = new ArrayList<>();
        String continuationToken = null;

        do {
            ListObjectsV2Request request = ListObjectsV2Request.builder()
                .bucket(bucketName)
                .prefix(prefix)
                .continuationToken(continuationToken)
                .build();

            ListObjectsV2Response response = s3Client.listObjectsV2(request);

            List<S3FileInfo> batchFiles = response.contents().stream()
                .filter(s3Object -> !s3Object.key().endsWith("/"))
                .map(s3Object -> S3FileInfo.fromS3Object(s3Object, bucketName))
                .collect(Collectors.toList());

            allFiles.addAll(batchFiles);
            continuationToken = response.nextContinuationToken();

        } while (continuationToken != null);

        return allFiles;
    }

    /**
     * 获取目录统计信息
     */
    public Map<String, Object> getDirectoryStats(String prefix) {
        List<S3FileInfo> files = listAllFiles(prefix);

        long totalSize = files.stream()
            .mapToLong(S3FileInfo::getSize)
            .sum();

        long fileCount = files.size();

        Instant oldestFileTime = files.stream()
            .map(S3FileInfo::getLastModified)
            .min(Instant::compareTo)
            .orElse(Instant.now());

        Instant newestFileTime = files.stream()
            .map(S3FileInfo::getLastModified)
            .max(Instant::compareTo)
            .orElse(Instant.now());

        Map<String, Object> stats = new HashMap<>();
        stats.put("directory", prefix);
        stats.put("fileCount", fileCount);
        stats.put("totalSize", totalSize);
        stats.put("totalSizeMB", totalSize / (1024.0 * 1024.0));
        stats.put("oldestFileTime", oldestFileTime);
        stats.put("newestFileTime", newestFileTime);
        stats.put("averageFileSize", fileCount > 0 ? totalSize / fileCount : 0);

        return stats;
    }

package com.example.s3processor.util;

import lombok.extern.slf4j.Slf4j;
import org.apache.commons.io.FilenameUtils;
import org.springframework.util.StringUtils;

import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.io.InputStream;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.Arrays;
import java.util.List;
import java.util.UUID;

@Slf4j
public class FileUtils {

    private static final List<String> ALLOWED_EXTENSIONS = Arrays.asList(
        "txt", "csv", "json", "xml", "log",
        "pdf", "doc", "docx", "xls", "xlsx"
    );

    private static final long MAX_FILE_SIZE = 100 * 1024 * 1024; // 100MB

    /**
     * 验证文件扩展名
     */
    public static boolean isValidExtension(String fileName) {
        if (!StringUtils.hasText(fileName)) {
            return false;
        }

        String extension = FilenameUtils.getExtension(fileName).toLowerCase();
        return ALLOWED_EXTENSIONS.contains(extension);
    }

    /**
     * 验证文件大小
     */
    public static boolean isValidFileSize(long fileSize) {
        return fileSize > 0 && fileSize <= MAX_FILE_SIZE;
    }

    /**
     * 生成文件处理ID
     */
    public static String generateFileId(String fileName) {
        String timestamp = LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyyMMddHHmmss"));
        String uuid = UUID.randomUUID().toString().substring(0, 8);
        return String.format("%s_%s_%s",
            FilenameUtils.getBaseName(fileName),
            timestamp,
            uuid
        );
    }

    /**
     * 计算文件的MD5哈希
     */
    public static String calculateMd5(byte[] content) {
        try {
            MessageDigest md = MessageDigest.getInstance("MD5");
            byte[] digest = md.digest(content);

            StringBuilder sb = new StringBuilder();
            for (byte b : digest) {
                sb.append(String.format("%02x", b));
            }
            return sb.toString();

        } catch (NoSuchAlgorithmException e) {
            log.error("MD5算法不可用", e);
            return null;
        }
    }

    /**
     * 计算文件的MD5哈希（流式）
     */
    public static String calculateMd5(InputStream inputStream) throws IOException {
        try {
            MessageDigest md = MessageDigest.getInstance("MD5");
            byte[] buffer = new byte[8192];
            int bytesRead;

            while ((bytesRead = inputStream.read(buffer)) != -1) {
                md.update(buffer, 0, bytesRead);
            }

            byte[] digest = md.digest();
            StringBuilder sb = new StringBuilder();
            for (byte b : digest) {
                sb.append(String.format("%02x", b));
            }
            return sb.toString();

        } catch (NoSuchAlgorithmException e) {
            log.error("MD5算法不可用", e);
            return null;
        }
    }

    /**
     * 安全获取文件扩展名
     */
    public static String getSafeExtension(String fileName) {
        if (!StringUtils.hasText(fileName)) {
            return "";
        }

        int lastDotIndex = fileName.lastIndexOf('.');
        if (lastDotIndex > 0 && lastDotIndex < fileName.length() - 1) {
            return fileName.substring(lastDotIndex + 1).toLowerCase();
        }

        return "";
    }

    /**
     * 获取文件基本名（无扩展名）
     */
    public static String getBaseName(String fileName) {
        if (!StringUtils.hasText(fileName)) {
            return "";
        }

        int lastDotIndex = fileName.lastIndexOf('.');
        if (lastDotIndex > 0) {
            return fileName.substring(0, lastDotIndex);
        }

        return fileName;
    }

    /**
     * 生成唯一的文件名
     */
    public static String generateUniqueFileName(String originalFileName) {
        String extension = getSafeExtension(originalFileName);
        String timestamp = LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyyMMdd_HHmmss"));
        String random = UUID.randomUUID().toString().substring(0, 6);

        if (StringUtils.hasText(extension)) {
            return String.format("%s_%s_%s.%s",
                getBaseName(originalFileName),
                timestamp,
                random,
                extension
            );
        } else {
            return String.format("%s_%s_%s",
                originalFileName,
                timestamp,
                random
            );
        }
    }

    /**
     * 验证文件内容是否为文本
     */
    public static boolean isTextFile(byte[] content) {
        if (content == null || content.length == 0) {
            return false;
        }

        // 检查前几个字节是否是文本
        int bytesToCheck = Math.min(content.length, 1024);
        int textBytes = 0;

        for (int i = 0; i < bytesToCheck; i++) {
            byte b = content[i];
            // ASCII可打印字符或常见控制字符
            if ((b >= 0x20 && b <= 0x7E) || b == 0x09 || b == 0x0A || b == 0x0D) {
                textBytes++;
            }
        }

        // 如果超过90%的字节是文本，则认为是文本文件
        return (double) textBytes / bytesToCheck > 0.9;
    }

    /**
     * 从字节数组创建输入流
     */
    public static InputStream createInputStream(byte[] content) {
        if (content == null) {
            return new ByteArrayInputStream(new byte[0]);
        }
        return new ByteArrayInputStream(content);
    }

    /**
     * 安全关闭流
     */
    public static void closeQuietly(InputStream inputStream) {
        if (inputStream != null) {
            try {
                inputStream.close();
            } catch (IOException e) {
                log.warn("关闭输入流失败", e);
            }
        }
    }

package com.example.s3processor.monitor;

import com.example.s3processor.model.FileStatus;
import com.example.s3processor.repository.ProcessedFileRepository;
import io.micrometer.core.instrument.Counter;
import io.micrometer.core.instrument.Gauge;
import io.micrometer.core.instrument.MeterRegistry;
import io.micrometer.core.instrument.Timer;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

import javax.annotation.PostConstruct;
import java.time.LocalDateTime;
import java.util.HashMap;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicLong;

@Slf4j
@Component
public class MetricsCollector {

    @Autowired
    private MeterRegistry meterRegistry;

    @Autowired
    private ProcessedFileRepository processedFileRepository;

    private final Map<FileStatus, AtomicLong> fileStatusCounters = new ConcurrentHashMap<>();
    private final Counter filesProcessedCounter;
    private final Counter filesFailedCounter;
    private final Counter filesRetriedCounter;
    private final Timer fileProcessingTimer;

    public MetricsCollector(MeterRegistry meterRegistry) {
        this.meterRegistry = meterRegistry;

        // 初始化计数器
        filesProcessedCounter = Counter.builder("s3.files.processed")
            .description("已处理的文件总数")
            .register(meterRegistry);

        filesFailedCounter = Counter.builder("s3.files.failed")
            .description("处理失败的文件总数")
            .register(meterRegistry);

        filesRetriedCounter = Counter.builder("s3.files.retried")
            .description("重试的文件总数")
            .register(meterRegistry);

        fileProcessingTimer = Timer.builder("s3.files.processing.time")
            .description("文件处理时间")
            .publishPercentiles(0.5, 0.9, 0.95, 0.99)
            .register(meterRegistry);

        // 初始化状态计数器
        for (FileStatus status : FileStatus.values()) {
            fileStatusCounters.put(status, new AtomicLong(0));

            Gauge.builder("s3.files.status",
                    () -> fileStatusCounters.get(status).get())
                .tag("status", status.name())
                .description("按状态统计的文件数量")
                .register(meterRegistry);
        }
    }

    @PostConstruct
    public void init() {
        // 初始加载统计信息
        refreshMetrics();
    }

    @Scheduled(fixedDelay = 60000) // 每分钟刷新一次
    public void refreshMetrics() {
        try {
            for (FileStatus status : FileStatus.values()) {
                long count = processedFileRepository.countByStatus(status);
                fileStatusCounters.get(status).set(count);
            }
        } catch (Exception e) {
            log.error("刷新监控指标失败", e);
        }
    }

    /**
     * 记录文件处理开始
     */
    public void recordFileProcessingStart() {
        filesProcessedCounter.increment();
    }

    /**
     * 记录文件处理完成
     */
    public void recordFileProcessingComplete(long durationMs) {
        fileProcessingTimer.record(durationMs, TimeUnit.MILLISECONDS);
    }

    /**
     * 记录文件处理失败
     */
    public void recordFileProcessingFailure() {
        filesFailedCounter.increment();
    }

    /**
     * 记录文件重试
     */
    public void recordFileRetry() {
        filesRetriedCounter.increment();
    }

    /**
     * 获取当前指标
     */
    public Map<String, Object> getCurrentMetrics() {
        Map<String, Object> metrics = new HashMap<>();

        metrics.put("filesProcessed", filesProcessedCounter.count());
        metrics.put("filesFailed", filesFailedCounter.count());
        metrics.put("filesRetried", filesRetriedCounter.count());

        Map<String, Long> statusCounts = new HashMap<>();
        for (FileStatus status : FileStatus.values()) {
            statusCounts.put(status.name(), fileStatusCounters.get(status).get());
        }
        metrics.put("statusCounts", statusCounts);

        metrics.put("processingTime.p50", fileProcessingTimer.percentile(0.5, TimeUnit.MILLISECONDS));
        metrics.put("processingTime.p90", fileProcessingTimer.percentile(0.9, TimeUnit.MILLISECONDS));
        metrics.put("processingTime.p95", fileProcessingTimer.percentile(0.95, TimeUnit.MILLISECONDS));
        metrics.put("processingTime.p99", fileProcessingTimer.percentile(0.99, TimeUnit.MILLISECONDS));

        metrics.put("timestamp", LocalDateTime.now());

        return metrics;
    }

import com.example.s3processor.model.FileStatus;

import com.example.s3processor.repository.ProcessedFileRepository;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.CommandLineRunner;

import org.springframework.data.mongodb.core.index.MongoPersistentEntityIndexResolver;
import org.springframework.data.mongodb.core.mapping.MongoMappingContext;
import org.springframework.stereotype.Component;

import java.time.LocalDateTime;
import java.util.Arrays;
import java.util.List;

@Slf4j
@Component
public class MongoDBInitializer implements CommandLineRunner {

    @Autowired
    private MongoTemplate mongoTemplate;

    @Autowired
    private MongoMappingContext mongoMappingContext;

    @Autowired
    private ProcessedFileRepository processedFileRepository;

    @Override
    public void run(String... args) throws Exception {
        log.info("初始化MongoDB...");

        // 创建索引
        createIndexes();

        // 初始化测试数据（开发环境）
        if (isDevelopmentEnvironment()) {
            initializeTestData();
        }

        log.info("MongoDB初始化完成");
    }

    /**
     * 创建索引
     */
    private void createIndexes() {
        try {
            // 为ProcessedFile创建索引
            IndexOperations indexOps = mongoTemplate.indexOps(ProcessedFile.class);
            IndexResolver resolver = new MongoPersistentEntityIndexResolver(mongoMappingContext);

            resolver.resolveIndexFor(ProcessedFile.class).forEach(indexOps::ensureIndex);

            log.info("MongoDB索引创建完成");
        } catch (Exception e) {
            log.error("创建MongoDB索引失败", e);
        }
    }

    /**
     * 初始化测试数据
     */
    private void initializeTestData() {
        if (processedFileRepository.count() > 0) {
            log.info("数据库中已有数据，跳过测试数据初始化");
            return;
        }

        try {
            List<ProcessedFile> testFiles = Arrays.asList(
                createTestFile("test1.csv", FileStatus.SUCCESS, LocalDateTime.now().minusDays(1)),
                createTestFile("test2.csv", FileStatus.FAILED, LocalDateTime.now().minusHours(12)),
                createTestFile("test3.csv", FileStatus.PROCESSING, LocalDateTime.now().minusHours(1))
            );

            processedFileRepository.saveAll(testFiles);
            log.info("初始化了 {} 条测试数据", testFiles.size());

        } catch (Exception e) {
            log.error("初始化测试数据失败", e);
        }
    }

    /**
     * 创建测试文件记录

    private ProcessedFile createTestFile(String fileName, FileStatus status, LocalDateTime processTime) {
        return ProcessedFile.builder()
            .fileKey("uploads/" + fileName)
            .fileName(fileName)
            .originalPath("uploads/" + fileName)
            .bucket("test-bucket")
            .fileSize(1024L)
            .contentType("text/csv")
            .md5Hash("d41d8cd98f00b204e9800998ecf8427e")
            .status(status)
            .processStartTime(processTime)
            .processEndTime(status == FileStatus.SUCCESS ? processTime.plusSeconds(5) : null)
            .processingDuration(status == FileStatus.SUCCESS ? 5000L : null)
            .retryCount(0)
            .backupKey("backup/" + fileName)
            .processedKey(status == FileStatus.SUCCESS ? "processed/" + fileName : null)
            .metadata(java.util.Map.of("test", true))
            .createdAt(processTime)
            .updatedAt(processTime)
            .processedBy("test-init")
            .build();
    }

    /**
     * 判断是否是开发环境
     */
    private boolean isDevelopmentEnvironment() {
        String env = System.getProperty("spring.profiles.active", "");
        return "dev".equals(env) || "development".equals(env) || env.isEmpty();
    }

package com.example.s3processor.aspect;

import com.example.s3processor.monitor.MetricsCollector;
import lombok.extern.slf4j.Slf4j;
import org.aspectj.lang.ProceedingJoinPoint;
import org.aspectj.lang.annotation.Around;
import org.aspectj.lang.annotation.Aspect;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

@Slf4j
@Aspect
@Component
public class FileProcessingAspect {

    @Autowired
    private MetricsCollector metricsCollector;

    /**
     * 监控文件处理方法
     */
    @Around("execution(* com.example.s3processor.service.FileProcessorService.processFile(..))")
    public Object monitorFileProcessing(ProceedingJoinPoint joinPoint) throws Throwable {
        String fileKey = (String) joinPoint.getArgs()[0];
        long startTime = System.currentTimeMillis();

        log.info("开始处理文件: {}", fileKey);
        metricsCollector.recordFileProcessingStart();

        try {
            Object result = joinPoint.proceed();
            long duration = System.currentTimeMillis() - startTime;

            log.info("文件处理完成: {}，耗时: {}ms", fileKey, duration);
            metricsCollector.recordFileProcessingComplete(duration);

            return result;

        } catch (Exception e) {
            long duration = System.currentTimeMillis() - startTime;

            log.error("文件处理失败: {}，耗时: {}ms", fileKey, duration, e);
            metricsCollector.recordFileProcessingFailure();

            throw e;
        }
    }

    /**
     * 监控文件重试方法
     */
    @Around("execution(* com.example.s3processor.service.FileRetryService.retryFailedFile(..))")
    public Object monitorFileRetry(ProceedingJoinPoint joinPoint) throws Throwable {
        String fileName = (String) joinPoint.getArgs()[0];

        log.info("开始重试文件: {}", fileName);
        metricsCollector.recordFileRetry();

        try {
            Object result = joinPoint.proceed();
            log.info("文件重试完成: {}", fileName);
            return result;

        } catch (Exception e) {
            log.error("文件重试失败: {}", fileName, e);
            throw e;
        }
    }
}
