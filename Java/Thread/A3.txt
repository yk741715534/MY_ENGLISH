package com.example.priorityqueue.core.domain;

import com.fasterxml.jackson.annotation.JsonCreator;
import com.fasterxml.jackson.annotation.JsonValue;

/**
 * 任务优先级枚举
 */
public enum TaskPriority {

    IMMEDIATE(0, "IMMEDIATE", "立即执行", 100),     // 最高优先级
    HIGH(1, "HIGH", "高优先级", 80),                // 高优先级
    NORMAL(2, "NORMAL", "普通优先级", 50),          // 普通优先级
    LOW(3, "LOW", "低优先级", 20),                  // 低优先级
    BACKGROUND(4, "BACKGROUND", "后台任务", 5);     // 后台任务

    private final int code;
    private final String name;
    private final String description;
    private final int weight;  // 权重值，用于比较

    TaskPriority(int code, String name, String description, int weight) {
        this.code = code;
        this.name = name;
        this.description = description;
        this.weight = weight;
    }

    public int getCode() {
        return code;
    }

    public String getName() {
        return name;
    }

    public String getDescription() {
        return description;
    }

    public int getWeight() {
        return weight;
    }

    @JsonValue
    public String toValue() {
        return this.name;
    }

    @JsonCreator
    public static TaskPriority fromValue(String value) {
        for (TaskPriority priority : TaskPriority.values()) {
            if (priority.name.equalsIgnoreCase(value)) {
                return priority;
            }
        }
        throw new IllegalArgumentException("无效的优先级: " + value);
    }

    public static TaskPriority fromCode(int code) {
        for (TaskPriority priority : TaskPriority.values()) {
            if (priority.code == code) {
                return priority;
            }
        }
        throw new IllegalArgumentException("无效的优先级代码: " + code);
    }
}

package com.example.priorityqueue.core.domain;

import lombok.Data;
import lombok.Builder;
import lombok.AllArgsConstructor;
import lombok.NoArgsConstructor;
import com.fasterxml.jackson.annotation.JsonIgnore;
import com.fasterxml.jackson.annotation.JsonInclude;
import java.time.LocalDateTime;
import java.util.Map;
import java.util.concurrent.atomic.AtomicLong;

/**
 * 优先级任务包装
 */
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
@JsonInclude(JsonInclude.Include.NON_NULL)
public class PriorityTask implements Runnable, Comparable<PriorityTask> {

    // 任务ID生成器
    private static final AtomicLong ID_GENERATOR = new AtomicLong(0);

    // 任务ID
    private String taskId;

    // 任务名称
    private String taskName;

    // 实际要执行的任务
    @JsonIgnore
    private Runnable task;

    // 可返回结果的任务
    @JsonIgnore
    private java.util.concurrent.Callable<?> callableTask;

    // 任务优先级
    private TaskPriority priority;

    // 业务数据
    private Map<String, Object> businessData;

    // 任务状态
    private TaskStatus status;

    // 提交时间
    private LocalDateTime submitTime;

    // 开始执行时间
    private LocalDateTime startTime;

    // 完成时间
    private LocalDateTime finishTime;

    // 执行时长(毫秒)
    private Long executionTime;

    // 序列号(用于比较)
    private long sequenceNumber;

    // 任务结果
    private Object result;

    // 异常信息
    private String errorMessage;

    // 重试次数
    private int retryCount;

    // 最大重试次数
    private int maxRetryCount;

    // 自定义比较器
    private java.util.Comparator<PriorityTask> customComparator;

    public PriorityTask(Runnable task, TaskPriority priority,
                       Map<String, Object> businessData) {
        this.taskId = generateTaskId();
        this.task = task;
        this.priority = priority;
        this.businessData = businessData;
        this.status = TaskStatus.PENDING;
        this.submitTime = LocalDateTime.now();
        this.sequenceNumber = ID_GENERATOR.getAndIncrement();
        this.maxRetryCount = 3;
    }

    public PriorityTask(Callable<?> callableTask, TaskPriority priority,
                       Map<String, Object> businessData) {
        this.taskId = generateTaskId();
        this.callableTask = callableTask;
        this.priority = priority;
        this.businessData = businessData;
        this.status = TaskStatus.PENDING;
        this.submitTime = LocalDateTime.now();
        this.sequenceNumber = ID_GENERATOR.getAndIncrement();
        this.maxRetryCount = 3;
    }

    private String generateTaskId() {
        return "TASK-" + System.currentTimeMillis() + "-" + ID_GENERATOR.get();
    }

    @Override
    public void run() {
        this.status = TaskStatus.RUNNING;
        this.startTime = LocalDateTime.now();

        try {
            if (task != null) {
                task.run();
            } else if (callableTask != null) {
                this.result = callableTask.call();
            }

            this.status = TaskStatus.COMPLETED;
            this.finishTime = LocalDateTime.now();
            this.executionTime = java.time.Duration.between(startTime, finishTime).toMillis();

        } catch (Exception e) {
            this.status = TaskStatus.FAILED;
            this.errorMessage = e.getMessage();
            this.finishTime = LocalDateTime.now();
            this.executionTime = java.time.Duration.between(startTime, finishTime).toMillis();

            if (retryCount < maxRetryCount) {
                this.status = TaskStatus.RETRYING;
                this.retryCount++;
            }
            throw new RuntimeException("任务执行失败: " + taskId, e);
        }
    }

    @Override
    public int compareTo(PriorityTask other) {
        if (customComparator != null) {
            return customComparator.compare(this, other);
        }

        // 默认比较：优先级权重(数字小的优先级高) > 提交时间
        int priorityCompare = Integer.compare(this.priority.getWeight(), other.priority.getWeight());
        if (priorityCompare != 0) {
            return priorityCompare;
        }

        // 如果优先级相同，按提交顺序(FIFO)
        return Long.compare(this.sequenceNumber, other.sequenceNumber);
    }

    /**
     * 任务状态枚举
     */
    public enum TaskStatus {
        PENDING,     // 等待中
        RUNNING,     // 执行中
        COMPLETED,   // 已完成
        FAILED,      // 失败
        RETRYING,    // 重试中
        CANCELLED    // 已取消
    }
}

    private String taskId;

    // 是否成功
    private boolean success;

    // 返回数据
    private T data;

    // 错误信息
    private String errorMessage;

    // 错误码
    private String errorCode;

    // 执行开始时间
    private LocalDateTime startTime;

    // 执行结束时间
    private LocalDateTime endTime;

    // 执行耗时(毫秒)
    private Long executionTime;

    // 任务优先级
    private TaskPriority priority;

    // 创建成功结果
    public static <T> TaskResult<T> success(String taskId, T data,
                                          LocalDateTime startTime,
                                          LocalDateTime endTime,
                                          TaskPriority priority) {
        return TaskResult.<T>builder()
                .taskId(taskId)
                .success(true)
                .data(data)
                .startTime(startTime)
                .endTime(endTime)
                .executionTime(java.time.Duration.between(startTime, endTime).toMillis())
                .priority(priority)
                .build();
    }

    public static <T> TaskResult<T> failure(String taskId, String errorMessage,
                                           String errorCode,
                                           LocalDateTime startTime,
                                           LocalDateTime endTime,
                                           TaskPriority priority) {
        return TaskResult.<T>builder()
                .taskId(taskId)
                .success(false)
                .errorMessage(errorMessage)
                .errorCode(errorCode)
                .startTime(startTime)
                .endTime(endTime)
                .executionTime(java.time.Duration.between(startTime, endTime).toMillis())

                .build();

package com.example.priorityqueue.core.evaluator;

import com.example.priorityqueue.core.domain.TaskPriority;
import org.springframework.stereotype.Component;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

/**
 * 业务优先级评估器
 * 可以根据具体的业务规则进行评估
 */
@Component
public class BusinessPriorityEvaluator implements PriorityEvaluator {

    // 缓存评估结果
    private final Map<String, TaskPriority> cache = new ConcurrentHashMap<>();

    // 业务规则配置
    private final Map<String, Object> ruleConfig = new ConcurrentHashMap<>();

    public BusinessPriorityEvaluator() {
        // 初始化默认规则
        initDefaultRules();
    }

    private void initDefaultRules() {
        // 任务类型权重
        ruleConfig.put("TASK_TYPE_WEIGHTS", Map.of(
            "PAYMENT", 100,
            "ORDER", 90,
            "NOTIFY", 70,
            "REPORT", 40,
            "CLEAN", 20,
            "BACKUP", 10
        ));

        // 用户等级权重
        ruleConfig.put("USER_LEVEL_WEIGHTS", Map.of(
            "VIP", 100,
            "GOLD", 80,
            "SILVER", 60,
            "NORMAL", 40,
            "GUEST", 20
        ));

        // 紧急程度映射
        ruleConfig.put("URGENCY_MAPPING", Map.of(
            "CRITICAL", 100,
            "HIGH", 80,
            "MEDIUM", 50,
            "LOW", 20
        ));
    }

    @Override
    public TaskPriority evaluate(Map<String, Object> businessData) {
        if (businessData == null || businessData.isEmpty()) {
            return TaskPriority.NORMAL;
        }

        // 生成缓存key
        String cacheKey = generateCacheKey(businessData);

        // 检查缓存
        TaskPriority cached = cache.get(cacheKey);
        if (cached != null) {
            return cached;
        }

        // 计算优先级
        TaskPriority priority = calculatePriority(businessData);

        // 缓存结果
        cache.put(cacheKey, priority);

        return priority;
    }

    private String generateCacheKey(Map<String, Object> businessData) {
        StringBuilder keyBuilder = new StringBuilder();
        businessData.forEach((k, v) ->
            keyBuilder.append(k).append(":").append(v).append(";"));
        return keyBuilder.toString();
    }

    private TaskPriority calculatePriority(Map<String, Object> businessData) {
        int score = 0;

        // 1. 根据任务类型评分
        String taskType = (String) businessData.get("taskType");
        if (taskType != null) {
            Map<String, Integer> weights = (Map<String, Integer>) ruleConfig.get("TASK_TYPE_WEIGHTS");
            score += weights.getOrDefault(taskType, 50);
        }

        // 2. 根据用户等级评分
        String userLevel = (String) businessData.get("userLevel");
        if (userLevel != null) {
            Map<String, Integer> weights = (Map<String, Integer>) ruleConfig.get("USER_LEVEL_WEIGHTS");
            score += weights.getOrDefault(userLevel, 50);
        }

        // 3. 根据紧急程度评分
        String urgency = (String) businessData.get("urgency");
        if (urgency != null) {
            Map<String, Integer> mapping = (Map<String, Integer>) ruleConfig.get("URGENCY_MAPPING");
            score += mapping.getOrDefault(urgency, 50);
        }

        // 4. 其他业务规则
        Boolean isImportant = (Boolean) businessData.get("isImportant");
        if (Boolean.TRUE.equals(isImportant)) {
            score += 30;
        }

        // 5. 时间因素（如果指定了截止时间）
        String deadlineStr = (String) businessData.get("deadline");
        if (deadlineStr != null) {
            try {
                java.time.LocalDateTime deadline = java.time.LocalDateTime.parse(deadlineStr);
                java.time.Duration duration = java.time.Duration.between(
                    java.time.LocalDateTime.now(), deadline);
                if (duration.toHours() < 1) {
                    score += 50;  // 1小时内截止
                } else if (duration.toHours() < 24) {
                    score += 20;  // 24小时内截止
                }
            } catch (Exception e) {
                // 解析失败，忽略
            }
        }

        // 根据总分确定优先级
        if (score >= 180) {
            return TaskPriority.IMMEDIATE;
        } else if (score >= 120) {
            return TaskPriority.HIGH;
        } else if (score >= 60) {
            return TaskPriority.NORMAL;
        } else if (score >= 20) {
            return TaskPriority.LOW;
        } else {
            return TaskPriority.BACKGROUND;
        }
    }

    /**
     * 清空缓存
     */
    public void clearCache() {
        cache.clear();
    }

    /**
     * 更新规则配置
     */
    public void updateRuleConfig(String key, Object value) {
        ruleConfig.put(key, value);
        clearCache();  // 更新规则后清空缓存
    }

    /**
     * 获取当前规则配置
     */
    public Map<String, Object> getRuleConfig() {
        return new ConcurrentHashMap<>(ruleConfig);
    }
}

package com.example.priorityqueue.core.executor;

import com.example.priorityqueue.core.domain.PriorityTask;
import com.example.priorityqueue.core.domain.TaskPriority;
import com.example.priorityqueue.core.evaluator.PriorityEvaluator;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.DisposableBean;
import org.springframework.beans.factory.InitializingBean;
import org.springframework.scheduling.concurrent.ThreadPoolTaskExecutor;
import org.springframework.stereotype.Component;
import javax.annotation.PreDestroy;
import java.util.Map;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicLong;

/**
 * 优先级线程队列管理器
 */
@Slf4j
@Component
public class PriorityThreadQueue implements InitializingBean, DisposableBean {

    // 优先级队列
    private final PriorityBlockingQueue<PriorityTask> taskQueue;

    // 执行器线程池
    private final ThreadPoolTaskExecutor executor;

    // 优先级评估器
    private final PriorityEvaluator priorityEvaluator;

    // 监控统计
    private final Map<TaskPriority, AtomicLong> submittedTasks = new ConcurrentHashMap<>();
    private final Map<TaskPriority, AtomicLong> completedTasks = new ConcurrentHashMap<>();
    private final Map<TaskPriority, AtomicLong> failedTasks = new ConcurrentHashMap<>();
    private final AtomicLong totalProcessingTime = new AtomicLong(0);
    private final AtomicLong totalProcessedTasks = new AtomicLong(0);

    // 控制标志
    private final AtomicBoolean running = new AtomicBoolean(false);
    private Thread queueProcessorThread;

    public PriorityThreadQueue(PriorityEvaluator priorityEvaluator) {
        this.taskQueue = new PriorityBlockingQueue<>();
        this.priorityEvaluator = priorityEvaluator;
        this.executor = createThreadPoolExecutor();

        // 初始化统计计数器
        for (TaskPriority priority : TaskPriority.values()) {
            submittedTasks.put(priority, new AtomicLong(0));
            completedTasks.put(priority, new AtomicLong(0));
            failedTasks.put(priority, new AtomicLong(0));
        }
    }

    private ThreadPoolTaskExecutor createThreadPoolExecutor() {
        ThreadPoolTaskExecutor executor = new ThreadPoolTaskExecutor();
        executor.setCorePoolSize(Runtime.getRuntime().availableProcessors() * 2);
        executor.setMaxPoolSize(Runtime.getRuntime().availableProcessors() * 4);
        executor.setQueueCapacity(1000);
        executor.setThreadNamePrefix("priority-executor-");
        executor.setRejectedExecutionHandler(new ThreadPoolExecutor.CallerRunsPolicy());
        executor.setWaitForTasksToCompleteOnShutdown(true);
        executor.setAwaitTerminationSeconds(60);

        // 自定义任务装饰器
        executor.setTaskDecorator(new PriorityTaskDecorator());

        return executor;
    }

    @Override
    public void afterPropertiesSet() throws Exception {
        this.executor.initialize();
        startQueueProcessor();
        log.info("PriorityThreadQueue 初始化完成，核心线程数: {}, 最大线程数: {}",
                executor.getCorePoolSize(), executor.getMaxPoolSize());
    }

    @PreDestroy
    @Override
    public void destroy() throws Exception {
        stopQueueProcessor();
        executor.shutdown();
        log.info("PriorityThreadQueue 已关闭");
    }

    /**
     * 提交任务
     */
    public <T> CompletableFuture<T> submit(Callable<T> task, Map<String, Object> businessData) {
        TaskPriority priority = priorityEvaluator.evaluate(businessData);
        return submit(task, priority, businessData);
    }

    /**
     * 提交任务（指定优先级）
     */
    public <T> CompletableFuture<T> submit(Callable<T> task, TaskPriority priority,
                                          Map<String, Object> businessData) {
        PriorityTask priorityTask = new PriorityTask(task, priority, businessData);

        // 更新统计
        submittedTasks.get(priority).incrementAndGet();

        // 创建CompletableFuture
        CompletableFuture<T> future = new CompletableFuture<>();

        // 包装任务以处理结果
        PriorityTask wrappedTask = PriorityTask.builder()
                .taskId(priorityTask.getTaskId())
                .callableTask(() -> {
                    try {
                        T result = task.call();
                        future.complete(result);
                        return result;
                    } catch (Exception e) {
                        future.completeExceptionally(e);
                        throw e;
                    }
                })
                .priority(priority)
                .businessData(businessData)
                .submitTime(priorityTask.getSubmitTime())
                .sequenceNumber(priorityTask.getSequenceNumber())
                .build();

        // 加入队列
        taskQueue.offer(wrappedTask);

        log.debug("任务提交成功: taskId={}, priority={}, queueSize={}",
                wrappedTask.getTaskId(), priority, taskQueue.size());

        return future;
    }

    /**
     * 提交Runnable任务
     */
    public CompletableFuture<Void> submit(Runnable task, Map<String, Object> businessData) {
        return submit(Executors.callable(task, null), businessData);
    }

    /**
     * 立即执行任务（不通过队列）
     */
    public <T> CompletableFuture<T> executeNow(Callable<T> task, TaskPriority priority,
                                              Map<String, Object> businessData) {
        PriorityTask priorityTask = new PriorityTask(task, priority, businessData);
        submittedTasks.get(priority).incrementAndGet();

        return CompletableFuture.supplyAsync(() -> {
            try {
                return task.call();
            } catch (Exception e) {
                throw new CompletionException(e);
            }
        }, executor);
    }

    /**
     * 启动队列处理器
     */
    private void startQueueProcessor() {
        if (running.compareAndSet(false, true)) {
            queueProcessorThread = new Thread(this::processQueue, "queue-processor");
            queueProcessorThread.setDaemon(true);
            queueProcessorThread.start();
            log.info("队列处理器已启动");
        }
    }

    /**
     * 停止队列处理器
     */
    private void stopQueueProcessor() {
        if (running.compareAndSet(true, false)) {
            if (queueProcessorThread != null && queueProcessorThread.isAlive()) {
                queueProcessorThread.interrupt();
                try {
                    queueProcessorThread.join(5000);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            }
            log.info("队列处理器已停止");
        }
    }

    /**
     * 队列处理逻辑
     */
    private void processQueue() {
        while (running.get() && !Thread.currentThread().isInterrupted()) {
            try {
                PriorityTask task = taskQueue.poll(100, TimeUnit.MILLISECONDS);
                if (task != null) {
                    processTask(task);
                }
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                break;
            } catch (Exception e) {
                log.error("处理任务时发生异常", e);
            }
        }
    }

    /**
     * 处理单个任务
     */
    private void processTask(PriorityTask task) {
        long startTime = System.currentTimeMillis();

        try {
            executor.execute(() -> {
                try {
                    task.run();

                    // 更新统计
                    completedTasks.get(task.getPriority()).incrementAndGet();
                    totalProcessedTasks.incrementAndGet();

                    long processingTime = System.currentTimeMillis() - startTime;
                    totalProcessingTime.addAndGet(processingTime);

                    log.debug("任务执行完成: taskId={}, priority={}, time={}ms",
                            task.getTaskId(), task.getPriority(), processingTime);

                } catch (Exception e) {
                    failedTasks.get(task.getPriority()).incrementAndGet();
                    log.error("任务执行失败: taskId=" + task.getTaskId(), e);

                    // 重试逻辑
                    if (task.getRetryCount() < task.getMaxRetryCount()) {
                        log.info("任务重试: taskId={}, retryCount={}",
                                task.getTaskId(), task.getRetryCount() + 1);
                        taskQueue.offer(task);
                    }
                }
            });
        } catch (RejectedExecutionException e) {
            log.warn("线程池已满，任务重新入队: taskId={}", task.getTaskId());
            taskQueue.offer(task);  // 重新加入队列
        }
    }

    /**
     * 获取队列统计信息
     */
    public QueueMetrics getMetrics() {
        int activeThreads = executor.getActiveCount();
        int poolSize = executor.getPoolSize();
        int queueSize = taskQueue.size();
        int corePoolSize = executor.getCorePoolSize();
        int maxPoolSize = executor.getMaxPoolSize();

        Map<TaskPriority, Long> submittedCounts = new ConcurrentHashMap<>();
        Map<TaskPriority, Long> completedCounts = new ConcurrentHashMap<>();
        Map<TaskPriority, Long> failedCounts = new ConcurrentHashMap<>();

        for (TaskPriority priority : TaskPriority.values()) {
            submittedCounts.put(priority, submittedTasks.get(priority).get());
            completedCounts.put(priority, completedTasks.get(priority).get());
            failedCounts.put(priority, failedTasks.get(priority).get());
        }

        long totalSubmitted = submittedCounts.values().stream().mapToLong(Long::longValue).sum();
        long avgProcessingTime = totalProcessedTasks.get() > 0 ?
                totalProcessingTime.get() / totalProcessedTasks.get() : 0;

        return QueueMetrics.builder()
                .activeThreads(activeThreads)
                .poolSize(poolSize)
                .queueSize(queueSize)
                .corePoolSize(corePoolSize)
                .maxPoolSize(maxPoolSize)
                .submittedTasks(submittedCounts)
                .completedTasks(completedCounts)
                .failedTasks(failedCounts)
                .totalSubmittedTasks(totalSubmitted)
                .averageProcessingTime(avgProcessingTime)
                .timestamp(java.time.LocalDateTime.now())
                .build();
    }

    /**
     * 获取队列大小
     */
    public int getQueueSize() {
        return taskQueue.size();
    }

    /**
     * 获取活动线程数
     */
    public int getActiveThreadCount() {
        return executor.getActiveCount();
    }

    /**
     * 清空队列
     */
    public void clearQueue() {
        taskQueue.clear();
        log.info("队列已清空");
    }

    /**
     * 调整线程池大小
     */
    public void resizeThreadPool(int corePoolSize, int maxPoolSize) {
        executor.setCorePoolSize(corePoolSize);
        executor.setMaxPoolSize(maxPoolSize);
        log.info("线程池大小已调整: corePoolSize={}, maxPoolSize={}", corePoolSize, maxPoolSize);
    }

public class PriorityTaskDecorator implements TaskDecorator {

    @Override
    public Runnable decorate(Runnable runnable) {
        // 保存调用线程的上下文
        Map<String, String> contextMap = MDC.getCopyOfContextMap();

        return () -> {
            try {
                // 恢复上下文
                if (contextMap != null) {
                    MDC.setContextMap(contextMap);
                }

                // 执行前的逻辑
                long startTime = System.currentTimeMillis();
                String threadName = Thread.currentThread().getName();

                try {
                    runnable.run();
                } finally {
                    // 执行后的逻辑
                    long executionTime = System.currentTimeMillis() - startTime;

                    if (executionTime > 1000) {  // 超过1秒的记录警告日志
                        System.out.println("任务执行时间较长: " + threadName +
                                         ", time=" + executionTime + "ms");
                    }

                    // 清理上下文
                    MDC.clear();
                }
            } catch (Exception e) {
                System.err.println("任务执行异常: " + e.getMessage());
                throw e;
            }
        };
    }

package com.example.priorityqueue.core.executor;

import com.example.priorityqueue.core.domain.PriorityTask;
import com.example.priorityqueue.core.domain.TaskPriority;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.locks.ReentrantReadWriteLock;

/**
 * 动态优先级线程队列
 * 支持运行时调整任务优先级
 */
@Slf4j
@Component
public class DynamicPriorityThreadQueue extends PriorityThreadQueue {

    // 任务跟踪器
    private final ConcurrentHashMap<String, PriorityTask> taskTracker =
        new ConcurrentHashMap<>();

    // 读写锁，用于保护优先级调整
    private final ReentrantReadWriteLock lock = new ReentrantReadWriteLock();

    public DynamicPriorityThreadQueue(PriorityEvaluator priorityEvaluator) {
        super(priorityEvaluator);
    }

    /**
     * 提交任务并返回任务ID
     */
    public String submitWithTracking(Runnable task, Map<String, Object> businessData) {
        PriorityTask priorityTask = new PriorityTask(task,
            priorityEvaluator.evaluate(businessData), businessData);

        taskTracker.put(priorityTask.getTaskId(), priorityTask);
        submit(task, businessData);

        return priorityTask.getTaskId();
    }

    /**
     * 动态调整任务优先级
     */
    public boolean changePriority(String taskId, TaskPriority newPriority) {
        lock.writeLock().lock();
        try {
            PriorityTask task = taskTracker.get(taskId);
            if (task == null) {
                log.warn("任务不存在: taskId={}", taskId);
                return false;
            }

            if (task.getStatus() != PriorityTask.TaskStatus.PENDING) {
                log.warn("任务状态不支持调整优先级: taskId={}, status={}",
                        taskId, task.getStatus());
                return false;
            }

            task.setPriority(newPriority);
            log.info("任务优先级已调整: taskId={}, newPriority={}", taskId, newPriority);
            return true;

        } finally {
            lock.writeLock().unlock();
        }
    }

    /**
     * 取消任务
     */
    public boolean cancelTask(String taskId) {
        PriorityTask task = taskTracker.get(taskId);
        if (task == null) {
            return false;
        }

        task.setStatus(PriorityTask.TaskStatus.CANCELLED);
        taskTracker.remove(taskId);
        log.info("任务已取消: taskId={}", taskId);
        return true;
    }

    /**
     * 获取任务状态
     */
    public PriorityTask.TaskStatus getTaskStatus(String taskId) {
        PriorityTask task = taskTracker.get(taskId);
        return task != null ? task.getStatus() : null;
    }

    /**
     * 获取任务信息
     */
    public PriorityTask getTask(String taskId) {
        return taskTracker.get(taskId);
    }

    /**
     * 清理已完成的任务
     */
    public void cleanupCompletedTasks(long timeoutMinutes) {
        long timeoutMillis = timeoutMinutes * 60 * 1000;
        long currentTime = System.currentTimeMillis();

        taskTracker.entrySet().removeIf(entry -> {
            PriorityTask task = entry.getValue();
            if (task.getFinishTime() != null) {
                long finishTime = task.getFinishTime().atZone(
                    java.time.ZoneId.systemDefault()).toInstant().toEpochMilli();
                return (currentTime - finishTime) > timeoutMillis;
            }
            return false;
        });

        log.info("已清理超时任务，当前跟踪任务数: {}", taskTracker.size());
    }
}

/**

@Slf4j
@Service
@RequiredArgsConstructor
public class PriorityTaskService {

    private final DynamicPriorityThreadQueue dynamicPriorityThreadQueue;
    private final Map<String, CompletableFuture<?>> pendingFutures = new ConcurrentHashMap<>();

    /**
     * 提交任务
     */
    public <T> CompletableFuture<TaskResult<T>> submitTask(
            String taskName,
            java.util.concurrent.Callable<T> task,
            Map<String, Object> businessData) {

        String taskId = dynamicPriorityThreadQueue.submitWithTracking(() -> {
            try {
                return task.call();
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }, businessData);

        // 创建并返回CompletableFuture
        CompletableFuture<TaskResult<T>> resultFuture = new CompletableFuture<>();

        // 监控任务完成
        monitorTaskCompletion(taskId, taskName, businessData, resultFuture);

        return resultFuture;
    }

    /**
     * 提交任务（指定优先级）
     */
    public <T> CompletableFuture<TaskResult<T>> submitTaskWithPriority(
            String taskName,
            java.util.concurrent.Callable<T> task,
            TaskPriority priority,
            Map<String, Object> businessData) {

        // 使用动态队列
        String taskId = dynamicPriorityThreadQueue.submitWithTracking(() -> {
            try {
                return task.call();
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }, businessData);

        // 设置指定优先级
        dynamicPriorityThreadQueue.changePriority(taskId, priority);

        // 创建并返回CompletableFuture
        CompletableFuture<TaskResult<T>> resultFuture = new CompletableFuture<>();

        // 监控任务完成
        monitorTaskCompletion(taskId, taskName, businessData, resultFuture);

        return resultFuture;
    }

    /**
     * 监控任务完成
     */
    private <T> void monitorTaskCompletion(String taskId, String taskName,
                                         Map<String, Object> businessData,
                                         CompletableFuture<TaskResult<T>> resultFuture) {

        // 定期检查任务状态
        CompletableFuture.runAsync(() -> {
            while (!resultFuture.isDone()) {
                try {
                    PriorityTask task = dynamicPriorityThreadQueue.getTask(taskId);
                    if (task == null) {
                        resultFuture.complete(TaskResult.<T>failure(
                            taskId, "任务不存在", "TASK_NOT_FOUND",
                            LocalDateTime.now(), LocalDateTime.now(), TaskPriority.NORMAL));
                        break;
                    }

                    switch (task.getStatus()) {
                        case COMPLETED:
                            resultFuture.complete(TaskResult.<T>success(
                                taskId, (T) task.getResult(),
                                task.getStartTime(), task.getFinishTime(),
                                task.getPriority()));
                            break;
                        case FAILED:
                            resultFuture.complete(TaskResult.<T>failure(
                                taskId, task.getErrorMessage(), "TASK_FAILED",
                                task.getStartTime(), task.getFinishTime(),
                                task.getPriority()));
                            break;
                        case CANCELLED:
                            resultFuture.complete(TaskResult.<T>failure(
                                taskId, "任务被取消", "TASK_CANCELLED",
                                task.getStartTime(), task.getFinishTime(),
                                task.getPriority()));
                            break;
                        default:
                            // 任务还在执行中，继续等待
                            TimeUnit.MILLISECONDS.sleep(100);
                    }
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    resultFuture.completeExceptionally(e);
                } catch (Exception e) {
                    resultFuture.completeExceptionally(e);
                }
            }
        });
    }

    /**
     * 调整任务优先级
     */
    public boolean adjustTaskPriority(String taskId, TaskPriority newPriority) {
        return dynamicPriorityThreadQueue.changePriority(taskId, newPriority);
    }

    /**
     * 取消任务
     */
    public boolean cancelTask(String taskId) {
        return dynamicPriorityThreadQueue.cancelTask(taskId);
    }

    /**
     * 获取任务状态
     */
    public PriorityTask.TaskStatus getTaskStatus(String taskId) {
        return dynamicPriorityThreadQueue.getTaskStatus(taskId);
    }

    /**
     * 批量提交任务
     */
    public Map<String, CompletableFuture<TaskResult<?>>> submitBatch(
            Map<String, java.util.concurrent.Callable<?>> tasks,
            Map<String, Object> commonBusinessData) {

        Map<String, CompletableFuture<TaskResult<?>>> results = new ConcurrentHashMap<>();

        tasks.forEach((taskName, task) -> {
            Map<String, Object> businessData = new ConcurrentHashMap<>(commonBusinessData);
            businessData.put("batchTaskName", taskName);

            CompletableFuture<TaskResult<?>> future = submitTask(taskName, task, businessData);
            results.put(taskName, future);
        });

        return results;
    }

    @PostConstruct
    public void init() {
        log.info("PriorityTaskService 初始化完成");
    }

    @PreDestroy
    public void cleanup() {
        // 清理资源
        pendingFutures.clear();
        log.info("PriorityTaskService 资源清理完成");
    }

        health.put("status", "UP");
        health.put("timestamp", java.time.LocalDateTime.now());
        health.put("queueSize", metrics.getQueueSize());
        health.put("activeThreads", metrics.getActiveThreads());
        health.put("totalSubmittedTasks", metrics.getTotalSubmittedTasks());

        // 检查队列健康状况
        boolean isHealthy = metrics.getQueueSize() < 1000 &&
                           metrics.getActiveThreads() < metrics.getMaxPoolSize() * 0.8;

        health.put("isHealthy", isHealthy);
        health.put("message", isHealthy ? "队列运行正常" : "队列负载较高");

        return ResponseEntity.ok(health);

    private int poolSize;
    private int corePoolSize;
    private int maxPoolSize;

    // 队列信息
    private int queueSize;

    // 任务统计

    private Map<TaskPriority, Long> completedTasks;
    private Map<TaskPriority, Long> failedTasks;

    // 汇总统计
    private long totalSubmittedTasks;
    private long totalCompletedTasks;
    private long totalFailedTasks;
    private long averageProcessingTime;  // 平均处理时间(毫秒)

    // 时间戳
    private LocalDateTime timestamp;

    // 计算成功率
    public double getSuccessRate() {
        if (totalCompletedTasks + totalFailedTasks == 0) {
            return 0.0;
        }
        return (double) totalCompletedTasks / (totalCompletedTasks + totalFailedTasks) * 100;
    }

    // 计算队列使用率
    public double getQueueUsageRate() {
        if (queueSize == 0) {
            return 0.0;
        }
        return (double) queueSize / 1000 * 100;  // 假设队列容量为1000
    }

    // 计算线程池使用率
    public double getThreadPoolUsageRate() {
        if (maxPoolSize == 0) {
            return 0.0;
        }
        return (double) activeThreads / maxPoolSize * 100;
    }

package com.example.priorityqueue.exception;

import com.example.priorityqueue.dto.TaskResponse;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.context.request.WebRequest;
import java.time.LocalDateTime;
import java.util.HashMap;
import java.util.Map;

/**
 * 全局异常处理器
 */
@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(TaskQueueException.class)
    public ResponseEntity<TaskResponse<?>> handleTaskQueueException(
            TaskQueueException ex, WebRequest request) {

        log.error("任务队列异常: {}", ex.getMessage(), ex);

        TaskResponse<?> response = TaskResponse.builder()
                .success(false)
                .code(ex.getErrorCode())
                .message(ex.getMessage())
                .timestamp(LocalDateTime.now())
                .build();

        return ResponseEntity.badRequest().body(response);
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<TaskResponse<?>> handleValidationExceptions(
            MethodArgumentNotValidException ex) {

        Map<String, String> errors = new HashMap<>();
        ex.getBindingResult().getAllErrors().forEach((error) -> {
            String fieldName = ((FieldError) error).getField();
            String errorMessage = error.getDefaultMessage();
            errors.put(fieldName, errorMessage);
        });

        TaskResponse<Map<String, String>> response = TaskResponse.<Map<String, String>>builder()
                .success(false)
                .code("VALIDATION_ERROR")
                .message("参数验证失败")
                .data(errors)
                .timestamp(LocalDateTime.now())
                .build();

        return ResponseEntity.badRequest().body(response);
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<TaskResponse<?>> handleGlobalException(
            Exception ex, WebRequest request) {

        log.error("系统异常: {}", ex.getMessage(), ex);

        TaskResponse<?> response = TaskResponse.builder()
                .success(false)
                .code("INTERNAL_SERVER_ERROR")
                .message("服务器内部错误: " + ex.getMessage())
                .timestamp(LocalDateTime.now())
                .build();

        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
                .body(response);
    }
}
