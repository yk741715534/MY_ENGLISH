import lombok.Getter;

/**
 * 任务优先级枚举
 */
@Getter
public enum TaskPriority {

    CRITICAL(0, "CRITICAL", "关键任务", 100),   // 立即执行
    HIGH(1, "HIGH", "高优先级", 80),            // 高优先级
    NORMAL(2, "NORMAL", "普通优先级", 50),      // 普通优先级
    LOW(3, "LOW", "低优先级", 20),             // 低优先级
    BACKGROUND(4, "BACKGROUND", "后台任务", 5); // 后台任务

    private final int code;
    private final String name;
    private final String description;
    private final int weight;

    TaskPriority(int code, String name, String description, int weight) {
        this.code = code;
        this.name = name;
        this.description = description;
        this.weight = weight;
    }

    /**
     * 根据权重获取优先级
     */
    public static TaskPriority fromWeight(int weight) {
        if (weight >= 90) return CRITICAL;
        if (weight >= 70) return HIGH;
        if (weight >= 30) return NORMAL;
        if (weight >= 10) return LOW;
        return BACKGROUND;
    }
}

package com.example.asyncqueue.core.task;

import lombok.Data;
import lombok.Builder;
import lombok.NoArgsConstructor;
import lombok.AllArgsConstructor;
import java.time.LocalDateTime;
import java.util.Map;
import java.util.concurrent.atomic.AtomicLong;

/**
 * 异步任务实体
 */
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class AsyncTask implements Runnable, Comparable<AsyncTask> {

    // 任务ID生成器
    private static final AtomicLong ID_GENERATOR = new AtomicLong(0);

    // 任务基本信息
    private String taskId;
    private String taskName;
    private String taskGroup;

    // 执行信息
    private Runnable task;
    private TaskPriority priority;
    private TaskStatus status;

    // 业务数据
    private Map<String, Object> metadata;

    // 时间信息
    private LocalDateTime createTime;
    private LocalDateTime queueTime;
    private LocalDateTime startTime;
    private LocalDateTime endTime;

    // 执行统计
    private Long executionTime;
    private Integer retryCount;
    private Integer maxRetryCount;

    // 错误信息
    private String errorMessage;
    private String errorStack;

    // 排序用序列号
    private long sequenceNumber;

    // 回调监听器
    private transient TaskListener listener;

    public AsyncTask(Runnable task, TaskPriority priority, Map<String, Object> metadata) {
        this.taskId = generateTaskId();
        this.task = task;
        this.priority = priority;
        this.status = TaskStatus.PENDING;
        this.metadata = metadata;
        this.createTime = LocalDateTime.now();
        this.retryCount = 0;
        this.maxRetryCount = 3;
        this.sequenceNumber = ID_GENERATOR.getAndIncrement();
    }

    private String generateTaskId() {
        return String.format("TASK-%d-%d",
            System.currentTimeMillis(),
            ID_GENERATOR.get());
    }

    @Override
    public void run() {
        this.queueTime = LocalDateTime.now();
        this.status = TaskStatus.PROCESSING;

        try {
            // 执行前回调
            if (listener != null) {
                listener.beforeExecute(this);
            }

            this.startTime = LocalDateTime.now();
            task.run();
            this.endTime = LocalDateTime.now();

            this.status = TaskStatus.SUCCESS;
            this.executionTime = java.time.Duration.between(startTime, endTime).toMillis();

            // 执行成功回调
            if (listener != null) {
                listener.onSuccess(this);
            }

        } catch (Exception e) {
            this.endTime = LocalDateTime.now();
            this.errorMessage = e.getMessage();
            this.errorStack = getStackTrace(e);

            if (retryCount < maxRetryCount) {
                this.status = TaskStatus.RETRYING;
                this.retryCount++;
            } else {
                this.status = TaskStatus.FAILED;
            }

            this.executionTime = java.time.Duration.between(startTime, endTime).toMillis();

            // 执行失败回调
            if (listener != null) {
                listener.onFailure(this, e);
            }

            throw new RuntimeException("任务执行失败: " + taskId, e);
        }
    }

    @Override
    public int compareTo(AsyncTask other) {
        // 优先级高的先执行（权重值小的优先级高）
        int priorityCompare = Integer.compare(this.priority.getWeight(), other.priority.getWeight());
        if (priorityCompare != 0) {
            return priorityCompare;
        }

        // 同优先级按创建时间排序
        return this.createTime.compareTo(other.createTime);
    }

    private String getStackTrace(Exception e) {
        java.io.StringWriter sw = new java.io.StringWriter();
        java.io.PrintWriter pw = new java.io.PrintWriter(sw);
        e.printStackTrace(pw);
        return sw.toString();
    }

    /**
     * 任务监听器接口
     */
    public interface TaskListener {
        void beforeExecute(AsyncTask task);
        void onSuccess(AsyncTask task);
        void onFailure(AsyncTask task, Exception e);
    }
}

package com.example.asyncqueue.core.evaluator;

import com.example.asyncqueue.core.task.TaskPriority;
import org.springframework.stereotype.Component;
import java.util.Map;

/**
 * 优先级评估器
 */
@Component
public class PriorityEvaluator {

    // 默认评估策略
    public TaskPriority evaluateDefault(Map<String, Object> metadata) {
        if (metadata == null || metadata.isEmpty()) {
            return TaskPriority.NORMAL;
        }

        String taskType = (String) metadata.get("taskType");
        if (taskType != null) {
            switch (taskType) {
                case "PAYMENT_CALLBACK":
                case "ORDER_CALLBACK":
                case "NOTIFY":
                    return TaskPriority.CRITICAL;
                case "LOG":
                case "AUDIT":
                    return TaskPriority.HIGH;
                case "REPORT":
                case "SYNC":
                    return TaskPriority.NORMAL;
                case "CLEAN":
                case "BACKUP":
                case "STATISTICS":
                    return TaskPriority.LOW;
                default:
                    return TaskPriority.NORMAL;
            }
        }

        return TaskPriority.NORMAL;
    }

    /**
     * 基于业务规则的评估
     */
    public TaskPriority evaluateByRule(Map<String, Object> metadata) {
        int score = 0;

        // 任务类型评分
        String taskType = (String) metadata.get("taskType");
        if (taskType != null) {
            score += getTaskTypeScore(taskType);
        }

        // 用户等级评分
        String userLevel = (String) metadata.get("userLevel");
        if (userLevel != null) {
            score += getUserLevelScore(userLevel);
        }

        // 时间敏感度评分
        Boolean isTimeSensitive = (Boolean) metadata.get("timeSensitive");
        if (Boolean.TRUE.equals(isTimeSensitive)) {
            score += 20;
        }

        // 数据重要性评分
        Boolean isImportant = (Boolean) metadata.get("isImportant");
        if (Boolean.TRUE.equals(isImportant)) {
            score += 15;
        }

        // 转换为优先级
        return TaskPriority.fromWeight(score);
    }

    private int getTaskTypeScore(String taskType) {
        switch (taskType) {
            case "PAYMENT": return 40;
            case "ORDER": return 35;
            case "LOGIN": return 30;
            case "REGISTER": return 25;
            case "NOTIFY": return 20;
            case "REPORT": return 15;
            case "CLEAN": return 5;
            default: return 10;
        }
    }

    private int getUserLevelScore(String userLevel) {
        switch (userLevel) {
            case "VIP": return 30;
            case "GOLD": return 20;
            case "SILVER": return 15;
            case "NORMAL": return 10;
            default: return 5;
        }
    }

package com.example.asyncqueue.core.queue;

import com.example.asyncqueue.core.task.AsyncTask;
import com.example.asyncqueue.core.task.TaskPriority;
import lombok.Data;
import lombok.Builder;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;
import javax.annotation.PostConstruct;
import javax.annotation.PreDestroy;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicLong;
import java.util.concurrent.locks.ReentrantLock;
import java.util.Map;
import java.util.HashMap;
import java.util.concurrent.ConcurrentHashMap;

/**
 * 优先级阻塞队列
 */
@Slf4j
@Component
public class PriorityTaskQueue {

    // 核心队列
    private final PriorityBlockingQueue<AsyncTask> taskQueue;

    // 线程池执行器
    private final ThreadPoolExecutor executor;

    // 统计信息
    private final AtomicLong totalSubmitted = new AtomicLong(0);
    private final AtomicLong totalProcessed = new AtomicLong(0);
    private final AtomicLong totalFailed = new AtomicLong(0);
    private final ConcurrentHashMap<TaskPriority, AtomicLong> priorityStats = new ConcurrentHashMap<>();

    // 控制标志
    private final AtomicBoolean running = new AtomicBoolean(true);
    private Thread queueConsumerThread;
    private final ReentrantLock lock = new ReentrantLock();

    public PriorityTaskQueue() {
        // 初始化优先级队列
        this.taskQueue = new PriorityBlockingQueue<>();

        // 初始化线程池
        int corePoolSize = Runtime.getRuntime().availableProcessors() * 2;
        int maxPoolSize = Runtime.getRuntime().availableProcessors() * 4;

        this.executor = new ThreadPoolExecutor(
            corePoolSize,
            maxPoolSize,
            60L, TimeUnit.SECONDS,
            new LinkedBlockingQueue<>(1000),
            new TaskThreadFactory(),
            new ThreadPoolExecutor.CallerRunsPolicy()
        );

        // 初始化统计
        for (TaskPriority priority : TaskPriority.values()) {
            priorityStats.put(priority, new AtomicLong(0));
        }
    }

    @PostConstruct
    public void init() {
        startQueueConsumer();
        log.info("优先级任务队列初始化完成，核心线程数: {}, 最大线程数: {}",
                executor.getCorePoolSize(), executor.getMaximumPoolSize());
    }

    @PreDestroy
    public void destroy() {
        shutdown();
    }

    /**
     * 提交任务
     */
    public void submit(Runnable task, Map<String, Object> metadata) {
        submit(task, null, metadata);
    }

    /**
     * 提交任务（指定优先级）
     */
    public void submit(Runnable task, TaskPriority priority, Map<String, Object> metadata) {
        if (!running.get()) {
            throw new IllegalStateException("任务队列已关闭");
        }

        AsyncTask asyncTask = new AsyncTask(task, priority, metadata);
        asyncTask.setStatus(com.example.asyncqueue.core.task.TaskStatus.QUEUED);

        lock.lock();
        try {
            taskQueue.offer(asyncTask);

            // 更新统计
            totalSubmitted.incrementAndGet();
            priorityStats.get(asyncTask.getPriority()).incrementAndGet();

            log.debug("任务已提交: taskId={}, priority={}, queueSize={}",
                    asyncTask.getTaskId(), asyncTask.getPriority(), taskQueue.size());

        } finally {
            lock.unlock();
        }
    }

    /**
     * 批量提交任务
     */
    public void submitBatch(Map<String, Runnable> tasks, Map<String, Object> commonMetadata) {
        for (Map.Entry<String, Runnable> entry : tasks.entrySet()) {
            Map<String, Object> metadata = new ConcurrentHashMap<>(commonMetadata);
            metadata.put("taskName", entry.getKey());
            submit(entry.getValue(), metadata);
        }
    }

    /**
     * 启动队列消费者
     */
    private void startQueueConsumer() {
        queueConsumerThread = new Thread(this::consumeTasks, "queue-consumer");
        queueConsumerThread.setDaemon(true);
        queueConsumerThread.start();
        log.info("队列消费者线程已启动");
    }

    /**
     * 消费任务
     */
    private void consumeTasks() {
        while (running.get() && !Thread.currentThread().isInterrupted()) {
            try {
                AsyncTask task = taskQueue.poll(100, TimeUnit.MILLISECONDS);
                if (task != null && task.getStatus().isExecutable()) {
                    processTask(task);
                }
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                break;
            } catch (Exception e) {
                log.error("消费任务时发生异常", e);
            }
        }
    }

    /**
     * 处理单个任务
     */
    private void processTask(AsyncTask task) {
        executor.execute(() -> {
            try {
                // 更新统计
                task.run();

                if (task.getStatus() == com.example.asyncqueue.core.task.TaskStatus.SUCCESS) {
                    totalProcessed.incrementAndGet();
                } else if (task.getStatus() == com.example.asyncqueue.core.task.TaskStatus.FAILED) {
                    totalFailed.incrementAndGet();
                }

                log.debug("任务处理完成: taskId={}, status={}, time={}ms",
                        task.getTaskId(), task.getStatus(), task.getExecutionTime());

            } catch (Exception e) {
                log.error("任务执行异常: taskId=" + task.getTaskId(), e);

                // 重试逻辑
                if (task.getRetryCount() < task.getMaxRetryCount()) {
                    log.info("任务重试: taskId={}, retryCount={}",
                            task.getTaskId(), task.getRetryCount());

                    // 重新加入队列
                    taskQueue.offer(task);
                } else {
                    totalFailed.incrementAndGet();
                }
            }
        });
    }

    /**
     * 获取队列状态
     */
    public QueueStatus getStatus() {
        return QueueStatus.builder()
                .queueSize(taskQueue.size())
                .activeThreads(executor.getActiveCount())
                .poolSize(executor.getPoolSize())
                .totalSubmitted(totalSubmitted.get())
                .totalProcessed(totalProcessed.get())
                .totalFailed(totalFailed.get())
                .queueCapacity(1000)
                .isRunning(running.get())
                .build();
    }

    /**
     * 获取优先级统计
     */
    public Map<TaskPriority, Long> getPriorityStatistics() {
        Map<TaskPriority, Long> stats = new ConcurrentHashMap<>();
        priorityStats.forEach((priority, count) ->
            stats.put(priority, count.get()));
        return stats;
    }

    /**
     * 调整线程池大小
     */
    public void resizeThreadPool(int coreSize, int maxSize) {
        executor.setCorePoolSize(coreSize);
        executor.setMaximumPoolSize(maxSize);
        log.info("线程池大小已调整: coreSize={}, maxSize={}", coreSize, maxSize);
    }

    /**
     * 清空队列
     */
    public void clear() {
        taskQueue.clear();
        log.info("任务队列已清空");
    }

    /**
     * 优雅关闭
     */
    public void shutdown() {
        if (running.compareAndSet(true, false)) {
            log.info("开始关闭任务队列...");

            // 停止消费者线程
            if (queueConsumerThread != null) {
                queueConsumerThread.interrupt();
            }

            // 关闭线程池
            executor.shutdown();
            try {
                if (!executor.awaitTermination(30, TimeUnit.SECONDS)) {
                    executor.shutdownNow();
                }
            } catch (InterruptedException e) {
                executor.shutdownNow();
                Thread.currentThread().interrupt();
            }

            log.info("任务队列已关闭");
        }
    }

    /**
     * 自定义线程工厂
     */
    private static class TaskThreadFactory implements ThreadFactory {
        private final AtomicInteger threadNumber = new AtomicInteger(1);

        @Override
        public Thread newThread(Runnable r) {
            Thread thread = new Thread(r, "task-worker-" + threadNumber.getAndIncrement());
            thread.setDaemon(false);
            thread.setPriority(Thread.NORM_PRIORITY);
            return thread;
        }
    }

    /**
     * 队列状态DTO
     */
    @Data
    @Builder
    public static class QueueStatus {
        private int queueSize;
        private int activeThreads;
        private int poolSize;
        private int queueCapacity;
        private long totalSubmitted;
        private long totalProcessed;
        private long totalFailed;
        private boolean isRunning;

        public double getSuccessRate() {
            if (totalProcessed + totalFailed == 0) return 0.0;
            return (double) totalProcessed / (totalProcessed + totalFailed) * 100;
        }

        public double getQueueUsage() {
            return (double) queueSize / queueCapacity * 100;
        }
    }
}

package com.example.asyncqueue.service;

import com.example.asyncqueue.core.evaluator.PriorityEvaluator;
import com.example.asyncqueue.core.queue.PriorityTaskQueue;
import com.example.asyncqueue.core.task.AsyncTask;
import com.example.asyncqueue.core.task.TaskPriority;
import lombok.Data;
import lombok.Builder;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.scheduling.annotation.Scheduled;
import javax.annotation.PostConstruct;
import javax.annotation.PreDestroy;
import java.time.LocalDateTime;
import java.util.Map;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.CopyOnWriteArrayList;

/**
 * 异步任务执行器服务
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class AsyncTaskExecutor {

    private final PriorityTaskQueue taskQueue;
    private final PriorityEvaluator priorityEvaluator;

    // 任务跟踪器
    private final ConcurrentMap<String, AsyncTask> taskTracker = new ConcurrentHashMap<>();

    // 任务监听器
    private final CopyOnWriteArrayList<TaskListener> listeners = new CopyOnWriteArrayList<>();

    // 默认重试策略
    private static final int DEFAULT_MAX_RETRIES = 3;
    private static final long DEFAULT_RETRY_DELAY = 1000; // 1秒

    @PostConstruct
    public void init() {
        log.info("异步任务执行器初始化完成");
    }

    @PreDestroy
    public void destroy() {
        cleanup();
    }

    /**
     * 提交异步任务
     */
    public String submitTask(String taskName, Runnable task, Map<String, Object> metadata) {
        TaskPriority priority = priorityEvaluator.evaluateDefault(metadata);
        return submitTask(taskName, task, priority, metadata);
    }

    /**
     * 提交异步任务（指定优先级）
     */
    public String submitTask(String taskName, Runnable task, TaskPriority priority,
                           Map<String, Object> metadata) {
        // 包装任务，添加监控
        Runnable monitoredTask = wrapWithMonitoring(taskName, task, metadata);

        // 创建异步任务
        AsyncTask asyncTask = new AsyncTask(monitoredTask, priority, metadata);
        asyncTask.setTaskName(taskName);

        // 注册监听器
        asyncTask.setListener(new AsyncTask.TaskListener() {
            @Override
            public void beforeExecute(AsyncTask task) {
                notifyBeforeExecute(task);
            }

            @Override
            public void onSuccess(AsyncTask task) {
                notifyOnSuccess(task);
                taskTracker.remove(task.getTaskId());
            }

            @Override
            public void onFailure(AsyncTask task, Exception e) {
                notifyOnFailure(task, e);
                handleRetry(task, e);
            }
        });

        // 跟踪任务
        taskTracker.put(asyncTask.getTaskId(), asyncTask);

        // 提交到队列
        taskQueue.submit(asyncTask.getTask(), priority, metadata);

        log.info("异步任务已提交: taskId={}, name={}, priority={}",
                asyncTask.getTaskId(), taskName, priority);

        return asyncTask.getTaskId();
    }

    /**
     * 提交延迟任务
     */
    public String submitDelayedTask(String taskName, Runnable task,
                                   long delay, TimeUnit unit,
                                   Map<String, Object> metadata) {
        ScheduledExecutorService scheduler = Executors.newSingleThreadScheduledExecutor();
        String taskId = "DELAYED-" + System.currentTimeMillis() + "-" + taskName;

        scheduler.schedule(() -> {
            try {
                submitTask(taskName, task, metadata);
            } finally {
                scheduler.shutdown();
            }
        }, delay, unit);

        return taskId;
    }

    /**
     * 提交周期性任务
     */
    public String submitPeriodicTask(String taskName, Runnable task,
                                    long initialDelay, long period, TimeUnit unit,
                                    Map<String, Object> metadata) {
        ScheduledExecutorService scheduler = Executors.newSingleThreadScheduledExecutor();
        String taskId = "PERIODIC-" + System.currentTimeMillis() + "-" + taskName;

        AtomicInteger executionCount = new AtomicInteger(0);

        scheduler.scheduleAtFixedRate(() -> {
            int count = executionCount.incrementAndGet();
            Map<String, Object> taskMetadata = new ConcurrentHashMap<>(metadata);
            taskMetadata.put("executionCount", count);
            taskMetadata.put("periodic", true);

            submitTask(taskName + "-" + count, task, taskMetadata);
        }, initialDelay, period, unit);

        return taskId;
    }

    /**
     * 批量提交任务
     */
    public void submitBatch(String batchName, Map<String, Runnable> tasks,
                          Map<String, Object> commonMetadata) {
        log.info("开始批量提交任务: batchName={}, taskCount={}", batchName, tasks.size());

        AtomicInteger completed = new AtomicInteger(0);

        for (Map.Entry<String, Runnable> entry : tasks.entrySet()) {
            Map<String, Object> metadata = new ConcurrentHashMap<>(commonMetadata);
            metadata.put("batchName", batchName);
            metadata.put("taskIndex", completed.get() + 1);

            submitTask(entry.getKey(), entry.getValue(), metadata);
            completed.incrementAndGet();
        }

        log.info("批量任务提交完成: batchName={}, submitted={}", batchName, completed.get());
    }

    /**
     * 包装任务，添加监控
     */
    private Runnable wrapWithMonitoring(String taskName, Runnable originalTask,
                                       Map<String, Object> metadata) {
        return () -> {
            String threadName = Thread.currentThread().getName();
            LocalDateTime startTime = LocalDateTime.now();

            log.debug("开始执行任务: name={}, thread={}, time={}",
                    taskName, threadName, startTime);

            try {
                originalTask.run();

                LocalDateTime endTime = LocalDateTime.now();
                long duration = java.time.Duration.between(startTime, endTime).toMillis();

                log.debug("任务执行成功: name={}, duration={}ms", taskName, duration);

            } catch (Exception e) {
                LocalDateTime endTime = LocalDateTime.now();
                long duration = java.time.Duration.between(startTime, endTime).toMillis();

                log.error("任务执行失败: name={}, duration={}ms, error={}",
                        taskName, duration, e.getMessage(), e);
                throw e;
            }
        };
    }

    /**
     * 处理重试
     */
    private void handleRetry(AsyncTask task, Exception e) {
        if (task.getRetryCount() < task.getMaxRetryCount()) {
            try {
                TimeUnit.MILLISECONDS.sleep(DEFAULT_RETRY_DELAY);

                log.info("任务重试: taskId={}, retryCount={}",
                        task.getTaskId(), task.getRetryCount() + 1);

                // 重新提交任务
                taskQueue.submit(task.getTask(), task.getPriority(), task.getMetadata());

            } catch (InterruptedException ie) {
                Thread.currentThread().interrupt();
                log.error("重试等待被中断", ie);
            }
        } else {
            log.error("任务达到最大重试次数: taskId={}, maxRetries={}",
                    task.getTaskId(), task.getMaxRetryCount());
        }
    }

    /**
     * 添加任务监听器
     */
    public void addListener(TaskListener listener) {
        listeners.add(listener);
    }

    /**
     * 移除任务监听器
     */
    public void removeListener(TaskListener listener) {
        listeners.remove(listener);
    }

    private void notifyBeforeExecute(AsyncTask task) {
        for (TaskListener listener : listeners) {
            try {
                listener.beforeExecute(task);
            } catch (Exception e) {
                log.error("任务监听器执行异常", e);
            }
        }
    }

    private void notifyOnSuccess(AsyncTask task) {
        for (TaskListener listener : listeners) {
            try {
                listener.onSuccess(task);
            } catch (Exception e) {
                log.error("任务监听器执行异常", e);
            }
        }
    }

    private void notifyOnFailure(AsyncTask task, Exception e) {
        for (TaskListener listener : listeners) {
            try {
                listener.onFailure(task, e);
            } catch (Exception e2) {
                log.error("任务监听器执行异常", e2);
            }
        }
    }

    /**
     * 获取执行统计
     */
    public ExecutionStats getStats() {
        PriorityTaskQueue.QueueStatus status = taskQueue.getStatus();

        return ExecutionStats.builder()
                .totalSubmitted(status.getTotalSubmitted())
                .totalProcessed(status.getTotalProcessed())
                .totalFailed(status.getTotalFailed())
                .successRate(status.getSuccessRate())
                .queueSize(status.getQueueSize())
                .activeThreads(status.getActiveThreads())
                .trackedTasks(taskTracker.size())
                .timestamp(LocalDateTime.now())
                .build();
    }

    /**
     * 清理已完成的任务
     */
    @Scheduled(fixedDelay = 300000) // 每5分钟清理一次
    public void cleanupCompletedTasks() {
        int beforeSize = taskTracker.size();

        taskTracker.entrySet().removeIf(entry -> {
            AsyncTask task = entry.getValue();
            return task.getStatus().isCompleted() &&
                   task.getEndTime() != null &&
                   task.getEndTime().plusMinutes(30).isBefore(LocalDateTime.now());
        });

        int removed = beforeSize - taskTracker.size();
        if (removed > 0) {
            log.debug("清理已完成任务: removed={}, remaining={}", removed, taskTracker.size());
        }
    }

    /**
     * 获取活动任务数
     */
    public int getActiveTaskCount() {
        return taskTracker.size();
    }

    /**
     * 优雅关闭
     */
    public void cleanup() {
        log.info("清理异步任务执行器，当前活动任务数: {}", taskTracker.size());
        listeners.clear();
    }

    /**
     * 任务监听器接口
     */
    public interface TaskListener {
        void beforeExecute(AsyncTask task);
        void onSuccess(AsyncTask task);
        void onFailure(AsyncTask task, Exception e);
    }

    /**
     * 执行统计DTO
     */
    @Data
    @Builder
    public static class ExecutionStats {
        private long totalSubmitted;
        private long totalProcessed;
        private long totalFailed;
        private double successRate;
        private int queueSize;
        private int activeThreads;
        private int trackedTasks;
        private LocalDateTime timestamp;
    }
}

@Slf4j
@RestController
@RequestMapping("/api/async/tasks")
@RequiredArgsConstructor
public class AsyncTaskController {

    private final AsyncTaskExecutor taskExecutor;

    @PostMapping("/submit")
    public ResponseEntity<Map<String, Object>> submitTask(
            @RequestParam String taskName,
            @RequestParam(required = false) Integer sleepSeconds,
            @RequestBody(required = false) Map<String, Object> metadata) {

        int sleepTime = sleepSeconds != null ? sleepSeconds : 1;

        String taskId = taskExecutor.submitTask(taskName, () -> {
            try {
                log.info("执行任务: {}", taskName);
                TimeUnit.SECONDS.sleep(sleepTime);
                log.info("任务完成: {}", taskName);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                log.error("任务被中断: {}", taskName, e);
            }
        }, metadata != null ? metadata : new ConcurrentHashMap<>());

        return ResponseEntity.ok(Map.of(
            "success", true,
            "message", "任务提交成功",
            "taskId", taskId,
            "taskName", taskName
        ));
    }

    @PostMapping("/submit-with-priority")
    public ResponseEntity<Map<String, Object>> submitTaskWithPriority(
            @RequestParam String taskName,
            @RequestParam TaskPriority priority,
            @RequestParam(required = false) Integer sleepSeconds,
            @RequestBody(required = false) Map<String, Object> metadata) {

        int sleepTime = sleepSeconds != null ? sleepSeconds : 1;

        String taskId = taskExecutor.submitTask(taskName, () -> {
            try {
                log.info("执行{}任务: {}", priority, taskName);
                TimeUnit.SECONDS.sleep(sleepTime);
                log.info("{}任务完成: {}", priority, taskName);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                log.error("任务被中断: {}", taskName, e);
            }
        }, priority, metadata != null ? metadata : new ConcurrentHashMap<>());

        return ResponseEntity.ok(Map.of(
            "success", true,
            "message", "任务提交成功",
            "taskId", taskId,
            "taskName", taskName,
            "priority", priority
        ));
    }

    @PostMapping("/delayed")
    public ResponseEntity<Map<String, Object>> submitDelayedTask(
            @RequestParam String taskName,
            @RequestParam Long delay,
            @RequestParam(defaultValue = "SECONDS") TimeUnit unit,
            @RequestBody(required = false) Map<String, Object> metadata) {

        String taskId = taskExecutor.submitDelayedTask(taskName, () -> {
            log.info("执行延迟任务: {}, delay={} {}", taskName, delay, unit);

    @PostMapping("/batch")
    public ResponseEntity<Map<String, Object>> submitBatch(
            @RequestParam String batchName,
            @RequestParam Integer taskCount,
            @RequestBody(required = false) Map<String, Object> metadata) {

        Map<String, Runnable> tasks = new ConcurrentHashMap<>();
        for (int i = 1; i <= taskCount; i++) {
            final int index = i;
            tasks.put("batch-task-" + i, () -> {
                try {
                    log.info("执行批量任务: {} - {}", batchName, index);
                    TimeUnit.MILLISECONDS.sleep(100);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            });
        }

        taskExecutor.submitBatch(batchName, tasks,
            metadata != null ? metadata : new ConcurrentHashMap<>());

        return ResponseEntity.ok(Map.of(
            "success", true,
            "message", "批量任务提交成功",
            "batchName", batchName,
            "taskCount", taskCount
        ));
    }

            "success", true,
            "stats", Map.of(
                "totalSubmitted", stats.getTotalSubmitted(),
                "totalProcessed", stats.getTotalProcessed(),
                "totalFailed", stats.getTotalFailed(),
                "successRate", String.format("%.2f%%", stats.getSuccessRate()),
                "queueSize", stats.getQueueSize(),
                "activeThreads", stats.getActiveThreads(),
                "trackedTasks", stats.getTrackedTasks(),
                "timestamp", stats.getTimestamp()
            )
        ));
    }
