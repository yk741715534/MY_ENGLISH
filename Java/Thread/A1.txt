import java.util.concurrent.*;
import java.util.Comparator;
import java.util.concurrent.atomic.AtomicLong;

// 任务优先级枚举
enum TaskPriority {
    HIGHEST(0),    // 最高优先级
    HIGH(1),       // 高优先级
    NORMAL(2),     // 普通优先级
    LOW(3),        // 低优先级
    LOWEST(4);     // 最低优先级

    private final int value;

    TaskPriority(int value) {
        this.value = value;
    }

    public int getValue() {
        return.value;
    }
}

// 任务包装类
class PriorityTask implements Runnable, Comparable<PriorityTask> {
    private final Runnable task;
    private final TaskPriority priority;
    private final long sequenceNumber;
    private final Object businessData;  // 业务数据
    private static final AtomicLong sequencer = new AtomicLong(0);

    public PriorityTask(Runnable task, TaskPriority priority, Object businessData) {
        this.task = task;
        this.priority = priority;
        this.businessData = businessData;
        this.sequenceNumber = sequencer.getAndIncrement();
    }

    @Override
    public void run() {
        task.run();
    }

    public TaskPriority getPriority() {
        return priority;
    }

    public Object getBusinessData() {
        return businessData;
    }

    @Override
    public int compareTo(PriorityTask other) {
        int priorityCompare = Integer.compare(this.priority.getValue(),
                                            other.priority.getValue());
        if (priorityCompare != 0) {
            return priorityCompare;
        }
        // 如果优先级相同，按FIFO顺序
        return Long.compare(this.sequenceNumber, other.sequenceNumber);
    }
}

import java.util.concurrent.*;
import java.util.function.Function;

/**
 * 可配置优先级的线程队列管理器
 */
public class PriorityThreadQueue {

    // 使用优先队列
    private final PriorityBlockingQueue<PriorityTask> taskQueue;

    // 线程池
    private final ExecutorService executorService;

    // 业务逻辑判断器
    private final Function<Object, TaskPriority> priorityEvaluator;

    /**
     * 构造函数
     * @param corePoolSize 核心线程数
     * @param maxPoolSize 最大线程数
     * @param keepAliveTime 线程保持存活时间
     * @param priorityEvaluator 业务逻辑优先级判断器
     */
    public PriorityThreadQueue(int corePoolSize, int maxPoolSize,
                              long keepAliveTime,
                              Function<Object, TaskPriority> priorityEvaluator) {
        this.taskQueue = new PriorityBlockingQueue<>();
        this.priorityEvaluator = priorityEvaluator;

        // 创建自定义线程池
        this.executorService = new ThreadPoolExecutor(
            corePoolSize,
            maxPoolSize,
            keepAliveTime,
            TimeUnit.SECONDS,
            new LinkedBlockingQueue<>(),  // 工作队列
            new PriorityThreadFactory(),
            new ThreadPoolExecutor.AbortPolicy()
        ) {
            @Override
            protected void beforeExecute(Thread t, Runnable r) {
                // 可以在执行前进行一些操作
                if (r instanceof PriorityTask) {
                    PriorityTask task = (PriorityTask) r;
                    System.out.println("开始执行任务，优先级: " + task.getPriority());
                }
            }
        };

        // 启动队列处理器
        startQueueProcessor();
    }

    /**
     * 提交任务
     * @param task 具体任务
     * @param businessData 业务数据（用于判断优先级）
     */
    public void submit(Runnable task, Object businessData) {
        TaskPriority priority = priorityEvaluator.apply(businessData);
        PriorityTask priorityTask = new PriorityTask(task, priority, businessData);
        taskQueue.offer(priorityTask);
    }

    /**
     * 启动队列处理器
     */
    private void startQueueProcessor() {
        Thread processorThread = new Thread(() -> {
            while (!executorService.isShutdown()) {
                try {
                    PriorityTask task = taskQueue.take();
                    executorService.execute(task);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    break;
                } catch (Exception e) {
                    e.printStackTrace();
                }
            }
        });
        processorThread.setDaemon(true);
        processorThread.start();
    }

    /**
     * 关闭线程池
     */
    public void shutdown() {
        executorService.shutdown();
        try {
            if (!executorService.awaitTermination(60, TimeUnit.SECONDS)) {
                executorService.shutdownNow();
            }
        } catch (InterruptedException e) {
            executorService.shutdownNow();
        }
    }

    /**
     * 自定义线程工厂
     */
    private static class PriorityThreadFactory implements ThreadFactory {
        private final AtomicInteger threadNumber = new AtomicInteger(1);
        private final String namePrefix = "PriorityThread-";

        @Override
        public Thread newThread(Runnable r) {
            Thread t = new Thread(r, namePrefix + threadNumber.getAndIncrement());
            t.setDaemon(false);
            t.setPriority(Thread.NORM_PRIORITY);
            return t;
        }
    }

    /**
     * 获取队列大小
     */
    public int getQueueSize() {
        return taskQueue.size();
    }

    /**
     * 获取活动线程数
     */
    public int getActiveThreadCount() {
        if (executorService instanceof ThreadPoolExecutor) {
            return ((ThreadPoolExecutor) executorService).getActiveCount();
        }
        return 0;
    }
}

                case "BACKUP":      // 备份任务
                    return TaskPriority.LOW;
                default:
                    return TaskPriority.NORMAL;
            }
        }
        return TaskPriority.NORMAL;
    }

        if (businessData instanceof Map) {

            String userLevel = (String) data.get("userLevel");

            switch (userLevel) {
                case "VIP":         // VIP用户
                    return TaskPriority.HIGHEST;
                case "MEMBER":      // 会员用户
                    return TaskPriority.HIGH;
                case "NORMAL":      // 普通用户
                    return TaskPriority.NORMAL;
                case "GUEST":       // 游客
                    return TaskPriority.LOW;
                default:
                    return TaskPriority.NORMAL;
            }

        return TaskPriority.NORMAL;

public class PriorityThreadQueueExample {

    public static void main(String[] args) {
        // 1. 创建优先级判断器
        Function<Object, TaskPriority> priorityEvaluator = businessData -> {
            // 这里可以实现复杂的业务逻辑判断
            if (businessData instanceof Map) {
                Map<String, Object> data = (Map<String, Object>) businessData;

                // 示例：根据任务类型和用户级别判断优先级
                String taskType = (String) data.get("taskType");
                boolean isVIP = "VIP".equals(data.get("userLevel"));

                if ("PAYMENT".equals(taskType) && isVIP) {
                    return TaskPriority.HIGHEST;
                } else if ("PAYMENT".equals(taskType)) {
                    return TaskPriority.HIGH;
                } else if ("NOTIFY".equals(taskType)) {
                    return TaskPriority.NORMAL;
                } else {
                    return TaskPriority.LOW;
                }
            }
            return TaskPriority.NORMAL;
        };

        // 2. 创建优先级线程队列
        PriorityThreadQueue queue = new PriorityThreadQueue(
            5,      // 核心线程数
            20,     // 最大线程数
            60,     // 空闲线程存活时间
            priorityEvaluator
        );

        // 3. 提交不同优先级的任务
        for (int i = 1; i <= 10; i++) {
            final int taskId = i;
            Map<String, Object> businessData = new HashMap<>();

            // 模拟不同的业务场景
            if (i % 3 == 0) {
                businessData.put("taskType", "PAYMENT");
                businessData.put("userLevel", "VIP");
            } else if (i % 3 == 1) {
                businessData.put("taskType", "NOTIFY");
                businessData.put("userLevel", "NORMAL");
            } else {
                businessData.put("taskType", "REPORT");
                businessData.put("userLevel", "GUEST");
            }

            queue.submit(() -> {
                System.out.println(Thread.currentThread().getName() +
                                 " 执行任务 " + taskId +
                                 "，业务数据: " + businessData);
                try {
                    Thread.sleep(1000); // 模拟任务执行
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            }, businessData);

            System.out.println("已提交任务 " + taskId +
                             "，当前队列大小: " + queue.getQueueSize());
        }

        // 4. 等待所有任务完成
        try {
            Thread.sleep(15000);
        } catch (InterruptedException e) {
            e.printStackTrace();
        }

        // 5. 关闭队列
        queue.shutdown();
    }
}

设计特点

灵活性：通过Function<Object, TaskPriority>接口，可以轻松实现任何复杂的业务逻辑判断

可扩展性：支持动态调整优先级、任务监控、统计等功能扩展

线程安全：使用PriorityBlockingQueue保证线程安全

可管理性：提供队列监控、状态查询等管理接口

性能优化：使用线程池管理，避免频繁创建销毁线程

你可以根据具体业务需求，调整优先级判断逻辑、线程池参数和任务处理策略。

/**
 * 事件监听器处理优先级任务

@Component
public class PriorityTaskEventListener {

    private final Map<TaskPriority, BlockingQueue<Runnable>> priorityQueues =
        new ConcurrentHashMap<>();

    private final ExecutorService[] executors;

    public PriorityTaskEventListener() {
        // 为每个优先级创建独立的线程池
        executors = new ExecutorService[TaskPriority.values().length];
        for (TaskPriority priority : TaskPriority.values()) {
            executors[priority.getValue()] = Executors.newFixedThreadPool(2);
            priorityQueues.put(priority, new LinkedBlockingQueue<>());
        }
    }

    @EventListener
    @Async("taskExecutor")
    public void handlePriorityTask(PriorityTaskEvent event) {
        TaskPriority priority = event.getPriority();
        priorityQueues.get(priority).offer(event.getTask());

        // 从对应优先级的队列中获取任务执行
        executors[priority.getValue()].submit(() -> {
            try {
                Runnable task = priorityQueues.get(priority).take();
                task.run();
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        });
    }
}

/**
 * 基于 @Order 注解的优先级任务执行器
 */
@Component
public class OrderedTaskExecutor {

    private final List<OrderedTaskHandler> handlers = new CopyOnWriteArrayList<>();

    /**
     * 收集所有 OrderedTaskHandler
     */
    @Autowired(required = false)
    public void setHandlers(List<OrderedTaskHandler> handlers) {
        this.handlers.addAll(handlers);
        // 按 @Order 注解排序
        this.handlers.sort(AnnotationAwareOrderComparator.INSTANCE);
    }

    /**
     * 按顺序执行任务
     */
    public void executeWithOrder(Runnable task, Object context) {
        for (OrderedTaskHandler handler : handlers) {
            if (handler.canHandle(context)) {
                handler.handle(task, context);
                break;
            }
        }
    }
}

/**
 * 优先级任务处理器
 */
public interface OrderedTaskHandler {
    boolean canHandle(Object context);
    void handle(Runnable task, Object context);
}

/**
 * 高优先级处理器
 */
@Component
@Order(Ordered.HIGHEST_PRECEDENCE)
class HighPriorityHandler implements OrderedTaskHandler {
    @Override
    public boolean canHandle(Object context) {
        return "HIGH".equals(((Map) context).get("priority"));
    }

    @Override
    public void handle(Runnable task, Object context) {
        // 立即执行
        task.run();
    }
}

        // 从消息头获取优先级
        Integer priority = message.getHeaders().get("priority", Integer.class);

        if (priority != null && priority > 5) {
            // 高优先级任务立即处理
            return processImmediately(message);
        } else {
            // 低优先级任务放入队列延迟处理
            return processLater(message);
        }
    }
}

@Service
public class PriorityTaskManager {

    @Autowired
    @Qualifier("mainPriorityExecutor")
    private ThreadPoolTaskExecutor executor;

    @Autowired
    private ApplicationEventPublisher eventPublisher;

    /**
     * 提交任务，根据业务逻辑自动判断优先级
     */
    public <T> CompletableFuture<T> submit(Callable<T> task, Object businessData) {
        // 判断优先级
        TaskPriority priority = evaluatePriority(businessData);

        // 包装为优先级任务
        PriorityCallable<T> priorityTask = new PriorityCallable<>(task, priority, businessData);

        // 发布事件（可选）
        eventPublisher.publishEvent(
            new PriorityTaskEvent(this, priority,
                () -> priorityTask.call(), businessData));

        // 提交执行
        return CompletableFuture.supplyAsync(() -> {
            try {
                return priorityTask.call();
            } catch (Exception e) {
                throw new CompletionException(e);
            }
        }, executor);
    }

    private TaskPriority evaluatePriority(Object businessData) {
        // 实现你的业务逻辑
        return TaskPriority.NORMAL;
    }

总结

Spring 框架中可以直接使用或借鉴的组件：

ThreadPoolTaskExecutor​ - Spring 封装的线程池，推荐使用

@Async注解​ - 声明式异步执行

TaskExecutorBuilder​ - Spring Boot 的线程池构建器

PriorityChannel​ - Spring Integration 的优先级通道

@Order和 Ordered接口​ - 优先级排序机制

ApplicationEvent 机制​ - 事件驱动的任务调度

Spring Batch 的 TaskExecutor​ - 批处理任务的线程池管理

DeferredResult 和 Callable​ - Web 异步处理

推荐方案：

对于 Web 应用，使用 ThreadPoolTaskExecutor+ @Async

对于批处理，使用 Spring Batch

对于消息驱动，使用 Spring Integration

对于微服务，使用 Spring Cloud Stream

Spring 提供的这些组件已经很好地封装了线程池的复杂性，你只需要关注业务逻辑的优先级判断即可。
